dword home_star;
str expected_name;
int planet_exists;
int planet_count;
dword existing_planet;

// Гархарл — домашняя система малоков.
home_star = GalaxyStar(Maloc);
if(home_star)
{
    expected_name = CT('Script.XenoPlanets.0');
    planet_exists = 0;
    planet_count = StarPlanets(home_star);
    for(i = 0; i < planet_count; i = i + 1)
    {
        existing_planet = StarPlanets(home_star, i);
        if(existing_planet)
        {
            if(lowercase(PlanetName(existing_planet)) == lowercase(expected_name)) planet_exists = 1;
        }
    }
    if(!planet_exists)
    {
        planet = CreatePlanet(home_star);
        if(planet)
        {
            SetName(planet, expected_name);
            PlanetSetGraph(planet, 208);
            PlanetTerrain(planet, 0, 400);
            PlanetTerrain(planet, 1, 600);
            PlanetTerrain(planet, 2, 0);
            for(i = PlanetItems(planet) - 1; i >= 0; i = i - 1) AddItemToPlanet(planet, GetItemFromPlanet(planet, i));
        }
    }
}
