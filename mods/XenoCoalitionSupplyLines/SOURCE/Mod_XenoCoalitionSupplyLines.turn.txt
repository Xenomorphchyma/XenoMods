// Clean release build: only explicitly versioned storage migrations are kept.
// General compatibility between arbitrary development saves is not promised.

function IdToStar(int star_id)
{
    result = 0;
    if(!star_id) exit;
    for(int i = 0; i < GalaxyStars(); i = i + 1)
    {
        dword star = GalaxyStar(i);
        if(!star) continue;
        if(Id(star) == star_id)
        {
            result = star;
            exit;
        }
    }
}

function CSL_ShipCurrentStar(dword ship)
{
    result = 0;
    // ShipInNormalSpace may become true one engine tick before a completed
    // OrderTakeOff has attached every internal space pointer. ShipStar and
    // ShipCanJump are still unsafe during that transition.
    if(!ship) exit;
    if(ShipIsTakeoff(ship)) exit;
    // ShipStar is unsafe for a docked ship. Resolve the dock first and call
    // ShipStar only when the ship is proven to be in normal space.
    dword ship_planet = GetShipPlanet(ship);
    if(ship_planet)
    {
        result = PlanetToStar(ship_planet);
        exit;
    }
    dword ship_ruins = GetShipRuins(ship);
    if(ship_ruins)
    {
        if(ShipInNormalSpace(ship_ruins)) result = ShipStar(ship_ruins);
        exit;
    }
    if(ShipInNormalSpace(ship)) result = ShipStar(ship);
}

// ShipGetBad may retain a raw handle after its target has left the system or
// been removed. Never call spatial or ship APIs on that handle directly.
// Resolve it only by equality with a ship that the current star enumerates.
function CSL_ResolveNormalShipInStar(dword raw_ship, dword expected_star)
{
    result = 0;
    if(!raw_ship || !expected_star) exit;
    dword live_player = Player();
    if(raw_ship == live_player)
    {
        if(ShipIsTakeoff(live_player) || ShipInHyperSpace(live_player, 1) || !ShipInNormalSpace(live_player)) exit;
        if(CSL_ShipCurrentStar(live_player) != expected_star) exit;
        result = live_player;
        exit;
    }
    int live_star_ship_count = StarShips(expected_star);
    for(int live_star_ship_index = 0; live_star_ship_index < live_star_ship_count; live_star_ship_index = live_star_ship_index + 1)
    {
        dword live_star_ship = StarShips(expected_star, live_star_ship_index);
        if(live_star_ship != raw_ship) continue;
        if(!live_star_ship) exit;
        if(!ShipInNormalSpace(live_star_ship)) exit;
        // StarShips also enumerates stations. ShipIsTakeoff is valid only for
        // mobile ships, whose type range ends immediately before t_RC.
        if(ShipTypeN(live_star_ship) >= t_RC) exit;
        if(ShipIsTakeoff(live_star_ship) || ShipInHyperSpace(live_star_ship, 1)) exit;
        result = live_star_ship;
        exit;
    }
}

function CSL_ClearPlayerJoin()
{
    csl_player_joined = 0;
    csl_player_wait_ship_id = 0;
}

// Joining the departure is represented only by the follow order of Player().
// Clicking another destination replaces that order and therefore cancels the
// registration without ever placing Player() in the scripted strike group.
function CSL_PlayerWaitingForStrike(dword ready_star)
{
    result = 0;
    if(!csl_player_joined || !ready_star) exit;
    if(!ShipInNormalSpace(Player()) || ShipInHyperSpace(Player(), 1) || ShipIsTakeoff(Player()))
    {
        CSL_ClearPlayerJoin();
        exit;
    }
    dword wait_ship = 0;
    if(csl_player_wait_ship_id > 1) wait_ship = IdToShip(csl_player_wait_ship_id);
    if(!wait_ship)
    {
        CSL_ClearPlayerJoin();
        exit;
    }
    if(!ShipInNormalSpace(wait_ship) || ShipInHyperSpace(wait_ship, 1) || ShipIsTakeoff(wait_ship))
    {
        CSL_ClearPlayerJoin();
        exit;
    }
    if(CSL_ShipCurrentStar(Player()) != ready_star || CSL_ShipCurrentStar(wait_ship) != ready_star)
    {
        CSL_ClearPlayerJoin();
        exit;
    }
    // A defensive sortie may temporarily replace the player follow order.
    // Remaining in the rally system keeps the accepted invitation alive until
    // the military group has driven the attacker away and re-formed.
    if(!csl_strike_defense_active && (ShipOrder(Player()) != 6 || ShipOrderObj(Player()) != wait_ship))
    {
        CSL_ClearPlayerJoin();
        exit;
    }
    result = 1;
}

function CSL_IsStrategicStar(dword star)
{
    result = 0;
    if(!star) exit;
    dword strategic_pirate_planet = PlanetPirateClan();
    dword strategic_pirate_star = 0;
    if(strategic_pirate_planet) strategic_pirate_star = PlanetToStar(strategic_pirate_planet);
    if(strategic_pirate_star && StarToCon(star) == StarToCon(strategic_pirate_star)) exit;
    if(StarCustomFaction(star) != '') exit;
    int strategic_owner = StarOwner(star);
    if(strategic_owner == 0 || strategic_owner == 1 || strategic_owner == 2) result = 1;
}

function CSL_IsFreeStar(dword star)
{
    result = 0;
    if(!CSL_IsStrategicStar(star)) exit;
    if(StarOwner(star) != 0) exit;
    if(StarBattle(star)) exit;
    if(StarEnemyThreatLevel(star, 1) >= 2) exit;
    result = 1;
}

function CSL_RefreshStrategy()
{
    if(csl_strategy_refresh_turn == CurTurn()) exit;
    csl_free_systems = 0;
    csl_strategic_systems = 0;
    csl_hostile_systems = 0;
    for(int strategic_index = 0; strategic_index < GalaxyStars(); strategic_index = strategic_index + 1)
    {
        dword strategic_star = GalaxyStar(strategic_index);
        if(!CSL_IsStrategicStar(strategic_star)) continue;
        csl_strategic_systems = csl_strategic_systems + 1;
        if(CSL_IsFreeStar(strategic_star)) csl_free_systems = csl_free_systems + 1;
        else if(StarOwner(strategic_star) == 1 || StarOwner(strategic_star) == 2)
            csl_hostile_systems = csl_hostile_systems + 1;
    }
    csl_strategy_percent = 0;
    if(csl_strategic_systems > 0)
        csl_strategy_percent = max(0, min(100, round(100.0 * csl_free_systems / csl_strategic_systems)));
    csl_strategy_refresh_turn = CurTurn();
}

function CSL_StrategyBand()
{
    result = 0;
    if(csl_free_systems < 3) { result = 7; exit; }
    if(csl_strategy_percent >= 80) result = 1;
    else if(csl_strategy_percent >= 60) result = 2;
    else if(csl_strategy_percent >= 40) result = 3;
    else if(csl_strategy_percent >= 20) result = 4;
    else if(csl_strategy_percent >= 11) result = 5;
    else result = 6;
}

// Values are read once by the optional Native Loader plugin from the INI next
// to its DLL.  If the loader is absent or the plugin is disabled, the exported
// getter returns -1 and the script keeps these built-in release defaults.
function CSL_ConfigValue(int config_key, int fallback_value)
{
    result = fallback_value;
    unknown config_getter = ImportedFunction('XenoCoalitionSupplyLinesConfig', 'XenoCoalitionSupplyLines_GetConfigInt');
    int configured_value = config_getter(config_key);
    if(configured_value >= 0) result = configured_value;
}

function CSL_IsFrontier(dword star)
{
    result = 0;
    if(!star) exit;
    if(!CSL_IsFreeStar(star)) exit;
    dword frontier_jump_ship = CSL_ActiveTransportLeader();
    if(!frontier_jump_ship) exit;
    for(int frontier_enemy_index = 0; frontier_enemy_index < GalaxyStars(); frontier_enemy_index = frontier_enemy_index + 1)
    {
        dword frontier_enemy = GalaxyStar(frontier_enemy_index);
        if(!CSL_IsStrategicStar(frontier_enemy)) continue;
        int frontier_owner = StarOwner(frontier_enemy);
        if((frontier_owner == 1 || frontier_owner == 2) && ShipCanJump(frontier_jump_ship, star, frontier_enemy, 1))
        {
            result = 1;
            exit;
        }
    }
}

function CSL_FindSourceForPreparation(int safe_count)
{
    dword selected = 0;
    int source_star_count = GalaxyStars();
    // Collect exact candidate/hostile subsets once.  The previous flattened
    // S*S pass repeated strategic predicates for every unrelated pair.
    unknown source_candidate_planets = newarray(1);
    unknown source_candidate_stars = newarray(1);
    unknown source_hostile_stars = newarray(1);
    unknown source_safety_values = newarray(1);
    ArrayClear(source_candidate_planets);
    ArrayClear(source_candidate_stars);
    ArrayClear(source_hostile_stars);
    ArrayClear(source_safety_values);
    for(int source_collect_index = 0; source_collect_index < source_star_count; source_collect_index = source_collect_index + 1)
    {
        dword source_collect_star = GalaxyStar(source_collect_index);
        if(!source_collect_star) continue;
        if(CSL_IsStrategicStar(source_collect_star))
        {
            int source_collect_owner = StarOwner(source_collect_star);
            if(source_collect_owner == 1 || source_collect_owner == 2) ArrayAdd(source_hostile_stars, source_collect_star);
        }
        if(CSL_IsFreeStar(source_collect_star))
        {
            dword source_collect_planet = CSL_FindOwnedPlanet(source_collect_star, 1);
            if(source_collect_planet)
            {
                ArrayAdd(source_candidate_planets, source_collect_planet);
                ArrayAdd(source_candidate_stars, source_collect_star);
            }
        }
    }
    if(ArrayDim(source_candidate_planets) <= 1) { result = 0; exit; }

    int source_candidate_count = ArrayDim(source_candidate_stars) - 1;
    int source_hostile_count = ArrayDim(source_hostile_stars) - 1;
    for(int source_safety_init = 1; source_safety_init <= source_candidate_count; source_safety_init = source_safety_init + 1)
        ArrayAdd(source_safety_values, 1000000);
    // Exact candidate/hostile matrix, flattened so it cannot create a nested
    // world traversal in the turn call graph.
    int source_score_pair_limit = source_candidate_count * source_hostile_count;
    for(int source_score_pair = 0; source_score_pair < source_score_pair_limit; source_score_pair = source_score_pair + 1)
    {
        int source_score_index = source_score_pair / source_hostile_count + 1;
        int source_hostile_index = source_score_pair - (source_score_index - 1) * source_hostile_count + 1;
        dword source_score_star = source_candidate_stars[source_score_index];
        dword source_hostile_star = source_hostile_stars[source_hostile_index];
        int source_hostile_distance = Dist(source_score_star, source_hostile_star);
        if(source_hostile_distance < source_safety_values[source_score_index]) source_safety_values[source_score_index] = source_hostile_distance;
    }

    int maximum_source_safety = -1;
    for(int source_max_index = 1; source_max_index <= source_candidate_count; source_max_index = source_max_index + 1)
    {
        int source_score_safety = source_safety_values[source_max_index];
        if(source_score_safety > maximum_source_safety) maximum_source_safety = source_score_safety;
    }

    int equal_choices = 0;
    for(int source_pick_index = 1; source_pick_index < ArrayDim(source_candidate_planets); source_pick_index = source_pick_index + 1)
    {
        dword source_planet = source_candidate_planets[source_pick_index];
        int source_safety = source_safety_values[source_pick_index];
        if(safe_count >= 5 && source_safety * 100 < maximum_source_safety * 60) continue;
        equal_choices = equal_choices + 1;
        if(RndObject(1, equal_choices, source_planet) == 1) selected = source_planet;
    }
    result = selected;
}

function CSL_FindTarget(dword frontier_star, dword jump_ship)
{
    dword selected = 0;
    int best_threat = -1;
    int best_distance = 1000000;
    if(!frontier_star || !jump_ship) { result = 0; exit; }
    for(int i = 0; i < GalaxyStars(); i = i + 1)
    {
        dword star = GalaxyStar(i);
        if(!CSL_IsStrategicStar(star)) continue;
        int target_owner = StarOwner(star);
        if(target_owner != 1 && target_owner != 2) continue;
        if(!ShipCanJump(jump_ship, frontier_star, star, 1)) continue;
        int strike_group_reachable = 1;
        for(int strike_route_index = 0; strike_route_index < GroupCount(SupplyStrike); strike_route_index = strike_route_index + 1)
        {
            dword strike_route_ship = GroupShip(SupplyStrike, strike_route_index);
            if(!strike_route_ship) continue;
            if(!ShipCanJump(strike_route_ship, frontier_star, star, 1))
            {
                strike_group_reachable = 0;
                break;
            }
        }
        if(!strike_group_reachable) continue;
        int threat = StarEnemyThreatLevel(star, 1);
        int enemy_distance = Dist(frontier_star, star);
        if(threat > best_threat || (threat == best_threat && enemy_distance < best_distance))
        {
            best_threat = threat;
            best_distance = enemy_distance;
            selected = star;
        }
    }
    result = selected;
}

function CSL_InflationFactor()
{
    int factor = 100;
    int start_capital = 0;
    int current_capital = 0;
    if(IsScriptActive('Mod_EvoFreeInflation'))
    {
        start_capital = GetValueFromScript('Mod_EvoFreeInflation', 'start_capital_amount');
        current_capital = GetValueFromScript('Mod_EvoFreeInflation', 'cur_capital_amount');
        if(start_capital > 0 && current_capital > 0)
        {
            int max_gtl = max(GalaxyTechLevel(), GetValueFromScript('Mod_EvoFreeInflation', 'max_reached_GTL'));
            int tech_factor = round(PortionInDiapason(max(1, min(8, max_gtl)), 1, 8, 100, 400));
            int combined = max(100, round(100.0 * current_capital / start_capital));
            factor = max(100, round(100.0 * combined / tech_factor));
        }
    }
    else
    {
        start_capital = round(PortionInDiapason(GalaxyDiffLevels(1), 50, 500, 75000, 10000));
        current_capital = RangersCapital();
        if(start_capital > 0 && current_capital > 0)
            factor = max(100, round(100.0 * current_capital / start_capital));
    }
    result = factor;
}

function CSL_ClearTransportCargo(dword ship)
{
    if(!ship) exit;
    // Vanilla trade goods are stored separately from hull equipment.
    // Remove only those goods and keep the ordinary transport loadout intact.
    for(int goods_type = t_Food; goods_type <= t_Narcotics; goods_type = goods_type + 1)
    {
        int goods_count = GoodsCount(ship, goods_type);
        if(goods_count > 0) GoodsAdd(ship, goods_type, -goods_count);
    }
    ShipCalcParam(ship);
}

function CSL_RemoveTransportItem(dword ship, dword item)
{
    if(!ship || !item) exit;
    ItemIsInUse(item, ship, 0);
    int item_position = FindItemInShip(ship, item);
    dword removed_item = 0;
    if(item_position >= 0) removed_item = GetItemFromShip(ship, item_position);
    if(removed_item) FreeItem(removed_item);
}

// RScript does not guarantee short-circuit evaluation for && and ||. Keep the
// nullable item guard in a separate control-flow step so GetEquipmentStats is
// never called with the zero handle returned by ShipEqInSlot.
function CSL_EquipmentStatOrZero(dword equipment, int stat_index)
{
    result = 0;
    if(!equipment) exit;
    result = GetEquipmentStats(equipment, stat_index);
}

function CSL_ConfigureTransport(dword ship)
{
    result = 0;
    if(!ship) exit;
    // The hull and pilot remain the ordinary BuyTransport result. Coalition
    // money is invested mainly in cargo: retain exactly one ordinary weapon
    // generated by BuyTransport, remove the rest and all optional systems,
    // then guarantee engine, tank, repair droid and shield.
    // Weapon slots are fixed and limited to five. Unrolling keeps convoy
    // construction bounded even when several transports are created together.
    dword kept_weapon = 0;
    dword weapon_item = ShipEqInSlot(ship, t_Weapon1, 1); if(weapon_item) kept_weapon = weapon_item;
    weapon_item = ShipEqInSlot(ship, t_Weapon1, 2); if(weapon_item) { if(!kept_weapon) kept_weapon = weapon_item; else CSL_RemoveTransportItem(ship, weapon_item); }
    weapon_item = ShipEqInSlot(ship, t_Weapon1, 3); if(weapon_item) { if(!kept_weapon) kept_weapon = weapon_item; else CSL_RemoveTransportItem(ship, weapon_item); }
    weapon_item = ShipEqInSlot(ship, t_Weapon1, 4); if(weapon_item) { if(!kept_weapon) kept_weapon = weapon_item; else CSL_RemoveTransportItem(ship, weapon_item); }
    weapon_item = ShipEqInSlot(ship, t_Weapon1, 5); if(weapon_item) { if(!kept_weapon) kept_weapon = weapon_item; else CSL_RemoveTransportItem(ship, weapon_item); }
    dword optional_item = ShipEqInSlot(ship, t_Radar);
    if(optional_item) CSL_RemoveTransportItem(ship, optional_item);
    optional_item = ShipEqInSlot(ship, t_Scaner);
    if(optional_item) CSL_RemoveTransportItem(ship, optional_item);
    optional_item = ShipEqInSlot(ship, t_CargoHook);
    if(optional_item) CSL_RemoveTransportItem(ship, optional_item);

    int equipment_level = max(1, min(8, GalaxyTechLevel()));
    int equipment_owner = ShipOwner(ship);
    dword transport_hull = ShipEqInSlot(ship, t_Hull);
    if(transport_hull)
    {
        int transport_droid_slots = CSL_EquipmentStatOrZero(transport_hull, 5) + ShipSpecialBonuses(ship, bonSlotDroid);
        if(transport_droid_slots < 1) ShipSpecialBonuses(ship, bonSlotDroid, 1);
        int transport_def_slots = CSL_EquipmentStatOrZero(transport_hull, 7) + ShipSpecialBonuses(ship, bonSlotDef);
        if(transport_def_slots < 1) ShipSpecialBonuses(ship, bonSlotDef, 1);
    }
    dword required_item = 0;
    if(!ShipEqInSlot(ship, t_Engine))
    {
        required_item = CreateEquipment(t_Engine, 30 + 5 * equipment_level, equipment_level, equipment_owner);
        if(required_item) { AddItemToShip(ship, required_item); ItemIsInUse(required_item, ship, 1); }
    }
    if(!ShipEqInSlot(ship, t_FuelTanks))
    {
        required_item = CreateEquipment(t_FuelTanks, 35 + 6 * equipment_level, equipment_level, equipment_owner);
        if(required_item) { AddItemToShip(ship, required_item); ItemIsInUse(required_item, ship, 1); }
    }
    if(!ShipEqInSlot(ship, t_RepairRobot))
    {
        required_item = CreateEquipment(t_RepairRobot, 25 + 4 * equipment_level, equipment_level, equipment_owner);
        if(required_item) { AddItemToShip(ship, required_item); ItemIsInUse(required_item, ship, 1); }
    }
    if(!ShipEqInSlot(ship, t_DefGenerator))
    {
        required_item = CreateEquipment(t_DefGenerator, 35 + 6 * equipment_level, equipment_level, equipment_owner);
        if(required_item) { AddItemToShip(ship, required_item); ItemIsInUse(required_item, ship, 1); }
    }
    ShipCalcParam(ship);
    ShipRefuel(ship);
    ShipRepairEq(ship);
    if(!ShipEqInSlot(ship, t_Engine) || !ShipEqInSlot(ship, t_FuelTanks) ||
       !ShipEqInSlot(ship, t_RepairRobot) || !ShipEqInSlot(ship, t_DefGenerator)) exit;
    int equipped_weapon_count = 0;
    if(ShipEqInSlot(ship, t_Weapon1, 1)) equipped_weapon_count = equipped_weapon_count + 1;
    if(ShipEqInSlot(ship, t_Weapon1, 2)) equipped_weapon_count = equipped_weapon_count + 1;
    if(ShipEqInSlot(ship, t_Weapon1, 3)) equipped_weapon_count = equipped_weapon_count + 1;
    if(ShipEqInSlot(ship, t_Weapon1, 4)) equipped_weapon_count = equipped_weapon_count + 1;
    if(ShipEqInSlot(ship, t_Weapon1, 5)) equipped_weapon_count = equipped_weapon_count + 1;
    if(equipped_weapon_count != 1) exit;
    result = 1;
}

function CSL_EnsureAmbushPirateHook(dword pirate)
{
    result = 0;
    if(!pirate) exit;
    dword pirate_hook = ShipEqInSlot(pirate, t_CargoHook);
    int pirate_hook_capacity = CSL_EquipmentStatOrZero(pirate_hook, 0);
    if(pirate_hook_capacity >= 60)
    {
        result = 1;
        exit;
    }
    if(pirate_hook) CSL_RemoveTransportItem(pirate, pirate_hook);
    int pirate_hook_level = max(1, min(8, GalaxyTechLevel()));
    dword new_pirate_hook = CreateEquipment(t_CargoHook, 60, pirate_hook_level, ShipOwner(pirate));
    if(!new_pirate_hook) exit;
    AddItemToShip(pirate, new_pirate_hook);
    ItemIsInUse(new_pirate_hook, pirate, 1);
    pirate_hook = ShipEqInSlot(pirate, t_CargoHook);
    if(!pirate_hook)
    {
        int pirate_hook_bonus = ShipSpecialBonuses(pirate, bonSlotHook);
        ShipSpecialBonuses(pirate, bonSlotHook, pirate_hook_bonus + 1);
        ItemIsInUse(new_pirate_hook, pirate, 1);
        pirate_hook = ShipEqInSlot(pirate, t_CargoHook);
    }
    ShipCalcParam(pirate);
    pirate_hook_capacity = CSL_EquipmentStatOrZero(pirate_hook, 0);
    if(pirate_hook_capacity >= 60) result = 1;
}

function CSL_ThreatWithinConvoyLeash(dword threat, dword leader)
{
    result = 0;
    if(!threat || !leader) exit;
    if(!ShipInNormalSpace(threat)) exit;
    if(ShipInHyperSpace(leader)) exit;
    int threat_id = Id(threat);
    if(CSL_IsTransportId(threat_id) || CSL_IsEscortId(threat_id)) exit;
    dword convoy_anchor = CSL_ConvoyAnchor(leader);
    if(!convoy_anchor) exit;
    dword convoy_star = CSL_ShipCurrentStar(leader);
    if(!convoy_star || CSL_ShipCurrentStar(threat) != convoy_star || Dist(threat, convoy_anchor) > 1000) exit;
    result = 1;
}

function CSL_FindNearestTransportToShip(dword anchor_ship)
{
    result = 0;
    if(!anchor_ship) exit;
    dword fallback_transport = 0;
    dword anchor_place = GetShipPlanet(anchor_ship);
    if(!anchor_place) anchor_place = GetShipRuins(anchor_ship);
    dword anchor_star = CSL_ShipCurrentStar(anchor_ship);
    dword anchor_object = anchor_ship;
    if(anchor_place) anchor_object = anchor_place;
    int anchor_can_measure = anchor_place || ShipInNormalSpace(anchor_ship);
    int nearest_transport_distance = 1000000;
    for(int nearest_transport_index = 0; nearest_transport_index < GroupCount(SupplyTransports); nearest_transport_index = nearest_transport_index + 1)
    {
        dword nearest_transport = GroupShip(SupplyTransports, nearest_transport_index);
        if(!nearest_transport) continue;
        if(!CSL_IsTransportId(Id(nearest_transport))) continue;
        dword transport_anchor = CSL_ConvoyAnchor(nearest_transport);
        dword transport_star = CSL_ShipCurrentStar(nearest_transport);
        if(!transport_anchor || !transport_star) continue;
        if(!fallback_transport) fallback_transport = nearest_transport;
        if(anchor_can_measure && anchor_object && anchor_star && transport_star == anchor_star)
        {
            int transport_distance = Dist(anchor_object, transport_anchor);
            if(transport_distance < nearest_transport_distance)
            {
                nearest_transport_distance = transport_distance;
                result = nearest_transport;
            }
        }
    }
    if(!result) result = fallback_transport;
}

function CSL_FindNearestEscortToShip(dword anchor_ship)
{
    result = 0;
    if(!anchor_ship) exit;
    dword anchor_star = CSL_ShipCurrentStar(anchor_ship);
    if(!anchor_star) exit;
    int nearest_escort_distance = 1000000;
    for(int nearest_escort_index = 0; nearest_escort_index < GroupCount(SupplyEscorts); nearest_escort_index = nearest_escort_index + 1)
    {
        dword nearest_escort = GroupShip(SupplyEscorts, nearest_escort_index);
        if(!nearest_escort) continue;
        if(!CSL_IsEscortId(Id(nearest_escort))) continue;
        if(ShipIsTakeoff(nearest_escort) || ShipInHyperSpace(nearest_escort, 1) || !ShipInNormalSpace(nearest_escort)) continue;
        if(CSL_ShipCurrentStar(nearest_escort) != anchor_star) continue;
        int escort_distance = Dist(anchor_ship, nearest_escort);
        if(escort_distance >= nearest_escort_distance) continue;
        nearest_escort_distance = escort_distance;
        result = nearest_escort;
    }
}

// A planet moves along its orbit every day, so its current coordinates cannot
// be used as a persistent space rally. Ask the engine for a real jump point,
// preserve those safe absolute coordinates, then replace the temporary jump
// with a move order. ShuWarriors issues OrderJump directly after a completed
// takeoff; probing ShipCanJump here is unsafe during the engine transition.
function CSL_EnsureSpaceRally(dword rally_anchor_ship)
{
    result = 0;
    if(csl_strike_rally_ready || !rally_anchor_ship || !csl_destination) exit;
    if(ShipIsTakeoff(rally_anchor_ship) || !ShipInNormalSpace(rally_anchor_ship) || ShipInHyperSpace(rally_anchor_ship, 1)) exit;
    dword rally_star = PlanetToStar(csl_destination);
    if(CSL_ShipCurrentStar(rally_anchor_ship) != rally_star) exit;
    dword rally_guide_star = StarNearbyStars(rally_star, 1);
    if(!rally_guide_star || rally_guide_star == rally_star) exit;
    int fallback_rally_x = CoordX(rally_anchor_ship);
    int fallback_rally_y = CoordY(rally_anchor_ship);
    OrderLock(rally_anchor_ship, 1);
    OrderJump(rally_anchor_ship, rally_guide_star);
    if(ShipOrder(rally_anchor_ship) != 3)
    {
        OrderMove(rally_anchor_ship, fallback_rally_x, fallback_rally_y);
        exit;
    }
    int engine_rally_x = ShipOrder(rally_anchor_ship, 'X');
    int engine_rally_y = ShipOrder(rally_anchor_ship, 'Y');
    if(!engine_rally_x && !engine_rally_y)
    {
        OrderMove(rally_anchor_ship, fallback_rally_x, fallback_rally_y);
        exit;
    }
    csl_strike_rally_x = engine_rally_x;
    csl_strike_rally_y = engine_rally_y;
    csl_strike_rally_ready = 1;
    OrderMove(rally_anchor_ship, csl_strike_rally_x, csl_strike_rally_y);
    result = 1;
}

function CSL_FindEnemyShipInStar(dword enemy_star)
{
    result = 0;
    if(!enemy_star) exit;
    for(int enemy_ship_index = 0; enemy_ship_index < StarShips(enemy_star); enemy_ship_index = enemy_ship_index + 1)
    {
        dword enemy_ship = StarShips(enemy_star, enemy_ship_index);
        if(!enemy_ship) continue;
        if(ShipInNormalSpace(enemy_ship))
        {
            int enemy_standing = ShipStanding(enemy_ship);

            {
                result = enemy_ship;
                exit;
            }
        }
    }
}

// The ShuWarriors attack state first restores the ordinary military AI and only
// then assigns an explicit same-system focus. This keeps the group advancing
// when a target dies and lets vanilla logic handle an enemy system whose next
// valid target has not appeared yet.
function CSL_CommandStrikeCombat(dword combat_star)
{
    if(!combat_star) exit;
    dword combat_target = CSL_FindEnemyShipInStar(combat_star);
    int player_in_battle = ShipInNormalSpace(Player()) && !ShipInHyperSpace(Player(), 1) && !ShipIsTakeoff(Player()) && CSL_ShipCurrentStar(Player()) == combat_star;
    for(int command_index = 0; command_index < GroupCount(SupplyStrike); command_index = command_index + 1)
    {
        dword command_ship = GroupShip(SupplyStrike, command_index);
        if(!command_ship) continue;
        if(ShipInHyperSpace(command_ship, 1) || ShipIsTakeoff(command_ship) || !ShipInNormalSpace(command_ship)) continue;
        if(CSL_ShipCurrentStar(command_ship) != combat_star) continue;
        if(player_in_battle && ShipGetBad(command_ship) == Player()) csl_disqualified = 1;
        if(combat_target)
        {
            if(ShipOrder(command_ship) != 6 || ShipOrderObj(command_ship) != combat_target)
            {
                OrderLock(command_ship, 1);
                OrderFollowShip(command_ship, combat_target, 1);
                ShipSetBad(command_ship, combat_target);
            }
        }
        else
        {
            OrderLock(command_ship, 0);
            ShipFreeFlight(command_ship);
        }
    }
}

function CSL_ResetVolunteerRangerIds()
{
    // The operation accepts no more than six NPC rangers. Slot 7 is an
    // unused guard and no code relies on a persisted live ArrayDim value.
    csl_volunteer_ranger_ids = newarray(8);
    csl_volunteer_ranger_ids[0] = 0;
    csl_volunteer_ranger_ids[1] = 0;
    csl_volunteer_ranger_ids[2] = 0;
    csl_volunteer_ranger_ids[3] = 0;
    csl_volunteer_ranger_ids[4] = 0;
    csl_volunteer_ranger_ids[5] = 0;
    csl_volunteer_ranger_ids[6] = 0;
    csl_volunteer_ranger_ids[7] = 0;
}

function CSL_VolunteerCount()
{
    result = 0;
    for(int volunteer_count_index = 1; volunteer_count_index <= 6; volunteer_count_index = volunteer_count_index + 1)
        if(IdToShip(csl_volunteer_ranger_ids[volunteer_count_index])) result = result + 1;
}

function CSL_RecruitVolunteerRanger(dword volunteer_star, dword volunteer_target)
{
    result = 0;
    if(!volunteer_star || !volunteer_target || CSL_VolunteerCount() >= 6) exit;
    dword volunteer_leader = GroupToShip(SupplyStrike);
    if(!volunteer_leader) exit;
    int volunteer_store_index = 0;
    for(int volunteer_slot_index = 1; volunteer_slot_index <= 6; volunteer_slot_index = volunteer_slot_index + 1)
    {
        if(csl_volunteer_ranger_ids[volunteer_slot_index] && IdToShip(csl_volunteer_ranger_ids[volunteer_slot_index])) continue;
        csl_volunteer_ranger_ids[volunteer_slot_index] = 0;
        volunteer_store_index = volunteer_slot_index;
        break;
    }
    if(!volunteer_store_index) exit;
    for(int volunteer_candidate_index = 0; volunteer_candidate_index < StarShips(volunteer_star); volunteer_candidate_index = volunteer_candidate_index + 1)
    {
        dword volunteer_candidate = StarShips(volunteer_star, volunteer_candidate_index);
        if(!volunteer_candidate) continue;
        if(volunteer_candidate == Player()) continue;
        if(ShipTypeN(volunteer_candidate) >= t_RC) continue;
        if(ShipTypeN(volunteer_candidate) != t_Ranger) continue;
        if(ShipOnSidePirateClan(volunteer_candidate) || ShipInGameEvent(volunteer_candidate) || ShipInScript(volunteer_candidate, 0)) continue;
        if(!ShipInNormalSpace(volunteer_candidate) || ShipIsTakeoff(volunteer_candidate) || ShipInHyperSpace(volunteer_candidate, 1)) continue;
        if(!ShipCanJump(volunteer_candidate, volunteer_star, volunteer_target, 1)) continue;
        // RScript 4.10f accepts no more than four real TGroup objects in one
        // project. Volunteers therefore share the strike group while their
        // identities remain separate in csl_volunteer_ranger_ids.
        ShipJoin(SupplyStrike, volunteer_candidate);
        csl_volunteer_ranger_ids[volunteer_store_index] = Id(volunteer_candidate);
        OrderLock(volunteer_candidate, 1);
        OrderFollowShip(volunteer_candidate, volunteer_leader, 0, 1);
        result = 1;
        exit;
    }
}

function CSL_PruneVolunteerRangers(dword volunteer_from_star, dword volunteer_target_star)
{
    result = 0;
    for(int volunteer_prune_index = 6; volunteer_prune_index >= 1; volunteer_prune_index = volunteer_prune_index - 1)
    {
        dword volunteer_prune_ship = IdToShip(csl_volunteer_ranger_ids[volunteer_prune_index]);
        if(!volunteer_prune_ship)
        {
            csl_volunteer_ranger_ids[volunteer_prune_index] = 0;
            continue;
        }
        if(volunteer_from_star && volunteer_target_star)
        {
            if(ShipCanJump(volunteer_prune_ship, volunteer_from_star, volunteer_target_star, 1)) continue;
        }
        csl_volunteer_ranger_ids[volunteer_prune_index] = 0;
        ShipOut(volunteer_prune_ship);
        OrderLock(volunteer_prune_ship, 0);
        ShipFreeFlight(volunteer_prune_ship);
        result = 1;
        exit;
    }
}

function CSL_FindStrikeAssemblyThreat()
{
    result = 0;
    if(!csl_destination) exit;
    dword rally_star = PlanetToStar(csl_destination);
    for(int threat_index = 0; threat_index < GroupCount(SupplyStrike); threat_index = threat_index + 1)
    {
        dword threatened_ship = GroupShip(SupplyStrike, threat_index);
        if(!threatened_ship) continue;
        if(ShipIsTakeoff(threatened_ship) || ShipInHyperSpace(threatened_ship, 1) || !ShipInNormalSpace(threatened_ship)) continue;
        dword raw_strike_threat = ShipGetBad(threatened_ship);
        dword strike_threat = CSL_ResolveNormalShipInStar(raw_strike_threat, rally_star);
        if(!strike_threat) continue;
        if(ShipOwner(strike_threat) != 6 && ShipOwner(strike_threat) != 7)
        {
            if(!ShipInNormalSpace(threatened_ship) || Dist(strike_threat, threatened_ship) > 1200) continue;
        }
        result = strike_threat;
        exit;
    }
    // Vanilla military-operation news does not seize NPC rangers into a
    // scenario group. Treat every non-pirate ranger close to the rally as an
    // autonomous volunteer and protect it when another ship targets it.
    dword rally_base = CSL_FindStrikeRallyBase();
    for(int attacker_index = 0; attacker_index < StarShips(rally_star); attacker_index = attacker_index + 1)
    {
        dword attacker = StarShips(rally_star, attacker_index);
        if(!attacker) continue;
        if(!ShipInNormalSpace(attacker) || ShipTypeN(attacker) >= t_RC) continue;
        if(ShipIsTakeoff(attacker)) continue;
        int attacker_id = Id(attacker);
        if(CSL_IsStrikeId(attacker_id)) continue;
        int attacker_standing = ShipStanding(attacker);
        if(attacker != Player() && attacker_standing != 0 && attacker_standing < 6) continue;
        dword raw_protected_ranger = ShipGetBad(attacker);
        dword protected_ranger = CSL_ResolveNormalShipInStar(raw_protected_ranger, rally_star);
        if(!protected_ranger) continue;
        int eligible_ranger = protected_ranger == Player() && csl_player_joined;
        if(protected_ranger != Player() && ShipTypeN(protected_ranger) == t_Ranger && !ShipOnSidePirateClan(protected_ranger))
            eligible_ranger = 1;
        if(!eligible_ranger) continue;
        int near_rally = 0;
        if(rally_base)
        {
            if(ShipInNormalSpace(rally_base) && Dist(protected_ranger, rally_base) <= 1200) near_rally = 1;
        }
        if(!rally_base && csl_strike_rally_ready &&
           DistCoords(CoordX(protected_ranger), CoordY(protected_ranger), csl_strike_rally_x, csl_strike_rally_y) <= 1200) near_rally = 1;
        if(!near_rally && GroupCount(SupplyStrike))
        {
            dword rally_leader = GroupToShip(SupplyStrike);
            if(rally_leader)
            {
                if(ShipInNormalSpace(rally_leader) && CSL_ShipCurrentStar(rally_leader) == rally_star && Dist(protected_ranger, rally_leader) <= 1200)
                    near_rally = 1;
            }
        }
        if(!near_rally) continue;
        result = attacker;
        exit;
    }
}

// Returns 1 while the destination system or one of the assembled ships is
// under attack. Production and recruitment readiness pause until the threat is
// gone; docked ships take off one per date and the whole group retaliates.
function CSL_DefendStrikeAssembly()
{
    result = 0;
    dword strike_threat = CSL_FindStrikeAssemblyThreat();
    if(!strike_threat) exit;
    result = 1;
    if(strike_threat == Player()) csl_disqualified = 1;
    dword rally_star = PlanetToStar(csl_destination);
    // First arm ships that are already in normal space. Only after all group
    // reads are finished may one docked member take off and mutate live state.
    for(int defend_strike_index = 0; defend_strike_index < GroupCount(SupplyStrike); defend_strike_index = defend_strike_index + 1)
    {
        dword defend_strike = GroupShip(SupplyStrike, defend_strike_index);
        if(!defend_strike) continue;
        if(ShipInHyperSpace(defend_strike, 1) || !ShipInNormalSpace(defend_strike)) continue;
        if(CSL_ShipCurrentStar(defend_strike) != rally_star) continue;
        if(ShipOrder(defend_strike) != 6 || ShipOrderObj(defend_strike) != strike_threat)
        {
            OrderLock(defend_strike, 1);
            OrderFollowShip(defend_strike, strike_threat, 1, 1);
            ShipSetBad(defend_strike, strike_threat);
        }
    }
    for(int launch_defender_index = 0; launch_defender_index < GroupCount(SupplyStrike); launch_defender_index = launch_defender_index + 1)
    {
        dword launch_defender = GroupShip(SupplyStrike, launch_defender_index);
        if(!launch_defender || ShipInHyperSpace(launch_defender, 1)) continue;
        dword launch_defender_place = GetShipPlanet(launch_defender);
        if(!launch_defender_place) launch_defender_place = GetShipRuins(launch_defender);
        if(!launch_defender_place) continue;
        ShipRefuel(launch_defender);
        ShipRepairEq(launch_defender);
        CSL_TakeOff(launch_defender);
        exit;
    }
}

function CSL_StrikeAssembled()
{
    result = 0;
    if(!csl_destination || !GroupCount(SupplyStrike)) exit;
    dword rally_star = PlanetToStar(csl_destination);
    dword rally_base = CSL_FindStrikeRallyBase();
    if(!rally_base && !csl_strike_rally_ready) exit;
    for(int assembled_index = 0; assembled_index < GroupCount(SupplyStrike); assembled_index = assembled_index + 1)
    {
        dword assembled_ship = GroupShip(SupplyStrike, assembled_index);
        if(!assembled_ship || ShipInHyperSpace(assembled_ship, 1)) exit;
        if(rally_base)
        {
            if(GetShipRuins(assembled_ship) != rally_base) exit;
            continue;
        }
        if(!ShipInNormalSpace(assembled_ship) || CSL_ShipCurrentStar(assembled_ship) != rally_star) exit;
        int slot_x = CSL_StrikeRallySlotX(assembled_index);
        int slot_y = CSL_StrikeRallySlotY(assembled_index);
        if(DistCoords(CoordX(assembled_ship), CoordY(assembled_ship), slot_x, slot_y) > 100) exit;
    }
    result = 1;
}

// Moves at most one ship per date toward the common rally. A live military
// base is preferred and used for actual landing/refuel; without it the same
// formation gathers at fixed coordinates near the destination planet.
function CSL_MoveStrikeToAssembly()
{
    result = 0;
    if(!csl_destination) exit;
    dword rally_star = PlanetToStar(csl_destination);
    dword rally_base = CSL_FindStrikeRallyBase();
    for(int rally_index = 0; rally_index < GroupCount(SupplyStrike); rally_index = rally_index + 1)
    {
        dword rally_ship = GroupShip(SupplyStrike, rally_index);
        if(!rally_ship) continue;
        if(ShipInHyperSpace(rally_ship)) { result = 1; exit; }
        if(ShipInHyperSpace(rally_ship, 1)) { result = 1; exit; }
        dword rally_ship_planet = GetShipPlanet(rally_ship);
        dword rally_ship_ruins = GetShipRuins(rally_ship);
        if(rally_base && rally_ship_ruins == rally_base)
        {
            ShipRefuel(rally_ship);
            ShipRepairEq(rally_ship);
            continue;
        }
        if(rally_ship_planet || rally_ship_ruins)
        {
            ShipRefuel(rally_ship);
            ShipRepairEq(rally_ship);
            CSL_TakeOff(rally_ship);
            result = 1;
            exit;
        }
        if(!ShipInNormalSpace(rally_ship)) { result = 1; exit; }
        if(CSL_ShipCurrentStar(rally_ship) != rally_star)
        {
            if(ShipCanJump(rally_ship, CSL_ShipCurrentStar(rally_ship), rally_star, 1)) CSL_Jump(rally_ship, rally_star);
            result = 1;
            exit;
        }
        if(!rally_base && !csl_strike_rally_ready) CSL_EnsureSpaceRally(rally_ship);
        if(!rally_base && !csl_strike_rally_ready) { result = 1; exit; }
        ShipSetBad(rally_ship, 0);
        OrderLock(rally_ship, 1);
        if(rally_base)
        {
            OrderLanding(rally_ship, rally_base, 1);
            result = 1;
            exit;
        }
        int rally_slot_x = CSL_StrikeRallySlotX(rally_index);
        int rally_slot_y = CSL_StrikeRallySlotY(rally_index);
        if(DistCoords(CoordX(rally_ship), CoordY(rally_ship), rally_slot_x, rally_slot_y) > 70)
        {
            OrderMove(rally_ship, rally_slot_x, rally_slot_y);
            result = 1;
            exit;
        }
        // Keep an explicit fixed-position order even for a ship already inside
        // the rally radius. Otherwise the completed takeoff order may let it
        // coast through the system after this function reports success.
        OrderMove(rally_ship, rally_slot_x, rally_slot_y);
    }
}

// Re-forms every surviving strike ship after a defensive sortie. Normal-space
// ships receive their return order on the same date; only docked ships retain
// the engine-safe one-takeoff-per-date gate.
function CSL_ReturnStrikeToAssemblyAll()
{
    result = 0;
    if(!csl_destination || !GroupCount(SupplyStrike)) exit;
    dword rally_star = PlanetToStar(csl_destination);
    dword rally_base = CSL_FindStrikeRallyBase();
    if(!rally_base && !csl_strike_rally_ready)
    {
        for(int rally_anchor_index = 0; rally_anchor_index < GroupCount(SupplyStrike); rally_anchor_index = rally_anchor_index + 1)
        {
            dword rally_anchor = GroupShip(SupplyStrike, rally_anchor_index);
            if(!rally_anchor) continue;
            if(ShipInNormalSpace(rally_anchor) && !ShipInHyperSpace(rally_anchor, 1) &&
               CSL_ShipCurrentStar(rally_anchor) == rally_star)
            {
                CSL_EnsureSpaceRally(rally_anchor);
                break;
            }
        }
        if(!csl_strike_rally_ready) { result = 1; exit; }
    }

    dword docked_return_ship = 0;
    for(int return_index = 0; return_index < GroupCount(SupplyStrike); return_index = return_index + 1)
    {
        dword return_ship = GroupShip(SupplyStrike, return_index);
        if(!return_ship) continue;
        if(ShipInHyperSpace(return_ship) || ShipInHyperSpace(return_ship, 1) || ShipIsTakeoff(return_ship))
        {
            result = 1;
            continue;
        }
        dword return_planet = GetShipPlanet(return_ship);
        dword return_ruins = GetShipRuins(return_ship);
        if(rally_base && return_ruins == rally_base)
        {
            ShipRefuel(return_ship);
            ShipRepairEq(return_ship);
            ShipSetBad(return_ship, 0);
            continue;
        }
        if(return_planet || return_ruins)
        {
            if(!docked_return_ship) docked_return_ship = return_ship;
            result = 1;
            continue;
        }
        if(!ShipInNormalSpace(return_ship)) { result = 1; continue; }
        dword return_star = CSL_ShipCurrentStar(return_ship);
        ShipSetBad(return_ship, 0);
        OrderLock(return_ship, 1);
        if(return_star != rally_star)
        {
            if(return_star)
            {
                if(ShipCanJump(return_ship, return_star, rally_star, 1)) CSL_Jump(return_ship, rally_star);
            }
            result = 1;
            continue;
        }
        if(rally_base)
        {
            OrderLanding(return_ship, rally_base, 1);
            result = 1;
            continue;
        }
        int return_slot_x = CSL_StrikeRallySlotX(return_index);
        int return_slot_y = CSL_StrikeRallySlotY(return_index);
        OrderMove(return_ship, return_slot_x, return_slot_y);
        if(DistCoords(CoordX(return_ship), CoordY(return_ship), return_slot_x, return_slot_y) > 70) result = 1;
    }
    if(docked_return_ship)
    {
        ShipRefuel(docked_return_ship);
        ShipRepairEq(docked_return_ship);
        CSL_TakeOff(docked_return_ship);
        result = 1;
        exit;
    }
}

// Snapshot the script group before ShipOut mutates it, then release every
// normal-space member in one handler invocation.
function CSL_ReleaseStrikeGroupAtOnce(int create_garrison)
{
    result = 0;
    if(!GroupCount(SupplyStrike)) exit;
    unknown release_ids = newarray(1);
    dword docked_release_ship = 0;
    for(int release_probe = 0; release_probe < GroupCount(SupplyStrike); release_probe = release_probe + 1)
    {
        dword release_probe_ship = GroupShip(SupplyStrike, release_probe);
        if(!release_probe_ship) continue;
        if(ShipInHyperSpace(release_probe_ship) || ShipInHyperSpace(release_probe_ship, 1) || ShipIsTakeoff(release_probe_ship))
        {
            result = 1;
            exit;
        }
        dword release_place = GetShipPlanet(release_probe_ship);
        if(!release_place) release_place = GetShipRuins(release_probe_ship);
        if(release_place)
        {
            docked_release_ship = release_probe_ship;
            break;
        }
        ArrayAdd(release_ids, Id(release_probe_ship));
    }
    if(docked_release_ship)
    {
        ShipRefuel(docked_release_ship);
        ShipRepairEq(docked_release_ship);
        CSL_TakeOff(docked_release_ship);
        result = 1;
        exit;
    }
    int release_count = ArrayDim(release_ids) - 1;
    int home_count = min(release_count, max(3, (release_count + 1) / 2));
    dword liberated_planet = 0;
    if(create_garrison && csl_target_star)
    {
        if(StarOwner(csl_target_star) == 0)
            liberated_planet = CSL_FindOwnedPlanet(csl_target_star, 0);
    }
    for(int release_index = 1; release_index < ArrayDim(release_ids); release_index = release_index + 1)
    {
        dword release_ship = IdToShip(release_ids[release_index]);
        if(!release_ship) continue;
        // At victory, half of the survivors (at least three, or all when fewer
        // survived) become ordinary warriors of a planet in the liberated
        // system. This is a real vanilla home assignment, not a scripted leash.
        if(liberated_planet && release_index <= home_count)
            ShipStatistic(release_ship, 10, liberated_planet);
        ShipSetBad(release_ship, 0);
        ShipOut(release_ship);
        OrderLock(release_ship, 0);
        ShipFreeFlight(release_ship);
    }
    result = 1;
}

function CSL_ApplyViolation(int new_severity)
{
    if(new_severity <= csl_violation) exit;
    int old_severity = csl_violation;
    int maloc_penalty = 8;
    int peleng_penalty = 4;
    int people_penalty = 10;
    int fei_penalty = 6;
    int gaal_penalty = 10;
    if(csl_source_race == Maloc) maloc_penalty = 15;
    else if(csl_source_race == Peleng) peleng_penalty = 15;
    else if(csl_source_race == People) people_penalty = 15;
    else if(csl_source_race == Fei) fei_penalty = 15;
    else if(csl_source_race == Gaal) gaal_penalty = 15;
    int maloc_step = round(1.0 * maloc_penalty * new_severity / 100) - round(1.0 * maloc_penalty * old_severity / 100);
    int peleng_step = round(1.0 * peleng_penalty * new_severity / 100) - round(1.0 * peleng_penalty * old_severity / 100);
    int people_step = round(1.0 * people_penalty * new_severity / 100) - round(1.0 * people_penalty * old_severity / 100);
    int fei_step = round(1.0 * fei_penalty * new_severity / 100) - round(1.0 * fei_penalty * old_severity / 100);
    int gaal_step = round(1.0 * gaal_penalty * new_severity / 100) - round(1.0 * gaal_penalty * old_severity / 100);
    if(maloc_step > 0) ChangeGlobalRelationsPlanets(Player(), 0, 3, -maloc_step, 1);
    if(peleng_step > 0) ChangeGlobalRelationsPlanets(Player(), 0, 3, -peleng_step, 2);
    if(people_step > 0) ChangeGlobalRelationsPlanets(Player(), 0, 3, -people_step, 4);
    if(fei_step > 0) ChangeGlobalRelationsPlanets(Player(), 0, 3, -fei_step, 8);
    if(gaal_step > 0) ChangeGlobalRelationsPlanets(Player(), 0, 3, -gaal_step, 16);
    int source_extra = 70 - 15;
    int destination_base = 10;
    if(PlanetRace(csl_destination) == csl_source_race) destination_base = 15;
    else if(PlanetRace(csl_destination) == Maloc) destination_base = 8;
    else if(PlanetRace(csl_destination) == Peleng) destination_base = 4;
    else if(PlanetRace(csl_destination) == Fei) destination_base = 6;
    int destination_extra = 25 - destination_base;
    int source_step = round(1.0 * source_extra * new_severity / 100) - round(1.0 * source_extra * old_severity / 100);
    int destination_step = round(1.0 * destination_extra * new_severity / 100) - round(1.0 * destination_extra * old_severity / 100);
    if(source_step > 0) RelationToRanger(csl_source, Player(), max(-100, RelationToRanger(csl_source, Player()) - source_step));
    if(destination_step > 0) RelationToRanger(csl_destination, Player(), max(-100, RelationToRanger(csl_destination, Player()) - destination_step));
    csl_violation = new_severity;
}

function CSL_ProcessConvoyThreats()
{
    result = 0;
    dword selected_threat = 0;
    int selected_threat_distance = 1000000;
    for(int threat_transport_index = 0; threat_transport_index < GroupCount(SupplyTransports); threat_transport_index = threat_transport_index + 1)
    {
        dword threat_transport = GroupShip(SupplyTransports, threat_transport_index);
        if(!threat_transport) continue;
        if(!CSL_IsTransportId(Id(threat_transport))) continue;
        dword raw_transport_attacker = ShipGetBad(threat_transport);
        if(!raw_transport_attacker) continue;
        if(raw_transport_attacker == Player()) CSL_ApplyViolation(40);
        dword threat_transport_star = CSL_ShipCurrentStar(threat_transport);
        if(!threat_transport_star) continue;
        dword transport_attacker = CSL_ResolveNormalShipInStar(raw_transport_attacker, threat_transport_star);
        if(!CSL_ThreatWithinConvoyLeash(transport_attacker, threat_transport))
        {
            ShipSetBad(threat_transport, 0);
            continue;
        }
        dword transport_threat_anchor = CSL_ConvoyAnchor(threat_transport);
        int transport_threat_distance = Dist(transport_attacker, transport_threat_anchor);
        if(transport_threat_distance < selected_threat_distance)
        {
            selected_threat = transport_attacker;
            selected_threat_distance = transport_threat_distance;
        }
    }
    for(int threat_escort_index = 0; threat_escort_index < GroupCount(SupplyEscorts); threat_escort_index = threat_escort_index + 1)
    {
        dword threat_escort = GroupShip(SupplyEscorts, threat_escort_index);
        if(!threat_escort) continue;
        if(!CSL_IsEscortId(Id(threat_escort))) continue;
        dword raw_escort_attacker = ShipGetBad(threat_escort);
        if(!raw_escort_attacker) continue;
        if(raw_escort_attacker == Player()) CSL_ApplyViolation(40);
        dword escorted_transport = CSL_FindNearestTransportToShip(threat_escort);
        dword threat_escort_star = CSL_ShipCurrentStar(threat_escort);
        if(!threat_escort_star) continue;
        dword escort_attacker = CSL_ResolveNormalShipInStar(raw_escort_attacker, threat_escort_star);
        if(!CSL_ThreatWithinConvoyLeash(escort_attacker, escorted_transport))
        {
            ShipSetBad(threat_escort, 0);
            continue;
        }
        dword escort_threat_anchor = CSL_ConvoyAnchor(escorted_transport);
        int escort_threat_distance = Dist(escort_attacker, escort_threat_anchor);
        if(escort_threat_distance < selected_threat_distance)
        {
            selected_threat = escort_attacker;
            selected_threat_distance = escort_threat_distance;
        }
    }
    // Script ambushers remain explicit convoy threats even after the pirate
    // currently stored in ShipGetBad has been destroyed. Without this pass,
    // escorts clear their stale target and stand still until another pirate
    // happens to hit a freighter.
    dword ambush_guard_leader = CSL_ActiveTransportLeader();
    if(ambush_guard_leader && csl_ambush_target_id > 1)
    {
        dword ambush_guard_star = CSL_ShipCurrentStar(ambush_guard_leader);
        dword ambush_guard_anchor = CSL_ConvoyAnchor(ambush_guard_leader);
        if(ambush_guard_star && ambush_guard_anchor)
        {
            for(int ambush_guard_index = 1; ambush_guard_index <= 8; ambush_guard_index = ambush_guard_index + 1)
            {
                dword ambush_guard_raw = IdToShip(CSL_GetPendingPirateId(ambush_guard_index));
                dword ambush_guard_pirate = CSL_ResolveNormalShipInStar(ambush_guard_raw, ambush_guard_star);
                if(!ambush_guard_pirate) continue;
                if(!CSL_AmbushPirateDataSafe(ambush_guard_pirate)) continue;
                if(CSL_PendingPirateRetreating(ambush_guard_index)) continue;
                int ambush_guard_distance = Dist(ambush_guard_pirate, ambush_guard_anchor);
                if(ambush_guard_distance < selected_threat_distance)
                {
                    selected_threat = ambush_guard_pirate;
                    selected_threat_distance = ambush_guard_distance;
                }
            }
        }
    }
    for(int threat_cargo_index = 1; threat_cargo_index <= 30; threat_cargo_index = threat_cargo_index + 1)
    {
        if(CSL_CargoIn(threat_cargo_index, Player())) CSL_ApplyViolation(70);
    }
    result = selected_threat;
}

function CSL_PublishStrikeAlert(str notice)
{
    if(csl_destination_id <= 1) exit;
    // EtherDelete removes a visible message but the engine keeps its ID in a
    // uniqueness set. Use a monotonic serial so retargeting can publish again.
    EtherDelete('XenoCoalitionSupplyLines_Operation' + csl_destination_id);
    if(csl_strike_alert_serial > 0)
        EtherDelete('XenoCoalitionSupplyLines_Operation_' + csl_strike_alert_serial);
    csl_strike_alert_serial = csl_strike_alert_serial + 1;
    Ether(0, 'XenoCoalitionSupplyLines_Operation_' + csl_strike_alert_serial,
          '<color=255,240,100>' + GameDateTxtByTurn(CurTurn()) + Format('</color><br> <br>') + notice);
    Sound('Sound.NewMsg');
}

function CSL_ClearStrikeAlert()
{
    if(csl_destination_id > 1) EtherDelete('XenoCoalitionSupplyLines_Operation' + csl_destination_id);
    if(csl_strike_alert_serial > 0)
        EtherDelete('XenoCoalitionSupplyLines_Operation_' + csl_strike_alert_serial);
}

function CSL_ScheduleCooldown()
{
    if(csl_runtime_context_kind == 2)
    {
        // Strike operations are serialized separately from the three convoy
        // schedulers.  Their completion frees the operation context instead
        // of silently consuming one convoy timer.
        csl_phase = -1;
        csl_replan_pending = 0;
        csl_strike_build_due = 0;
        csl_procurement_active = 0;
        exit;
    }
    csl_phase = 0;
    csl_replan_pending = 0;
    csl_strike_build_due = 0;
    csl_procurement_active = 0;
    if(csl_hostile_systems <= 0)
    {
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    int cooldown_band = CSL_StrategyBand();
    int cooldown_min = 450;
    int cooldown_max = 700;
    if(cooldown_band == 2) { cooldown_min = 550; cooldown_max = 850; }
    else if(cooldown_band == 3) { cooldown_min = 700; cooldown_max = 1050; }
    else if(cooldown_band == 4) { cooldown_min = 900; cooldown_max = 1350; }
    else if(cooldown_band == 5) { cooldown_min = 1200; cooldown_max = 1800; }
    else if(cooldown_band == 6) { cooldown_min = 1500; cooldown_max = 2200; }
    else if(cooldown_band == 7) { cooldown_min = 1800; cooldown_max = 2600; }
    int difficulty = max(50, min(500, round(0.5 * (GalaxyDiffLevels(4) + GalaxyDiffLevels(5)))));
    csl_source = 0;
    csl_source_id = 0;
    csl_procurement_mask = RndObject(1, 7, Player());
    csl_procurement_start_turn = CurTurn();
    csl_next_turn = CurTurn() + max(1, round(1.0 * RndObject(cooldown_min, cooldown_max, Player()) * difficulty / 100));
    csl_procurement_mid_turn = csl_procurement_start_turn + (csl_next_turn - csl_procurement_start_turn) / 2;
    csl_procurement_end_turn = csl_next_turn;
    csl_procurement_last_turn = CurTurn() - 1;
}

function CSL_ProcessProcurement()
{
    if(!csl_procurement_active || !csl_source || CurTurn() >= csl_procurement_end_turn) exit;
    if(csl_procurement_last_turn >= CurTurn()) exit;
    csl_procurement_last_turn = CurTurn();
    int procurement_daily = 2;
    if(CurTurn() < csl_procurement_mid_turn) procurement_daily = 5;
    if(csl_procurement_mask - (csl_procurement_mask / 2) * 2 == 1)
    {
        int procurement_technics_stock = GoodsAdd(csl_source, t_Technics, 0);
        if(procurement_technics_stock > 0) GoodsAdd(csl_source, t_Technics, -min(procurement_technics_stock, procurement_daily));
    }
    if((csl_procurement_mask / 2) - (csl_procurement_mask / 4) * 2 == 1)
    {
        int procurement_minerals_stock = GoodsAdd(csl_source, t_Minerals, 0);
        if(procurement_minerals_stock > 0) GoodsAdd(csl_source, t_Minerals, -min(procurement_minerals_stock, procurement_daily));
    }
    if(csl_procurement_mask >= 4)
    {
        int procurement_arms_stock = GoodsAdd(csl_source, t_Arms, 0);
        if(procurement_arms_stock > 0) GoodsAdd(csl_source, t_Arms, -min(procurement_arms_stock, procurement_daily));
    }
}

function CSL_QueueStrikeOperation()
{
    if(csl_destination_id <= 1 || csl_delivered_cargo <= 0) exit;
    int queued_build_due = csl_strike_build_due;
    if(queued_build_due <= 0) queued_build_due = CurTurn() + RndObject(15, 30, csl_destination);
    ArrayAdd(csl_strike_queue_destination_ids, csl_destination_id);
    ArrayAdd(csl_strike_queue_delivered_cargo, csl_delivered_cargo);
    ArrayAdd(csl_strike_queue_delivered_escorts, csl_delivered_escorts);
    ArrayAdd(csl_strike_queue_composition_bands, csl_composition_band);
    ArrayAdd(csl_strike_queue_source_races, csl_source_race);
    ArrayAdd(csl_strike_queue_build_due_turns, queued_build_due);
}

function CSL_FinishConvoyCleanup()
{
    csl_pending_transport_cleanup_id = 0;
    CSL_ResetTransportIds();
    CSL_ResetEscortIds();
    ArrayClear(csl_route_planets);
    csl_replan_pending = 0;
    if(csl_delivered_cargo <= 0)
    {
        // An ambush publishes its own loss report as soon as the last
        // transport is destroyed. Do not publish the generic report again.
        if(csl_ambush_origin_planet_id != -1)
            AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Lost'), '#destination#', Name(csl_destination)));
        csl_ambush_origin_planet_id = 0;
        csl_ambush_origin_base_id = 0;
        csl_ambush_target_id = 0;
        CSL_ScheduleCooldown();
    }
    else
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Arrived'), '#destination#', Name(csl_destination), '#delivered#', csl_delivered_cargo, '#total#', csl_total_cargo));
        CSL_QueueStrikeOperation();
        CSL_ScheduleCooldown();
    }
}

function CSL_CompositionMatchesSystems(int safe_systems)
{
    result = CSL_StrategyBand() == csl_composition_band &&
             CSL_LiveTransportCount() == csl_transport_count &&
             CSL_LiveEscortCount() == csl_escort_count;
}

function CSL_ConvoyCanJump(dword route_from_star, dword route_to_star)
{
    result = 0;
    if(!route_from_star || !route_to_star) exit;
    if(CSL_LiveTransportCount() <= 0) exit;
    for(int convoy_route_transport_index = 0; convoy_route_transport_index < GroupCount(SupplyTransports); convoy_route_transport_index = convoy_route_transport_index + 1)
    {
        dword convoy_route_transport = GroupShip(SupplyTransports, convoy_route_transport_index);
        if(!convoy_route_transport) continue;
        if(!CSL_IsTransportId(Id(convoy_route_transport))) continue;
        if(!ShipCanJump(convoy_route_transport, route_from_star, route_to_star, 1)) exit;
    }
    for(int convoy_route_escort_index = 0; convoy_route_escort_index < GroupCount(SupplyEscorts); convoy_route_escort_index = convoy_route_escort_index + 1)
    {
        dword convoy_route_escort = GroupShip(SupplyEscorts, convoy_route_escort_index);
        if(!convoy_route_escort) continue;
        if(!CSL_IsEscortId(Id(convoy_route_escort))) continue;
        if(!ShipCanJump(convoy_route_escort, route_from_star, route_to_star, 1)) exit;
    }
    result = 1;
}

// Build one shared breadth-first route through peaceful Coalition planets.
// Planet IDs are persisted; no engine object survives across a turn boundary.
function CSL_BuildConvoyRoute(dword route_ship, dword route_source_star, int minimum_hops)
{
    result = 0;
    ArrayClear(csl_route_planets);
    csl_route_index = 0;
    if(!route_ship || !route_source_star) exit;
    // Keep only peaceful planet systems and hostile strategic systems in the
    // matrices below. BigGalaxy can expose hundreds of unrelated stars; the
    // old full S by S passes evaluated all of them twice and stalled a turn at
    // 600k expressions. Candidate order remains the GalaxyStar order, so the
    // selected route and all tie-breaks stay identical.
    unknown route_stars = newarray(1);
    unknown route_parent_indices = newarray(1);
    unknown route_depths = newarray(1);
    unknown route_candidate_indices = newarray(1);
    unknown route_candidate_stars = newarray(1);
    unknown route_candidate_planets = newarray(1);
    unknown route_hostile_stars = newarray(1);
    unknown route_target_by_candidate = newarray(1);
    unknown route_target_threat_by_candidate = newarray(1);
    unknown route_target_distance_by_candidate = newarray(1);
    unknown route_seen_by_candidate = newarray(1);
    ArrayClear(route_stars);
    ArrayClear(route_parent_indices);
    ArrayClear(route_depths);
    ArrayClear(route_candidate_indices);
    ArrayClear(route_candidate_stars);
    ArrayClear(route_candidate_planets);
    ArrayClear(route_hostile_stars);
    ArrayClear(route_target_by_candidate);
    ArrayClear(route_target_threat_by_candidate);
    ArrayClear(route_target_distance_by_candidate);
    ArrayClear(route_seen_by_candidate);
    int route_galaxy_star_count = GalaxyStars();
    int route_source_found = 0;
    int route_source_candidate_index = -1;
    for(int route_seen_init_index = 0; route_seen_init_index < route_galaxy_star_count; route_seen_init_index = route_seen_init_index + 1)
    {
        dword route_seen_init_star = GalaxyStar(route_seen_init_index);
        if(!route_seen_init_star) continue;
        if(CSL_IsStrategicStar(route_seen_init_star))
        {
            int route_seen_init_owner = StarOwner(route_seen_init_star);
            if(route_seen_init_owner == 1 || route_seen_init_owner == 2)
                ArrayAdd(route_hostile_stars, route_seen_init_star);
        }
        if(!CSL_IsFreeStar(route_seen_init_star)) continue;
        dword route_seen_init_planet = CSL_FindOwnedPlanet(route_seen_init_star, 0);
        if(!route_seen_init_planet) continue;
        ArrayAdd(route_candidate_stars, route_seen_init_star);
        ArrayAdd(route_candidate_planets, route_seen_init_planet);
        ArrayAdd(route_target_by_candidate, 0);
        ArrayAdd(route_target_threat_by_candidate, -1);
        ArrayAdd(route_target_distance_by_candidate, 1000000);
        ArrayAdd(route_seen_by_candidate, 0);
        if(route_seen_init_star == route_source_star)
        {
            route_source_found = 1;
            route_source_candidate_index = ArrayDim(route_candidate_stars) - 1;
        }
    }
    // A reroute may start in a system that ceased to be peaceful during this
    // very turn. It remains a valid BFS origin but never a destination node.
    if(!route_source_found)
    {
        for(int route_source_find_index = 0; route_source_find_index < route_galaxy_star_count; route_source_find_index = route_source_find_index + 1)
        {
            if(GalaxyStar(route_source_find_index) == route_source_star)
            {
                route_source_found = 1;
                break;
            }
        }
    }
    if(!route_source_found) exit;

    int route_candidate_count = ArrayDim(route_candidate_stars) - 1;
    int route_hostile_count = ArrayDim(route_hostile_stars) - 1;
    // Exact candidate/hostile comparison in one flat pass.
    int route_target_pair_limit = route_candidate_count * route_hostile_count;
    for(int route_target_pair = 0; route_target_pair < route_target_pair_limit; route_target_pair = route_target_pair + 1)
    {
        int route_target_destination_index = route_target_pair / route_hostile_count + 1;
        int route_target_enemy_index = route_target_pair - (route_target_destination_index - 1) * route_hostile_count + 1;
        dword route_target_destination_star = route_candidate_stars[route_target_destination_index];
        dword route_target_enemy_star = route_hostile_stars[route_target_enemy_index];
        if(!ShipCanJump(route_ship, route_target_destination_star, route_target_enemy_star, 1)) continue;
        int route_target_threat = StarEnemyThreatLevel(route_target_enemy_star, 1);
        int route_target_distance = Dist(route_target_destination_star, route_target_enemy_star);
        if(route_target_threat > route_target_threat_by_candidate[route_target_destination_index] ||
           (route_target_threat == route_target_threat_by_candidate[route_target_destination_index] && route_target_distance < route_target_distance_by_candidate[route_target_destination_index]))
        {
            route_target_by_candidate[route_target_destination_index] = route_target_enemy_star;
            route_target_threat_by_candidate[route_target_destination_index] = route_target_threat;
            route_target_distance_by_candidate[route_target_destination_index] = route_target_distance;
        }
    }

    if(route_source_candidate_index >= 1)
        route_seen_by_candidate[route_source_candidate_index] = 1;
    ArrayAdd(route_stars, route_source_star);
    ArrayAdd(route_parent_indices, 0);
    ArrayAdd(route_depths, 0);
    ArrayAdd(route_candidate_indices, route_source_candidate_index);
    int selected_route_node = -1;
    dword selected_route_target = 0;
    int selected_route_threat = -1;
    int selected_route_depth = 1000000;
    int route_bfs_pair_limit = (route_candidate_count + 1) * route_candidate_count;
    // Flatten BFS node/candidate pairs. Visited lookup is O(1) in the compact
    // candidate array, so unrelated stars never enter the hot loop.
    for(int route_bfs_pair = 0; route_bfs_pair < route_bfs_pair_limit; route_bfs_pair = route_bfs_pair + 1)
    {
        int route_cursor = route_bfs_pair / route_candidate_count + 1;
        int route_candidate_index = route_bfs_pair - (route_cursor - 1) * route_candidate_count + 1;
        if(route_cursor >= ArrayDim(route_stars)) break;
        dword route_current_star = route_stars[route_cursor];
        if(!route_current_star) continue;
        int route_current_depth = route_depths[route_cursor];
        if(selected_route_node >= 1 && route_current_depth > selected_route_depth) break;
        int route_current_candidate_index = route_candidate_indices[route_cursor];

        if(route_candidate_index == 1 && route_current_depth >= minimum_hops)
        {
            dword route_current_target = route_target_by_candidate[route_current_candidate_index];
            if(route_current_target)
            {
                int route_current_threat = route_target_threat_by_candidate[route_current_candidate_index];
                if(route_current_depth < selected_route_depth ||
                    (route_current_depth == selected_route_depth && route_current_threat > selected_route_threat))
                {
                    selected_route_node = route_cursor;
                    selected_route_target = route_current_target;
                    selected_route_threat = route_current_threat;
                    selected_route_depth = route_current_depth;
            }
        }
        }
        if(selected_route_node >= 1) continue;

        if(route_seen_by_candidate[route_candidate_index]) continue;
        dword route_candidate_star = route_candidate_stars[route_candidate_index];
        // A cheap leader check rejects most impossible edges before the full
        // convoy equipment check walks all transports and escorts.
        if(!ShipCanJump(route_ship, route_current_star, route_candidate_star, 1)) continue;
        if(!CSL_ConvoyCanJump(route_current_star, route_candidate_star)) continue;
        route_seen_by_candidate[route_candidate_index] = 1;
        ArrayAdd(route_stars, route_candidate_star);
        ArrayAdd(route_parent_indices, route_cursor);
        ArrayAdd(route_depths, route_current_depth + 1);
        ArrayAdd(route_candidate_indices, route_candidate_index);
    }
    if(selected_route_node < 1) exit;
    int selected_route_candidate_index = route_candidate_indices[selected_route_node];
    csl_destination = route_candidate_planets[selected_route_candidate_index];
    if(!csl_destination || !selected_route_target) exit;

    unknown reverse_route_planets = newarray(1);
    ArrayClear(reverse_route_planets);
    int route_backtrack_node = selected_route_node;
    for(int route_backtrack_step = 0; route_backtrack_step <= route_candidate_count; route_backtrack_step = route_backtrack_step + 1)
    {
        if(route_backtrack_node <= 1) break;
        int route_backtrack_candidate_index = route_candidate_indices[route_backtrack_node];
        dword route_backtrack_planet = route_candidate_planets[route_backtrack_candidate_index];
        if(!route_backtrack_planet) { ArrayClear(csl_route_planets); exit; }
        ArrayAdd(reverse_route_planets, Id(route_backtrack_planet));
        route_backtrack_node = route_parent_indices[route_backtrack_node];
    }
    if(route_backtrack_node != 1) { ArrayClear(csl_route_planets); exit; }
    for(int route_reverse_index = ArrayDim(reverse_route_planets) - 1; route_reverse_index >= 1; route_reverse_index = route_reverse_index - 1)
        ArrayAdd(csl_route_planets, reverse_route_planets[route_reverse_index]);
    if(ArrayDim(csl_route_planets) - 1 < minimum_hops) { ArrayClear(csl_route_planets); exit; }
    csl_target_star = selected_route_target;
    if(IdToPlanet(csl_route_planets[ArrayDim(csl_route_planets) - 1]) != csl_destination) { ArrayClear(csl_route_planets); exit; }
    csl_destination_id = Id(csl_destination);
    csl_target_star_id = Id(csl_target_star);
    csl_route_index = 1;
    result = ArrayDim(csl_route_planets) - 1;
}

function CSL_RoutePlanetSafe(dword route_planet)
{
    result = 0;
    if(!route_planet || !CSL_PlanetEligible(route_planet, 0)) exit;
    if(!CSL_IsFreeStar(PlanetToStar(route_planet))) exit;
    result = 1;
}

function CSL_ActiveTransportLeader()
{
    result = 0;
    // Keep one deterministic formation leader. Selecting whichever transport
    // happened to be nearest to an escort made the leader change in flight.
    for(int active_leader_slot = 1; active_leader_slot <= 6; active_leader_slot = active_leader_slot + 1)
    {
        int active_leader_id = CSL_GetTransportId(active_leader_slot);
        if(active_leader_id <= 1) continue;
        dword active_leader = 0;
        if(active_leader_id > 1) active_leader = IdToShip(active_leader_id);
        if(!active_leader) continue;
        if(!CSL_IsTransportId(Id(active_leader))) continue;
        result = active_leader;
        exit;
    }
}

function CSL_LiveConvoyInHyperSpace()
{
    result = 0;
    for(int hyper_transport_index = 0; hyper_transport_index < GroupCount(SupplyTransports); hyper_transport_index = hyper_transport_index + 1)
    {
        dword hyper_transport = GroupShip(SupplyTransports, hyper_transport_index);
        if(!hyper_transport) continue;
        if(CSL_IsTransportId(Id(hyper_transport)) && ShipInHyperSpace(hyper_transport)) { result = 1; exit; }
    }
    for(int hyper_escort_index = 0; hyper_escort_index < GroupCount(SupplyEscorts); hyper_escort_index = hyper_escort_index + 1)
    {
        dword hyper_escort = GroupShip(SupplyEscorts, hyper_escort_index);
        if(!hyper_escort) continue;
        if(CSL_IsEscortId(Id(hyper_escort)) && ShipInHyperSpace(hyper_escort)) { result = 1; exit; }
    }
}

function CSL_LiveConvoyCanReach(dword candidate_star)
{
    result = 0;
    if(!candidate_star) exit;
    int active_transports = 0;
    for(int reach_transport_index = 0; reach_transport_index < GroupCount(SupplyTransports); reach_transport_index = reach_transport_index + 1)
    {
        dword current_ship = GroupShip(SupplyTransports, reach_transport_index);
        if(!current_ship) continue;
        if(!CSL_IsTransportId(Id(current_ship))) continue;
        if(ShipInHyperSpace(current_ship)) exit;
        dword current_star = CSL_ShipCurrentStar(current_ship);
        if(!current_star) exit;
        if(current_star != candidate_star)
        {
            if(!ShipCanJump(current_ship, current_star, candidate_star, 1)) exit;
        }
        active_transports = active_transports + 1;
    }
    if(active_transports <= 0) exit;
    for(int reach_escort_index = 0; reach_escort_index < GroupCount(SupplyEscorts); reach_escort_index = reach_escort_index + 1)
    {
        dword current_escort = GroupShip(SupplyEscorts, reach_escort_index);
        if(!current_escort) continue;
        if(!CSL_IsEscortId(Id(current_escort))) continue;
        if(ShipInHyperSpace(current_escort)) exit;
        dword escort_star = CSL_ShipCurrentStar(current_escort);
        if(!escort_star) exit;
        if(escort_star != candidate_star)
        {
            if(!ShipCanJump(current_escort, escort_star, candidate_star, 1)) exit;
        }
    }
    result = 1;
}

function CSL_FindNearestReroutePlanet(int frontier_only)
{
    result = 0;
    dword reroute_leader = CSL_ActiveTransportLeader();
    if(!reroute_leader || ShipInHyperSpace(reroute_leader)) exit;
    dword reroute_origin_star = CSL_ShipCurrentStar(reroute_leader);
    if(!reroute_origin_star) exit;
    int reroute_star_count = GalaxyStars();
    unknown reroute_free_stars = newarray(1);
    unknown reroute_hostile_stars = newarray(1);
    unknown reroute_frontier_by_candidate = newarray(1);
    ArrayClear(reroute_free_stars);
    ArrayClear(reroute_hostile_stars);
    ArrayClear(reroute_frontier_by_candidate);
    for(int reroute_frontier_init = 0; reroute_frontier_init < reroute_star_count; reroute_frontier_init = reroute_frontier_init + 1)
    {
        dword reroute_frontier_init_star = GalaxyStar(reroute_frontier_init);
        if(!reroute_frontier_init_star) continue;
        if(CSL_IsStrategicStar(reroute_frontier_init_star))
        {
            int reroute_frontier_init_owner = StarOwner(reroute_frontier_init_star);
            if(reroute_frontier_init_owner == 1 || reroute_frontier_init_owner == 2)
                ArrayAdd(reroute_hostile_stars, reroute_frontier_init_star);
        }
        if(CSL_IsFreeStar(reroute_frontier_init_star))
        {
            ArrayAdd(reroute_free_stars, reroute_frontier_init_star);
            ArrayAdd(reroute_frontier_by_candidate, 0);
        }
    }
    int reroute_free_count = ArrayDim(reroute_free_stars) - 1;
    int reroute_hostile_count = ArrayDim(reroute_hostile_stars) - 1;
    if(frontier_only)
    {
        dword reroute_frontier_ship = CSL_ActiveTransportLeader();
        if(!reroute_frontier_ship) exit;
        // Exact candidate/enemy comparison over compact subsets. The result is
        // unchanged, while BigGalaxy-only and hidden stars leave no hot-loop
        // cost behind.
        int reroute_frontier_pair_limit = reroute_free_count * reroute_hostile_count;
        for(int reroute_frontier_pair = 0; reroute_frontier_pair < reroute_frontier_pair_limit; reroute_frontier_pair = reroute_frontier_pair + 1)
        {
            int reroute_frontier_candidate_index = reroute_frontier_pair / reroute_hostile_count + 1;
            int reroute_frontier_enemy_index = reroute_frontier_pair - (reroute_frontier_candidate_index - 1) * reroute_hostile_count + 1;
            if(reroute_frontier_by_candidate[reroute_frontier_candidate_index]) continue;
            dword reroute_frontier_candidate = reroute_free_stars[reroute_frontier_candidate_index];
            dword reroute_frontier_enemy = reroute_hostile_stars[reroute_frontier_enemy_index];
            if(ShipCanJump(reroute_frontier_ship, reroute_frontier_candidate, reroute_frontier_enemy, 1))
                reroute_frontier_by_candidate[reroute_frontier_candidate_index] = 1;
        }
    }
    int best_reroute_distance = 1000000;
    for(int reroute_star_index = 1; reroute_star_index <= reroute_free_count; reroute_star_index = reroute_star_index + 1)
    {
        dword reroute_star = reroute_free_stars[reroute_star_index];
        // The first pass promises a real frontier destination. Without this
        // filter the current safe system wins at distance zero, is committed as
        // final, then is rejected on the following turn forever.
        if(frontier_only && !reroute_frontier_by_candidate[reroute_star_index]) continue;
        // Route rebuilding performs the exact frontier check after reaching
        // this safe regroup point. Avoid a nested full-galaxy scan here.
        dword reroute_planet = CSL_FindOwnedPlanet(reroute_star, 0);
        if(!reroute_planet || !CSL_LiveConvoyCanReach(reroute_star)) continue;
        int reroute_distance = Dist(reroute_origin_star, reroute_star);
        if(reroute_distance < best_reroute_distance)
        {
            best_reroute_distance = reroute_distance;
            result = reroute_planet;
        }
    }
}

function CSL_CommitReroute(dword reroute_planet, int needs_full_route)
{
    result = 0;
    if(!reroute_planet) exit;
    ArrayClear(csl_route_planets);
    ArrayAdd(csl_route_planets, Id(reroute_planet));
    csl_route_index = 1;
    csl_destination = reroute_planet;
    csl_destination_id = Id(reroute_planet);
    csl_target_star = 0;
    csl_target_star_id = 0;
    csl_replan_pending = needs_full_route;
    csl_next_turn = CurTurn() + 1;
    result = 1;
}

function CSL_PrepareRerouteFuel()
{
    // A ship already docked at a safe stop may use that planet before the
    // replacement leg is evaluated. Ships caught in space keep real fuel.
    for(int reroute_fuel_transport_index = 0; reroute_fuel_transport_index < GroupCount(SupplyTransports); reroute_fuel_transport_index = reroute_fuel_transport_index + 1)
    {
        dword reroute_fuel_transport = GroupShip(SupplyTransports, reroute_fuel_transport_index);
        if(!reroute_fuel_transport) continue;
        if(!CSL_IsTransportId(Id(reroute_fuel_transport))) continue;
        if(GetShipPlanet(reroute_fuel_transport) || GetShipRuins(reroute_fuel_transport))
        {
            ShipRefuel(reroute_fuel_transport);
            ShipRepairEq(reroute_fuel_transport);
        }
    }
    for(int reroute_fuel_escort_index = 0; reroute_fuel_escort_index < GroupCount(SupplyEscorts); reroute_fuel_escort_index = reroute_fuel_escort_index + 1)
    {
        dword reroute_fuel_escort = GroupShip(SupplyEscorts, reroute_fuel_escort_index);
        if(!reroute_fuel_escort) continue;
        if(!CSL_IsEscortId(Id(reroute_fuel_escort))) continue;
        if(GetShipPlanet(reroute_fuel_escort) || GetShipRuins(reroute_fuel_escort))
        {
            ShipRefuel(reroute_fuel_escort);
            ShipRepairEq(reroute_fuel_escort);
        }
    }
}

function CSL_TryImmediateReroute()
{
    result = 0;
    if(CSL_LiveConvoyInHyperSpace()) exit;
    CSL_PrepareRerouteFuel();
    dword new_frontier_planet = CSL_FindNearestReroutePlanet(1);
    if(new_frontier_planet && CSL_CommitReroute(new_frontier_planet, 0))
    {
        csl_delivered_cargo = 0;
        csl_arrived_transports = 0;
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.RerouteDirect'), '#destination#', Name(new_frontier_planet)));
        result = 1;
        exit;
    }
    dword regroup_planet = CSL_FindNearestReroutePlanet(0);
    if(regroup_planet && CSL_CommitReroute(regroup_planet, 1))
    {
        csl_delivered_cargo = 0;
        csl_arrived_transports = 0;
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.RerouteRegroup'), '#destination#', Name(regroup_planet)));
        result = 1;
    }
}

function CSL_SendShip(dword ship, dword destination)
{
    result = 0;
    if(!ship || !destination) exit;
    // Takeoff and hyperspace are immutable engine transitions. Reissuing
    // takeoff or clearing the order here can strand a convoy member.
    if(ShipIsTakeoff(ship) || ShipInHyperSpace(ship, 1)) exit;
    dword landed = GetShipPlanet(ship);
    if(!landed) landed = GetShipRuins(ship);
    if(landed)
    {
        if(landed == destination)
        {
            result = 1;
            exit;
        }
        ShipRefuel(ship);
        ShipRepairEq(ship);
        CSL_TakeOff(ship);
        exit;
    }
    dword destination_star = PlanetToStar(destination);
    dword current_star = CSL_ShipCurrentStar(ship);
    if(!current_star) exit;
    if(current_star == destination_star)
    {
        CSL_Land(ship, destination);
        exit;
    }
    if(ShipCanJump(ship, current_star, destination_star, 1)) CSL_Jump(ship, destination_star);
}

function CSL_LiveEscortCount()

    int live_escorts = 0;
    for(int live_escort_index = 0; live_escort_index < GroupCount(SupplyEscorts); live_escort_index = live_escort_index + 1)
    {
        dword live_escort = GroupShip(SupplyEscorts, live_escort_index);
        if(!live_escort) continue;
        if(CSL_IsEscortId(Id(live_escort))) live_escorts = live_escorts + 1;
    }
    result = live_escorts;
}

    }

function CSL_TransportSlotForId(int transport_id)

    result = 0;
    if(transport_id <= 1) exit;
    for(int transport_slot_index = 1; transport_slot_index <= 6; transport_slot_index = transport_slot_index + 1)
    {
        if(CSL_GetTransportId(transport_slot_index) != transport_id) continue;
        result = transport_slot_index;
        exit;
    }
}

function CSL_LandActiveConvoyAtDestination()
{
    result = 0;
    if(!csl_destination) exit;
    dword landing_star = PlanetToStar(csl_destination);
    if(!landing_star) exit;
    for(int landing_transport_index = 0; landing_transport_index < GroupCount(SupplyTransports); landing_transport_index = landing_transport_index + 1)
    {
        dword landing_transport = GroupShip(SupplyTransports, landing_transport_index);
        if(!landing_transport) continue;
        if(!CSL_IsTransportId(Id(landing_transport))) continue;
        if(GetShipPlanet(landing_transport) == csl_destination) continue;
        if(ShipIsTakeoff(landing_transport) || ShipInHyperSpace(landing_transport, 1) || !ShipInNormalSpace(landing_transport)) continue;
        if(CSL_ShipCurrentStar(landing_transport) != landing_star) continue;
        CSL_Land(landing_transport, csl_destination);
        result = 1;
    }
    for(int landing_escort_index = 0; landing_escort_index < GroupCount(SupplyEscorts); landing_escort_index = landing_escort_index + 1)
    {
        dword landing_escort = GroupShip(SupplyEscorts, landing_escort_index);
        if(!landing_escort) continue;
        if(!CSL_IsEscortId(Id(landing_escort))) continue;
        if(GetShipPlanet(landing_escort) == csl_destination) continue;
        if(ShipIsTakeoff(landing_escort) || ShipInHyperSpace(landing_escort, 1) || !ShipInNormalSpace(landing_escort)) continue;
        if(CSL_ShipCurrentStar(landing_escort) != landing_star) continue;
        OrderLock(landing_escort, 0);
        if(ShipGetBad(landing_escort)) ShipSetBad(landing_escort, 0);
        CSL_Land(landing_escort, csl_destination);
        result = 1;
    }
}

function CSL_HoldActiveTransports()
{
    for(int hold_transport_index = 0; hold_transport_index < GroupCount(SupplyTransports); hold_transport_index = hold_transport_index + 1)
    {
        dword hold_transport = GroupShip(SupplyTransports, hold_transport_index);
        if(!hold_transport) continue;
        if(!CSL_IsTransportId(Id(hold_transport))) continue;
        // OrderNone can cancel the engine transitional takeoff state even
        // after ShipInNormalSpace has already become true for one tick.
        if(ShipIsTakeoff(hold_transport)) continue;
        if(ShipInHyperSpace(hold_transport, 1)) continue;
        if(!ShipInNormalSpace(hold_transport)) continue;
        OrderLock(hold_transport, 0);
        OrderNone(hold_transport);
        ShipSetBad(hold_transport, 0);
        OrderLock(hold_transport, 1);
    }
}

function CSL_AllActiveTransportsSheltered()
{
    int sheltered_transport_count = 0;
    for(int sheltered_transport_index = 0; sheltered_transport_index < GroupCount(SupplyTransports); sheltered_transport_index = sheltered_transport_index + 1)
    {
        dword sheltered_transport = GroupShip(SupplyTransports, sheltered_transport_index);
        if(!sheltered_transport) continue;
        if(!CSL_IsTransportId(Id(sheltered_transport))) continue;
        sheltered_transport_count = sheltered_transport_count + 1;
        dword sheltered_transport_place = GetShipPlanet(sheltered_transport);
        if(!sheltered_transport_place) sheltered_transport_place = GetShipRuins(sheltered_transport);
        if(!sheltered_transport_place) { result = 0; exit; }
    }
    result = sheltered_transport_count > 0;
}

function CSL_CommonTransportPlanet()
{
    result = 0;
    dword common_transport_planet = 0;
    int common_transport_count = 0;
    for(int common_transport_index = 0; common_transport_index < GroupCount(SupplyTransports); common_transport_index = common_transport_index + 1)
    {
        dword common_transport = GroupShip(SupplyTransports, common_transport_index);
        if(!common_transport) continue;
        if(!CSL_IsTransportId(Id(common_transport))) continue;
        dword common_transport_place = GetShipPlanet(common_transport);
        if(!common_transport_place) exit;
        if(!common_transport_planet) common_transport_planet = common_transport_place;
        else if(common_transport_planet != common_transport_place) exit;
        common_transport_count = common_transport_count + 1;
    }
    if(common_transport_count > 0) result = common_transport_planet;
}

function CSL_ResumeConvoyAfterAmbush()
{
    result = 0;
    // Ending an ambush must also end the shelter order. If every surviving
    // freighter reached the same safe planet, make it a temporary regroup
    // point. The normal route code will land the escorts there, refuel the
    // whole formation and rebuild the complete route on the following turn.
    csl_next_turn = CurTurn() + 1;
    dword ambush_shelter_planet = CSL_CommonTransportPlanet();
    if(!ambush_shelter_planet) exit;
    if(!CSL_RoutePlanetSafe(ambush_shelter_planet)) exit;
    CSL_CommitReroute(ambush_shelter_planet, 1);
    csl_delivered_cargo = 0;
    csl_arrived_transports = 0;
    result = 1;
}

function CSL_EvacuateActiveTransports(dword evacuation_leader)
{
    if(!evacuation_leader) { CSL_HoldActiveTransports(); exit; }
    dword evacuation_star = CSL_ShipCurrentStar(evacuation_leader);
    if(!evacuation_star) { CSL_HoldActiveTransports(); exit; }
    dword evacuation_planet = CSL_FindOwnedPlanet(evacuation_star, 0);
    if(!evacuation_planet) evacuation_planet = NearCivilPlanet(evacuation_leader);
    if(!evacuation_planet) { CSL_HoldActiveTransports(); exit; }
    for(int evacuation_transport_index = 0; evacuation_transport_index < GroupCount(SupplyTransports); evacuation_transport_index = evacuation_transport_index + 1)
    {
        dword evacuation_transport = GroupShip(SupplyTransports, evacuation_transport_index);
        if(!evacuation_transport) continue;
        if(!CSL_IsTransportId(Id(evacuation_transport))) continue;
        dword evacuation_place = GetShipPlanet(evacuation_transport);
        if(!evacuation_place) evacuation_place = GetShipRuins(evacuation_transport);
        if(evacuation_place || ShipIsTakeoff(evacuation_transport) || ShipInHyperSpace(evacuation_transport, 1)) continue;
        if(!ShipInNormalSpace(evacuation_transport)) continue;
        if(CSL_ShipCurrentStar(evacuation_transport) != evacuation_star) continue;
        OrderLock(evacuation_transport, 0);
        if(ShipGetBad(evacuation_transport)) ShipSetBad(evacuation_transport, 0);
        CSL_Land(evacuation_transport, evacuation_planet);
    }
}

function CSL_EscortsRejoined(dword leader)
{
    result = 1;
    if(!leader) exit;
    if(ShipIsTakeoff(leader) || ShipInHyperSpace(leader, 1)) { result = 0; exit; }
    dword leader_place = GetShipPlanet(leader);
    if(!leader_place) leader_place = GetShipRuins(leader);
    for(int rejoin_escort_index = 0; rejoin_escort_index < GroupCount(SupplyEscorts); rejoin_escort_index = rejoin_escort_index + 1)
    {
        dword rejoin_escort = GroupShip(SupplyEscorts, rejoin_escort_index);
        if(!rejoin_escort) continue;
        if(!CSL_IsEscortId(Id(rejoin_escort))) continue;
        if(ShipIsTakeoff(rejoin_escort) || ShipInHyperSpace(rejoin_escort, 1)) { result = 0; exit; }
        if(leader_place)
        {
            dword escort_place = GetShipPlanet(rejoin_escort);
            if(!escort_place) escort_place = GetShipRuins(rejoin_escort);
            if(escort_place != leader_place) { result = 0; exit; }
            continue;
        }
        if(!ShipInNormalSpace(leader) || !ShipInNormalSpace(rejoin_escort) ||
           CSL_ShipCurrentStar(rejoin_escort) != CSL_ShipCurrentStar(leader) || Dist(rejoin_escort, leader) > 500)
        {
            result = 0;
            exit;
        }
    }
}

function CSL_TransportsRejoined(dword formation_leader)
{
    result = 0;
    if(!formation_leader) exit;
    if(ShipIsTakeoff(formation_leader)) exit;
    if(ShipInHyperSpace(formation_leader, 1)) exit;
    dword formation_leader_place = GetShipPlanet(formation_leader);
    if(!formation_leader_place) formation_leader_place = GetShipRuins(formation_leader);
    dword formation_leader_star = CSL_ShipCurrentStar(formation_leader);
    if(!formation_leader_star) exit;
    for(int formation_transport_slot = 1; formation_transport_slot <= 6; formation_transport_slot = formation_transport_slot + 1)
    {
        int formation_transport_id = CSL_GetTransportId(formation_transport_slot);
        if(formation_transport_id <= 1) continue;
        dword formation_transport = 0;
        if(formation_transport_id > 1) formation_transport = IdToShip(formation_transport_id);
        if(!formation_transport || formation_transport == formation_leader) continue;
        if(!CSL_IsTransportId(Id(formation_transport))) continue;
        if(ShipIsTakeoff(formation_transport)) exit;
        if(ShipInHyperSpace(formation_transport, 1)) exit;
        dword formation_transport_place = GetShipPlanet(formation_transport);
        if(!formation_transport_place) formation_transport_place = GetShipRuins(formation_transport);
        if(formation_leader_place)
        {
            if(formation_transport_place != formation_leader_place) exit;
            continue;
        }
        if(formation_transport_place) exit;
        if(!ShipInNormalSpace(formation_leader)) exit;
        if(!ShipInNormalSpace(formation_transport)) exit;
        dword formation_transport_star = CSL_ShipCurrentStar(formation_transport);
        if(formation_transport_star != formation_leader_star) exit;
        if(Dist(formation_transport, formation_leader) > 450) exit;
    }
    result = 1;
}

function CSL_ReturnTransportToLeader(dword regroup_transport, dword regroup_leader)
{
    if(!regroup_transport || !regroup_leader || regroup_transport == regroup_leader) exit;
    if(ShipIsTakeoff(regroup_transport) || ShipInHyperSpace(regroup_transport, 1)) exit;
    if(ShipIsTakeoff(regroup_leader) || ShipInHyperSpace(regroup_leader, 1)) exit;
    dword regroup_leader_place = GetShipPlanet(regroup_leader);
    if(!regroup_leader_place) regroup_leader_place = GetShipRuins(regroup_leader);
    dword regroup_transport_place = GetShipPlanet(regroup_transport);
    if(!regroup_transport_place) regroup_transport_place = GetShipRuins(regroup_transport);
    if(regroup_transport_place)
    {
        if(regroup_leader_place && regroup_transport_place == regroup_leader_place) exit;
        ShipRefuel(regroup_transport);
        ShipRepairEq(regroup_transport);
        CSL_TakeOff(regroup_transport);
        exit;
    }
    dword regroup_leader_star = CSL_ShipCurrentStar(regroup_leader);
    dword regroup_transport_star = CSL_ShipCurrentStar(regroup_transport);
    if(!regroup_leader_star || !regroup_transport_star) exit;
    if(regroup_transport_star == regroup_leader_star)
    {
        if(regroup_leader_place) CSL_Land(regroup_transport, regroup_leader_place);
        else if(ShipInNormalSpace(regroup_transport) && ShipInNormalSpace(regroup_leader))
        {
            OrderLock(regroup_transport, 0);
            if(ShipGetBad(regroup_transport)) ShipSetBad(regroup_transport, 0);
            OrderNone(regroup_transport);
            CSL_Follow(regroup_transport, regroup_leader);
        }
        exit;
    }
    if(ShipCanJump(regroup_transport, regroup_transport_star, regroup_leader_star, 1))
    {
        CSL_Jump(regroup_transport, regroup_leader_star);
        exit;
    }
    dword regroup_fuel_planet = CSL_FindOwnedPlanet(regroup_transport_star, 0);
    if(!regroup_fuel_planet) regroup_fuel_planet = NearCivilPlanet(regroup_transport);
    if(regroup_fuel_planet) CSL_Land(regroup_transport, regroup_fuel_planet);
}

function CSL_RegroupTransports(dword regroup_leader)
{
    if(!regroup_leader) exit;
    // Never replace an unfinished takeoff or jump order. The formation will be
    // checked again after the leader reaches a stable runtime state.
    if(ShipIsTakeoff(regroup_leader)) exit;
    if(ShipInHyperSpace(regroup_leader, 1)) exit;
    dword regroup_leader_place = GetShipPlanet(regroup_leader);
    if(!regroup_leader_place) regroup_leader_place = GetShipRuins(regroup_leader);
    if(!regroup_leader_place && ShipInNormalSpace(regroup_leader))
    {
        OrderLock(regroup_leader, 0);
        if(ShipGetBad(regroup_leader)) ShipSetBad(regroup_leader, 0);
        OrderNone(regroup_leader);
        OrderLock(regroup_leader, 1);
    }
    for(int regroup_transport_slot = 1; regroup_transport_slot <= 6; regroup_transport_slot = regroup_transport_slot + 1)
    {
        int regroup_member_id = CSL_GetTransportId(regroup_transport_slot);
        if(regroup_member_id <= 1) continue;
        dword regroup_member = 0;
        if(regroup_member_id > 1) regroup_member = IdToShip(regroup_member_id);
        if(!regroup_member || regroup_member == regroup_leader) continue;
        if(!CSL_IsTransportId(Id(regroup_member))) continue;
        CSL_ReturnTransportToLeader(regroup_member, regroup_leader);
    }
}

function CSL_ReturnEscortToConvoy(dword escort, dword leader)
{
    if(!escort || !leader) exit;
    // Never cancel a hyperspace order. OrderNone on a travelling ship can
    // strand it outside normal space, so regrouping resumes after emergence.
    if(ShipIsTakeoff(escort) || ShipIsTakeoff(leader)) exit;
    if(ShipInHyperSpace(escort, 1) || ShipInHyperSpace(leader, 1)) exit;
    ShipSetBad(escort, 0);
    OrderLock(escort, 0);
    OrderNone(escort);
    OrderLock(escort, 1);
    dword leader_place = GetShipPlanet(leader);
    if(!leader_place) leader_place = GetShipRuins(leader);
    dword escort_place = GetShipPlanet(escort);
    if(!escort_place) escort_place = GetShipRuins(escort);
    if(escort_place)
    {
        if(leader_place && escort_place == leader_place) exit;
        ShipRefuel(escort);
        ShipRepairEq(escort);
        CSL_TakeOff(escort);
        exit;
    }
    dword leader_star = CSL_ShipCurrentStar(leader);
    dword escort_star = CSL_ShipCurrentStar(escort);
    if(!leader_star || !escort_star) exit;
    if(escort_star == leader_star)
    {
        if(leader_place) CSL_Land(escort, leader_place);
        else if(ShipInNormalSpace(leader) && ShipInNormalSpace(escort)) CSL_Follow(escort, leader);
        exit;
    }
    if(ShipCanJump(escort, escort_star, leader_star, 1))
    {
        CSL_Jump(escort, leader_star);
        exit;
    }
    dword return_fuel_planet = CSL_FindOwnedPlanet(escort_star, 0);
    if(!return_fuel_planet) return_fuel_planet = NearCivilPlanet(escort);
    if(return_fuel_planet) CSL_Land(escort, return_fuel_planet);
}

function CSL_RallyEscorts(dword leader)
{
    if(!leader) exit;
    for(int rally_escort_index = 0; rally_escort_index < GroupCount(SupplyEscorts); rally_escort_index = rally_escort_index + 1)
    {
        dword rally_escort = GroupShip(SupplyEscorts, rally_escort_index);
        if(!rally_escort) continue;
        if(CSL_IsEscortId(Id(rally_escort))) CSL_ReturnEscortToConvoy(rally_escort, leader);
    }
}

function CSL_EngageConvoyThreat(dword threat, dword leader)
{
    if(!threat || !leader) exit;
    for(int engage_escort_index = 0; engage_escort_index < GroupCount(SupplyEscorts); engage_escort_index = engage_escort_index + 1)
    {
        dword engage_escort = GroupShip(SupplyEscorts, engage_escort_index);
        if(!engage_escort) continue;
        if(!CSL_IsEscortId(Id(engage_escort))) continue;
        dword engage_place = GetShipPlanet(engage_escort);
        if(!engage_place) engage_place = GetShipRuins(engage_escort);
        if(engage_place)
        {
            if(PlanetToStar(engage_place) != CSL_ShipCurrentStar(threat))
            {
                CSL_ReturnEscortToConvoy(engage_escort, leader);
                continue;
            }
            ShipRefuel(engage_escort);
            ShipRepairEq(engage_escort);
            CSL_TakeOff(engage_escort);
            continue;
        }
        if(ShipInHyperSpace(engage_escort) || !ShipInNormalSpace(engage_escort) || CSL_ShipCurrentStar(engage_escort) != CSL_ShipCurrentStar(threat))
        {
            CSL_ReturnEscortToConvoy(engage_escort, leader);
            continue;
        }
        OrderLock(engage_escort, 0);
        OrderNone(engage_escort);
        ShipSetBad(engage_escort, threat);
        OrderLock(engage_escort, 1);
        OrderFollowShip(engage_escort, threat, 1, 1);
    }
}

function CSL_FindAmbushTransport()
{
    result = 0;
    if(csl_ambush_checked_star_id <= 1) exit;
    for(int ambush_transport_index = 0; ambush_transport_index < GroupCount(SupplyTransports); ambush_transport_index = ambush_transport_index + 1)
    {
        dword ambush_transport = GroupShip(SupplyTransports, ambush_transport_index);
        if(!ambush_transport) continue;
        if(!CSL_IsTransportId(Id(ambush_transport))) continue;
        // During takeoff the engine has already detached part of the planet
        // state but has not attached a safe ShipStar yet. It is still the
        // convoy ship the pirates are waiting for in the ambush system.
        if(ShipIsTakeoff(ambush_transport))
        {
            dword ambush_takeoff_planet = GetShipPlanet(ambush_transport);
            if(ambush_takeoff_planet)
            {
                dword ambush_takeoff_star = PlanetToStar(ambush_takeoff_planet);
                if(ambush_takeoff_star)
                {
                    if(Id(ambush_takeoff_star) == csl_ambush_checked_star_id)
                    {
                        result = ambush_transport;
                        exit;
                    }
                }
            }
            continue;
        }
        if(ShipInHyperSpace(ambush_transport, 1)) continue;
        dword ambush_transport_star = CSL_ShipCurrentStar(ambush_transport);
        if(!ambush_transport_star) continue;
        if(Id(ambush_transport_star) == csl_ambush_checked_star_id)
        {
            result = ambush_transport;
            exit;
        }
    }
}

function CSL_FindAmbushCargoForPirate(dword cargo_pirate, dword cargo_star)
{
    result = 0;
    if(!cargo_pirate || !cargo_star) exit;
    dword cargo_pirate_hook = ShipEqInSlot(cargo_pirate, t_CargoHook);
    if(!cargo_pirate_hook) exit;
    int cargo_pirate_capacity = CSL_EquipmentStatOrZero(cargo_pirate_hook, 0);
    if(cargo_pirate_capacity <= 0) exit;
    int cargo_pirate_space = ShipFreeSpace(cargo_pirate);
    for(int ambush_cargo_index = 1; ambush_cargo_index <= 30; ambush_cargo_index = ambush_cargo_index + 1)
    {
        dword ambush_cargo = CSL_GetCargo(ambush_cargo_index);
        if(!ambush_cargo || !ItemIn(ambush_cargo, cargo_star)) continue;
        // ItemIn(item, star) may remain true when the item is actually held by
        // a ship in that system. Only a directly loose container is salvage.
        int ambush_cargo_holder_count = StarShips(cargo_star);
        int ambush_cargo_is_held = 0;
        for(int ambush_cargo_holder_index = 0; ambush_cargo_holder_index < ambush_cargo_holder_count; ambush_cargo_holder_index = ambush_cargo_holder_index + 1)
        {
            dword ambush_cargo_holder = StarShips(cargo_star, ambush_cargo_holder_index);
            if(!ambush_cargo_holder) continue;
            if(ItemIn(ambush_cargo, ambush_cargo_holder))
            {
                ambush_cargo_is_held = 1;
                break;
            }
        }
        if(ambush_cargo_is_held) continue;
        int ambush_cargo_size = ItemSize(ambush_cargo);
        if(ambush_cargo_size > cargo_pirate_capacity || ambush_cargo_size > cargo_pirate_space) continue;
        result = ambush_cargo;
        exit;
    }
}

function CSL_CancelAmbushCargoPickup(dword cargo_pirate)
{
    if(!cargo_pirate) exit;
    for(int cancel_cargo_index = 1; cancel_cargo_index <= 30; cancel_cargo_index = cancel_cargo_index + 1)
    {
        dword cancel_cargo = CSL_GetCargo(cancel_cargo_index);
        if(cancel_cargo) ShipPicksItem(cargo_pirate, cancel_cargo, 0);
    }
}

function CSL_OrderAmbushCargoPickup(dword cargo_pirate, dword cargo_star)
{
    result = 0;
    if(!cargo_pirate || !cargo_star) exit;
    if(ShipIsTakeoff(cargo_pirate)) exit;
    if(ShipInHyperSpace(cargo_pirate, 1)) exit;
    if(!ShipInNormalSpace(cargo_pirate)) exit;
    dword selected_ambush_cargo = CSL_FindAmbushCargoForPirate(cargo_pirate, cargo_star);
    if(!selected_ambush_cargo) exit;
    OrderLock(cargo_pirate, 0);
    if(ShipGetBad(cargo_pirate)) ShipSetBad(cargo_pirate, 0);
    OrderNone(cargo_pirate);
    ShipPicksItem(cargo_pirate, selected_ambush_cargo, 1);
    OrderLock(cargo_pirate, 1);
    OrderMove(cargo_pirate, CoordX(selected_ambush_cargo), CoordY(selected_ambush_cargo));
    result = 1;
}

function CSL_AmbushPirateHasCargo(dword cargo_pirate)
{
    result = 0;
    if(!cargo_pirate) exit;
    for(int carried_cargo_index = 1; carried_cargo_index <= 30; carried_cargo_index = carried_cargo_index + 1)
    {
        dword carried_cargo = CSL_GetCargo(carried_cargo_index);
        if(!carried_cargo || !ItemIn(carried_cargo, cargo_pirate)) continue;
        result = 1;
        exit;
    }
}

function CSL_AmbushPirateOrigin(int origin_slot, dword origin_pirate)
{
    result = 0;
    if(!origin_pirate) exit;
    if(!CSL_AmbushPirateDataSafe(origin_pirate)) exit;
    int pirate_origin_id = CSL_GetPendingPirateOriginId(origin_slot);
    int pirate_origin_type = CSL_GetPendingPirateOriginType(origin_slot);
    if(pirate_origin_id <= 1) exit;
    if(pirate_origin_type == 1)
    {
        dword pirate_origin_planet = IdToPlanet(pirate_origin_id);
        if(pirate_origin_planet) result = pirate_origin_planet;
        exit;
    }
    if(pirate_origin_type != 2) exit;
    dword pirate_current_base = GetShipRuins(origin_pirate);
    if(pirate_current_base)
    {
        if(Id(pirate_current_base) == pirate_origin_id)
        {
            result = pirate_current_base;
            exit;
        }
    }
    dword pirate_origin_star = CSL_ShipCurrentStar(origin_pirate);
    if(!pirate_origin_star) exit;
    dword pirate_origin_base = StarRuins(pirate_origin_star, 'PB');
    if(!pirate_origin_base) exit;
    if(Id(pirate_origin_base) == pirate_origin_id) result = pirate_origin_base;
}

// Result 0: this pirate has no stolen cargo.
// Result 1: the pirate has abandoned the attack and is returning home.
// Result 2: the pirate reached its own launch point and was released to AI.
function CSL_ProcessAmbushPirateRetreat(int retreat_slot, dword retreat_pirate)
{
    result = 0;
    if(!retreat_pirate) exit;
    if(!CSL_AmbushPirateDataSafe(retreat_pirate)) exit;
    int pirate_is_retreating = CSL_PendingPirateRetreating(retreat_slot);
    if(!pirate_is_retreating)
    {
        if(!CSL_AmbushPirateHasCargo(retreat_pirate)) exit;
        CSL_SetPendingPirateRetreating(retreat_slot, 1);
        pirate_is_retreating = 1;
    }
    if(!pirate_is_retreating) exit;
    result = 1;
    if(ShipIsTakeoff(retreat_pirate)) exit;
    if(ShipInHyperSpace(retreat_pirate, 1)) exit;
    dword pirate_retreat_origin = CSL_AmbushPirateOrigin(retreat_slot, retreat_pirate);
    if(!pirate_retreat_origin)
    {
        if(CSL_FreeAmbushPirate(retreat_pirate)) result = 2;
        exit;
    }
    dword retreat_ambush_star = IdToStar(csl_ambush_checked_star_id);
    dword retreat_normal_pirate = CSL_ResolveNormalShipInStar(retreat_pirate, retreat_ambush_star);
    if(retreat_normal_pirate)
    {
        CSL_CancelAmbushCargoPickup(retreat_normal_pirate);
        OrderLock(retreat_normal_pirate, 0);
        if(ShipGetBad(retreat_normal_pirate)) ShipSetBad(retreat_normal_pirate, 0);
        OrderNone(retreat_normal_pirate);
        OrderLanding(retreat_normal_pirate, pirate_retreat_origin, 1);
        OrderForsage(retreat_normal_pirate, 1);
        OrderLock(retreat_normal_pirate, 1);
        exit;
    }
    dword pirate_retreat_place = GetShipPlanet(retreat_pirate);
    if(!pirate_retreat_place) pirate_retreat_place = GetShipRuins(retreat_pirate);
    if(pirate_retreat_place == pirate_retreat_origin)
    {
        if(OrderForsage(retreat_pirate)) OrderForsage(retreat_pirate, 0);
        if(CSL_FreeAmbushPirate(retreat_pirate)) result = 2;
        exit;
    }
    if(pirate_retreat_place)
    {
        OrderTakeOff(retreat_pirate);
        exit;
    }
}

function CSL_AmbushHasRecoverableCargo()
{
    result = 0;
    if(csl_ambush_checked_star_id <= 1) exit;
    dword recoverable_cargo_star = IdToStar(csl_ambush_checked_star_id);
    if(!recoverable_cargo_star) exit;
    for(int recoverable_pirate_index = 1; recoverable_pirate_index <= 8; recoverable_pirate_index = recoverable_pirate_index + 1)
    {
        dword recoverable_pirate = IdToShip(CSL_GetPendingPirateId(recoverable_pirate_index));
        if(!recoverable_pirate) continue;
        if(!CSL_AmbushPirateDataSafe(recoverable_pirate)) continue;
        if(CSL_PendingPirateRetreating(recoverable_pirate_index)) continue;
        if(CSL_FindAmbushCargoForPirate(recoverable_pirate, recoverable_cargo_star))
        {
            result = 1;
            exit;
        }
    }
}

function CSL_SetAmbushPirateConsequences(dword consequence_pirate)
{
    if(!consequence_pirate) exit;
    // A successful or surviving ambusher keeps vanilla pirate AI and any
    // captured cargo, but every non-Peleng Coalition world now treats that
    // specific pirate as an outright enemy. Peleng worlds remain available.
    ChangeGlobalRelationsPlanets(consequence_pirate, 0, 0, 0, 29);
}

function CSL_FreeAmbushPirate(dword released_ambush_pirate)
{
    result = 0;
    if(!released_ambush_pirate) { result = 1; exit; }
    // Never mutate orders while the mobile ship is between runtime states.
    if(ShipIsTakeoff(released_ambush_pirate)) exit;
    if(ShipInHyperSpace(released_ambush_pirate, 1)) exit;
    CSL_SetAmbushPirateConsequences(released_ambush_pirate);
    NoTalkToShip(released_ambush_pirate, 0);
    OrderLock(released_ambush_pirate, 0);
    if(ShipGetBad(released_ambush_pirate)) ShipSetBad(released_ambush_pirate, 0);
    OrderNone(released_ambush_pirate);
    ShipFreeFlight(released_ambush_pirate);
    result = 1;
}

function CSL_ReleaseAmbushPirates()
{
    result = 1;
    for(int release_ambush_index = 8; release_ambush_index >= 1; release_ambush_index = release_ambush_index - 1)
    {
        dword release_ambush_pirate = IdToShip(CSL_GetPendingPirateId(release_ambush_index));
        if(!release_ambush_pirate)
        {
            CSL_SetPendingPirateId(release_ambush_index, 0);
            continue;
        }
        if(!CSL_AmbushPirateDataSafe(release_ambush_pirate))
        {
            result = 0;
            continue;
        }
        if(CSL_PendingPirateRetreating(release_ambush_index)) continue;
        if(!CSL_FreeAmbushPirate(release_ambush_pirate))
        {
            result = 0;
            continue;
        }
        CSL_SetPendingPirateId(release_ambush_index, 0);
    }
    if(CSL_ActiveAmbushPirateCount() > 0) result = 0;
}

function CSL_CheckConvoyAmbush(dword ambush_star, dword ambush_target)
{
    result = 0;
    // BuyPirate and ShipCalcParam may synchronously re-enter Turn code. While
    // a batch or equipment rebuild is in progress, the future takeoff date is
    // a re-entry barrier: nested calls must not touch partially built ships.
    if(csl_ambush_target_id > 1 && csl_ambush_takeoff_turn > CurTurn()) exit;
    // Retreat and release are isolated from the active-combat traversal.
    // ShipFreeFlight mutates script-group ownership, so after one completed
    // return this invocation must end before any old ship handle is reused.
    for(int retreat_pirate_index = 8; retreat_pirate_index >= 1; retreat_pirate_index = retreat_pirate_index - 1)
    {
        dword tracked_retreat_pirate = IdToShip(CSL_GetPendingPirateId(retreat_pirate_index));
        if(!tracked_retreat_pirate) continue;
        if(!CSL_AmbushPirateDataSafe(tracked_retreat_pirate)) continue;
        int tracked_retreat_state = CSL_ProcessAmbushPirateRetreat(retreat_pirate_index, tracked_retreat_pirate);
        if(tracked_retreat_state != 2) continue;
        CSL_SetPendingPirateId(retreat_pirate_index, 0);
        exit;
    }
    if(csl_ambush_target_id > 1)
    {
        dword ambush_outcome_target = IdToShip(csl_ambush_target_id);
        int ambush_outcome_target_valid = 0;
        if(ambush_outcome_target)
        {
            if(CSL_IsTransportId(Id(ambush_outcome_target)))
            {
                if(ShipIsTakeoff(ambush_outcome_target)) ambush_outcome_target_valid = 1;
                else if(!ShipInHyperSpace(ambush_outcome_target, 1))
                {
                    dword ambush_outcome_target_star = CSL_ShipCurrentStar(ambush_outcome_target);
                    if(ambush_outcome_target_star)
                    {
                        if(Id(ambush_outcome_target_star) == csl_ambush_checked_star_id) ambush_outcome_target_valid = 1;
                    }
                }
            }
        }
        if(!ambush_outcome_target_valid)
        {
            // Pirates continue after a remaining freighter in their system,
            // not after the one that happened to be selected at launch.
            ambush_outcome_target = CSL_FindAmbushTransport();
            if(ambush_outcome_target)
            {
                ambush_outcome_target_valid = 1;
                csl_ambush_target_id = Id(ambush_outcome_target);
            }
        }
        if(!ambush_outcome_target_valid)
        {
            // The encounter ends only after every surviving freighter has
            // left the system or every freighter has been destroyed. Before
            // becoming ordinary pirates, survivors collect every container
            // their installed hooks and remaining holds can carry.
            if(!CSL_AmbushHasRecoverableCargo())
            {
                if(!CSL_ReleaseAmbushPirates()) exit;
                dword ambush_surviving_transport = CSL_ActiveTransportLeader();
                if(ambush_surviving_transport || csl_delivered_cargo > 0 || csl_arrived_transports > 0)
                    AddPlanetNews(CT('XenoCoalitionSupplyLines.AmbushEscapedNoEscorts'));
                else AddPlanetNews(CT('XenoCoalitionSupplyLines.AmbushConvoyLost'));
                csl_ambush_target_id = 0;
                csl_ambush_origin_planet_id = 0;
                csl_ambush_origin_base_id = 0;
                csl_ambush_takeoff_turn = 0;
                exit;
            }
        }
        if(CSL_LiveEscortCount() <= 0)
        {
            int ambush_outcome_sheltered = CSL_AllActiveTransportsSheltered();
            if(ambush_outcome_sheltered && csl_ambush_takeoff_turn != -1)
            {
                // The pirates remain outside. Rebuild the route from shelter
                // and let the armed freighters attempt a breakout.
                csl_ambush_takeoff_turn = -1;
                result = 1;
                exit;
            }
        }
    }
    dword tracked_ambush_star = IdToStar(csl_ambush_checked_star_id);
    for(int pending_pirate_index = 8; pending_pirate_index >= 1; pending_pirate_index = pending_pirate_index - 1)
    {
        dword pending_pirate = IdToShip(CSL_GetPendingPirateId(pending_pirate_index));
        if(!pending_pirate)
        {
            CSL_SetPendingPirateId(pending_pirate_index, 0);
            continue;
        }
        dword pending_normal_pirate = CSL_ResolveNormalShipInStar(pending_pirate, tracked_ambush_star);
        int pending_pirate_in_normal_space = pending_normal_pirate != 0;
        if(pending_pirate_in_normal_space) pending_pirate = pending_normal_pirate;
        else if(!CSL_AmbushPirateDataSafe(pending_pirate)) continue;
        if(CSL_PendingPirateRetreating(pending_pirate_index)) continue;
        // A flying pirate is no longer a fresh BuyPirate result. Install its
        // cargo hook in an isolated invocation and do not reuse that handle
        // after the equipment mutation; combat begins on the following call.
        if(pending_pirate_in_normal_space)
        {
            dword active_pirate_hook = ShipEqInSlot(pending_pirate, t_CargoHook);
            int active_pirate_hook_capacity = CSL_EquipmentStatOrZero(active_pirate_hook, 0);
            if(active_pirate_hook_capacity < 60)
            {
                int hook_reentry_takeoff_turn = csl_ambush_takeoff_turn;
                csl_ambush_takeoff_turn = CurTurn() + 1;
                CSL_EnsureAmbushPirateHook(pending_pirate);
                csl_ambush_takeoff_turn = hook_reentry_takeoff_turn;
                continue;
            }
        }
        // Loot takes precedence as soon as a destroyed transport drops it.
        // The first pirate that actually secures a container abandons the
        // remaining raiders on the next invocation and runs for its own base.
        dword immediate_cargo_star = tracked_ambush_star;
        if(immediate_cargo_star && CSL_OrderAmbushCargoPickup(pending_pirate, immediate_cargo_star)) continue;
        // Every pirate chooses its own nearest live freighter. A single
        // convoy-wide target made distant pirates ignore closer transports.
        dword pending_target = CSL_FindNearestTransportToShip(pending_pirate);
        int pending_target_valid = 0;
        if(pending_target)
        {
            if(CSL_IsTransportId(Id(pending_target)))
            {
                if(ShipIsTakeoff(pending_target)) pending_target_valid = 1;
                else
                {
                    dword pending_pirate_star = CSL_ShipCurrentStar(pending_pirate);
                    dword pending_target_star = CSL_ShipCurrentStar(pending_target);
                    if(pending_pirate_star && pending_pirate_star == pending_target_star) pending_target_valid = 1;
                }
            }
        }
        dword pending_pirate_place = 0;
        if(!pending_pirate_in_normal_space)
        {
            pending_pirate_place = GetShipPlanet(pending_pirate);
            if(!pending_pirate_place) pending_pirate_place = GetShipRuins(pending_pirate);
        }
        if(!pending_target_valid)
        {
            if(ShipIsTakeoff(pending_pirate) || ShipInHyperSpace(pending_pirate, 1)) continue;
            dword abandoned_cargo_star = IdToStar(csl_ambush_checked_star_id);
            if(abandoned_cargo_star && CSL_OrderAmbushCargoPickup(pending_pirate, abandoned_cargo_star)) continue;
            // A pirate that has independently left the ambush system is no
            // longer held by the scenario and continues under vanilla AI.
            if(CSL_FreeAmbushPirate(pending_pirate))
                CSL_SetPendingPirateId(pending_pirate_index, 0);
            continue;
        }
        if(pending_pirate_place)
        {
            if(ShipIsTakeoff(pending_pirate) || ShipInHyperSpace(pending_pirate, 1)) continue;
            // BuyPirate returns a fresh runtime handle. Mutating equipment and
            // orders through that handle in the same batch-spawn invocation can
            // leave RScript rebuilding ship state recursively. Configure only a
            // stable ship resolved by its persistent ID on the following turn.
            dword pending_origin_planet = GetShipPlanet(pending_pirate);
            dword pending_origin_base = GetShipRuins(pending_pirate);
            if(pending_origin_base)
            {
                CSL_SetPendingPirateOrigin(pending_pirate_index, Id(pending_origin_base), 2);
            }
            else if(pending_origin_planet)
            {
                CSL_SetPendingPirateOrigin(pending_pirate_index, Id(pending_origin_planet), 1);
            }
            NoTalkToShip(pending_pirate, 1);
            if(CurTurn() >= csl_ambush_takeoff_turn) OrderTakeOff(pending_pirate);
            continue;
        }
        if(ShipIsTakeoff(pending_pirate) || ShipInHyperSpace(pending_pirate, 1)) continue;
        if(!pending_pirate_in_normal_space) continue;
        if(ShipIsTakeoff(pending_target))
        {
            OrderLock(pending_pirate, 0);
            if(ShipGetBad(pending_pirate)) ShipSetBad(pending_pirate, 0);
            OrderNone(pending_pirate);
            OrderLock(pending_pirate, 1);
            continue;
        }
        dword pending_target_place = GetShipPlanet(pending_target);
        if(!pending_target_place) pending_target_place = GetShipRuins(pending_target);
        if(pending_target_place)
        {
            dword shelter_cargo_star = CSL_ShipCurrentStar(pending_pirate);
            if(shelter_cargo_star && CSL_OrderAmbushCargoPickup(pending_pirate, shelter_cargo_star)) continue;
            // Once the freighters are sheltered, the raid does not become
            // passive.  Ambushers proactively engage the surviving convoy
            // guard; waiting for an escort to shoot first made half the raid
            // hover harmlessly around the planet.
            dword waiting_escort_target = CSL_FindNearestEscortToShip(pending_pirate);
            if(waiting_escort_target)
            {
                CSL_CancelAmbushCargoPickup(pending_pirate);
                OrderLock(pending_pirate, 0);
                OrderNone(pending_pirate);
                if(ShipGetBad(pending_pirate) != waiting_escort_target) ShipSetBad(pending_pirate, waiting_escort_target);
                OrderLock(pending_pirate, 1);
                OrderFollowShip(pending_pirate, waiting_escort_target, 1, 1);
                continue;
            }
            // With no live escort left, patrol around the shelter instead of
            // freezing at OrderNone. A real vanilla attacker remains a valid
            // temporary combat target.
            dword waiting_pirate_star = CSL_ShipCurrentStar(pending_pirate);
            dword waiting_raw_attacker = ShipGetBad(pending_pirate);
            dword waiting_live_attacker = CSL_ResolveNormalShipInStar(waiting_raw_attacker, waiting_pirate_star);
            if(waiting_live_attacker)
            {
                if(ShipOrderObj(pending_pirate) != waiting_live_attacker)
                {
                    OrderLock(pending_pirate, 0);
                    OrderNone(pending_pirate);
                    if(ShipGetBad(pending_pirate) != waiting_live_attacker) ShipSetBad(pending_pirate, waiting_live_attacker);
                    OrderLock(pending_pirate, 1);
                    OrderFollowShip(pending_pirate, waiting_live_attacker, 1, 1);
                }
                continue;
            }
            int ambush_patrol_phase = (CurTurn() + pending_pirate_index) - ((CurTurn() + pending_pirate_index) / 4) * 4;
            int ambush_patrol_radius = 900 + 120 * (pending_pirate_index - (pending_pirate_index / 3) * 3);
            int ambush_patrol_x = CoordX(pending_target_place);
            int ambush_patrol_y = CoordY(pending_target_place);
            if(ambush_patrol_phase == 0) ambush_patrol_x = ambush_patrol_x + ambush_patrol_radius;
            else if(ambush_patrol_phase == 1) ambush_patrol_y = ambush_patrol_y + ambush_patrol_radius;
            else if(ambush_patrol_phase == 2) ambush_patrol_x = ambush_patrol_x - ambush_patrol_radius;
            else ambush_patrol_y = ambush_patrol_y - ambush_patrol_radius;
            OrderLock(pending_pirate, 0);
            if(ShipGetBad(pending_pirate)) ShipSetBad(pending_pirate, 0);
            OrderNone(pending_pirate);
            OrderLock(pending_pirate, 1);
            OrderMove(pending_pirate, ambush_patrol_x, ambush_patrol_y);
            continue;
        }
        // Reassert the attack every turn. ShipOrderObj may retain the old
        // target even after vanilla AI has switched to salvage or landing.
        CSL_CancelAmbushCargoPickup(pending_pirate);
        OrderLock(pending_pirate, 0);
        OrderNone(pending_pirate);
        if(ShipGetBad(pending_pirate) != pending_target) ShipSetBad(pending_pirate, pending_target);
        OrderLock(pending_pirate, 1);
        OrderFollowShip(pending_pirate, pending_target, 1, 1);
    }
    if(csl_ambush_target_id > 1 && CSL_ActiveAmbushPirateCount() <= 0)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.AmbushRepelled'));
        csl_ambush_target_id = 0;
        csl_ambush_origin_planet_id = 0;
        csl_ambush_origin_base_id = 0;
        csl_ambush_takeoff_turn = 0;
        result = 1;
        exit;
    }
    if(!ambush_star || !ambush_target || csl_source_race == Peleng) exit;
    int ambush_star_id = Id(ambush_star);
    // The roll belongs to this visit, not to every day spent in the system.
    // A scalar is deliberate: the old persistent growing array could lose its
    // live dimension after a save and allow another pirate wave next turn.
    if(csl_ambush_checked_star_id == ambush_star_id) exit;
    csl_ambush_checked_star_id = ambush_star_id;
    dword ambush_peleng_planet = 0;
    for(int ambush_planet_index = 0; ambush_planet_index < StarPlanets(ambush_star); ambush_planet_index = ambush_planet_index + 1)
    {
        dword ambush_planet = StarPlanets(ambush_star, ambush_planet_index);
        if(!ambush_planet) continue;
        int ambush_planet_owner = PlanetOwner(ambush_planet);
        if(PlanetRace(ambush_planet) == Peleng && ambush_planet_owner >= 0 && ambush_planet_owner <= 4)
        {
            ambush_peleng_planet = ambush_planet;
            break;
        }
    }
    dword ambush_pirate_base = StarRuins(ambush_star, 'PB');
    int ambush_chance = 0;
    if(ambush_peleng_planet) ambush_chance = ambush_chance + 5;
    if(ambush_pirate_base) ambush_chance = ambush_chance + 5;
    if(ambush_chance <= 0 || Rnd(1, 100) > ambush_chance) exit;
    csl_ambush_origin_planet_id = 0;
    csl_ambush_origin_base_id = 0;
    if(ambush_peleng_planet) csl_ambush_origin_planet_id = Id(ambush_peleng_planet);
    if(ambush_pirate_base) csl_ambush_origin_base_id = Id(ambush_pirate_base);
    csl_ambush_target_id = Id(ambush_target);
    int ambush_spawn_count = max(1, min(8, CSL_LiveEscortCount() + Rnd(-1, 1)));
    csl_ambush_takeoff_turn = CurTurn() + 1;
    int ambush_created_count = 0;
    for(int ambush_create_index = 1; ambush_create_index <= ambush_spawn_count; ambush_create_index = ambush_create_index + 1)
    {
        // Reserve a persistent tracking slot before creating the hull. A thief
        // returning from an earlier encounter may still occupy one of the eight
        // slots, so creating first could leave an untracked scripted pirate.
        int created_ambush_slot = CSL_FindFreePendingPirateSlot();
        if(created_ambush_slot <= 0) break;
        dword created_ambush_pirate = 0;
        int created_ambush_pirate_id = 0;
        int create_from_pirate_base = ambush_pirate_base && !ambush_peleng_planet;
        if(ambush_pirate_base && ambush_peleng_planet) create_from_pirate_base = Rnd(0, 1);
        if(create_from_pirate_base)
        {
            if(!ambush_pirate_base) continue;
            dword created_ambush_factory = PlanetPirateClan();
            if(!created_ambush_factory) continue;
            created_ambush_pirate = BuyPirate(created_ambush_factory, 100);
            // This is the only same-turn mutation retained from the known-good
            // batch spawner: move the newly purchased pirate to its real base.
            // No equipment, order or script-data mutation follows this transfer.
            if(created_ambush_pirate)
            {
                created_ambush_pirate_id = Id(created_ambush_pirate);
                TransferShip(created_ambush_pirate, ambush_pirate_base);
            }
        }
        else
        {
            if(!ambush_peleng_planet) continue;
            created_ambush_pirate = BuyPirate(ambush_peleng_planet, 100);
            if(created_ambush_pirate) created_ambush_pirate_id = Id(created_ambush_pirate);
        }
        if(created_ambush_pirate_id > 1)
        {
            // Register the complete wave now, then yield. The next invocation
            // resolves each pirate by ID and performs all remaining setup.
            CSL_SetPendingPirateId(created_ambush_slot, created_ambush_pirate_id);
            if(create_from_pirate_base)
                CSL_SetPendingPirateOrigin(created_ambush_slot, Id(ambush_pirate_base), 2);
            else CSL_SetPendingPirateOrigin(created_ambush_slot, Id(ambush_peleng_planet), 1);
            ambush_created_count = ambush_created_count + 1;
        }
    }
    if(ambush_created_count <= 0)
    {
        csl_ambush_target_id = 0;
        csl_ambush_origin_planet_id = 0;
        csl_ambush_origin_base_id = 0;
        csl_ambush_takeoff_turn = 0;
        exit;
    }
    if(ambush_peleng_planet && ambush_pirate_base)
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.AmbushCombinedLaunch'), '#planet#', Name(ambush_peleng_planet), '#system#', StarName(ambush_star), '#count#', ambush_created_count));
    else if(ambush_pirate_base)
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.AmbushBaseLaunch'), '#system#', StarName(ambush_star), '#count#', ambush_created_count));
    else
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.AmbushPelengLaunch'), '#planet#', Name(ambush_peleng_planet), '#count#', ambush_created_count));
}

function CSL_ProcessDeferredCrisisItem()
{
    if(csl_crisis_pending_free_item_id <= 1) exit;
    int crisis_deferred_item_id = csl_crisis_pending_free_item_id;
    csl_crisis_pending_free_item_id = 0;
    dword crisis_deferred_item = IdToItem(crisis_deferred_item_id);
    if(crisis_deferred_item) FreeItem(crisis_deferred_item);
}

function CSL_ReloadCrisisCargo()
{
    result = 0;
    if(!csl_source || CSL_LiveTransportCount() <= 0) exit;
    for(int crisis_clear_slot = 1; crisis_clear_slot <= 30; crisis_clear_slot = crisis_clear_slot + 1) CSL_ReleaseCargoSlot(crisis_clear_slot);
    dword crisis_price_hull = CreateHull(0, 100 + 175 * max(1, GalaxyTechLevel()), max(1, min(8, GalaxyTechLevel())), csl_source_race, -1);
    int crisis_price_hull_cost = max(1, ItemCost(crisis_price_hull));
    FreeItem(crisis_price_hull);
    int crisis_reload_cargo_index = 1;
    for(int crisis_reload_transport_index = 0; crisis_reload_transport_index < GroupCount(SupplyTransports); crisis_reload_transport_index = crisis_reload_transport_index + 1)
    {
        dword crisis_reload_transport = GroupShip(SupplyTransports, crisis_reload_transport_index);
        if(!crisis_reload_transport) continue;
        if(!CSL_IsTransportId(Id(crisis_reload_transport))) continue;
        if(GetShipPlanet(crisis_reload_transport) != csl_source) exit;
        int crisis_reload_slot = CSL_TransportSlotForId(Id(crisis_reload_transport));
        if(crisis_reload_slot <= 0 || crisis_reload_slot >= ArrayDim(csl_transport_cargo_counts)) continue;
        int crisis_reload_count = csl_transport_cargo_counts[crisis_reload_slot];
        int crisis_reload_base = max(1000, round(2.0 * crisis_price_hull_cost / max(1, crisis_reload_count)));
        int crisis_reload_required = 0;
        unknown crisis_reload_sizes = newarray(1);
        ArrayClear(crisis_reload_sizes);
        for(int crisis_reload_size_index = 1; crisis_reload_size_index <= crisis_reload_count; crisis_reload_size_index = crisis_reload_size_index + 1)
        {
            int crisis_reload_size = Rnd(30, 60);
            ArrayAdd(crisis_reload_sizes, crisis_reload_size);
            crisis_reload_required = crisis_reload_required + crisis_reload_size;
        }
        if(!CSL_EnsureCargoCapacity(crisis_reload_transport, crisis_reload_required)) exit;
        for(int crisis_reload_create_index = 1; crisis_reload_create_index < ArrayDim(crisis_reload_sizes); crisis_reload_create_index = crisis_reload_create_index + 1)
        {
            CSL_CreateCargo(crisis_reload_transport, crisis_reload_cargo_index, crisis_reload_base, crisis_reload_sizes[crisis_reload_create_index]);
            crisis_reload_cargo_index = crisis_reload_cargo_index + 1;
        }
        ShipRefuel(crisis_reload_transport);
        ShipRepairEq(crisis_reload_transport);
    }
    csl_total_cargo = csl_total_cargo + crisis_reload_cargo_index - 1;
    result = 1;
}

function CSL_DeliverCargo(dword cargo_ship, int cargo_index)
{
    dword cargo = CSL_GetCargo(cargo_index);
    if(cargo && ItemIn(cargo, cargo_ship))
    {
        csl_delivered_cargo = csl_delivered_cargo + 1;
        // Do not detach and free inventory while the ship is docked. The
        // accepted crate stays aboard until deferred ShipDestroy removes the
        // transport. Script links are cleared only after the ship disappears.
        NoDropItem(cargo, 1);
    }
}

function CSL_DeliverTransport(dword ship)
{
    if(!ship) exit;
    for(int delivered_cargo_index = 1; delivered_cargo_index <= 30; delivered_cargo_index = delivered_cargo_index + 1)
        CSL_DeliverCargo(ship, delivered_cargo_index);
    csl_arrived_transports = csl_arrived_transports + 1;
    int delivered_id = Id(ship);
    CSL_ClearTransportIdValue(delivered_id);
    csl_pending_transport_cleanup_id = delivered_id;
    int remaining_tracked_transports = 0;
    for(int remaining_transport_index = 0; remaining_transport_index < GroupCount(SupplyTransports); remaining_transport_index = remaining_transport_index + 1)
    {
        dword remaining_transport = GroupShip(SupplyTransports, remaining_transport_index);
        if(!remaining_transport) continue;
        if(CSL_IsTransportId(Id(remaining_transport))) remaining_tracked_transports = remaining_tracked_transports + 1;
    }
    // Start the preparation delay at the actual final unloading, not after the
    // delivered hulls and escorts have been retired one per date.
    if(!remaining_tracked_transports && csl_delivered_cargo > 0 && csl_strike_build_due <= 0)
        csl_strike_build_due = CurTurn() + RndObject(15, 30, csl_destination);
    ShipCalcParam(ship);
    csl_next_turn = CurTurn() + 1;
}

function CSL_EnsureCargoCapacity(dword ship, int required_space)
{
    result = 0;
    if(!ship || required_space <= 0) exit;
    ShipCalcParam(ship);
    if(ShipFreeSpace(ship) < required_space)
    {
        dword cargo_hull = ShipEqInSlot(ship, t_Hull);
        if(!cargo_hull) exit;
        int missing_space = required_space - ShipFreeSpace(ship);
        ItemSize(cargo_hull, ItemSize(cargo_hull) + missing_space + 1);
        HullHP(ship, ItemSize(cargo_hull));
        ShipCalcParam(ship);
    }
    result = ShipFreeSpace(ship) >= required_space;
}

function CSL_CreateCargo(dword ship, int cargo_index, int crate_base, int crate_size)
{
    dword cargo = CreateQuestItem('XenoCoalitionSupplyCargo', csl_source_race);
    ItemSize(cargo, crate_size);
    ItemLevel(cargo, max(1, min(8, GalaxyTechLevel())));
    ItemCost(cargo, max(1, round(1.0 * crate_base * CSL_InflationFactor() / 100)));
    // UtilityFunctions.RndObject rejects quest items as anchors. A fresh
    // built-in roll keeps the 65% chance independent for every container.
    if(Rnd(1, 100) > 65) NoDropItem(cargo, 1);
    CSL_LinkCargo(cargo, cargo_index);
    AddItemToShip(ship, cargo);
    int cargo_id = Id(cargo);
    CSL_SetCargoId(cargo_index, cargo_id);
    ArrayAdd(csl_cargo_registry, cargo_id);
    ArrayAdd(csl_cargo_bases, crate_base);
}

function CSL_SelectComposition()
{
    csl_composition_band = CSL_StrategyBand();
    csl_crisis_mode = csl_composition_band == 7;
    int transport_min = max(1, min(6, CSL_ConfigBandValue(csl_composition_band, 1)));
    int transport_max = max(transport_min, min(6, CSL_ConfigBandValue(csl_composition_band, 2)));
    int escort_min = max(0, min(6, CSL_ConfigBandValue(csl_composition_band, 3)));
    int escort_max = max(escort_min, min(6, CSL_ConfigBandValue(csl_composition_band, 4)));
    csl_cargo_min = max(1, min(5, CSL_ConfigBandValue(csl_composition_band, 5)));
    csl_cargo_max = max(csl_cargo_min, min(5, CSL_ConfigBandValue(csl_composition_band, 6)));
    csl_transport_count = CSL_ConfigRoll(transport_min, transport_max, csl_source);
    csl_escort_count = CSL_ConfigRoll(escort_min, escort_max, csl_source);
}

function CSL_BuildCrisisRoute(dword crisis_source_star)
{
    result = 0;
    if(!crisis_source_star) exit;
    dword crisis_selected_planet = 0;
    int crisis_selected_distance = 1000000;
    for(int crisis_star_index = 0; crisis_star_index < GalaxyStars(); crisis_star_index = crisis_star_index + 1)
    {
        dword crisis_star = GalaxyStar(crisis_star_index);
        if(!CSL_IsFreeStar(crisis_star) || crisis_star == crisis_source_star) continue;
        dword crisis_planet = CSL_FindOwnedPlanet(crisis_star, 0);
        if(!crisis_planet || !CSL_ConvoyCanJump(crisis_source_star, crisis_star)) continue;
        int crisis_distance = Dist(crisis_source_star, crisis_star);
        if(crisis_distance < crisis_selected_distance)
        {
            crisis_selected_distance = crisis_distance;
            crisis_selected_planet = crisis_planet;
        }
    }
    if(!crisis_selected_planet) exit;
    csl_destination = crisis_selected_planet;
    csl_destination_id = Id(crisis_selected_planet);
    ArrayClear(csl_route_planets);
    ArrayAdd(csl_route_planets, csl_destination_id);
    csl_route_index = 1;
    result = 1;
}

function CSL_ReleaseVolunteerRangersAtOnce()
{
    result = 0;
    if(!CSL_VolunteerCount())
    {
        CSL_ResetVolunteerRangerIds();
        exit;
    }
    unknown volunteer_release_ids = newarray(1);
    ArrayClear(volunteer_release_ids);
    for(int volunteer_release_probe = 1; volunteer_release_probe <= 6; volunteer_release_probe = volunteer_release_probe + 1)
    {
        dword volunteer_release_probe_ship = IdToShip(csl_volunteer_ranger_ids[volunteer_release_probe]);
        if(!volunteer_release_probe_ship) continue;
        if(ShipInHyperSpace(volunteer_release_probe_ship) || ShipInHyperSpace(volunteer_release_probe_ship, 1) || ShipIsTakeoff(volunteer_release_probe_ship))
        {
            result = 1;
            exit;
        }
        dword volunteer_release_place = GetShipPlanet(volunteer_release_probe_ship);
        if(!volunteer_release_place) volunteer_release_place = GetShipRuins(volunteer_release_probe_ship);
        if(volunteer_release_place)
        {
            ShipRefuel(volunteer_release_probe_ship);
            ShipRepairEq(volunteer_release_probe_ship);
            CSL_TakeOff(volunteer_release_probe_ship);
            result = 1;
            exit;
        }
        ArrayAdd(volunteer_release_ids, Id(volunteer_release_probe_ship));
    }
    for(int volunteer_release_index = 1; volunteer_release_index < ArrayDim(volunteer_release_ids); volunteer_release_index = volunteer_release_index + 1)
    {
        dword volunteer_release_ship = IdToShip(volunteer_release_ids[volunteer_release_index]);
        if(!volunteer_release_ship) continue;
        ShipSetBad(volunteer_release_ship, 0);
        ShipOut(volunteer_release_ship);
        OrderLock(volunteer_release_ship, 0);
        ShipFreeFlight(volunteer_release_ship);
    }
    CSL_ResetVolunteerRangerIds();
    result = 1;
}

function CSL_SpawnConvoy()
{
    // Fail closed: any runtime error during construction must lead to cleanup
    // on the next invocation instead of duplicating a partially built convoy.
    csl_phase = 10;
    csl_next_turn = CurTurn() + 1;
    csl_replan_pending = 0;
    CSL_ResetTransportIds();
    CSL_ResetEscortIds();
    ArrayClear(csl_transport_cargo_counts);
    csl_ambush_checked_star_id = 0;
    csl_ambush_target_id = 0;
    csl_ambush_origin_planet_id = 0;
    csl_ambush_origin_base_id = 0;
    csl_pending_transport_cleanup_id = 0;
    for(int clear_index = 1; clear_index <= 30; clear_index = clear_index + 1) CSL_ReleaseCargoSlot(clear_index);
    if(csl_hostile_systems <= 0)
    {
        csl_phase = 0;
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    CSL_SelectComposition();
    csl_total_cargo = 0;
    csl_delivered_cargo = 0;
    csl_delivered_escorts = 0;
    csl_crisis_trips_completed = 0;
    csl_crisis_unload_pending = 0;
    csl_strike_build_due = 0;
    csl_arrived_transports = 0;
    csl_violation = 0;
    csl_source_race = PlanetRace(csl_source);
    dword first_transport = BuyTransport(csl_source, 0, 100);
    if(!first_transport)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 0;
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    ShipJoin(SupplyTransports, first_transport);
    OrderLock(first_transport, 1);
    CSL_ClearTransportCargo(first_transport);
    if(!CSL_ConfigureTransport(first_transport))
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        exit;
    }
    int cargo_index = 1;
    for(int transport_index = 1; transport_index <= csl_transport_count; transport_index = transport_index + 1)
    {
        dword new_transport = first_transport;
        if(transport_index > 1)
        {
            new_transport = BuyTransport(csl_source, 0, 100);
            if(!new_transport) continue;
            ShipJoin(SupplyTransports, new_transport);
            OrderLock(new_transport, 1);
            CSL_ClearTransportCargo(new_transport);
            if(!CSL_ConfigureTransport(new_transport))
            {
                AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
                exit;
            }
        }
        CSL_SetTransportId(transport_index, Id(new_transport));
        int transport_cargo_count = RndObject(csl_cargo_min, csl_cargo_max, new_transport);
        ArrayAdd(csl_transport_cargo_counts, transport_cargo_count);
        csl_total_cargo = csl_total_cargo + transport_cargo_count;
        ShipCalcParam(new_transport);
    }
    for(int escort_index = 1; escort_index <= csl_escort_count; escort_index = escort_index + 1)
    {
        dword new_escort = BuyWarrior(csl_source, 100);
        if(!new_escort) continue;
        ShipJoin(SupplyEscorts, new_escort);
        OrderLock(new_escort, 1);
        ShipCalcParam(new_escort);
        ShipRefuel(new_escort);
        ShipRepairEq(new_escort);
        CSL_SetEscortId(escort_index, Id(new_escort));
    }

    if(CSL_LiveTransportCount() != csl_transport_count || CSL_LiveEscortCount() != csl_escort_count)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 10;
        exit;
    }

    dword price_hull = CreateHull(0, 100 + 175 * max(1, GalaxyTechLevel()), max(1, min(8, GalaxyTechLevel())), csl_source_race, -1);
    int price_hull_cost = max(1, ItemCost(price_hull));
    FreeItem(price_hull);
    for(int cargo_transport_cursor = 0; cargo_transport_cursor < GroupCount(SupplyTransports); cargo_transport_cursor = cargo_transport_cursor + 1)
    {
        dword cargo_transport = GroupShip(SupplyTransports, cargo_transport_cursor);
        if(!cargo_transport) continue;
        if(!CSL_IsTransportId(Id(cargo_transport))) continue;
        int cargo_transport_slot = CSL_TransportSlotForId(Id(cargo_transport));
        if(cargo_transport_slot <= 0 || cargo_transport_slot >= ArrayDim(csl_transport_cargo_counts)) continue;
        int cargo_count_for_transport = csl_transport_cargo_counts[cargo_transport_slot];
        int crate_base_for_transport = max(1000, round(2.0 * price_hull_cost / max(1, cargo_count_for_transport)));
        unknown cargo_sizes_for_transport = newarray(1);
        ArrayClear(cargo_sizes_for_transport);
        int cargo_required_space = 0;
        for(int cargo_size_cursor = 1; cargo_size_cursor <= cargo_count_for_transport; cargo_size_cursor = cargo_size_cursor + 1)
        {
            int generated_cargo_size = Rnd(30, 60);
            ArrayAdd(cargo_sizes_for_transport, generated_cargo_size);
            cargo_required_space = cargo_required_space + generated_cargo_size;
        }
        if(!CSL_EnsureCargoCapacity(cargo_transport, cargo_required_space))
        {
            AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
            csl_phase = 10;
            exit;
        }
        for(int cargo_create_cursor = 1; cargo_create_cursor < ArrayDim(cargo_sizes_for_transport); cargo_create_cursor = cargo_create_cursor + 1)
        {
            CSL_CreateCargo(cargo_transport, cargo_index, crate_base_for_transport, cargo_sizes_for_transport[cargo_create_cursor]);
            cargo_index = cargo_index + 1;
        }
        ShipCalcParam(cargo_transport);
    }

    // Build the shared route only after every transport carries its final
    // final variable container count. ShipCanJump validates the real loaded
    // mass and range, not the lighter pre-loading configuration.
    // Route length follows the real installed engines and fuel tanks. A direct
    // frontier jump is valid at any strategic stage when every ship can make it.
    int convoy_route_built = 0;
    if(csl_crisis_mode) convoy_route_built = CSL_BuildCrisisRoute(PlanetToStar(csl_source));
    else convoy_route_built = CSL_BuildConvoyRoute(first_transport, PlanetToStar(csl_source), 1);
    if(!convoy_route_built)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 10;
        exit;
    }

    AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Assembled'), '#source#', Name(csl_source), '#transports#', csl_transport_count, '#escorts#', csl_escort_count, '#destination#', Name(csl_destination)));
    csl_phase = 11;
}

function CSL_BeginStrikeBuild()
{
    int cargo_per_enhanced_ship = max(1, min(10, CSL_ConfigValue(101, 2)));
    int cargo_remainder = csl_delivered_cargo - csl_delivered_cargo / cargo_per_enhanced_ship * cargo_per_enhanced_ship;
    csl_strike_pair_count = csl_delivered_cargo / cargo_per_enhanced_ship;
    csl_strike_odd_count = 0;
    if(cargo_remainder > 0 && CSL_ConfigValue(102, 1)) csl_strike_odd_count = 1;
    int escort_ship_multiplier = max(0, min(3, CSL_ConfigValue(103, 1)));
    int maximum_strike_ships = max(1, min(21, CSL_ConfigValue(104, 21)));
    csl_strike_planned_count = min(maximum_strike_ships, csl_strike_pair_count + csl_strike_odd_count + csl_delivered_escorts * escort_ship_multiplier);
    csl_strike_slots_started = 0;
    csl_strike_current_id = 0;
    csl_strike_batch_next_slot = 0;
    csl_strike_batch_end_slot = 0;
    csl_strike_ready_since = 0;
    csl_strike_launch_turn = 0;
    csl_strike_defense_active = 0;
    csl_strike_prepared_announced = 0;
    csl_strike_departed = 0;
    csl_strike_build_due = 0;
    csl_strike_rally_base_id = 0;
    csl_strike_rally_x = 0;
    csl_strike_rally_y = 0;
    csl_strike_rally_ready = 0;
    csl_strike_countdown_announced = 0;
    CSL_ClearPlayerJoin();
    csl_target_star = 0;
    csl_target_star_id = 0;
    csl_operation_announced = 0;
    csl_operation_deadline = CurTurn() + 300;
    csl_score = 0;
    csl_disqualified = 0;
    CSL_ResetStrikeIds();
    CSL_ResetVolunteerRangerIds();
    CSL_ResetScoredKillIds();
    dword strike_rally_base = StarRuins(PlanetToStar(csl_destination), 'WB');
    if(strike_rally_base) csl_strike_rally_base_id = Id(strike_rally_base);
    // Write the next phase and date before the first BuyWarrior. A re-entrant
    // world callback on this date therefore cannot duplicate the whole group.
    csl_phase = 13;
    csl_next_turn = CurTurn() + 1;
}

function CSL_CreateNextStrikeShip()
{
    result = 0;
    if(!csl_destination || csl_strike_slots_started >= csl_strike_planned_count) exit;
    int strike_slot = csl_strike_slots_started + 1;
    int simple_strike_slot = csl_strike_odd_count && strike_slot == csl_strike_pair_count + 1;
    csl_next_turn = CurTurn() + 1;
    dword ship = 0;
    if(simple_strike_slot) ship = BuyWarrior(csl_destination, 100);
    else ship = BuyWarrior(csl_destination, 1000);
    if(!ship) exit;
    // Register the freshly bought docked ship before any manual equipment
    // changes. This is the same safe lifecycle used by the vanilla military
    // operation scripts and prevents a half-created hull from being released
    // into space by an equipment failure callback.
    TransferShip(ship, csl_destination);
    OrderLock(ship, 1);
    csl_strike_current_id = Id(ship);
    csl_strike_ready_since = 0;
    int equipment_level = max(1, min(8, GalaxyTechLevel()));
    if(simple_strike_slot) equipment_level = max(1, equipment_level - 2);
    if(!simple_strike_slot)
    {
        ShipRefit(ship);
        dword item = ShipEqInSlot(ship, t_FuelTanks); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Engine); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Radar); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Scaner); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_RepairRobot); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Weapon1, 1); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Weapon1, 2); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Weapon1, 3); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Weapon1, 4); if(item) ItemLevel(item, equipment_level);
        item = ShipEqInSlot(ship, t_Weapon1, 5); if(item) ItemLevel(item, equipment_level);
    }
    dword strike_hull = ShipEqInSlot(ship, t_Hull);
    if(strike_hull)
    {
        int strike_def_bonus = ShipSpecialBonuses(ship, bonSlotDef);
        int strike_def_slots = GetEquipmentStats(strike_hull, 7) + strike_def_bonus;
        if(strike_def_slots < 1) ShipSpecialBonuses(ship, bonSlotDef, strike_def_bonus + 1 - strike_def_slots);
    }
    ShipCalcParam(ship);
    dword strike_shield = ShipEqInSlot(ship, t_DefGenerator);
    if(!strike_shield)
    {
        strike_shield = CreateEquipment(t_DefGenerator, 35 + 6 * equipment_level, equipment_level, ShipOwner(ship));
        if(strike_shield)
        {
            // Reserve cargo mass before activating the mandatory shield. The
            // previous order tried to activate it first and only enlarged an
            // overloaded hull afterwards.
            if(strike_hull)
            {
                int strike_shield_space_need = ItemSize(strike_shield) - ShipFreeSpace(ship);
                if(strike_shield_space_need > 0)
                {
                    ItemSize(strike_hull, ItemSize(strike_hull) + strike_shield_space_need + 1);
                    HullHP(ship, ItemSize(strike_hull));
                    ShipCalcParam(ship);
                }
            }
            AddItemToShip(ship, strike_shield);
            ItemIsInUse(strike_shield, ship, 1);
        }
    }
    ShipCalcParam(ship);
    strike_shield = ShipEqInSlot(ship, t_DefGenerator);
    if(!strike_shield)
    {
        // Leave the docked hull locked and identified. Phase 13 removes it on
        // the next date; same-handler ShipFreeFlight is unsafe for BuyWarrior.
        exit;
    }
    ItemLevel(strike_shield, equipment_level);
    if(strike_hull && ShipFreeSpace(ship) < 0)
    {
        ItemSize(strike_hull, ItemSize(strike_hull) + abs(ShipFreeSpace(ship)) + 1);
        HullHP(ship, ItemSize(strike_hull));
        ShipCalcParam(ship);
    }
    if(!simple_strike_slot)
    {
        int skill = RndObject(5, 6, ship);
        ShipSkill(ship, 0, skill); ShipSkill(ship, 1, skill); ShipSkill(ship, 2, skill);
        ShipSkill(ship, 3, skill); ShipSkill(ship, 4, skill); ShipSkill(ship, 5, skill);
        ShipImproveItems(ship, 2);
    }
    ShipCalcParam(ship);
    ShipRefuel(ship);
    ShipRepairEq(ship);
    if(!ShipEqInSlot(ship, t_DefGenerator))
    {
        // Do not let a strike ship without its mandatory shield take off.
        // Deferred cleanup is handled at the start of the next phase-13 date.
        exit;
    }
    ShipJoin(SupplyStrike, ship);
    CSL_SetStrikeId(strike_slot, Id(ship));
    csl_strike_slots_started = strike_slot;
    result = 1;
}

// Start one whole industrial batch on the same date.  The persistent next-day
// gate is written before the first BuyWarrior, so a nested world callback can
// observe only a closed batch and cannot start it twice.  Launched ships are
// then escorted to the common rally one by one; creation itself is simultaneous.
function CSL_CreateStrikeBatch()
{
    result = 0;
    if(!csl_destination || csl_strike_slots_started >= csl_strike_planned_count) exit;
    int configured_batch = max(1, min(4, CSL_ConfigBandValue(csl_composition_band, 7)));
    int batch_target = min(configured_batch, csl_strike_planned_count - csl_strike_slots_started);
    csl_strike_batch_next_slot = csl_strike_slots_started + 1;
    csl_strike_batch_end_slot = csl_strike_slots_started;
    csl_next_turn = CurTurn() + 1;
    for(int batch_create_index = 1; batch_create_index <= batch_target; batch_create_index = batch_create_index + 1)
    {
        csl_strike_current_id = 0;
        if(!CSL_CreateNextStrikeShip()) break;
        csl_strike_batch_end_slot = csl_strike_slots_started;
        csl_strike_current_id = 0;
        result = result + 1;
    }
    // A failed shield installation leaves its unregistered hull in current_id
    // for next-day destruction.  Earlier successful members stay in the batch.
    if(csl_strike_current_id > 1) exit;
    if(csl_strike_batch_end_slot >= csl_strike_batch_next_slot)
        csl_strike_current_id = CSL_GetStrikeId(csl_strike_batch_next_slot);
    else
    {
        csl_strike_batch_next_slot = 0;
        csl_strike_batch_end_slot = 0;
    }
}

function CSL_FinishCurrentStrikeBatchMember()
{
    int finished_strike_id = csl_strike_current_id;
    csl_strike_current_id = 0;
    csl_strike_ready_since = 0;
    if(csl_strike_batch_next_slot > 0 && csl_strike_batch_next_slot <= csl_strike_batch_end_slot)
    {
        if(CSL_GetStrikeId(csl_strike_batch_next_slot) == finished_strike_id)
            csl_strike_batch_next_slot = csl_strike_batch_next_slot + 1;
    }
    if(csl_strike_batch_next_slot > csl_strike_batch_end_slot)
    {
        csl_strike_batch_next_slot = 0;
        csl_strike_batch_end_slot = 0;
    }
}

function CSL_LoadNextStrikeBatchMember()
{
    result = 0;
    while(csl_strike_batch_next_slot > 0 && csl_strike_batch_next_slot <= csl_strike_batch_end_slot)
    {
        int queued_strike_id = CSL_GetStrikeId(csl_strike_batch_next_slot);
        if(queued_strike_id > 1)
        {
            dword queued_strike = IdToShip(queued_strike_id);
            if(queued_strike)
            {
                csl_strike_current_id = queued_strike_id;
                csl_strike_ready_since = 0;
                result = 1;
                exit;
            }
        }
        csl_strike_batch_next_slot = csl_strike_batch_next_slot + 1;
    }
    csl_strike_batch_next_slot = 0;
    csl_strike_batch_end_slot = 0;
}

function CSL_RewardPlayer()
{
    if(csl_disqualified)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.Disqualified'));
        exit;
    }
    if(csl_score <= 0) exit;
    int race = PlanetRace(csl_destination);
    int reward = GalaxyMoney(2, race) + csl_score * GalaxyMoney(0, race);
    reward = max(1, round(1.0 * reward * CSL_InflationFactor() / 100));
    ShipMoney(Player(), ShipMoney(Player()) + reward);
    int relation_gain = min(15, 3 + csl_score / 2);
    RelationToRanger(csl_destination, Player(), min(100, RelationToRanger(csl_destination, Player()) + relation_gain));
    RelationToRanger(csl_source, Player(), min(100, RelationToRanger(csl_source, Player()) + relation_gain / 2));
    AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Reward'), '#money#', reward, '#score#', csl_score));
}

function CSL_ProcessEvents()
{
    for(int i = GalaxyEvents() - 1; i >= 0; i = i - 1)
    {
        if(GalaxyEventDate(i) < CurTurn()) break;
        str event_type = GalaxyEventType(i);
        if(event_type == 'PlayerSellsEquipment')
        {
            int sold_cargo_id = GalaxyEventData(i, 3);
            int sold_registry_index = -1;
            for(int sold_search_index = ArrayDim(csl_cargo_registry) - 1; sold_search_index >= 1; sold_search_index = sold_search_index - 1)
            {
                if(csl_cargo_registry[sold_search_index] == sold_cargo_id)
                {
                    sold_registry_index = sold_search_index;
                    break;
                }
            }
            if(sold_registry_index >= 1)
            {
                int sold_is_current_context = CSL_IsCurrentCargoId(sold_cargo_id);
                int sold_owned_by_convoy_context = sold_is_current_context;
                // Galaxy events are global, while ProcessEvents runs once for
                // every parallel convoy. Do not let an unrelated context eat
                // the sale before the owning context can apply its violation.
                if(!sold_owned_by_convoy_context)
                {
                    for(int sold_owner_context = 1; sold_owner_context <= 3; sold_owner_context = sold_owner_context + 1)
                    {
                        for(int sold_owner_slot = 1; sold_owner_slot <= 30; sold_owner_slot = sold_owner_slot + 1)
                        {
                            int sold_owner_offset = (sold_owner_context - 1) * 32 + sold_owner_slot;
                            if(csl_context_active_cargo_ids[sold_owner_offset] == sold_cargo_id)
                            {
                                sold_owned_by_convoy_context = 1;
                                break;
                            }
                        }
                        if(sold_owned_by_convoy_context) break;
                    }
                }
                // Cargo from an older, already completed/destroyed convoy is a
                // legal find. Only a container belonging to the currently live
                // convoy may trigger the theft severity on sale.
                if(sold_is_current_context)
                {
                    if(csl_phase == 2 && (CSL_LiveTransportCount() || CSL_LiveEscortCount()))
                        CSL_ApplyViolation(70);
                    // A normal shop sale now remains a plain x1 sale. The
                    // special RC/PB reward belongs only to the explicit dialog.
                    CSL_ClearCargoIdValue(sold_cargo_id);
                    ArrayDelete(csl_cargo_registry, sold_registry_index);
                    ArrayDelete(csl_cargo_bases, sold_registry_index);
                }
                // Orphaned cargo has no live owning context. The strike context
                // runs last and performs the single shared-registry cleanup.
                else if(!sold_owned_by_convoy_context && csl_runtime_context_kind == 2)
                {
                    ArrayDelete(csl_cargo_registry, sold_registry_index);
                    ArrayDelete(csl_cargo_bases, sold_registry_index);
                }
            }
            continue;
        }
        int killed_id = GalaxyEventData(i, 2);
        int player_kill = event_type == 'PlayerKillsShip' || event_type == 'PlayerCompanionKillsShip' || event_type == 'PlayerTranclucatorKillsShip';
        int player_assist = event_type == 'PlayerAssistKillsShip' || event_type == 'PlayerCompanionAssistKillsShip' || event_type == 'PlayerTranclucatorAssistKillsShip';
        if(csl_phase == 2 && (player_kill || player_assist))
        {
            if(CSL_IsTransportId(killed_id)) CSL_ApplyViolation(100);
            else if(CSL_IsEscortId(killed_id)) CSL_ApplyViolation(70);
        }
        int strike_lifecycle_active = csl_phase == 13 || csl_phase == 14 || csl_phase == 15 || csl_phase == 4 || csl_phase == 5;
        if(strike_lifecycle_active && (player_kill || player_assist))
        {
            if(CSL_IsStrikeId(killed_id)) csl_disqualified = 1;
            else if(csl_phase == 4 && csl_operation_announced && csl_score < 20 && GalaxyEventData(i, 1) == csl_target_star_id &&
                    (GalaxyEventData(i, 0) == t_Kling || GalaxyEventData(i, 0) == t_Pirate))
            {
                int score_already_counted = 0;
                int score_store_index = 0;
                for(int score_event_index = 1; score_event_index <= 20; score_event_index = score_event_index + 1)
                {
                    if(csl_scored_kill_ids[score_event_index] == killed_id)
                    {
                        score_already_counted = 1;
                        break;
                    }
                    if(!score_store_index && !csl_scored_kill_ids[score_event_index]) score_store_index = score_event_index;
                }
                if(!score_already_counted && score_store_index)
                {
                    csl_scored_kill_ids[score_store_index] = killed_id;
                    if(player_kill) csl_score = min(20, csl_score + 2);
                    else csl_score = min(20, csl_score + 1);
                }
            }
        }
    }
}

function CSL_ProcessRangerQueue()
{
    if(ArrayDim(csl_rc_queue_due_turns) <= 1 || ArrayDim(csl_rc_queue_base_ids) != ArrayDim(csl_rc_queue_due_turns)) exit;
    int ranger_queue_has_ready_entry = 0;
    for(int ranger_queue_probe = ArrayDim(csl_rc_queue_due_turns) - 1; ranger_queue_probe >= 1; ranger_queue_probe = ranger_queue_probe - 1)
    {
        if(csl_rc_queue_base_ids[ranger_queue_probe] > 1 && csl_rc_queue_due_turns[ranger_queue_probe] <= CurTurn())
        {
            ranger_queue_has_ready_entry = 1;
            break;
        }
    }
    if(!ranger_queue_has_ready_entry) exit;
    dword fallback_ranger_factory_planet = 0;
    for(int ranger_factory_star_index = 0; ranger_factory_star_index < GalaxyStars(); ranger_factory_star_index = ranger_factory_star_index + 1)
    {
        fallback_ranger_factory_planet = CSL_FindOwnedPlanet(GalaxyStar(ranger_factory_star_index), 0);
        if(fallback_ranger_factory_planet) break;
    }
    if(!fallback_ranger_factory_planet) exit;
    // Process every entry that became ready this turn. Reverse traversal keeps
    // array deletion stable and lets 1, 3 or 9 containers sold together finish
    // on the same due date instead of leaking one extra day per container.
    for(int ranger_queue_index = ArrayDim(csl_rc_queue_due_turns) - 1; ranger_queue_index >= 1; ranger_queue_index = ranger_queue_index - 1)
    {
        if(csl_rc_queue_due_turns[ranger_queue_index] > CurTurn()) continue;
        if(csl_rc_queue_base_ids[ranger_queue_index] <= 1)
        {
            ArrayDelete(csl_rc_queue_base_ids, ranger_queue_index);
            ArrayDelete(csl_rc_queue_due_turns, ranger_queue_index);
            continue;
        }
        int ready_ranger_base_id = csl_rc_queue_base_ids[ranger_queue_index];
        dword ready_ranger_base = CSL_FindRangerCenterById(ready_ranger_base_id);
        if(!ready_ranger_base)
        {
            // The contracted Ranger Center was destroyed. Cancel this order
            // instead of retaining a ready entry and scanning the galaxy every
            // subsequent turn forever.
            ArrayDelete(csl_rc_queue_base_ids, ranger_queue_index);
            ArrayDelete(csl_rc_queue_due_turns, ranger_queue_index);
            continue;
        }
        dword ranger_factory_planet = CSL_FindOwnedPlanet(ShipStar(ready_ranger_base), 0);
        if(!ranger_factory_planet) ranger_factory_planet = fallback_ranger_factory_planet;
        dword released_ranger = BuyRanger(ranger_factory_planet, 100);
        if(!released_ranger) continue;
        TransferShip(released_ranger, ready_ranger_base);
        ArrayDelete(csl_rc_queue_base_ids, ranger_queue_index);
        ArrayDelete(csl_rc_queue_due_turns, ranger_queue_index);
    }
}

function CSL_InitializeConvoyContext(int context_slot, int delay_minimum, int delay_maximum)
{
    CSL_LoadContext(context_slot);
    csl_phase = 0;
    csl_route_index = 0;
    csl_replan_pending = 0;
    csl_procurement_mask = RndObject(1, 7, Player());
    csl_procurement_start_turn = CurTurn();
    csl_next_turn = max(301, CurTurn()) + RndObject(delay_minimum, delay_maximum, Player());
    csl_procurement_mid_turn = csl_procurement_start_turn + (csl_next_turn - csl_procurement_start_turn) / 2;
    csl_procurement_end_turn = csl_next_turn;
    csl_procurement_last_turn = CurTurn() - 1;
    CSL_RefreshStrategy();
    csl_source = CSL_FindSourceForPreparation(csl_free_systems);
    csl_source_id = 0;
    csl_procurement_active = 0;
    if(csl_source)
    {
        csl_source_id = Id(csl_source);
        csl_procurement_active = 1;
    }
    CSL_SaveContext(context_slot);
}

function CSL_StartQueuedStrikeOperation()
{
    if(csl_phase != -1 || ArrayDim(csl_strike_queue_destination_ids) <= 1) exit;
    if(ArrayDim(csl_strike_queue_delivered_cargo) != ArrayDim(csl_strike_queue_destination_ids) ||
       ArrayDim(csl_strike_queue_delivered_escorts) != ArrayDim(csl_strike_queue_destination_ids) ||
       ArrayDim(csl_strike_queue_composition_bands) != ArrayDim(csl_strike_queue_destination_ids) ||
       ArrayDim(csl_strike_queue_source_races) != ArrayDim(csl_strike_queue_destination_ids) ||
       ArrayDim(csl_strike_queue_build_due_turns) != ArrayDim(csl_strike_queue_destination_ids))
    {
        ArrayClear(csl_strike_queue_destination_ids);
        ArrayClear(csl_strike_queue_delivered_cargo);
        ArrayClear(csl_strike_queue_delivered_escorts);
        ArrayClear(csl_strike_queue_composition_bands);
        ArrayClear(csl_strike_queue_source_races);
        ArrayClear(csl_strike_queue_build_due_turns);
        exit;
    }
    csl_destination_id = csl_strike_queue_destination_ids[1];
    csl_delivered_cargo = csl_strike_queue_delivered_cargo[1];
    csl_delivered_escorts = csl_strike_queue_delivered_escorts[1];
    csl_composition_band = csl_strike_queue_composition_bands[1];
    csl_source_race = csl_strike_queue_source_races[1];
    csl_strike_build_due = csl_strike_queue_build_due_turns[1];
    csl_total_cargo = csl_delivered_cargo;
    csl_source_id = 0;
    csl_target_star_id = 0;
    csl_phase = 3;
    csl_next_turn = max(CurTurn() + 1, csl_strike_build_due);
    ArrayDelete(csl_strike_queue_destination_ids, 1);
    ArrayDelete(csl_strike_queue_delivered_cargo, 1);
    ArrayDelete(csl_strike_queue_delivered_escorts, 1);
    ArrayDelete(csl_strike_queue_composition_bands, 1);
    ArrayDelete(csl_strike_queue_source_races, 1);
    ArrayDelete(csl_strike_queue_build_due_turns, 1);
}

// World objects are restored from numeric IDs for this invocation only.
dword csl_source = 0;
dword csl_destination = 0;
dword csl_target_star = 0;
int csl_runtime_context_kind = 0;

// Persistent storage has its own schema gate. The gameplay/UI gate at the top
// already prevents this migration from running during galaxy generation, so
// normal initialization may safely continue in the same invocation.

    ArrayClear(csl_garrison_ship_ids);
    ArrayClear(csl_garrison_star_ids);
    ArrayClear(csl_garrison_until_turns);
    ArrayClear(csl_transport_cargo_counts);
    csl_ambush_checked_star_id = 0;
    csl_strike_batch_next_slot = 0;
    csl_strike_batch_end_slot = 0;
    csl_storage_version = 7;
}

if(csl_storage_version < 8)
{
    csl_context_state = newarray(401);
    csl_context_active_cargo_ids = newarray(128);
    csl_context_transport_cargo_counts = newarray(32);
    csl_context_pending_pirate_ids = newarray(36);
    csl_context_pending_pirate_origin_ids = newarray(36);
    csl_context_pending_pirate_origin_types = newarray(36);
    csl_context_pending_pirate_retreating = newarray(36);
    csl_context_volunteer_ranger_ids = newarray(32);
    csl_context_scored_kill_ids = newarray(88);
    csl_context_route_1 = newarray(256);
    csl_context_route_2 = newarray(256);
    csl_context_route_3 = newarray(256);
    csl_strike_queue_destination_ids = newarray(1);
    csl_strike_queue_delivered_cargo = newarray(1);
    csl_strike_queue_delivered_escorts = newarray(1);

        csl_context_state[context_state_init] = 0;
    for(int context_cargo_init = 0; context_cargo_init < 128; context_cargo_init = context_cargo_init + 1)
        csl_context_active_cargo_ids[context_cargo_init] = 0;
    for(int context_transport_cargo_init = 0; context_transport_cargo_init < 32; context_transport_cargo_init = context_transport_cargo_init + 1)
        csl_context_transport_cargo_counts[context_transport_cargo_init] = 0;
    for(int context_pirate_init = 0; context_pirate_init < 36; context_pirate_init = context_pirate_init + 1)
    {
        csl_context_pending_pirate_ids[context_pirate_init] = 0;
        csl_context_pending_pirate_origin_ids[context_pirate_init] = 0;
        csl_context_pending_pirate_origin_types[context_pirate_init] = 0;
        csl_context_pending_pirate_retreating[context_pirate_init] = 0;
    }
    for(int context_volunteer_init = 0; context_volunteer_init < 32; context_volunteer_init = context_volunteer_init + 1)
        csl_context_volunteer_ranger_ids[context_volunteer_init] = 0;

        csl_context_route_1[context_route_init] = 0;

        csl_context_route_3[context_route_init] = 0;
    }
    csl_strategy_refresh_turn = 0;
    csl_visible_context = 1;
    csl_initialized = 0;
    csl_storage_version = 8;
}

if(!csl_initialized)

    CSL_InitializeConvoyContext(1, 120, 200);
    CSL_InitializeConvoyContext(2, 180, 300);
    CSL_InitializeConvoyContext(3, 240, 400);
    CSL_LoadContext(4);

    csl_initialized = 1;

}

// The galaxy-wide strategic scan and source search run outside the large
// phase machine. Three concurrent contexts share one scan per game date, but
// each context still selects and stores its own industrial source immediately.
function CSL_PrepareConvoyStrategy()
{
    CSL_RefreshStrategy();
    if(csl_phase != 0 || csl_hostile_systems <= 0) exit;
    int prepared_source_valid = 0;
    if(csl_source)
    {
        dword prepared_source_star = PlanetToStar(csl_source);
        if(CSL_IsFreeStar(prepared_source_star) && PlanetEco(csl_source) == 2) prepared_source_valid = 1;
    }
    if(!prepared_source_valid) csl_source = CSL_FindSourceForPreparation(csl_free_systems);
    csl_source_id = 0;
    csl_procurement_active = 0;
    if(csl_source)
    {
        csl_source_id = Id(csl_source);
        if(csl_next_turn > CurTurn()) csl_procurement_active = 1;
    }
}

// A route rejected by the real first transport is cleaned up one script ship
// per turn, then the scheduler retries normally after 30 turns.
function CSL_ProcessCurrentContext(int csl_ambush_resume_route, int context_kind)
{
if(context_kind == 1 && csl_phase != 0 && csl_phase != 1 && csl_phase != 2 && csl_phase != 10 &&
   csl_phase != 11 && csl_phase != 12 && csl_phase != 16 && csl_phase != 17) exit;
if(context_kind == 2 && csl_phase != 3 && csl_phase != 4 && csl_phase != 5 &&
   csl_phase != 13 && csl_phase != 14 && csl_phase != 15) exit;
if(csl_phase == 10)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    if(CSL_LiveEscortCount())
    {
        dword failed_escort = 0;
        for(int failed_escort_index = GroupCount(SupplyEscorts) - 1; failed_escort_index >= 0; failed_escort_index = failed_escort_index - 1)
        {
            dword failed_escort_probe = GroupShip(SupplyEscorts, failed_escort_index);
            if(!failed_escort_probe) continue;
            if(CSL_IsEscortId(Id(failed_escort_probe))) { failed_escort = failed_escort_probe; break; }
        }
        if(!failed_escort) exit;
        OrderLock(failed_escort, 0);
        ShipDestroy(failed_escort);
        exit;
    }
    if(CSL_LiveTransportCount())
    {
        dword failed_transport = 0;
        for(int failed_transport_index = GroupCount(SupplyTransports) - 1; failed_transport_index >= 0; failed_transport_index = failed_transport_index - 1)
        {
            dword failed_transport_probe = GroupShip(SupplyTransports, failed_transport_index);
            if(!failed_transport_probe) continue;
            if(CSL_IsTransportId(Id(failed_transport_probe))) { failed_transport = failed_transport_probe; break; }
        }
        if(!failed_transport) exit;
        CSL_ProtectCargoOnShip(failed_transport);
        OrderLock(failed_transport, 0);
        ShipDestroy(failed_transport);
        exit;
    }
    for(int failed_cargo_index = 1; failed_cargo_index <= 30; failed_cargo_index = failed_cargo_index + 1)
        CSL_ReleaseCargoSlot(failed_cargo_index);
    csl_phase = 0;
    csl_next_turn = CurTurn() + 30;
    exit;
}

if(csl_phase == 0 && CurTurn() >= csl_next_turn)
{
    if(CSL_PendingPirateCount() > 0)
    {
        csl_ambush_target_id = 0;
        csl_next_turn = CurTurn() + 1;
        exit;
    }
    csl_procurement_active = 0;
    if(csl_hostile_systems <= 0)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoTargets'));
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    csl_destination = 0;
    csl_target_star = 0;
    if(!csl_source)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    csl_source_id = Id(csl_source);
    csl_destination_id = 0;
    csl_target_star_id = 0;
    csl_preparation_band = CSL_StrategyBand();
    int build_difficulty = max(50, min(500, round(0.5 * (GalaxyDiffLevels(2) + GalaxyDiffLevels(4)))));
    int build_turns = max(1, round(1.0 * RndObject(30, 60, csl_source) * build_difficulty / 100));
    csl_next_turn = CurTurn() + build_turns;
    csl_phase = 1;
    AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Preparing'), '#source#', Name(csl_source), '#turns#', build_turns));
    exit;
}

if(csl_phase == 1 && CurTurn() >= csl_next_turn)
{
    if(csl_hostile_systems <= 0)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoTargets'));
        csl_phase = 0;
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    if(CSL_StrategyBand() != csl_preparation_band)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 0;
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    if(!CSL_IsFreeStar(PlanetToStar(csl_source)) || PlanetEco(csl_source) != 2)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 0;
        csl_next_turn = CurTurn() + 30;
        exit;
    }
    CSL_SpawnConvoy();
    exit;
}

if(csl_phase == 11)
{
    if(CurTurn() < csl_next_turn) exit;
    if(csl_hostile_systems <= 0)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoTargets'));
        csl_phase = 10;
        exit;
    }
    // Nothing has taken off yet. A changed strategic bracket therefore
    // invalidates the assembled composition/source and is rebuilt later.
    if(!CSL_CompositionMatchesSystems(csl_free_systems))
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 10;
        exit;
    }
    if(CSL_LiveTransportCount() != csl_transport_count || CSL_LiveEscortCount() != csl_escort_count)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 10;
        exit;
    }
    for(int launch_transport_index = 0; launch_transport_index < GroupCount(SupplyTransports); launch_transport_index = launch_transport_index + 1)
    {
        dword launch_transport = GroupShip(SupplyTransports, launch_transport_index);
        if(!launch_transport) continue;
        if(CSL_IsTransportId(Id(launch_transport)) && GetShipPlanet(launch_transport) == csl_source) OrderTakeOff(launch_transport);
    }
    for(int launch_escort_index = 0; launch_escort_index < GroupCount(SupplyEscorts); launch_escort_index = launch_escort_index + 1)
    {
        dword launch_escort = GroupShip(SupplyEscorts, launch_escort_index);
        if(!launch_escort) continue;
        if(CSL_IsEscortId(Id(launch_escort)) && GetShipPlanet(launch_escort) == csl_source) OrderTakeOff(launch_escort);
    }
    AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Launched'), '#source#', Name(csl_source), '#transports#', csl_transport_count, '#escorts#', csl_escort_count, '#destination#', Name(csl_destination)));
    csl_phase = 2;
    exit;
}

if(csl_phase == 2)
{
    // A pending hyperjump is immutable. In particular, never let regrouping
    // call OrderNone for a ship that has not emerged yet.
    if(CSL_LiveConvoyInHyperSpace()) exit;
    // Loss/completion is resolved before threat and route code. Otherwise an
    // escort without a transport leader can lose its attacker or loop in
    // rerouting forever.
    if(!CSL_LiveTransportCount())
    {
        for(int lost_escort_index = 0; lost_escort_index < GroupCount(SupplyEscorts); lost_escort_index = lost_escort_index + 1)
        {
            dword lost_escort = GroupShip(SupplyEscorts, lost_escort_index);
            if(!lost_escort) continue;
            if(!CSL_IsEscortId(Id(lost_escort))) continue;
            if(ShipGetBad(lost_escort) == Player()) CSL_ApplyViolation(40);
        }
        for(int lost_cargo_index = 1; lost_cargo_index <= 30; lost_cargo_index = lost_cargo_index + 1)
            if(CSL_CargoIn(lost_cargo_index, Player())) CSL_ApplyViolation(70);
        csl_phase = 12;
        csl_next_turn = CurTurn() + 1;
        if(csl_delivered_cargo > 0 && csl_strike_build_due <= 0)
            csl_strike_build_due = CurTurn() + RndObject(15, 30, csl_destination);
        exit;
    }
    if(csl_ambush_resume_route)
    {
        // Value 2 means that the ambush is over, but one or more freighters
        // may still be completing their landing approach.
        csl_replan_pending = 2;
        csl_next_turn = CurTurn() + 1;
    }
    if(csl_replan_pending == 2)
    {
        // Keep the surviving guard moving with the freighters while the last
        // landing approaches finish. Previously this early branch prevented
        // the normal regroup code from ever issuing the return order.
        dword ambush_return_leader = CSL_ActiveTransportLeader();
        if(ambush_return_leader && CSL_LiveEscortCount() > 0 && !CSL_EscortsRejoined(ambush_return_leader))
            CSL_RallyEscorts(ambush_return_leader);
        CSL_ResumeConvoyAfterAmbush();
        exit;
    }
    dword active_convoy_threat = CSL_ProcessConvoyThreats();
    dword active_convoy_leader = CSL_ActiveTransportLeader();
    if(active_convoy_leader)
    {
        if(ShipInNormalSpace(active_convoy_leader) && !ShipIsTakeoff(active_convoy_leader))
        {
            dword active_ambush_star = CSL_ShipCurrentStar(active_convoy_leader);
            if(active_ambush_star)
            {
                int ambush_away_from_source = 1;
                if(csl_source)
                {
                    dword ambush_source_star = PlanetToStar(csl_source);
                    if(active_ambush_star == ambush_source_star) ambush_away_from_source = 0;
                }
                if(ambush_away_from_source) CSL_CheckConvoyAmbush(active_ambush_star, active_convoy_leader);
            }
        }
    }
    if(active_convoy_threat)
    {
        dword threatened_convoy_leader = CSL_FindNearestTransportToShip(active_convoy_threat);
        if(threatened_convoy_leader) active_convoy_leader = threatened_convoy_leader;
    }
    int active_convoy_escorts = CSL_LiveEscortCount();
    int active_convoy_breakout = csl_ambush_target_id > 1 && csl_ambush_takeoff_turn == -1 && active_convoy_escorts <= 0;
    int active_convoy_at_destination = 0;
    if(active_convoy_leader && csl_destination && ArrayDim(csl_route_planets) > 1 &&
       csl_route_index == ArrayDim(csl_route_planets) - 1)
    {
        dword active_destination_star = PlanetToStar(csl_destination);
        if(active_destination_star && CSL_ShipCurrentStar(active_convoy_leader) == active_destination_star)
            active_convoy_at_destination = 1;
    }
    if(active_convoy_at_destination && (active_convoy_threat || csl_ambush_target_id > 1))
    {
        // The cargo has reached its final system.  Everyone lands even while
        // the raiders remain in space; the delivery code below must continue
        // running, so this branch deliberately does not exit.
        CSL_LandActiveConvoyAtDestination();
    }
    if(!active_convoy_at_destination && active_convoy_threat && active_convoy_leader && !active_convoy_breakout)
    {
        CSL_EvacuateActiveTransports(active_convoy_leader);
        if(active_convoy_escorts > 0)
            CSL_EngageConvoyThreat(active_convoy_threat, active_convoy_leader);
        exit;
    }
    if(!active_convoy_at_destination && csl_ambush_target_id > 1 && active_convoy_leader && !active_convoy_breakout)
    {
        // The ambushers leave their planet or base on the following turn.
        // Start moving the freighters to shelter immediately and leave the
        // escorts in space to meet the attack.
        CSL_EvacuateActiveTransports(active_convoy_leader);
        exit;
    }
    if(!active_convoy_at_destination && active_convoy_leader && !CSL_TransportsRejoined(active_convoy_leader))
    {
        // Hold only the stable lead freighter. Every trailing transport keeps
        // a follow, jump or landing order until it rejoins the formation.
        CSL_RegroupTransports(active_convoy_leader);
        if(active_convoy_escorts > 0) CSL_RallyEscorts(active_convoy_leader);
        exit;
    }
    if(!active_convoy_at_destination && active_convoy_leader && active_convoy_escorts > 0 && !CSL_EscortsRejoined(active_convoy_leader))
    {
        CSL_HoldActiveTransports();
        CSL_RallyEscorts(active_convoy_leader);
        exit;
    }
    if(CurTurn() < csl_next_turn) exit;
    // Remove only the hull unloaded by this context. Other parallel convoys
    // share SupplyTransports but own different ID tables.
    if(csl_pending_transport_cleanup_id > 1)
    {
        dword early_cleanup_ship = IdToShip(csl_pending_transport_cleanup_id);
        csl_pending_transport_cleanup_id = 0;
        if(early_cleanup_ship)
        {
            CSL_ProtectCargoOnShip(early_cleanup_ship);
            OrderLock(early_cleanup_ship, 0);
            ShipDestroy(early_cleanup_ship);
        }
        csl_next_turn = CurTurn() + 1;
        exit;
    }
    if(ArrayDim(csl_route_planets) <= 1 || csl_route_index < 1 || csl_route_index >= ArrayDim(csl_route_planets))
    {
        if(!CSL_TryImmediateReroute())
        {
            AddPlanetNews(CT('XenoCoalitionSupplyLines.RerouteWaiting'));
            csl_next_turn = CurTurn() + 5;
        }
        exit;
    }
    dword active_route_planet = IdToPlanet(csl_route_planets[csl_route_index]);
    if(!active_route_planet)
    {
        if(!CSL_TryImmediateReroute())
        {
            AddPlanetNews(CT('XenoCoalitionSupplyLines.RerouteWaiting'));
            csl_next_turn = CurTurn() + 5;
        }
        exit;
    }
    int active_route_is_final = csl_route_index == ArrayDim(csl_route_planets) - 1;
    if(!CSL_RoutePlanetSafe(active_route_planet) ||
       (active_route_is_final && !csl_replan_pending && !csl_crisis_mode && !CSL_IsFrontier(PlanetToStar(active_route_planet))))
    {
        if(!CSL_TryImmediateReroute())
        {
            AddPlanetNews(CT('XenoCoalitionSupplyLines.RerouteWaiting'));
            csl_next_turn = CurTurn() + 5;
        }
        exit;
    }
    int active_route_transport_landed = 0;
    int active_route_escort_landed = 0;
    // Send transports first once the whole formation is already in the target
    // system. The engine otherwise considers Follow complete beside the still
    // stationary leader and does not wake the guard when that leader receives
    // its landing order later in the same turn.
    dword route_dispatch_leader = CSL_ActiveTransportLeader();
    dword route_delivery_candidate = 0;
    for(int convoy_transport_cursor = 0; convoy_transport_cursor < GroupCount(SupplyTransports); convoy_transport_cursor = convoy_transport_cursor + 1)
    {
        dword convoy_transport_ship = GroupShip(SupplyTransports, convoy_transport_cursor);
        if(!convoy_transport_ship) continue;
        if(!CSL_IsTransportId(Id(convoy_transport_ship))) continue;
        if(CSL_SendShip(convoy_transport_ship, active_route_planet))
        {
            if(active_route_is_final && !csl_replan_pending && !csl_crisis_mode)
            {
                if(!route_delivery_candidate) route_delivery_candidate = convoy_transport_ship;
                continue;
            }
            active_route_transport_landed = active_route_transport_landed + 1;
        }
    }
    for(int route_dispatch_guard_index = 0; route_dispatch_guard_index < GroupCount(SupplyEscorts); route_dispatch_guard_index = route_dispatch_guard_index + 1)
    {
        dword route_dispatch_guard = GroupShip(SupplyEscorts, route_dispatch_guard_index);
        if(!route_dispatch_guard) continue;
        if(!CSL_IsEscortId(Id(route_dispatch_guard))) continue;
        int route_dispatch_can_follow = 0;
        if(route_dispatch_leader)
        {
            if(ShipInNormalSpace(route_dispatch_guard) && ShipInNormalSpace(route_dispatch_leader) &&
               CSL_ShipCurrentStar(route_dispatch_leader) == PlanetToStar(active_route_planet) &&
               CSL_ShipCurrentStar(route_dispatch_guard) == CSL_ShipCurrentStar(route_dispatch_leader))
                route_dispatch_can_follow = 1;
        }
        if(route_dispatch_can_follow) CSL_Follow(route_dispatch_guard, route_dispatch_leader);
        else CSL_SendShip(route_dispatch_guard, active_route_planet);
    }
    if(route_delivery_candidate)
    {
        CSL_DeliverTransport(route_delivery_candidate);
        exit;
    }
    for(int active_route_escort_count_index = 0; active_route_escort_count_index < GroupCount(SupplyEscorts); active_route_escort_count_index = active_route_escort_count_index + 1)
    {
        dword active_route_escort_count_ship = GroupShip(SupplyEscorts, active_route_escort_count_index);
        if(!active_route_escort_count_ship) continue;
        if(!CSL_IsEscortId(Id(active_route_escort_count_ship))) continue;
        if(GetShipPlanet(active_route_escort_count_ship) == active_route_planet)
            active_route_escort_landed = active_route_escort_landed + 1;
    }
    int active_route_transport_count = CSL_LiveTransportCount();
    int active_route_escort_count = CSL_LiveEscortCount();
    if(csl_crisis_mode && active_route_is_final && active_route_transport_landed == active_route_transport_count)
    {
        for(int crisis_arrival_transport_index = 0; crisis_arrival_transport_index < GroupCount(SupplyTransports); crisis_arrival_transport_index = crisis_arrival_transport_index + 1)
        {
            dword crisis_arrival_transport = GroupShip(SupplyTransports, crisis_arrival_transport_index);
            if(!crisis_arrival_transport) continue;
            if(CSL_IsTransportId(Id(crisis_arrival_transport)))
            {
                for(int crisis_count_cargo_index = 1; crisis_count_cargo_index <= 30; crisis_count_cargo_index = crisis_count_cargo_index + 1)
                {
                    dword crisis_count_cargo = CSL_GetCargo(crisis_count_cargo_index);
                    if(crisis_count_cargo && ItemIn(crisis_count_cargo, crisis_arrival_transport))
                    {
                        csl_delivered_cargo = csl_delivered_cargo + 1;
                        NoDropItem(crisis_count_cargo, 1);
                    }
                }
            }
        }
        csl_crisis_trips_completed = csl_crisis_trips_completed + 1;
        csl_phase = 16;
        csl_next_turn = CurTurn() + 1;
        exit;
    }
    if(active_route_is_final && csl_replan_pending && active_route_transport_count > 0 &&
       active_route_transport_landed == active_route_transport_count && active_route_escort_landed == active_route_escort_count)
    {
        for(int regroup_transport_index = 0; regroup_transport_index < GroupCount(SupplyTransports); regroup_transport_index = regroup_transport_index + 1)
        {
            dword regroup_transport = GroupShip(SupplyTransports, regroup_transport_index);
            if(!regroup_transport) continue;
            if(CSL_IsTransportId(Id(regroup_transport))) { ShipRefuel(regroup_transport); ShipRepairEq(regroup_transport); }
        }
        for(int regroup_escort_index = 0; regroup_escort_index < GroupCount(SupplyEscorts); regroup_escort_index = regroup_escort_index + 1)
        {
            dword regroup_escort = GroupShip(SupplyEscorts, regroup_escort_index);
            if(!regroup_escort) continue;
            if(CSL_IsEscortId(Id(regroup_escort))) { ShipRefuel(regroup_escort); ShipRepairEq(regroup_escort); }
        }
        // The regroup point itself may already be a valid frontier destination.
        // In that case unload here instead of rebuilding a redundant route.
        if(CSL_IsFrontier(PlanetToStar(active_route_planet)))
        {
            CSL_CommitReroute(active_route_planet, 0);
            csl_delivered_cargo = 0;
            csl_arrived_transports = 0;
            AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.RerouteDirect'), '#destination#', Name(active_route_planet)));
            exit;
        }
        dword regroup_leader = CSL_ActiveTransportLeader();
        if(regroup_leader && CSL_BuildConvoyRoute(regroup_leader, PlanetToStar(active_route_planet), 1))
        {
            csl_replan_pending = 0;
            csl_next_turn = CurTurn() + 1;
            AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.RerouteComplete'), '#destination#', Name(csl_destination)));
        }
        else
        {
            CSL_CommitReroute(active_route_planet, 1);
            csl_next_turn = CurTurn() + 5;
            AddPlanetNews(CT('XenoCoalitionSupplyLines.RerouteWaiting'));
        }
        exit;
    }
    if(!active_route_is_final && active_route_transport_count > 0 &&
       active_route_transport_landed == active_route_transport_count && active_route_escort_landed == active_route_escort_count)
    {
        dword next_route_planet = IdToPlanet(csl_route_planets[csl_route_index + 1]);
        if(!CSL_RoutePlanetSafe(next_route_planet))
        {
            if(CSL_IsFrontier(PlanetToStar(active_route_planet)))
            {
                CSL_CommitReroute(active_route_planet, 0);
                csl_delivered_cargo = 0;
                csl_arrived_transports = 0;
                AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.RerouteDirect'), '#destination#', Name(active_route_planet)));
            }
            else
            {
                CSL_CommitReroute(active_route_planet, 1);
                AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.RerouteRegroup'), '#destination#', Name(active_route_planet)));
            }
            exit;
        }
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Refuelled'), '#planet#', Name(active_route_planet), '#next#', Name(next_route_planet)));
        for(int active_route_refuel_transport_index = 0; active_route_refuel_transport_index < GroupCount(SupplyTransports); active_route_refuel_transport_index = active_route_refuel_transport_index + 1)
        {
            dword active_route_refuel_transport = GroupShip(SupplyTransports, active_route_refuel_transport_index);
            if(!active_route_refuel_transport) continue;
            if(!CSL_IsTransportId(Id(active_route_refuel_transport))) continue;
            ShipRefuel(active_route_refuel_transport);
            ShipRepairEq(active_route_refuel_transport);
            OrderTakeOff(active_route_refuel_transport);
        }
        for(int active_route_refuel_escort_index = 0; active_route_refuel_escort_index < GroupCount(SupplyEscorts); active_route_refuel_escort_index = active_route_refuel_escort_index + 1)
        {
            dword active_route_refuel_escort = GroupShip(SupplyEscorts, active_route_refuel_escort_index);
            if(!active_route_refuel_escort) continue;
            if(!CSL_IsEscortId(Id(active_route_refuel_escort))) continue;
            ShipRefuel(active_route_refuel_escort);
            ShipRepairEq(active_route_refuel_escort);
            OrderTakeOff(active_route_refuel_escort);
        }
        csl_route_index = csl_route_index + 1;
        exit;
    }
    if(active_route_transport_count > 0 || active_route_escort_count > 0)
    {
        for(int theft_cursor = 1; theft_cursor <= 30; theft_cursor = theft_cursor + 1)
        {
            if(CSL_CargoIn(theft_cursor, Player())) CSL_ApplyViolation(70);
        }
    }
    exit;
}

if(csl_phase == 16)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    if(csl_crisis_trips_completed >= 3)
    {
        if(CSL_LiveTransportCount())
        {
            dword crisis_final_transport = 0;
            for(int crisis_final_index = GroupCount(SupplyTransports) - 1; crisis_final_index >= 0; crisis_final_index = crisis_final_index - 1)
            {
                dword crisis_final_probe = GroupShip(SupplyTransports, crisis_final_index);
                if(!crisis_final_probe) continue;
                if(CSL_IsTransportId(Id(crisis_final_probe))) { crisis_final_transport = crisis_final_probe; break; }
            }
            if(!crisis_final_transport) exit;
            int crisis_final_transport_id = Id(crisis_final_transport);
            CSL_ClearTransportIdValue(crisis_final_transport_id);
            CSL_ProtectCargoOnShip(crisis_final_transport);
            OrderLock(crisis_final_transport, 0);
            ShipDestroy(crisis_final_transport);
            exit;
        }
        csl_strike_build_due = CurTurn() + RndObject(15, 30, csl_destination);
        csl_phase = 12;
        exit;
    }
    dword crisis_docked_transport = 0;
    for(int crisis_takeoff_index = 0; crisis_takeoff_index < GroupCount(SupplyTransports); crisis_takeoff_index = crisis_takeoff_index + 1)
    {
        dword crisis_takeoff_transport = GroupShip(SupplyTransports, crisis_takeoff_index);
        if(!crisis_takeoff_transport) continue;
        if(!CSL_IsTransportId(Id(crisis_takeoff_transport))) continue;
        if(GetShipPlanet(crisis_takeoff_transport) == csl_destination)
        {
            crisis_docked_transport = crisis_takeoff_transport;
            break;
        }
        if(ShipIsTakeoff(crisis_takeoff_transport) || ShipInHyperSpace(crisis_takeoff_transport, 1) || !ShipInNormalSpace(crisis_takeoff_transport)) exit;
    }
    if(crisis_docked_transport)
    {
        ShipRefuel(crisis_docked_transport);
        ShipRepairEq(crisis_docked_transport);
        CSL_TakeOff(crisis_docked_transport);
        exit;
    }
    if(csl_crisis_pending_free_item_id <= 1)
    {
        for(int crisis_remove_transport_index = 0; crisis_remove_transport_index < GroupCount(SupplyTransports); crisis_remove_transport_index = crisis_remove_transport_index + 1)
        {
            dword crisis_remove_transport = GroupShip(SupplyTransports, crisis_remove_transport_index);
            if(!crisis_remove_transport) continue;
            if(!CSL_IsTransportId(Id(crisis_remove_transport))) continue;
            if(!ShipInNormalSpace(crisis_remove_transport) || ShipIsTakeoff(crisis_remove_transport)) continue;
            for(int crisis_remove_cargo_index = 1; crisis_remove_cargo_index <= 30; crisis_remove_cargo_index = crisis_remove_cargo_index + 1)
            {
                dword crisis_remove_cargo = CSL_GetCargo(crisis_remove_cargo_index);
                if(!crisis_remove_cargo || !ItemIn(crisis_remove_cargo, crisis_remove_transport)) continue;
                dword crisis_removed_cargo = GetItemFromShip(crisis_remove_transport, crisis_remove_cargo);
                if(crisis_removed_cargo)
                {
                    if(ItemInScript(crisis_removed_cargo)) ReleaseItemFromScript(crisis_removed_cargo);
                    csl_crisis_pending_free_item_id = Id(crisis_removed_cargo);
                    CSL_SetCargoId(crisis_remove_cargo_index, 0);
                    ShipCalcParam(crisis_remove_transport);
                    exit;
                }
            }
        }
    }
    csl_phase = 17;
    exit;
}

if(csl_phase == 17)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    int crisis_source_landed = 0;
    for(int crisis_return_index = 0; crisis_return_index < GroupCount(SupplyTransports); crisis_return_index = crisis_return_index + 1)
    {
        dword crisis_return_transport = GroupShip(SupplyTransports, crisis_return_index);
        if(!crisis_return_transport) continue;
        if(!CSL_IsTransportId(Id(crisis_return_transport))) continue;
        if(CSL_SendShip(crisis_return_transport, csl_source)) crisis_source_landed = crisis_source_landed + 1;
    }
    if(crisis_source_landed < CSL_LiveTransportCount()) exit;
    if(!CSL_ReloadCrisisCargo())
    {
        csl_phase = 10;
        exit;
    }
    ArrayClear(csl_route_planets);
    ArrayAdd(csl_route_planets, csl_destination_id);
    csl_route_index = 1;
    for(int crisis_relaunch_index = 0; crisis_relaunch_index < GroupCount(SupplyTransports); crisis_relaunch_index = crisis_relaunch_index + 1)
    {
        dword crisis_relaunch_transport = GroupShip(SupplyTransports, crisis_relaunch_index);
        if(!crisis_relaunch_transport) continue;
        if(CSL_IsTransportId(Id(crisis_relaunch_transport))) CSL_TakeOff(crisis_relaunch_transport);
    }
    csl_phase = 2;
    exit;
}

// Finish a convoy one world mutation per turn. Successful escorts disappear
// while docked, without the old visible takeoff-and-explosion sequence. If all
// transports were lost, escorts keep their current enemy and return to vanilla
// military AI instead of being destroyed.
if(csl_phase == 12)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    // Transports are already gone at this point. Releasing all surviving
    // script links is safe and lets the promised preparation timer run in
    // parallel with one-per-turn escort cleanup.
    for(int finished_cargo_index = 1; finished_cargo_index <= 30; finished_cargo_index = finished_cargo_index + 1)
        CSL_ReleaseCargoSlot(finished_cargo_index);
    if(CSL_LiveEscortCount())
    {
        int finished_last_escort = CSL_LiveEscortCount() == 1;
        dword finished_escort = 0;
        for(int finished_escort_index = GroupCount(SupplyEscorts) - 1; finished_escort_index >= 0; finished_escort_index = finished_escort_index - 1)
        {
            dword finished_escort_probe = GroupShip(SupplyEscorts, finished_escort_index);
            if(!finished_escort_probe) continue;
            if(CSL_IsEscortId(Id(finished_escort_probe))) { finished_escort = finished_escort_probe; break; }
        }
        if(!finished_escort) exit;
        if(ShipInHyperSpace(finished_escort)) exit;
        dword finished_escort_place = GetShipPlanet(finished_escort);
        if(!finished_escort_place) finished_escort_place = GetShipRuins(finished_escort);
        int finished_escort_id = Id(finished_escort);
        if(csl_delivered_cargo > 0)
        {
            if(finished_escort_place)
            {
                if(finished_escort_place == csl_destination) csl_delivered_escorts = csl_delivered_escorts + 1;
                CSL_ClearEscortIdValue(finished_escort_id);
                if(finished_last_escort) CSL_FinishConvoyCleanup();
                OrderLock(finished_escort, 0);
                ShipDestroy(finished_escort);
                exit;
            }
            int finished_escort_at_destination = 0;
            if(csl_destination)
            {
                dword finished_destination_star = PlanetToStar(csl_destination);
                if(CSL_ShipCurrentStar(finished_escort) == finished_destination_star && CSL_PlanetEligible(csl_destination, 0))
                    finished_escort_at_destination = 1;
            }
            if(finished_escort_at_destination)
            {
                CSL_Land(finished_escort, csl_destination);
                exit;
            }
        }
        else
        {
            if(finished_escort_place)
            {
                ShipRefuel(finished_escort);
                ShipRepairEq(finished_escort);
                CSL_TakeOff(finished_escort);
                exit;
            }
        }
        CSL_ClearEscortIdValue(finished_escort_id);
        csl_next_turn = CurTurn() + 1;
        if(finished_last_escort) CSL_FinishConvoyCleanup();
        ShipOut(finished_escort);
        OrderLock(finished_escort, 0);
        ShipFreeFlight(finished_escort);
        exit;
    }
    CSL_FinishConvoyCleanup();
    exit;
}

if(csl_phase == 3 && CurTurn() >= csl_next_turn)
{
    if(!csl_destination)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 5;
        csl_next_turn = CurTurn() + 1;
        exit;
    }
    if(StarOwner(PlanetToStar(csl_destination)) != 0 || !CSL_PlanetEligible(csl_destination, 0))
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
        csl_phase = 5;
        csl_next_turn = CurTurn() + 1;
        exit;
    }
    CSL_BeginStrikeBuild();
    exit;
}

// A strategic band starts up to four hulls on one production date.  Members
// of that closed batch are then launched and placed at the rally sequentially,
// which keeps every unsafe takeoff transition outside the BuyWarrior loop.
if(csl_phase == 13)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    if(CurTurn() >= csl_operation_deadline)
    {
        if(csl_destination) AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
        else AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 5;
        exit;
    }
    if(!csl_destination) { csl_phase = 5; exit; }
    if(StarOwner(PlanetToStar(csl_destination)) != 0 || !CSL_PlanetEligible(csl_destination, 0))
    {
        if(GroupCount(SupplyStrike)) CSL_DefendStrikeAssembly();
        if(CurTurn() >= csl_operation_deadline)
        {
            AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
            csl_phase = 5;
        }
        exit;
    }
    if(GroupCount(SupplyStrike) && CSL_DefendStrikeAssembly())
    {
        csl_strike_defense_active = 1;
        exit;
    }
    if(csl_strike_defense_active)
    {
        if(CSL_ReturnStrikeToAssemblyAll()) exit;
        csl_strike_defense_active = 0;
    }
    if(csl_strike_current_id <= 1 && csl_strike_batch_next_slot > 0)
    {
        CSL_LoadNextStrikeBatchMember();
        if(csl_strike_current_id > 1) exit;
    }
    if(csl_strike_current_id > 1)
    {
        dword current_strike = IdToShip(csl_strike_current_id);
        if(!current_strike)
        {
            CSL_FinishCurrentStrikeBatchMember();
            exit;
        }
        // A failed mandatory-shield installation is cleaned up one date after
        // BuyWarrior, never from the creation callback itself. No slot was
        // counted and the following date may safely retry the same slot.
        if(!ShipEqInSlot(current_strike, t_DefGenerator))
        {
            if(ShipInHyperSpace(current_strike, 1) || ShipIsTakeoff(current_strike)) exit;
            OrderLock(current_strike, 0);
            ShipDestroy(current_strike);
            csl_strike_current_id = 0;
            csl_strike_ready_since = 0;
            csl_next_turn = CurTurn() + 1;
            exit;
        }
        if(ShipInHyperSpace(current_strike, 1) || ShipIsTakeoff(current_strike)) exit;
        dword current_rally_base = CSL_FindStrikeRallyBase();
        if(current_rally_base && GetShipRuins(current_strike) == current_rally_base)
        {
            ShipRefuel(current_strike);
            ShipRepairEq(current_strike);
            CSL_FinishCurrentStrikeBatchMember();
            exit;
        }
        dword current_strike_place = GetShipPlanet(current_strike);
        if(!current_strike_place) current_strike_place = GetShipRuins(current_strike);
        if(current_strike_place)
        {
            ShipRefuel(current_strike);
            ShipRepairEq(current_strike);
            CSL_TakeOff(current_strike);
            exit;
        }
        dword current_strike_star = CSL_ShipCurrentStar(current_strike);
        dword build_star = PlanetToStar(csl_destination);
        if(current_strike_star != build_star)
        {
            if(current_strike_star)
            {
                if(ShipCanJump(current_strike, current_strike_star, build_star, 1)) CSL_Jump(current_strike, build_star);
            }
            exit;
        }
        ShipSetBad(current_strike, 0);
        OrderLock(current_strike, 1);
        if(current_rally_base) OrderLanding(current_strike, current_rally_base, 1);
        else
        {
            int new_space_rally = CSL_EnsureSpaceRally(current_strike);
            if(new_space_rally)
            {
                csl_strike_ready_since = 1;
                exit;
            }
            if(!csl_strike_rally_ready) exit;
            int current_rally_slot = csl_strike_slots_started - 1;
            if(csl_strike_batch_next_slot > 0) current_rally_slot = csl_strike_batch_next_slot - 1;
            int current_slot_x = CSL_StrikeRallySlotX(current_rally_slot);
            int current_slot_y = CSL_StrikeRallySlotY(current_rally_slot);
            if(!csl_strike_ready_since)
            {
                OrderMove(current_strike, current_slot_x, current_slot_y);
                csl_strike_ready_since = 1;
                exit;
            }
            if(DistCoords(CoordX(current_strike), CoordY(current_strike), current_slot_x, current_slot_y) > 70)
            {
                OrderMove(current_strike, current_slot_x, current_slot_y);
                exit;
            }
            OrderMove(current_strike, current_slot_x, current_slot_y);
            CSL_FinishCurrentStrikeBatchMember();
        }
        exit;
    }
    if(csl_strike_slots_started < csl_strike_planned_count)
    {
        CSL_CreateStrikeBatch();
        exit;
    }
    if(!GroupCount(SupplyStrike))
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
        csl_phase = 5;
        exit;
    }
    csl_strike_ready_since = 0;
    csl_phase = 14;
    exit;
}

// Assemble every survivor before selecting and announcing a target. A missing
// military base is a normal case: the same ships gather near the planet.
if(csl_phase == 14)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    csl_strike_countdown_announced = 0;
    if(!csl_destination)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 5;
        exit;
    }
    if(CurTurn() >= csl_operation_deadline)
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
        csl_phase = 5;
        exit;
    }
    if(!GroupCount(SupplyStrike))
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
        csl_phase = 5;
        exit;
    }
    if(StarOwner(PlanetToStar(csl_destination)) != 0 || !CSL_PlanetEligible(csl_destination, 0))
    {
        if(CSL_DefendStrikeAssembly()) exit;
        if(CurTurn() >= csl_operation_deadline)
        {
            AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
            csl_phase = 5;
        }
        exit;
    }
    if(CSL_DefendStrikeAssembly())
    {
        csl_strike_defense_active = 1;
        csl_strike_ready_since = 0;
        exit;
    }
    if(csl_strike_defense_active)
    {
        if(CSL_ReturnStrikeToAssemblyAll()) exit;
        csl_strike_defense_active = 0;
    }
    if(CSL_MoveStrikeToAssembly()) { csl_strike_ready_since = 0; exit; }
    if(!CSL_StrikeAssembled()) { csl_strike_ready_since = 0; exit; }
    dword assembled_leader = GroupToShip(SupplyStrike);
    dword assembled_target = CSL_FindTarget(PlanetToStar(csl_destination), assembled_leader);
    if(!assembled_target)
    {
        csl_target_star_id = 0;
        csl_next_turn = CurTurn() + 5;
        exit;
    }
    int previous_assembled_target_id = csl_target_star_id;
    csl_target_star = assembled_target;
    csl_target_star_id = Id(assembled_target);
    csl_strike_ready_since = CurTurn();
    if(!csl_strike_launch_turn) csl_strike_launch_turn = CurTurn() + RndObject(25, 50, csl_destination);
    if(!csl_strike_prepared_announced)
    {
        str strike_prepared_notice = Format(CT('XenoCoalitionSupplyLines.StrikePrepared'), '#destination#', Name(csl_destination), '#target#', StarName(csl_target_star), '#date#', GameDateTxtByTurn(csl_strike_launch_turn));
        AddPlanetNews(strike_prepared_notice, 26);
        CSL_PublishStrikeAlert(strike_prepared_notice);
        csl_strike_prepared_announced = 1;
    }
    else if(previous_assembled_target_id != csl_target_star_id)
    {
        str assembled_retarget_notice = Format(CT('XenoCoalitionSupplyLines.StrikeRetargeted'), '#destination#', Name(csl_destination), '#target#', StarName(csl_target_star), '#date#', GameDateTxtByTurn(csl_strike_launch_turn));
        AddPlanetNews(assembled_retarget_notice, 26);
        CSL_PublishStrikeAlert(assembled_retarget_notice);
    }
    csl_phase = 15;
    exit;
}

// The main version keeps a 60-100 day public recruitment window. A defensive
// sortie pauses physical readiness but does not discard accepted volunteers or
// restart the whole recruitment period.
if(csl_phase == 15)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    if(!csl_destination)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 5;
        exit;
    }
    if(CurTurn() >= csl_operation_deadline)
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.OperationCancelled'), '#destination#', Name(csl_destination)));
        csl_phase = 5;
        exit;
    }
    if(!GroupCount(SupplyStrike)) { csl_phase = 5; exit; }
    dword ready_star = PlanetToStar(csl_destination);
    if(StarOwner(PlanetToStar(csl_destination)) != 0 || !CSL_PlanetEligible(csl_destination, 0))
    {
        csl_strike_ready_since = 0;
        csl_phase = 14;
        exit;
    }
    if(CSL_DefendStrikeAssembly())
    {
        csl_strike_defense_active = 1;
        exit;
    }
    if(csl_strike_defense_active)
    {
        if(CSL_ReturnStrikeToAssemblyAll()) exit;
        if(!CSL_StrikeAssembled()) exit;
        csl_strike_defense_active = 0;
        if(csl_player_joined)
        {
            dword recovery_wait_ship = 0;
            if(csl_player_wait_ship_id > 1) recovery_wait_ship = IdToShip(csl_player_wait_ship_id);
            if(recovery_wait_ship)
            {
                if(ShipInNormalSpace(recovery_wait_ship) && ShipInNormalSpace(Player()) &&
                   CSL_ShipCurrentStar(recovery_wait_ship) == ready_star && CSL_ShipCurrentStar(Player()) == ready_star)
                    OrderFollowShip(Player(), recovery_wait_ship, 1);
            }
        }
    }
    CSL_PlayerWaitingForStrike(ready_star);
    if(CSL_MoveStrikeToAssembly()) exit;
    if(!CSL_StrikeAssembled()) exit;
    int strike_target_valid = 0;
    if(csl_target_star)
    {
        if(CSL_IsStrategicStar(csl_target_star))
        {
            int strike_target_owner = StarOwner(csl_target_star);
            if((strike_target_owner == 1 || strike_target_owner == 2) && CSL_StrikeAllCanJump(ready_star, csl_target_star))
                strike_target_valid = 1;
        }
    }
    if(!strike_target_valid)
    {
        dword replacement_target = CSL_FindTarget(ready_star, GroupToShip(SupplyStrike));
        if(!replacement_target)
        {
            csl_target_star_id = 0;
            csl_strike_ready_since = 0;
            csl_phase = 14;
            csl_next_turn = CurTurn() + 5;
            exit;
        }
        csl_target_star = replacement_target;
        csl_target_star_id = Id(replacement_target);
        csl_strike_ready_since = CurTurn();
        csl_strike_launch_turn = max(csl_strike_launch_turn, CurTurn() + 10);
        csl_strike_countdown_announced = 0;
        str waiting_retarget_notice = Format(CT('XenoCoalitionSupplyLines.StrikeRetargeted'), '#destination#', Name(csl_destination), '#target#', StarName(csl_target_star), '#date#', GameDateTxtByTurn(csl_strike_launch_turn));
        AddPlanetNews(waiting_retarget_notice, 26);
        CSL_PublishStrikeAlert(waiting_retarget_notice);
    }
    if(CSL_PruneVolunteerRangers(ready_star, csl_target_star)) exit;
    if(CSL_VolunteerCount() < 6 && CSL_RecruitVolunteerRanger(ready_star, csl_target_star)) exit;
    if(!csl_strike_countdown_announced && CurTurn() < csl_strike_launch_turn && CurTurn() + 1 >= csl_strike_launch_turn)
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.StrikeCountdown'), '#destination#', Name(csl_destination), '#target#', StarName(csl_target_star)), 26);
        csl_strike_countdown_announced = 1;
    }
    if(CurTurn() < csl_strike_launch_turn) exit;
    csl_operation_announced = 0;
    csl_strike_departed = 0;
    csl_phase = 4;
    csl_next_turn = CurTurn() + 1;
    exit;
}

if(csl_phase == 4)
{
    if(CurTurn() < csl_next_turn) exit;
    csl_next_turn = CurTurn() + 1;
    if(!csl_destination || !csl_target_star)
    {
        AddPlanetNews(CT('XenoCoalitionSupplyLines.NoRoute'));
        csl_phase = 5;
        exit;
    }
    if(!GroupCount(SupplyStrike))
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Failure'), '#target#', StarName(csl_target_star)));
        csl_phase = 5;
        exit;
    }
    dword operation_start_star = PlanetToStar(csl_destination);
    // The staging world may be attacked while ships leave a military base one
    // by one. Defend it and suspend the external operation; never launch from
    // a captured or otherwise unsafe origin.
    if(!csl_strike_departed && (StarOwner(operation_start_star) != 0 || !CSL_PlanetEligible(csl_destination, 0)))
    {
        csl_strike_ready_since = 0;
        if(CSL_DefendStrikeAssembly()) exit;
        if(CurTurn() >= csl_operation_deadline)
        {
            AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Failure'), '#target#', StarName(csl_target_star)));
            csl_phase = 5;
            exit;
        }
        csl_operation_announced = 0;
        csl_phase = 14;
        exit;
    }
    if(!csl_strike_departed && (!CSL_IsStrategicStar(csl_target_star) || (StarOwner(csl_target_star) != 1 && StarOwner(csl_target_star) != 2)))
    {
        dword launch_replacement_target = CSL_FindTarget(PlanetToStar(csl_destination), GroupToShip(SupplyStrike));
        if(launch_replacement_target)
        {
            csl_target_star = launch_replacement_target;
            csl_target_star_id = Id(launch_replacement_target);
            csl_strike_ready_since = CurTurn();
            csl_strike_launch_turn = CurTurn() + 10;
            csl_operation_announced = 0;
            csl_phase = 15;
            str launch_retarget_notice = Format(CT('XenoCoalitionSupplyLines.StrikeRetargeted'), '#destination#', Name(csl_destination), '#target#', StarName(csl_target_star), '#date#', GameDateTxtByTurn(csl_strike_launch_turn));
            AddPlanetNews(launch_retarget_notice, 26);
            CSL_PublishStrikeAlert(launch_retarget_notice);
            exit;
        }
        // The old target was liberated before launch and no valid replacement
        // exists. Return to target search; this is not a completed operation.
        csl_target_star = 0;
        csl_target_star_id = 0;
        csl_strike_ready_since = 0;
        csl_operation_announced = 0;
        csl_phase = 14;
        csl_next_turn = CurTurn() + 5;
        exit;
    }
    if(csl_strike_departed && StarOwner(csl_target_star) == 0)
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Victory'), '#target#', StarName(csl_target_star)));
        CSL_RewardPlayer();
        csl_phase = 5;
        exit;
    }
    if(CurTurn() >= csl_operation_deadline || (StarOwner(csl_target_star) != 1 && StarOwner(csl_target_star) != 2))
    {
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Failure'), '#target#', StarName(csl_target_star)));
        csl_phase = 5;
        exit;
    }
    if(!csl_strike_departed)
    {
        if(CSL_DefendStrikeAssembly()) exit;
        for(int operation_takeoff_index = 0; operation_takeoff_index < GroupCount(SupplyStrike); operation_takeoff_index = operation_takeoff_index + 1)
        {
            dword operation_takeoff_ship = GroupShip(SupplyStrike, operation_takeoff_index);
            if(!operation_takeoff_ship) continue;
            if(ShipInHyperSpace(operation_takeoff_ship, 1)) exit;
            dword operation_takeoff_place = GetShipPlanet(operation_takeoff_ship);
            if(!operation_takeoff_place) operation_takeoff_place = GetShipRuins(operation_takeoff_ship);
            if(operation_takeoff_place)
            {
                ShipRefuel(operation_takeoff_ship);
                ShipRepairEq(operation_takeoff_ship);
                CSL_TakeOff(operation_takeoff_ship);
                exit;
            }
            if(!ShipInNormalSpace(operation_takeoff_ship)) exit;
            if(CSL_ShipCurrentStar(operation_takeoff_ship) != operation_start_star)
            {
                dword takeoff_current_star = CSL_ShipCurrentStar(operation_takeoff_ship);
                if(takeoff_current_star)
                {
                    if(ShipCanJump(operation_takeoff_ship, takeoff_current_star, operation_start_star, 1)) CSL_Jump(operation_takeoff_ship, operation_start_star);
                }
                exit;
            }
        }
        if(!CSL_StrikeAllCanJump(operation_start_star, csl_target_star))
        {
            csl_strike_ready_since = 0;
            csl_phase = 14;
            csl_operation_announced = 0;
            exit;
        }
        // Mark departure before issuing the first jump so a callback cannot
        // launch only part of the group through the preparation branch again.
        int joined_player_ready = CSL_PlayerWaitingForStrike(operation_start_star);
        if(joined_player_ready)
        {
            OrderJump(Player(), csl_target_star);
            if(ShipOrder(Player()) == 3 && ShipOrderObj(Player()) == csl_target_star)
                AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.PlayerRouteSet'), '#target#', StarName(csl_target_star)), 26);
            else AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.PlayerRouteFailed'), '#target#', StarName(csl_target_star)), 26);
        }
        CSL_ClearPlayerJoin();
        CSL_ClearStrikeAlert();
        csl_strike_departed = 1;
        csl_operation_announced = 1;
        csl_operation_deadline = CurTurn() + 300;
        AddPlanetNews(Format(CT('XenoCoalitionSupplyLines.Operation'), '#destination#', Name(csl_destination), '#target#', StarName(csl_target_star)), 26);
        for(int operation_jump_index = 0; operation_jump_index < GroupCount(SupplyStrike); operation_jump_index = operation_jump_index + 1)
        {
            dword operation_jump_ship = GroupShip(SupplyStrike, operation_jump_index);
            if(operation_jump_ship) CSL_Jump(operation_jump_ship, csl_target_star);
        }
        exit;
    }
    int strike_waiting_in_hyper = 0;
    for(int transit_index = 0; transit_index < GroupCount(SupplyStrike); transit_index = transit_index + 1)
    {
        dword transit_ship = GroupShip(SupplyStrike, transit_index);
        if(transit_ship && ShipInHyperSpace(transit_ship)) strike_waiting_in_hyper = 1;
    }
    CSL_CommandStrikeCombat(csl_target_star);
    if(strike_waiting_in_hyper)
    {
        exit;
    }
    for(int laggard_index = 0; laggard_index < GroupCount(SupplyStrike); laggard_index = laggard_index + 1)
    {
        dword laggard = GroupShip(SupplyStrike, laggard_index);
        if(!laggard) continue;
        if(!ShipInNormalSpace(laggard)) continue;
        dword laggard_star = CSL_ShipCurrentStar(laggard);
        if(laggard_star != csl_target_star)
        {
            if(laggard_star)
            {
                if(ShipCanJump(laggard, laggard_star, csl_target_star, 1)) CSL_Jump(laggard, csl_target_star);
            }
            exit;
        }
    }
}

if(csl_phase == 5)
{
    CSL_ClearStrikeAlert();
    if(CSL_VolunteerCount())
    {
        if(CurTurn() < csl_next_turn) exit;
        csl_next_turn = CurTurn() + 1;
        CSL_ReleaseVolunteerRangersAtOnce();
        exit;
    }
    if(GroupCount(SupplyStrike))
    {
        if(CurTurn() < csl_next_turn) exit;
        csl_next_turn = CurTurn() + 1;
        int create_final_garrison = 0;
        if(csl_target_star)
        {
            if(StarOwner(csl_target_star) == 0) create_final_garrison = 1;
        }
        CSL_ReleaseStrikeGroupAtOnce(create_final_garrison);
        exit;
    }
    CSL_ResetStrikeIds();
    csl_strike_planned_count = 0;
    csl_strike_slots_started = 0;
    csl_strike_current_id = 0;
    csl_strike_batch_next_slot = 0;
    csl_strike_batch_end_slot = 0;
    csl_strike_rally_base_id = 0;
    csl_strike_rally_x = 0;
    csl_strike_rally_y = 0;
    csl_strike_rally_ready = 0;
    csl_strike_build_due = 0;
    csl_strike_ready_since = 0;
    csl_strike_launch_turn = 0;
    csl_strike_defense_active = 0;
    csl_strike_prepared_announced = 0;
    csl_strike_departed = 0;
    csl_strike_countdown_announced = 0;
    CSL_ClearPlayerJoin();
    CSL_ScheduleCooldown();
    exit;
}
}

// Persist any dialogue-side changes made since the previous day before the
// working globals are reused for another context.
CSL_EnsureRuntimeStorage();
if(csl_visible_context >= 1 && csl_visible_context <= 4) CSL_SaveContext(csl_visible_context);
CSL_ProcessRangerQueue();

int configured_convoy_contexts = CSL_MaxConcurrentConvoys();
CSL_LoadContext(1);
csl_runtime_context_kind = 1;
CSL_RestoreWorldRefs();
CSL_PrepareConvoyStrategy();
CSL_ProcessDeferredCrisisItem();
if(csl_pending_trade_violation > csl_violation && csl_source && csl_destination)
    CSL_ApplyViolation(csl_pending_trade_violation);
csl_pending_trade_violation = 0;
CSL_ProcessEvents();
CSL_ProcessProcurement();
int convoy_context_ambush_resume_1 = CSL_CheckConvoyAmbush(0, 0);
CSL_ProcessCurrentContext(convoy_context_ambush_resume_1, 1);
CSL_PrepareConvoyStrategy();
CSL_SaveContext(1);

CSL_LoadContext(2);
csl_runtime_context_kind = 1;
if(configured_convoy_contexts >= 2 || csl_phase != 0)
{
    CSL_RestoreWorldRefs();
    CSL_PrepareConvoyStrategy();
    CSL_ProcessDeferredCrisisItem();
    if(csl_pending_trade_violation > csl_violation && csl_source && csl_destination)
        CSL_ApplyViolation(csl_pending_trade_violation);
    csl_pending_trade_violation = 0;
    CSL_ProcessEvents();
    CSL_ProcessProcurement();
    int convoy_context_ambush_resume_2 = CSL_CheckConvoyAmbush(0, 0);
    CSL_ProcessCurrentContext(convoy_context_ambush_resume_2, 1);
    CSL_PrepareConvoyStrategy();
}
else csl_procurement_active = 0;
CSL_SaveContext(2);

CSL_LoadContext(3);
csl_runtime_context_kind = 1;
if(configured_convoy_contexts >= 3 || csl_phase != 0)
{
    CSL_RestoreWorldRefs();
    CSL_PrepareConvoyStrategy();
    CSL_ProcessDeferredCrisisItem();
    if(csl_pending_trade_violation > csl_violation && csl_source && csl_destination)
        CSL_ApplyViolation(csl_pending_trade_violation);
    csl_pending_trade_violation = 0;
    CSL_ProcessEvents();
    CSL_ProcessProcurement();
    int convoy_context_ambush_resume_3 = CSL_CheckConvoyAmbush(0, 0);
    CSL_ProcessCurrentContext(convoy_context_ambush_resume_3, 1);
    CSL_PrepareConvoyStrategy();
}
else csl_procurement_active = 0;
CSL_SaveContext(3);

CSL_LoadContext(4);
csl_runtime_context_kind = 2;
CSL_StartQueuedStrikeOperation();
CSL_RestoreWorldRefs();
CSL_ProcessEvents();
CSL_ProcessCurrentContext(0, 2);
CSL_SaveContext(4);

if(ArrayDim(csl_cargo_registry) > 1 && ArrayDim(csl_cargo_bases) == ArrayDim(csl_cargo_registry))
{
    int cargo_factor = CSL_InflationFactor();
    for(int cargo_cursor = ArrayDim(csl_cargo_registry) - 1; cargo_cursor >= 1; cargo_cursor = cargo_cursor - 1)
    {
        dword indexed_cargo = IdToItem(csl_cargo_registry[cargo_cursor]);
        if(!indexed_cargo)
        {
            ArrayDelete(csl_cargo_registry, cargo_cursor);
            ArrayDelete(csl_cargo_bases, cargo_cursor);
            continue;
        }
        int indexed_player_position = FindItemInShip(Player(), indexed_cargo);
        int indexed_storage_position = -1;
        if(indexed_player_position < 0) indexed_storage_position = FindItemInStorage(indexed_cargo);
        if(ItemInScript(indexed_cargo) && (indexed_player_position >= 0 || indexed_storage_position >= 0))
            ReleaseItemFromScript(indexed_cargo);
        ItemCost(indexed_cargo, max(1, round(1.0 * csl_cargo_bases[cargo_cursor] * cargo_factor / 100)));
    }
}

    csl_visible_context = 4;

    CSL_LoadContext(1);
