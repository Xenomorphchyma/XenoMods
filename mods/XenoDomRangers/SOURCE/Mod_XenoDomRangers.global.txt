// XenoDomRangers shared new-game runtime source.
// Persistent data contains only primitive values and engine object IDs.

function XDR_NativeReady()
{
    unknown native_ready_function = ImportedFunction('XenoDomRangersNative', 'XenoDomRangers_IsReady');
    result = native_ready_function();
}

function XDR_NativeSetMask(dword native_ranger, int native_series, int native_resume_existing)
{
    result = 0;
    if(!native_ranger || native_series < 0 || native_series > 2) exit;
    unknown native_set_function = ImportedFunction('XenoDomRangersNative', 'XenoDomRangers_SetNpcChameleon');
    result = native_set_function(native_ranger, Id(native_ranger), native_series, t_Ranger, t_Kling, native_resume_existing);
}

function XDR_NativeClearMask(dword native_ranger)
{
    result = 0;
    if(!native_ranger) exit;
    unknown native_clear_function = ImportedFunction('XenoDomRangersNative', 'XenoDomRangers_ClearNpcChameleon');
    result = native_clear_function(native_ranger, Id(native_ranger), t_Ranger);
}

function XDR_NativeReset()
{
    unknown native_reset_function = ImportedFunction('XenoDomRangersNative', 'XenoDomRangers_ResetSessions');
    result = native_reset_function();
}

function XDR_NativeWorldTick()
{
    result = 0;
    if(!XDR_NativeReady()) exit;
    dword world_player = Player();
    if(!world_player) exit;
    unknown native_world_function = ImportedFunction('XenoDomRangersNative', 'XenoDomRangers_WorldTick');
    result = native_world_function(world_player, Id(world_player), CurTurn());
}

function XDR_ConfigInt(int config_key, int config_fallback)
{
    result = config_fallback;
    if(!XDR_NativeReady()) exit;
    unknown native_config_function = ImportedFunction('XenoDomRangersNative', 'XenoDomRangers_GetConfigInt');
    int config_value = native_config_function(config_key);
    if(config_value >= 0) result = config_value;
}

function XDR_CamouflageCost(dword config_cost_ship)
{
    result = 0;
    if(!config_cost_ship) exit;
    int config_cost_base = GalaxyMoney(1, ShipPilotRace(config_cost_ship));
    int config_cost_percent = XDR_ConfigInt(10, 100);
    result = max(0, round(config_cost_base * config_cost_percent / 100));
}

function XDR_MinLootCost(dword config_loot_ship)
{
    result = 1;
    if(!config_loot_ship) exit;
    int config_loot_base = GalaxyMoney(1, ShipPilotRace(config_loot_ship));
    int config_loot_percent = XDR_ConfigInt(15, 100);
    result = max(0, round(config_loot_base * config_loot_percent / 100));
}

function XDR_FindStarById(int wanted_star_id)
{
    result = 0;
    if(wanted_star_id <= 1) exit;
    for(int find_star_index = 0; find_star_index < GalaxyStars(); find_star_index = find_star_index + 1)
    {
        dword find_star_candidate = GalaxyStar(find_star_index);
        if(!find_star_candidate) continue;
        if(Id(find_star_candidate) == wanted_star_id)
        {
            result = find_star_candidate;
            exit;
        }
    }
}

function XDR_BaseStar(dword wanted_base)
{
    result = 0;
    if(!wanted_base) exit;
    for(int base_star_index = 0; base_star_index < GalaxyStars(); base_star_index = base_star_index + 1)
    {
        dword base_star_candidate = GalaxyStar(base_star_index);
        if(!base_star_candidate) continue;
        dword base_star_ruin = StarRuins(base_star_candidate, 'PB');
        if(base_star_ruin == wanted_base)
        {
            result = base_star_candidate;
            exit;
        }
    }
}

function XDR_FindPirateBaseById(int wanted_base_id)
{
    result = 0;
    if(wanted_base_id <= 1) exit;
    dword wanted_base_ship = IdToShip(wanted_base_id);
    if(!wanted_base_ship) exit;
    if(!XDR_BaseStar(wanted_base_ship)) exit;
    result = wanted_base_ship;
}

function XDR_CurrentStar(dword current_ship)
{
    result = 0;
    if(!current_ship) exit;
    dword current_planet = GetShipPlanet(current_ship);
    if(current_planet)
    {
        result = PlanetToStar(current_planet);
        exit;
    }
    dword current_base = GetShipRuins(current_ship);
    if(current_base)
    {
        result = XDR_BaseStar(current_base);
        exit;
    }
    if(ShipInHyperSpace(current_ship, 1) || ShipIsTakeoff(current_ship) || !ShipInNormalSpace(current_ship)) exit;
    result = ShipStar(current_ship);
}

function XDR_IsHiddenPirateSector(dword checked_star)
{
    result = 0;
    if(!checked_star) exit;
    dword hidden_clan_planet = PlanetPirateClan();
    if(!hidden_clan_planet) exit;
    dword hidden_clan_star = PlanetToStar(hidden_clan_planet);
    if(!hidden_clan_star) exit;
    if(StarToCon(checked_star) == StarToCon(hidden_clan_star)) result = 1;
}

function XDR_IsStrategicStar(dword checked_star)
{
    result = 0;
    if(!checked_star || XDR_IsHiddenPirateSector(checked_star)) exit;
    if(StarCustomFaction(checked_star) != '') exit;
    int checked_owner = StarOwner(checked_star);
    if(checked_owner == 0 || checked_owner == 1 || checked_owner == 2) result = 1;
}

function XDR_CountPirateBases()
{
    int base_count = 0;
    if(xdr_test_mode && xdr_test_no_bases) { result = 0; exit; }
    for(int count_base_star_index = 0; count_base_star_index < GalaxyStars(); count_base_star_index = count_base_star_index + 1)
    {
        dword count_base_star = GalaxyStar(count_base_star_index);
        if(!count_base_star) continue;
        dword count_base = StarRuins(count_base_star, 'PB');
        if(count_base) base_count = base_count + 1;
    }
    result = base_count;
}

function XDR_UsablePirateBaseAtStar(dword usable_base_star, dword usable_base_ship)
{
    result = 0;
    if(!usable_base_star || !usable_base_ship) exit;
    // An intact PB inside a Dominator-owned or actively contested system is
    // not a usable camouflage shop: reaching it unmasked would be suicidal.
    if(StarOwner(usable_base_star) == 1 || StarBattle(usable_base_star)) exit;
    if(XDR_IsHiddenPirateSector(usable_base_star)) exit;
    dword usable_base = StarRuins(usable_base_star, 'PB');
    if(!usable_base) exit;
    if(RelationToRanger(usable_base, usable_base_ship) < 10) exit;
    result = usable_base;
}

function XDR_IsSafeReturnPlanet(dword safe_return_planet)
{
    result = 0;
    if(!safe_return_planet) exit;
    int safe_return_owner = PlanetOwner(safe_return_planet);
    if(safe_return_owner >= 0 && safe_return_owner <= 4) result = 1;
}

function XDR_FindSafePlanet(dword safe_star)
{
    result = 0;
    if(!safe_star) exit;
    if(StarBattle(safe_star) || StarOwner(safe_star) != 0) exit;
    // This helper is called by galaxy-wide BFS code. Keep it loop-free so a
    // route scan never becomes GalaxyStars * StarPlanets in one NextDay.
    // Ordinary generated systems stay well below this bounded planet count.
    int safe_planet_count = StarPlanets(safe_star);
    dword safe_planet = 0;
    if(safe_planet_count > 0) safe_planet = StarPlanets(safe_star, 0);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 1) safe_planet = StarPlanets(safe_star, 1);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 2) safe_planet = StarPlanets(safe_star, 2);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 3) safe_planet = StarPlanets(safe_star, 3);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 4) safe_planet = StarPlanets(safe_star, 4);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 5) safe_planet = StarPlanets(safe_star, 5);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 6) safe_planet = StarPlanets(safe_star, 6);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 7) safe_planet = StarPlanets(safe_star, 7);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 8) safe_planet = StarPlanets(safe_star, 8);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 9) safe_planet = StarPlanets(safe_star, 9);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 10) safe_planet = StarPlanets(safe_star, 10);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
    if(safe_planet_count > 11) safe_planet = StarPlanets(safe_star, 11);
    if(XDR_IsSafeReturnPlanet(safe_planet)) { result = safe_planet; exit; }
}

function XDR_FindUninhabitedShelter(dword shelter_star)
{
    result = 0;
    if(!shelter_star) exit;
    // This scan is limited to the current system. PlanetOwner < 0 is
    // the only script-level indication that the planet has no settlement.
    // The engine exposes no CanLand predicate, so state 17 verifies the actual
    // docking result and falls back to a bounded safe-space wait after 5 days.
    int shelter_planet_count = StarPlanets(shelter_star);
    for(int shelter_planet_index = 0; shelter_planet_index < shelter_planet_count; shelter_planet_index = shelter_planet_index + 1)
    {
        dword shelter_planet = StarPlanets(shelter_star, shelter_planet_index);
        if(!shelter_planet) continue;
        if(PlanetOwner(shelter_planet) >= 0) continue;
        result = shelter_planet;
        exit;
    }
}

function XDR_GroupContains(dword checked_group_ship)
{
    result = 0;
    if(!checked_group_ship) exit;
    for(int checked_group_index = 0; checked_group_index < GroupCount(XenoDomSpecialists); checked_group_index = checked_group_index + 1)
    {
        if(GroupShip(XenoDomSpecialists, checked_group_index) == checked_group_ship)
        {
            result = 1;
            exit;
        }
    }
}

function XDR_SlotByShip(dword checked_slot_ship)
{
    result = 0;
    if(!checked_slot_ship || ArrayDim(xdr_ids) != 4) exit;
    int checked_slot_id = Id(checked_slot_ship);
    if(checked_slot_id <= 1) exit;
    if(xdr_ids[1] == checked_slot_id) { result = 1; exit; }
    if(xdr_ids[2] == checked_slot_id) { result = 2; exit; }
    if(xdr_ids[3] == checked_slot_id) { result = 3; exit; }
}

function XDR_IsMaskedShip(dword checked_mask_ship)

    int checked_mask_slot = XDR_SlotByShip(checked_mask_ship);
    if(checked_mask_slot < 1 || ArrayDim(xdr_ghost_flags) != 4) exit;
    result = xdr_ghost_flags[checked_mask_slot];
}

function XDR_ResetRouteArrays()
{
    // Keep the legacy persistent arrays and use only one fixed slot per
    // specialist. Reusing this storage preserves every old object/dialog ID.
    xdr_route_slots = newarray(1);
    xdr_route_orders = newarray(1);
    xdr_route_star_ids = newarray(1);
    ArrayClear(xdr_route_slots);
    ArrayClear(xdr_route_orders);
    ArrayClear(xdr_route_star_ids);
    for(int reset_route_slot = 1; reset_route_slot <= 3; reset_route_slot = reset_route_slot + 1)
    {
        ArrayAdd(xdr_route_slots, 0);
        ArrayAdd(xdr_route_orders, 0);
        ArrayAdd(xdr_route_star_ids, 0);
    }
}

function XDR_SetNextRouteStarId(int route_slot, int route_star_id)
{
    if(route_slot < 1 || route_slot > 3) exit;
    if(ArrayDim(xdr_route_star_ids) != 4) XDR_ResetRouteArrays();
    xdr_route_star_ids[route_slot] = route_star_id;
}

function XDR_GetNextRouteStarId(int route_slot)
{
    result = 0;
    if(route_slot < 1 || route_slot > 3) exit;
    if(ArrayDim(xdr_route_star_ids) != 4) exit;
    result = xdr_route_star_ids[route_slot];
}

function XDR_ClearRoute(int route_slot)
{
    if(route_slot < 1 || route_slot > 3) exit;
    XDR_SetNextRouteStarId(route_slot, 0);
    XDR_SetReturnFuelNeed(route_slot, -1);
}

function XDR_NextRouteStar(int route_slot)
{
    result = 0;
    if(route_slot < 1 || route_slot > 3) exit;
    int next_route_star_id = XDR_GetNextRouteStarId(route_slot);
    if(next_route_star_id > 1) result = XDR_FindStarById(next_route_star_id);
}

function XDR_OrderLockedJump(dword jump_ship, dword jump_star)
{
    if(!jump_ship || !jump_star) exit;
    // Clear stale vanilla pickup/combat intent before giving the next mission
    // order. Releasing and immediately reclaiming only the stalker lock is the
    // working convoy pattern and does not interfere with any Dominator AI.
    OrderLock(jump_ship, 0);
    if(ShipGetBad(jump_ship)) ShipSetBad(jump_ship, 0);
    OrderNone(jump_ship);
    OrderLock(jump_ship, 1);
    OrderJump(jump_ship, jump_star, 1);
}

function XDR_PopRouteStar(int route_slot, dword reached_star)
{
    if(!reached_star) exit;
    if(route_slot < 1 || route_slot > 3) exit;
    int reached_star_id = Id(reached_star);
    if(XDR_GetNextRouteStarId(route_slot) != reached_star_id) exit;
    XDR_ClearRoute(route_slot);
}

function XDR_BuildRouteToBase(int route_slot, dword route_ship, dword route_source_star)
{
    result = 0;
    XDR_ClearRoute(route_slot);
    if(!route_ship || !route_source_star || XDR_CountPirateBases() <= 0) exit;

    // BFS previously failed when the ranger was spawned in the same system as
    // a pirate base: the reconstructed path had zero edges and was rejected.
    // Keep the current star as a one-step marker so state 1 pops it and lands.
    dword route_local_base = XDR_UsablePirateBaseAtStar(route_source_star, route_ship);
    if(route_local_base)
    {
        XDR_SetNextRouteStarId(route_slot, Id(route_source_star));
        xdr_home_base_ids[route_slot] = Id(route_local_base);
        result = 1;
        exit;
    }

    unknown route_galaxy_indices = newarray(1);
    unknown route_parent_indices = newarray(1);
    unknown route_depths = newarray(1);
    unknown route_seen_by_galaxy = newarray(1);
    ArrayClear(route_galaxy_indices);
    ArrayClear(route_parent_indices);
    ArrayClear(route_depths);
    ArrayClear(route_seen_by_galaxy);
    int route_galaxy_count = GalaxyStars();
    int route_source_index = -1;
    for(int route_source_probe = 0; route_source_probe < route_galaxy_count; route_source_probe = route_source_probe + 1)
    {
        ArrayAdd(route_seen_by_galaxy, 0);
        if(GalaxyStar(route_source_probe) == route_source_star) route_source_index = route_source_probe;
    }
    if(route_source_index < 0) exit;
    ArrayAdd(route_galaxy_indices, route_source_index);
    ArrayAdd(route_parent_indices, 0);
    ArrayAdd(route_depths, 0);
    route_seen_by_galaxy[route_source_index + 1] = 1;

    int found_route_node = -1;
    int found_route_base_id = 0;
    for(int route_pair_cursor = 0; route_pair_cursor < route_galaxy_count * route_galaxy_count; route_pair_cursor = route_pair_cursor + 1)
    {
        int route_cursor = route_pair_cursor / route_galaxy_count + 1;
        int route_candidate_index = route_pair_cursor - (route_cursor - 1) * route_galaxy_count;
        if(route_cursor >= ArrayDim(route_galaxy_indices)) continue;
        dword route_current_star = GalaxyStar(route_galaxy_indices[route_cursor]);
        if(!route_current_star) continue;
        if(route_candidate_index == 0)
        {
            dword route_current_base = XDR_UsablePirateBaseAtStar(route_current_star, route_ship);
            if(route_current_base)
            {
                found_route_node = route_cursor;
                found_route_base_id = Id(route_current_base);
                break;
            }
        }
        dword route_candidate_star = GalaxyStar(route_candidate_index);
        if(!route_candidate_star) continue;
        if(route_candidate_star == route_source_star || route_seen_by_galaxy[route_candidate_index + 1]) continue;
        dword route_candidate_base = XDR_UsablePirateBaseAtStar(route_candidate_star, route_ship);
        int route_is_base_star = 0;
        if(route_candidate_base) route_is_base_star = 1;
        int route_is_safe_star = XDR_IsStrategicStar(route_candidate_star) && StarOwner(route_candidate_star) == 0 && !StarBattle(route_candidate_star) && StarPlanets(route_candidate_star) > 0;
        if(!route_is_base_star && !route_is_safe_star) continue;
        if(!ShipCanJump(route_ship, route_current_star, route_candidate_star)) continue;
        route_seen_by_galaxy[route_candidate_index + 1] = 1;
        ArrayAdd(route_galaxy_indices, route_candidate_index);
        ArrayAdd(route_parent_indices, route_cursor);
        ArrayAdd(route_depths, route_depths[route_cursor] + 1);
    }
    if(found_route_node < 1 || found_route_base_id <= 1) exit;

    unknown reverse_route_ids = newarray(1);
    ArrayClear(reverse_route_ids);
    int route_backtrack_node = found_route_node;
    for(int route_backtrack_step = 0; route_backtrack_step < route_galaxy_count; route_backtrack_step = route_backtrack_step + 1)
    {
        if(route_backtrack_node <= 1) break;
        dword route_backtrack_star = GalaxyStar(route_galaxy_indices[route_backtrack_node]);
        if(!route_backtrack_star) exit;
        ArrayAdd(reverse_route_ids, Id(route_backtrack_star));
        route_backtrack_node = route_parent_indices[route_backtrack_node];
    }
    if(route_backtrack_node != 1) exit;
    int route_first_index = ArrayDim(reverse_route_ids) - 1;
    if(route_first_index < 1) exit;
    XDR_SetNextRouteStarId(route_slot, reverse_route_ids[route_first_index]);
    xdr_home_base_ids[route_slot] = found_route_base_id;
    result = 1;
}

function XDR_HasPurchasedCharge(int charge_slot)
{
    result = 0;
    if(charge_slot < 1 || charge_slot > 3) exit;
    if(ArrayDim(xdr_target_star_ids) != 4 || ArrayDim(xdr_series) != 4 || ArrayDim(xdr_charge_expire_turns) != 4) exit;
    if(xdr_target_star_ids[charge_slot] <= 1) exit;
    if(xdr_series[charge_slot] < 0) exit;
    if(xdr_charge_expire_turns[charge_slot] <= 0) exit;
    result = 1;
}

function XDR_IsSafeSalvageItem(dword safe_salvage_item)
{
    result = 0;
    if(!safe_salvage_item) exit;
    // Vanilla stars use SafeRadius 250..300 and DamageRadius 240..290.
    // Mission loot is never worth crossing either boundary: solar charging
    // has its own controlled radial approach below.
    if(DistCoords(CoordX(safe_salvage_item), CoordY(safe_salvage_item), 0, 0) >= XDR_ConfigInt(20, 400)) result = 1;
}

function XDR_OrderRadialMove(dword radial_ship, int radial_target_radius)
{
    if(!radial_ship || radial_target_radius < 1) exit;
    if(ShipInHyperSpace(radial_ship, 1)) exit;
    if(ShipIsTakeoff(radial_ship)) exit;
    if(!ShipInNormalSpace(radial_ship)) exit;
    int radial_source_x = CoordX(radial_ship);
    int radial_source_y = CoordY(radial_ship);
    int radial_source_radius = DistCoords(radial_source_x, radial_source_y, 0, 0);
    int radial_target_x = radial_target_radius;
    int radial_target_y = 0;
    if(radial_source_radius > 0)
    {
        radial_target_x = round(radial_source_x * radial_target_radius / radial_source_radius);
        radial_target_y = round(radial_source_y * radial_target_radius / radial_source_radius);
        if(!radial_target_x && !radial_target_y) radial_target_x = radial_target_radius;
    }
    OrderMove(radial_ship, radial_target_x, radial_target_y);
}

function XDR_OrderSolarApproach(dword solar_approach_ship, int solar_damage_boundary)
{
    if(!solar_approach_ship) exit;
    int solar_approach_radius = DistCoords(CoordX(solar_approach_ship), CoordY(solar_approach_ship), 0, 0);
    // The system edge can be thousands of units away, so fly normally to a
    // wide radius 600 first. All vanilla star damage/safe radii are far inside
    // this perimeter. Only the final approach probes inward by at most 100
    // units per day, allowing state 7 to record the first real HullDamage tick.
    int solar_approach_floor = 200;
    if(solar_damage_boundary > solar_approach_floor) solar_approach_floor = solar_damage_boundary;
    int solar_careful_start_radius = max(XDR_ConfigInt(21, 600), solar_approach_floor);
    if(solar_approach_radius > solar_careful_start_radius)
    {
        XDR_OrderRadialMove(solar_approach_ship, solar_careful_start_radius);
        exit;
    }
    if(solar_approach_radius <= solar_approach_floor)
    {
        OrderNone(solar_approach_ship);
        exit;
    }
    XDR_OrderRadialMove(solar_approach_ship, max(solar_approach_floor, solar_approach_radius - XDR_ConfigInt(22, 100)));
}

function XDR_OrderSolarRetreat(dword solar_retreat_ship)
{
    if(!solar_retreat_ship) exit;
    int solar_retreat_radius = DistCoords(CoordX(solar_retreat_ship), CoordY(solar_retreat_ship), 0, 0);
    // 450 is outside every vanilla SafeRadius with a margin, but still well
    // inside ordinary tactical jump points and cannot send a ship off-map.
    int solar_retreat_target = XDR_ConfigInt(23, 450);
    if(solar_retreat_radius >= solar_retreat_target)
    {
        OrderNone(solar_retreat_ship);
        exit;
    }
    XDR_OrderRadialMove(solar_retreat_ship, solar_retreat_target);
}

function XDR_FuelTankIsFull(dword full_tank_ship)
{
    result = 0;
    if(!full_tank_ship) exit;
    dword full_tank = ShipEqInSlot(full_tank_ship, t_FuelTanks);
    if(!full_tank) exit;
    int full_tank_capacity = GetEquipmentStats(full_tank, 0);
    if(full_tank_capacity <= 0) exit;
    if(ShipFuel(full_tank_ship) >= full_tank_capacity) result = 1;
}

function XDR_EnterCriticalShelter(int shelter_slot, dword shelter_ship, dword shelter_star)
{
    if(shelter_slot < 1 || shelter_slot > 3 || !shelter_ship || !shelter_star) exit;
    XDR_CancelPickup(shelter_slot, shelter_ship);
    XDR_ClearBattleWait(shelter_slot);
    // A negative no-loot counter is a state-local flag: after critical solar
    // damage the ranger may use real cisterns, but it must never retry the sun.
    xdr_no_loot_counts[shelter_slot] = -1;
    xdr_pending_since[shelter_slot] = -ShipFuel(shelter_ship) - 1;
    dword shelter_planet = XDR_FindUninhabitedShelter(shelter_star);
    if(shelter_planet)
    {
        xdr_route_slots[shelter_slot] = Id(shelter_planet);
        xdr_route_orders[shelter_slot] = CurTurn();
        OrderLanding(shelter_ship, shelter_planet, 1);
        if(xdr_test_mode) xdr_test_last_event = 'критический корпус: сталкер пытается сесть на необитаемую планету и ждать реального топлива';
    }
    else
    {
        xdr_route_slots[shelter_slot] = 0;
        xdr_route_orders[shelter_slot] = 0;
        XDR_OrderSolarRetreat(shelter_ship);
        if(xdr_test_mode) xdr_test_last_event = 'критический корпус: подходящей необитаемой планеты нет, сталкер ждёт в безопасном космосе';
    }
    xdr_states[shelter_slot] = 17;
}

function XDR_StarHasValuableLoot(dword valuable_ship, dword valuable_star)
{
    result = 0;
    if(!valuable_ship || !valuable_star) exit;
    dword valuable_hook = ShipEqInSlot(valuable_ship, t_CargoHook);
    if(!valuable_hook) exit;
    int valuable_hook_size = GetEquipmentStats(valuable_hook, 0);
    int valuable_free_space = ShipFreeSpace(valuable_ship);
    int valuable_cost_floor = XDR_MinLootCost(valuable_ship);
    int valuable_item_count = StarItems(valuable_star);
    for(int valuable_item_index = 0; valuable_item_index < valuable_item_count; valuable_item_index = valuable_item_index + 1)
    {
        dword valuable_item = StarItems(valuable_star, valuable_item_index);
        if(!valuable_item) continue;
        if(!ItemIn(valuable_item, valuable_star)) continue;
        if(ItemInScript(valuable_item) == 1) continue;
        if(!XDR_IsSafeSalvageItem(valuable_item)) continue;
        int valuable_item_size = ItemSize(valuable_item);
        if(valuable_item_size > valuable_free_space || valuable_item_size > valuable_hook_size) continue;
        int valuable_item_type = ItemType(valuable_item);
        int valuable_item_has_usable_fuel = 0;
        if(valuable_item_type == t_Cistern || valuable_item_type == t_FuelTanks)
        {
            if(ContainerFuel(valuable_item) > 0) valuable_item_has_usable_fuel = 1;
        }
        if(!valuable_item_has_usable_fuel && ItemCost(valuable_item) < valuable_cost_floor) continue;
        result = 1;
        exit;
    }
}

function XDR_BuildRouteToTarget(int route_slot, dword route_ship, dword route_source_star, int required_series)
{
    result = 0;
    XDR_ClearRoute(route_slot);
    if(!route_ship || !route_source_star) exit;

    // The quick test deliberately creates its loot only at the real ranger
    // entry point. Until then this one direct route is allowed without the
    // ordinary valuable-loot pre-scan.
    if(xdr_test_mode && xdr_test_quick_stage == 3 && route_slot == xdr_test_quick_slot)
    {
        if(xdr_target_star_ids[route_slot] == xdr_test_quick_target_star_id && required_series == xdr_series[route_slot])
        {
            dword target_route_quick_star = XDR_FindStarById(xdr_test_quick_target_star_id);
            if(target_route_quick_star)
            {
                if(target_route_quick_star != route_source_star)
                {
                    if(XDR_IsTargetValid(target_route_quick_star, required_series, 0))
                    {
                        if(ShipCanJump(route_ship, route_source_star, target_route_quick_star))
                        {
                            XDR_SetNextRouteStarId(route_slot, Id(target_route_quick_star));
                            result = 1;
                            exit;
                        }
                    }
                }
            }
        }
    }

    unknown target_route_galaxy_indices = newarray(1);
    unknown target_route_parent_indices = newarray(1);
    unknown target_route_seen_by_galaxy = newarray(1);
    ArrayClear(target_route_galaxy_indices);
    ArrayClear(target_route_parent_indices);
    ArrayClear(target_route_seen_by_galaxy);
    int target_route_galaxy_count = GalaxyStars();

    unknown target_route_valuable_star_ids = newarray(1);
    ArrayClear(target_route_valuable_star_ids);
    dword target_route_hook = ShipEqInSlot(route_ship, t_CargoHook);
    if(!target_route_hook) exit;
    int target_route_hook_size = GetEquipmentStats(target_route_hook, 0);
    int target_route_free_space = ShipFreeSpace(route_ship);
    int target_route_cost_floor = XDR_MinLootCost(route_ship);
    int target_route_max_items = 0;
    for(int target_route_value_star_probe = 0; target_route_value_star_probe < target_route_galaxy_count; target_route_value_star_probe = target_route_value_star_probe + 1)
    {
        dword target_route_value_probe_star = GalaxyStar(target_route_value_star_probe);
        if(!target_route_value_probe_star) continue;
        if(!XDR_IsTargetValid(target_route_value_probe_star, required_series, 0)) continue;
        target_route_max_items = max(target_route_max_items, StarItems(target_route_value_probe_star));
    }
    if(target_route_max_items <= 0) exit;
    for(int target_route_value_cursor = 0; target_route_value_cursor < target_route_galaxy_count * target_route_max_items; target_route_value_cursor = target_route_value_cursor + 1)
    {
        int target_route_value_star_index = target_route_value_cursor / target_route_max_items;
        int target_route_value_item_index = target_route_value_cursor - target_route_value_star_index * target_route_max_items;
        dword target_route_value_star = GalaxyStar(target_route_value_star_index);
        if(!target_route_value_star) continue;
        if(!XDR_IsTargetValid(target_route_value_star, required_series, 0)) continue;
        int target_route_value_star_id = Id(target_route_value_star);
        if(ArrayFind(target_route_valuable_star_ids, target_route_value_star_id) >= 1) continue;
        if(target_route_value_item_index >= StarItems(target_route_value_star)) continue;
        dword target_route_value_item = StarItems(target_route_value_star, target_route_value_item_index);
        if(!target_route_value_item) continue;
        if(!ItemIn(target_route_value_item, target_route_value_star)) continue;
        if(ItemInScript(target_route_value_item) == 1) continue;
        if(!XDR_IsSafeSalvageItem(target_route_value_item)) continue;
        int target_route_value_item_size = ItemSize(target_route_value_item);
        if(target_route_value_item_size > target_route_free_space || target_route_value_item_size > target_route_hook_size) continue;
        int target_route_value_item_type = ItemType(target_route_value_item);
        int target_route_value_has_usable_fuel = 0;
        if(target_route_value_item_type == t_Cistern || target_route_value_item_type == t_FuelTanks)
        {
            if(ContainerFuel(target_route_value_item) > 0) target_route_value_has_usable_fuel = 1;
        }
        if(!target_route_value_has_usable_fuel && ItemCost(target_route_value_item) < target_route_cost_floor) continue;
        ArrayAdd(target_route_valuable_star_ids, target_route_value_star_id);
    }
    if(ArrayDim(target_route_valuable_star_ids) <= 1) exit;

    int target_route_source_index = -1;
    for(int target_route_source_probe = 0; target_route_source_probe < target_route_galaxy_count; target_route_source_probe = target_route_source_probe + 1)
    {
        ArrayAdd(target_route_seen_by_galaxy, 0);
        if(GalaxyStar(target_route_source_probe) == route_source_star) target_route_source_index = target_route_source_probe;
    }
    if(target_route_source_index < 0) exit;
    ArrayAdd(target_route_galaxy_indices, target_route_source_index);
    ArrayAdd(target_route_parent_indices, 0);
    target_route_seen_by_galaxy[target_route_source_index + 1] = 1;

    int target_route_found_node = -1;
    int target_route_found_star_id = 0;
    dword target_route_found_star = 0;
    for(int target_route_pair_cursor = 0; target_route_pair_cursor < target_route_galaxy_count * target_route_galaxy_count; target_route_pair_cursor = target_route_pair_cursor + 1)
    {
        int target_route_cursor = target_route_pair_cursor / target_route_galaxy_count + 1;
        int target_route_candidate_index = target_route_pair_cursor - (target_route_cursor - 1) * target_route_galaxy_count;
        if(target_route_cursor >= ArrayDim(target_route_galaxy_indices)) continue;
        dword target_route_current_star = GalaxyStar(target_route_galaxy_indices[target_route_cursor]);
        if(!target_route_current_star) continue;
        if(target_route_candidate_index == 0 && target_route_current_star != route_source_star)
        {
            if(XDR_IsTargetValid(target_route_current_star, required_series, 0) && ArrayFind(target_route_valuable_star_ids, Id(target_route_current_star)) >= 1)
            {
                target_route_found_node = target_route_cursor;
                target_route_found_star_id = Id(target_route_current_star);
                target_route_found_star = target_route_current_star;
                break;
            }
        }
        dword target_route_candidate_star = GalaxyStar(target_route_candidate_index);
        if(!target_route_candidate_star) continue;
        if(target_route_candidate_star == route_source_star || target_route_seen_by_galaxy[target_route_candidate_index + 1]) continue;
        int target_route_is_target = XDR_IsTargetValid(target_route_candidate_star, required_series, 0) && ArrayFind(target_route_valuable_star_ids, Id(target_route_candidate_star)) >= 1;
        int target_route_owner = StarOwner(target_route_candidate_star);
        int target_route_is_safe_transit = XDR_IsStrategicStar(target_route_candidate_star) && !StarBattle(target_route_candidate_star) && (target_route_owner == 0 || target_route_owner == 2);
        if(!target_route_is_target && !target_route_is_safe_transit) continue;
        if(!ShipCanJump(route_ship, target_route_current_star, target_route_candidate_star)) continue;
        target_route_seen_by_galaxy[target_route_candidate_index + 1] = 1;
        ArrayAdd(target_route_galaxy_indices, target_route_candidate_index);
        ArrayAdd(target_route_parent_indices, target_route_cursor);
    }
    if(target_route_found_node < 1 || target_route_found_star_id <= 1) exit;

    unknown target_route_reverse_ids = newarray(1);
    ArrayClear(target_route_reverse_ids);
    int target_route_backtrack_node = target_route_found_node;
    for(int target_route_backtrack_step = 0; target_route_backtrack_step < target_route_galaxy_count; target_route_backtrack_step = target_route_backtrack_step + 1)
    {
        if(target_route_backtrack_node <= 1) break;
        dword target_route_backtrack_star = GalaxyStar(target_route_galaxy_indices[target_route_backtrack_node]);
        if(!target_route_backtrack_star) exit;
        ArrayAdd(target_route_reverse_ids, Id(target_route_backtrack_star));
        target_route_backtrack_node = target_route_parent_indices[target_route_backtrack_node];
    }
    if(target_route_backtrack_node != 1) exit;
    int target_route_first_index = ArrayDim(target_route_reverse_ids) - 1;
    if(target_route_first_index < 1) exit;
    XDR_SetNextRouteStarId(route_slot, target_route_reverse_ids[target_route_first_index]);
    if(!target_route_found_star) { XDR_ClearRoute(route_slot); exit; }
    xdr_target_star_ids[route_slot] = target_route_found_star_id;
    if(required_series < 0) xdr_series[route_slot] = StarSeries(target_route_found_star);
    result = 1;
}

function XDR_ServiceRefuel(dword service_ship, int service_reserve)
{
    if(!service_ship) exit;
    dword service_tank = ShipEqInSlot(service_ship, t_FuelTanks);
    if(!service_tank || !ShipFuelLow(service_ship)) exit;
    int service_capacity = GetEquipmentStats(service_tank, 0);
    int service_missing = max(0, service_capacity - ShipFuel(service_ship));
    if(service_missing <= 0) exit;
    int service_unit_cost = max(1, round(PortionInDiapason(CurTurn(), 1000, 15000, 1, 10)));
    int service_budget = max(0, ShipMoney(service_ship) - max(0, service_reserve));
    int service_amount = min(service_missing, service_budget / service_unit_cost);
    if(service_amount <= 0) exit;
    ShipMoney(service_ship, ShipMoney(service_ship) - service_amount * service_unit_cost);
    ShipFuel(service_ship, ShipFuel(service_ship) + service_amount);
}

function XDR_IsCoalitionAssaultShip(dword assault_star, int assault_ship_index, int assault_ship_count)
{
    result = 0;
    if(!assault_star) exit;
    if(assault_ship_index < 0 || assault_ship_index >= assault_ship_count) exit;
    dword assault_ship = StarShips(assault_star, assault_ship_index);
    if(!assault_ship) exit;
    if(ShipTypeN(assault_ship) >= t_RC) exit;
    if(ShipTypeN(assault_ship) == t_Ranger) exit;
    if(assault_ship == Player()) exit;
    if(XDR_SlotByShip(assault_ship) > 0) exit;
    if(ShipInHyperSpace(assault_ship, 1)) exit;
    if(ShipIsTakeoff(assault_ship)) exit;
    if(!ShipInNormalSpace(assault_ship)) exit;
    if(HullDamage(assault_ship) >= 100) exit;
    int assault_owner = ShipOwner(assault_ship);
    if(assault_owner < 0 || assault_owner > 4) exit;
    if(ShipStanding(assault_ship) != 2) exit;
    result = 1;
}

function XDR_IsMissionBlockingBattle(dword blocking_star)
{
    result = 0;
    if(!blocking_star) exit;
    if(StarOwner(blocking_star) == 1)
    {
        result = XDR_IsCoalitionAssault(blocking_star);
        exit;
    }
    result = StarBattle(blocking_star);
}

function XDR_IsTargetValid(dword target_star, int target_series, int require_loot)
{
    result = 0;
    if(!target_star || !XDR_IsStrategicStar(target_star)) exit;
    if(StarOwner(target_star) != 1) exit;
    if(XDR_IsCoalitionAssault(target_star)) exit;
    if(target_series >= 0 && StarSeries(target_star) != target_series) exit;
    if(require_loot && StarItems(target_star) <= 0) exit;
    result = 1;
}

function XDR_FindTarget(dword target_ship, dword target_origin_star, int required_series)
{
    result = 0;
    if(!target_ship || !target_origin_star) exit;
    dword target_hook = ShipEqInSlot(target_ship, t_CargoHook);
    if(!target_hook) exit;
    int target_hook_size = GetEquipmentStats(target_hook, 0);
    int target_free_space = ShipFreeSpace(target_ship);
    int target_cost_floor = XDR_MinLootCost(target_ship);
    int target_galaxy_count = GalaxyStars();
    int target_max_items = 0;
    for(int target_value_star_probe = 0; target_value_star_probe < target_galaxy_count; target_value_star_probe = target_value_star_probe + 1)
    {
        dword target_value_probe_star = GalaxyStar(target_value_star_probe);
        if(!target_value_probe_star) continue;
        if(!XDR_IsTargetValid(target_value_probe_star, required_series, 0)) continue;
        target_max_items = max(target_max_items, StarItems(target_value_probe_star));
    }
    if(target_max_items <= 0) exit;
    unknown target_valuable_star_ids = newarray(1);
    ArrayClear(target_valuable_star_ids);
    for(int target_value_cursor = 0; target_value_cursor < target_galaxy_count * target_max_items; target_value_cursor = target_value_cursor + 1)
    {
        int target_value_star_index = target_value_cursor / target_max_items;
        int target_value_item_index = target_value_cursor - target_value_star_index * target_max_items;
        dword target_value_star = GalaxyStar(target_value_star_index);
        if(!target_value_star) continue;
        if(!XDR_IsTargetValid(target_value_star, required_series, 0)) continue;
        int target_value_star_id = Id(target_value_star);
        if(ArrayFind(target_valuable_star_ids, target_value_star_id) >= 1) continue;
        if(target_value_item_index >= StarItems(target_value_star)) continue;
        dword target_value_item = StarItems(target_value_star, target_value_item_index);
        if(!target_value_item) continue;
        if(!ItemIn(target_value_item, target_value_star)) continue;
        if(ItemInScript(target_value_item) == 1) continue;
        if(!XDR_IsSafeSalvageItem(target_value_item)) continue;
        int target_value_item_size = ItemSize(target_value_item);
        if(target_value_item_size > target_free_space || target_value_item_size > target_hook_size) continue;
        int target_value_item_type = ItemType(target_value_item);
        int target_value_has_usable_fuel = 0;
        if(target_value_item_type == t_Cistern || target_value_item_type == t_FuelTanks)
        {
            if(ContainerFuel(target_value_item) > 0) target_value_has_usable_fuel = 1;
        }
        if(!target_value_has_usable_fuel && ItemCost(target_value_item) < target_cost_floor) continue;
        ArrayAdd(target_valuable_star_ids, target_value_star_id);
    }
    if(ArrayDim(target_valuable_star_ids) <= 1) exit;
    int best_target_distance = 1000000;
    int equal_target_count = 0;
    for(int target_index = 0; target_index < target_galaxy_count; target_index = target_index + 1)
    {
        dword target_candidate = GalaxyStar(target_index);
        if(!target_candidate) continue;
        if(ArrayFind(target_valuable_star_ids, Id(target_candidate)) < 1) continue;
        int target_distance = Dist(target_origin_star, target_candidate);
        if(target_distance < best_target_distance)
        {
            best_target_distance = target_distance;
            equal_target_count = 1;
            result = target_candidate;
        }
        else if(target_distance == best_target_distance)
        {
            equal_target_count = equal_target_count + 1;
            if(RndObject(1, equal_target_count, target_candidate) == 1) result = target_candidate;
        }
    }
}

function XDR_EnableGhost(int ghost_slot, dword ghost_ranger)
{
    if(!ghost_ranger) { result = 0; exit; }
    if(!XDR_NativeReady()) { result = 0; exit; }
    if(xdr_old_names[ghost_slot] == '') xdr_old_names[ghost_slot] = ShipName(ghost_ranger);
    int ghost_applied = XDR_NativeSetMask(ghost_ranger, xdr_series[ghost_slot], 0);
    if(!ghost_applied) { result = 0; exit; }
    xdr_ghost_flags[ghost_slot] = 1;
    xdr_dialog_markers[ghost_slot] = 0;
    result = 1;
}

function XDR_ReassertGhost(int ghost_slot, dword ghost_ranger)
{
    result = 0;
    if(!ghost_ranger) exit;
    if(!XDR_NativeSetMask(ghost_ranger, xdr_series[ghost_slot], 1)) exit;
    xdr_ghost_flags[ghost_slot] = 1;
    if(xdr_old_names[ghost_slot] == '') xdr_old_names[ghost_slot] = ShipName(ghost_ranger);
    result = 1;
}

function XDR_DisableGhostOnly(int unghost_slot, dword unghost_ranger)
{
    if(unghost_ranger) XDR_NativeClearMask(unghost_ranger);
    xdr_dialog_markers[unghost_slot] = 0;
    xdr_ghost_flags[unghost_slot] = 0;
}

function XDR_SuppressMaskedOffense(dword passive_ranger, dword passive_star)
{
    if(!passive_ranger || !passive_star) exit;
    if(!XDR_IsMaskedShip(passive_ranger)) exit;
    if(ShipInHyperSpace(passive_ranger, 1) || ShipIsTakeoff(passive_ranger)) exit;
    if(!ShipInNormalSpace(passive_ranger)) exit;
    if(ShipStar(passive_ranger) != passive_star) exit;

    // Native camouflage changes how TKling sees the ranger, but it does not
    // disable automatic weapon acquisition by the ranger. Clear all three
    // combat channels without touching an explicit salvage/navigation order:
    // the hostile pointer, a combat order aimed at that pointer and every
    // independently acquired proximity target of an installed weapon.
    dword passive_bad = ShipGetBad(passive_ranger);
    dword passive_order_target = ShipOrderObj(passive_ranger);
    if(passive_bad) ShipSetBad(passive_ranger, 0);
    if(passive_bad && passive_order_target == passive_bad) OrderNone(passive_ranger);
    int passive_weapon_count = ShipCntWeapon(passive_ranger);
    for(int passive_weapon_index = 1; passive_weapon_index <= passive_weapon_count; passive_weapon_index = passive_weapon_index + 1)
    {
        dword passive_weapon = ShipWeapon(passive_ranger, passive_weapon_index);
        if(passive_weapon && WeaponTarget(passive_weapon)) WeaponTarget(passive_weapon, 0);
    }
}

function XDR_FindActualAttacker(dword defense_ranger, dword defense_star)
{
    result = 0;
    if(!defense_ranger || !defense_star) exit;
    int defense_ship_count = StarShips(defense_star);
    for(int defense_ship_index = 0; defense_ship_index < defense_ship_count; defense_ship_index = defense_ship_index + 1)
    {
        dword defense_candidate = StarShips(defense_star, defense_ship_index);
        if(!defense_candidate || defense_candidate == defense_ranger) continue;
        if(ShipTypeN(defense_candidate) >= t_RC) continue;
        if(ShipInHyperSpace(defense_candidate, 1)) continue;
        if(ShipIsTakeoff(defense_candidate)) continue;
        if(!ShipInNormalSpace(defense_candidate)) continue;
        if(ShipGetBad(defense_candidate) == defense_ranger)
        {
            result = defense_candidate;
            exit;
        }
    }
}

function XDR_KeepDefensiveTargetOnly(dword defense_only_ranger)
{
    if(!defense_only_ranger) exit;
    if(ShipInHyperSpace(defense_only_ranger, 1)) exit;
    if(ShipIsTakeoff(defense_only_ranger)) exit;
    if(!ShipInNormalSpace(defense_only_ranger)) exit;
    dword defense_only_star = ShipStar(defense_only_ranger);
    if(!defense_only_star) exit;
    int defense_only_masked = XDR_IsMaskedShip(defense_only_ranger);
    if(defense_only_masked)
    {
        XDR_SuppressMaskedOffense(defense_only_ranger, defense_only_star);
        exit;
    }
    dword defense_only_attacker = XDR_FindActualAttacker(defense_only_ranger, defense_only_star);
    dword defense_only_old_target = ShipGetBad(defense_only_ranger);
    if(defense_only_attacker)
    {
        int defense_only_attacker_owner = ShipOwner(defense_only_attacker);
        if(defense_only_attacker_owner == 5)
        {
            // A masked specialist never fights Dominators. Clear both halves of
            // a target pair directly; TruceBetweenShips is forbidden here
            // because it synchronously re-enters ship state code.
            ShipSetBad(defense_only_attacker, 0);
            XDR_SetDialogMarker(defense_only_ranger, 0);
            if(defense_only_old_target) ShipSetBad(defense_only_ranger, 0);
            exit;
        }
        if(defense_only_old_target != defense_only_attacker) ShipSetBad(defense_only_ranger, defense_only_attacker);
        exit;
    }
    if(defense_only_old_target) ShipSetBad(defense_only_ranger, 0);
}

function XDR_FindMilitaryAttacker(dword military_ranger, dword military_star)
{
    result = 0;
    if(!military_ranger || !military_star) exit;
    int military_ship_count = StarShips(military_star);
    for(int military_ship_index = 0; military_ship_index < military_ship_count; military_ship_index = military_ship_index + 1)
    {
        dword military_candidate = StarShips(military_star, military_ship_index);
        if(!military_candidate || military_candidate == military_ranger) continue;
        if(ShipTypeN(military_candidate) >= t_RC) continue;
        if(ShipInHyperSpace(military_candidate, 1)) continue;
        if(ShipIsTakeoff(military_candidate)) continue;
        if(!ShipInNormalSpace(military_candidate)) continue;
        int military_owner = ShipOwner(military_candidate);
        if(military_owner < 0 || military_owner > 4) continue;
        if(ShipStanding(military_candidate) != 2) continue;
        if(ShipGetBad(military_candidate) == military_ranger)
        {
            result = military_candidate;
            exit;
        }
    }
}

function XDR_ClearBattleWait(int wait_clear_slot)
{
    if(wait_clear_slot < 1 || wait_clear_slot > 3) exit;
    if(ArrayDim(xdr_route_slots) != 4) exit;
    if(ArrayDim(xdr_route_orders) != 4) exit;
    // These two legacy, otherwise unused per-slot arrays now hold the bounded
    // battle-wait coordinates without changing the persistent save schema.
    xdr_route_slots[wait_clear_slot] = 0;
    xdr_route_orders[wait_clear_slot] = 0;
}

function XDR_RecordSystemEntryPoint(int entry_slot, dword entry_ranger)
{
    if(entry_slot < 1 || entry_slot > 3) exit;
    if(!entry_ranger) exit;
    if(ShipInHyperSpace(entry_ranger, 1) || ShipIsTakeoff(entry_ranger) || !ShipInNormalSpace(entry_ranger)) exit;
    if(ArrayDim(xdr_route_slots) != 4) exit;
    if(ArrayDim(xdr_route_orders) != 4) exit;
    // A ship materializes at a tactical jump point selected by the engine.
    // Keeping those exact arrival coordinates gives us a real, finite system
    // edge without inventing a radius or issuing temporary movement orders.
    xdr_route_slots[entry_slot] = CoordX(entry_ranger);
    xdr_route_orders[entry_slot] = CoordY(entry_ranger);
}

function XDR_BattleClearance(dword clearance_star, dword clearance_ignore_ship, int clearance_x, int clearance_y)
{
    result = 30000;
    if(!clearance_star) exit;
    int clearance_ship_count = StarShips(clearance_star);
    int clearance_found = 0;
    for(int clearance_ship_index = 0; clearance_ship_index < clearance_ship_count; clearance_ship_index = clearance_ship_index + 1)
    {
        dword clearance_ship = StarShips(clearance_star, clearance_ship_index);
        if(!clearance_ship || clearance_ship == clearance_ignore_ship) continue;
        if(ShipTypeN(clearance_ship) >= t_RC) continue;
        if(ShipInHyperSpace(clearance_ship, 1) || ShipIsTakeoff(clearance_ship) || !ShipInNormalSpace(clearance_ship)) continue;
        int clearance_combatant = ShipGetBad(clearance_ship) != 0;
        int clearance_owner = ShipOwner(clearance_ship);
        if(clearance_owner == 5) clearance_combatant = 1;
        if(clearance_owner >= 0 && clearance_owner <= 4 && ShipStanding(clearance_ship) == 2) clearance_combatant = 1;
        if(!clearance_combatant) continue;
        int clearance_distance = DistCoords(clearance_x, clearance_y, CoordX(clearance_ship), CoordY(clearance_ship));
        if(!clearance_found || clearance_distance < result) result = clearance_distance;
        clearance_found = 1;
    }
}

function XDR_OrderBattleWait(int wait_slot, dword wait_ranger)
{
    if(wait_slot < 1 || wait_slot > 3) exit;
    if(!wait_ranger) exit;
    if(ShipInHyperSpace(wait_ranger, 1)) exit;
    if(ShipIsTakeoff(wait_ranger)) exit;
    if(!ShipInNormalSpace(wait_ranger)) exit;
    if(ArrayDim(xdr_route_slots) != 4) exit;
    if(ArrayDim(xdr_route_orders) != 4) exit;
    int wait_x = xdr_route_slots[wait_slot];
    int wait_y = xdr_route_orders[wait_slot];
    if(!wait_x && !wait_y)
    {
        // Compatibility fallback for an old save already inside state 5/16:
        // ask the engine for one genuine jump point. Unlike the old code this
        // never rewrites OrderJump repeatedly while comparing arbitrary exits.
        wait_x = CoordX(wait_ranger);
        wait_y = CoordY(wait_ranger);
        dword wait_star = ShipStar(wait_ranger);
        if(wait_star)
        {
            int wait_nearby_count = min(24, GalaxyStars() - 1);
            for(int wait_nearby_index = 1; wait_nearby_index <= wait_nearby_count; wait_nearby_index = wait_nearby_index + 1)
            {
                dword wait_guide_star = StarNearbyStars(wait_star, wait_nearby_index);
                if(!wait_guide_star || wait_guide_star == wait_star) continue;
                if(!ShipCanJump(wait_ranger, wait_star, wait_guide_star)) continue;
                OrderJump(wait_ranger, wait_guide_star);
                if(ShipOrder(wait_ranger) == 3 && ShipOrderObj(wait_ranger) == wait_guide_star)
                {
                    int wait_jump_x = ShipOrder(wait_ranger, 'X');
                    int wait_jump_y = ShipOrder(wait_ranger, 'Y');
                    if(wait_jump_x || wait_jump_y)
                    {
                        wait_x = wait_jump_x;
                        wait_y = wait_jump_y;
                    }
                }
                break;
            }
        }
        xdr_route_slots[wait_slot] = wait_x;
        xdr_route_orders[wait_slot] = wait_y;
    }
    int wait_point_radius = DistCoords(wait_x, wait_y, 0, 0);
    if(wait_point_radius < 450)
    {
        int wait_direction_x = wait_x;
        int wait_direction_y = wait_y;
        if(!wait_direction_x && !wait_direction_y)
        {
            wait_direction_x = CoordX(wait_ranger);
            wait_direction_y = CoordY(wait_ranger);
        }
        int wait_direction_radius = DistCoords(wait_direction_x, wait_direction_y, 0, 0);
        if(wait_direction_radius > 0)
        {
            wait_x = round(wait_direction_x * 500 / wait_direction_radius);
            wait_y = round(wait_direction_y * 500 / wait_direction_radius);
        }
        else
        {
            wait_x = 500;
            wait_y = 0;
        }
        xdr_route_slots[wait_slot] = wait_x;
        xdr_route_orders[wait_slot] = wait_y;
    }
    if(ShipGetBad(wait_ranger)) ShipSetBad(wait_ranger, 0);
    if(DistCoords(CoordX(wait_ranger), CoordY(wait_ranger), wait_x, wait_y) > 250)
        OrderMove(wait_ranger, wait_x, wait_y);
    else
        OrderNone(wait_ranger);
}

function XDR_ApplyMask(int mask_slot, dword mask_ranger, dword mask_star)
{
    result = 0;
    if(!mask_ranger || !mask_star) exit;
    if(xdr_test_mode && xdr_test_force_mask_failure) exit;
    result = XDR_NativeSetMask(mask_ranger, xdr_series[mask_slot], xdr_ghost_flags[mask_slot]);
}

function XDR_ApplyVisualIdentity(int visual_slot, dword visual_ranger, dword visual_star)
{
    if(!visual_ranger || !visual_star) exit;
    str visual_graph = '';
    str visual_name = '';
    str visual_fallback_graph = '';
    str visual_fallback_name = '';
    int visual_ranger_hull_size = 0;
    int visual_best_size_delta = 2000000000;
    dword visual_ranger_hull = ShipEqInSlot(visual_ranger, t_Hull);
    if(visual_ranger_hull) visual_ranger_hull_size = ItemSize(visual_ranger_hull);
    int visual_ship_count = StarShips(visual_star);
    for(int visual_ship_index = 0; visual_ship_index < visual_ship_count; visual_ship_index = visual_ship_index + 1)
    {
        dword visual_dom = StarShips(visual_star, visual_ship_index);
        if(!visual_dom || visual_dom == visual_ranger) continue;
        if(ShipTypeN(visual_dom) >= t_RC) continue;
        if(ShipOwner(visual_dom) != 5) continue;
        if(!ShipInNormalSpace(visual_dom) || ShipIsTakeoff(visual_dom) || ShipInHyperSpace(visual_dom, 1)) continue;
        str visual_candidate_graph = Chameleon(visual_dom, 'GraphName');
        if(visual_candidate_graph == '') continue;
        if(visual_fallback_graph == '')
        {
            visual_fallback_graph = visual_candidate_graph;
            visual_fallback_name = ShipName(visual_dom);
        }
        dword visual_dom_hull = ShipEqInSlot(visual_dom, t_Hull);
        if(!visual_dom_hull) continue;
        if(ShipSubrace(visual_dom, -1) != xdr_series[visual_slot]) continue;
        int visual_dom_hull_size = ItemSize(visual_dom_hull);
        int visual_size_delta = visual_dom_hull_size - visual_ranger_hull_size;
        if(visual_size_delta < 0) visual_size_delta = -visual_size_delta;
        if(visual_graph == '' || visual_size_delta < visual_best_size_delta)
        {
            visual_best_size_delta = visual_size_delta;
            visual_graph = visual_candidate_graph;
            visual_name = ShipName(visual_dom);
        }
    }
    if(visual_graph == '')
    {
        visual_graph = visual_fallback_graph;
        visual_name = visual_fallback_name;
    }
    if(visual_graph == '') exit;

    // Never request a synchronous StarMap re-entry here. Doing that from a turn
    // callback skips visible ship animations and can leave two renderer copies.
    // The outbound state applies this identity while the ranger is in hyper.
    if(IsChameleon(visual_ranger)) Chameleon(visual_ranger);
    Chameleon(visual_ranger, visual_graph);
    if(visual_name != '') SetName(visual_ranger, visual_name);
    xdr_visual_flags[visual_slot] = 1;
}

function XDR_ArmMaskForJump(int arm_slot, dword arm_ranger, dword arm_star)
{
    result = 0;
    if(!arm_ranger || !arm_star) exit;
    if(ShipInHyperSpace(arm_ranger, 1)) exit;
    if(ShipIsTakeoff(arm_ranger)) exit;
    if(!ShipInNormalSpace(arm_ranger)) exit;
    if(ShipGetBad(arm_ranger)) ShipSetBad(arm_ranger, 0);
    if(!XDR_EnableGhost(arm_slot, arm_ranger)) exit;
    if(!XDR_ReassertGhost(arm_slot, arm_ranger)) exit;
    if(!XDR_ApplyMask(arm_slot, arm_ranger, arm_star)) exit;
    result = 1;
}

function XDR_RefreshMaskAfterJump(int refresh_slot, dword refresh_ranger, dword refresh_star)
{
    result = 0;
    if(!refresh_ranger || !refresh_star) exit;
    if(ShipGetBad(refresh_ranger)) ShipSetBad(refresh_ranger, 0);
    if(!XDR_ReassertGhost(refresh_slot, refresh_ranger)) exit;
    if(!XDR_ApplyMask(refresh_slot, refresh_ranger, refresh_star)) exit;
    if(!IsChameleon(refresh_ranger)) xdr_visual_flags[refresh_slot] = 0;
    if(!xdr_visual_flags[refresh_slot]) XDR_ApplyVisualIdentity(refresh_slot, refresh_ranger, refresh_star);
    XDR_KeepDefensiveTargetOnly(refresh_ranger);
    result = 1;
}

function XDR_RestoreMask(int restore_slot, dword restore_ranger)
{
    if(!restore_ranger)
    {
        xdr_ghost_flags[restore_slot] = 0;
        xdr_visual_flags[restore_slot] = 0;
        exit;
    }
    XDR_DisableGhostOnly(restore_slot, restore_ranger);
    if(xdr_visual_flags[restore_slot] || IsChameleon(restore_ranger)) Chameleon(restore_ranger);
    if(xdr_old_names[restore_slot] != '') SetName(restore_ranger, xdr_old_names[restore_slot]);
    if(!ShipInHyperSpace(restore_ranger, 1) && !ShipIsTakeoff(restore_ranger) && ShipInNormalSpace(restore_ranger))
    {
        if(ShipGetBad(restore_ranger)) ShipSetBad(restore_ranger, 0);
    }
    xdr_visual_flags[restore_slot] = 0;
    xdr_threat_counts[restore_slot] = 0;
    xdr_old_names[restore_slot] = '';
    xdr_old_factions[restore_slot] = '';
}

function XDR_CancelPickup(int pickup_slot, dword pickup_ship)
{
    if(xdr_pending_item_ids[pickup_slot] > 1)
    {
        dword pickup_item = IdToItem(xdr_pending_item_ids[pickup_slot]);
        if(pickup_item && pickup_ship) ShipPicksItem(pickup_ship, pickup_item, 0);
    }
    xdr_pending_item_ids[pickup_slot] = 0;
    xdr_pending_since[pickup_slot] = 0;
}

function XDR_StartMarkedPickup(int marked_pickup_slot, dword marked_pickup_ship, dword marked_pickup_item, int marked_pickup_force_logic)
{
    if(marked_pickup_slot < 1 || marked_pickup_slot > 3) exit;
    if(!marked_pickup_ship || !marked_pickup_item) exit;
    if(ShipInHyperSpace(marked_pickup_ship, 1)) exit;
    if(ShipIsTakeoff(marked_pickup_ship)) exit;
    if(!ShipInNormalSpace(marked_pickup_ship)) exit;

    dword marked_pickup_star = ShipStar(marked_pickup_ship);
    if(!marked_pickup_star) exit;
    if(!ItemIn(marked_pickup_item, marked_pickup_star)) exit;
    if(ItemInScript(marked_pickup_item) == 1) exit;
    if(!XDR_IsSafeSalvageItem(marked_pickup_item)) exit;

    dword marked_pickup_hook = ShipEqInSlot(marked_pickup_ship, t_CargoHook);
    if(!marked_pickup_hook) exit;
    int marked_pickup_size = ItemSize(marked_pickup_item);
    if(marked_pickup_size > ShipFreeSpace(marked_pickup_ship)) exit;
    if(marked_pickup_size > GetEquipmentStats(marked_pickup_hook, 0)) exit;

    XDR_SuppressMaskedOffense(marked_pickup_ship, marked_pickup_star);
    int marked_pickup_x = CoordX(marked_pickup_item);
    int marked_pickup_y = CoordY(marked_pickup_item);
    int marked_pickup_distance = DistCoords(CoordX(marked_pickup_ship), CoordY(marked_pickup_ship), marked_pickup_x, marked_pickup_y);
    int marked_pickup_capture_distance = min(120, max(40, GetEquipmentStats(marked_pickup_hook, 1) / 8));
    if(marked_pickup_distance <= marked_pickup_capture_distance)
    {
        // Keep the mission lock while the tractor is working. The old code ran
        // ShipFreeFlight here, which synchronously woke the vanilla ranger AI:
        // it could replace the salvage order with a flight to the system edge
        // or acquire the player/Dominators as a combat target.
        int marked_pickup_start_turn = xdr_pending_since[marked_pickup_slot];
        if(marked_pickup_start_turn < 0) marked_pickup_start_turn = -marked_pickup_start_turn - 1;
        if(xdr_pending_since[marked_pickup_slot] >= 0)
        {
            if(ShipGetBad(marked_pickup_ship)) ShipSetBad(marked_pickup_ship, 0);
            ShipPicksItem(marked_pickup_ship, marked_pickup_item, 1);
            OrderMove(marked_pickup_ship, marked_pickup_x, marked_pickup_y);
            xdr_pending_since[marked_pickup_slot] = -CurTurn() - 1;
            exit;
        }
        if(CurTurn() == marked_pickup_start_turn) exit;
        if(CurTurn() - marked_pickup_start_turn <= 1)
        {
            if(ShipGetBad(marked_pickup_ship)) ShipSetBad(marked_pickup_ship, 0);
            ShipPicksItem(marked_pickup_ship, marked_pickup_item, 1);
            OrderMove(marked_pickup_ship, marked_pickup_x, marked_pickup_y);
            exit;
        }

        // A broken or unsuitable class-AI pickup must not deadlock the whole
        // mission. Clear the vanilla marker before the bounded direct fallback.
        ShipPicksItem(marked_pickup_ship, marked_pickup_item, 0);
        dword marked_pickup_taken = 0;
        if(ItemIn(marked_pickup_item, marked_pickup_star))
            marked_pickup_taken = GetItemFromStar(marked_pickup_star, marked_pickup_item);
        if(marked_pickup_taken)
        {
            AddItemToShip(marked_pickup_ship, marked_pickup_taken);
            XDR_RecordScavengedFuel(marked_pickup_slot, marked_pickup_taken);
            xdr_diag_direct_pickups = xdr_diag_direct_pickups + 1;
        }
        exit;
    }

    int marked_pickup_run_logic = marked_pickup_force_logic;
    if(ShipOrder(marked_pickup_ship) == 0) marked_pickup_run_logic = 1;
    if(ShipTurnBeforeEndOrder(marked_pickup_ship) <= 0) marked_pickup_run_logic = 1;
    if(ShipSpeed(marked_pickup_ship) <= 0) marked_pickup_run_logic = 1;
    if(ShipOrder(marked_pickup_ship) == 1)
    {
        int marked_pickup_order_distance = DistCoords(ShipOrder(marked_pickup_ship, 'X'), ShipOrder(marked_pickup_ship, 'Y'), marked_pickup_x, marked_pickup_y);
        if(marked_pickup_order_distance > 30) marked_pickup_run_logic = 1;
    }
    if(!marked_pickup_run_logic) exit;
    if(ShipGetBad(marked_pickup_ship)) ShipSetBad(marked_pickup_ship, 0);
    ShipPicksItem(marked_pickup_ship, marked_pickup_item, 1);
    // ShipPicksItem only marks the tractor target. Navigation remains an
    // explicit mission order so the class AI is never allowed to improvise.
    int marked_pickup_has_move_order = 0;
    if(ShipOrder(marked_pickup_ship) == 1)
    {
        int marked_pickup_final_order_distance = DistCoords(ShipOrder(marked_pickup_ship, 'X'), ShipOrder(marked_pickup_ship, 'Y'), marked_pickup_x, marked_pickup_y);
        if(marked_pickup_final_order_distance <= 30) marked_pickup_has_move_order = 1;
    }
    if(!marked_pickup_has_move_order) OrderMove(marked_pickup_ship, marked_pickup_x, marked_pickup_y);
}

function XDR_IsEjectedFuelItem(int ejected_slot, dword ejected_item)
{
    result = 0;
    if(ejected_slot < 1 || ejected_slot > 3 || !ejected_item) exit;
    if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66) exit;
    int ejected_item_id = Id(ejected_item);
    for(int ejected_index = 1; ejected_index <= 64; ejected_index = ejected_index + 1)
    {
        if(xdr_scavenged_fuel_slots[ejected_index] != ejected_slot) continue;
        if(xdr_scavenged_fuel_item_ids[ejected_index] != ejected_item_id) continue;
        result = 1;
        exit;
    }
}

function XDR_SelectLoot(dword loot_ship, dword loot_star, int fuel_only)
{
    result = 0;
    if(!loot_ship || !loot_star) exit;
    int loot_slot = XDR_SlotByShip(loot_ship);
    dword loot_hook = ShipEqInSlot(loot_ship, t_CargoHook);
    if(!loot_hook) exit;
    int loot_hook_size = GetEquipmentStats(loot_hook, 0);
    int loot_cost_floor = XDR_MinLootCost(loot_ship);
    dword loot_installed_tank = ShipEqInSlot(loot_ship, t_FuelTanks);
    int loot_fuel_room = 0;
    if(loot_installed_tank) loot_fuel_room = max(0, GetEquipmentStats(loot_installed_tank, 0) - ShipFuel(loot_ship));
    int loot_only_fuel = fuel_only == 1;
    int loot_ignore_cisterns = fuel_only == 2;
    int loot_best_priority = -1;
    int loot_best_score = -1;
    for(int loot_index = 0; loot_index < StarItems(loot_star); loot_index = loot_index + 1)
    {
        dword loot_item = StarItems(loot_star, loot_index);
        if(!loot_item || !ItemIn(loot_item, loot_star)) continue;
        if(loot_slot > 0 && XDR_IsEjectedFuelItem(loot_slot, loot_item)) continue;
        if(ItemInScript(loot_item) == 1) continue;
        if(!XDR_IsSafeSalvageItem(loot_item)) continue;
        int loot_size = ItemSize(loot_item);
        if(loot_size > ShipFreeSpace(loot_ship) || loot_size > loot_hook_size) continue;
        int loot_type = ItemType(loot_item);
        int loot_is_usable_cistern = 0;
        if(loot_type == t_Cistern || loot_type == t_FuelTanks)
        {
            if(ContainerFuel(loot_item) > 0 && loot_fuel_room > 0) loot_is_usable_cistern = 1;
        }
        // Mode 2 means the complete return route is already covered by the
        // installed fuel. Disposable cisterns are then ignored completely;
        // an expensive equipment tank remains ordinary valuable cargo.
        if(loot_ignore_cisterns && loot_type == t_Cistern) continue;
        if(loot_ignore_cisterns && loot_type == t_FuelTanks) loot_is_usable_cistern = 0;
        if(loot_only_fuel && !loot_is_usable_cistern) continue;
        if(loot_type == t_Cistern && !loot_is_usable_cistern) continue;
        if(!loot_is_usable_cistern && ItemCost(loot_item) < loot_cost_floor) continue;
        int loot_priority = 1;
        int loot_score = ItemCost(loot_item);
        if(loot_is_usable_cistern)
        {
            loot_priority = 3;
            loot_score = ContainerFuel(loot_item);
        }
        else if(ItemOwner(loot_item) == 5 || loot_type == t_UselessItem) loot_priority = 2;
        if(loot_priority > loot_best_priority || (loot_priority == loot_best_priority && loot_score > loot_best_score))
        {
            loot_best_priority = loot_priority;
            loot_best_score = loot_score;
            result = loot_item;
        }
    }
}

function XDR_SelectBattleEdgeLoot(int edge_slot, dword edge_ship, dword edge_star, int edge_ignore_cisterns)
{
    result = 0;
    if(edge_slot < 1 || edge_slot > 3) exit;
    if(!edge_ship || !edge_star) exit;
    if(ArrayDim(xdr_route_slots) != 4) exit;
    if(ArrayDim(xdr_route_orders) != 4) exit;
    int edge_wait_x = xdr_route_slots[edge_slot];
    int edge_wait_y = xdr_route_orders[edge_slot];
    if(!edge_wait_x && !edge_wait_y) exit;
    // If the battle has drifted onto the waiting point, do not start a loot
    // run at all. The point remains the least dangerous bounded position, but
    // scavenging pauses until the fighting moves away.
    if(XDR_BattleClearance(edge_star, edge_ship, edge_wait_x, edge_wait_y) < 1600) exit;
    dword edge_hook = ShipEqInSlot(edge_ship, t_CargoHook);
    if(!edge_hook) exit;
    int edge_hook_size = GetEquipmentStats(edge_hook, 0);
    int edge_cost_floor = XDR_MinLootCost(edge_ship);
    dword edge_installed_tank = ShipEqInSlot(edge_ship, t_FuelTanks);
    int edge_fuel_room = 0;
    if(edge_installed_tank) edge_fuel_room = max(0, GetEquipmentStats(edge_installed_tank, 0) - ShipFuel(edge_ship));
    int edge_wait_radius = DistCoords(edge_wait_x, edge_wait_y, 0, 0);
    int edge_best_priority = -1;
    int edge_best_score = -1;
    int edge_item_count = StarItems(edge_star);
    for(int edge_item_index = 0; edge_item_index < edge_item_count; edge_item_index = edge_item_index + 1)
    {
        dword edge_item = StarItems(edge_star, edge_item_index);
        if(!edge_item) continue;
        if(!ItemIn(edge_item, edge_star)) continue;
        if(XDR_IsEjectedFuelItem(edge_slot, edge_item)) continue;
        if(ItemInScript(edge_item) == 1) continue;
        if(!XDR_IsSafeSalvageItem(edge_item)) continue;
        int edge_item_x = CoordX(edge_item);
        int edge_item_y = CoordY(edge_item);
        // Collect only debris close to the holding point and no more than 500
        // units inward toward the central battle cluster.
        if(DistCoords(edge_item_x, edge_item_y, edge_wait_x, edge_wait_y) > 800) continue;
        if(XDR_BattleClearance(edge_star, edge_ship, edge_item_x, edge_item_y) < 1400) continue;
        int edge_item_radius = DistCoords(edge_item_x, edge_item_y, 0, 0);
        if(edge_item_radius + 500 < edge_wait_radius) continue;
        int edge_item_size = ItemSize(edge_item);
        if(edge_item_size > ShipFreeSpace(edge_ship) || edge_item_size > edge_hook_size) continue;
        int edge_item_type = ItemType(edge_item);
        int edge_usable_cistern = 0;
        if(edge_item_type == t_Cistern || edge_item_type == t_FuelTanks)
        {
            if(ContainerFuel(edge_item) > 0 && edge_fuel_room > 0) edge_usable_cistern = 1;
        }
        if(edge_ignore_cisterns && edge_item_type == t_Cistern) continue;
        if(edge_ignore_cisterns && edge_item_type == t_FuelTanks) edge_usable_cistern = 0;
        if(edge_item_type == t_Cistern && !edge_usable_cistern) continue;
        if(!edge_usable_cistern && ItemCost(edge_item) < edge_cost_floor) continue;
        int edge_priority = 1;
        int edge_score = ItemCost(edge_item);
        if(edge_usable_cistern)
        {
            edge_priority = 3;
            edge_score = ContainerFuel(edge_item);
        }
        else if(ItemOwner(edge_item) == 5 || edge_item_type == t_UselessItem) edge_priority = 2;
        if(edge_priority > edge_best_priority || (edge_priority == edge_best_priority && edge_score > edge_best_score))
        {
            edge_best_priority = edge_priority;
            edge_best_score = edge_score;
            result = edge_item;
        }
    }
}

function XDR_ResetScavengedFuelArrays()
{
    // Fixed arrays avoid live ArrayDim drift after save/load. Slot 65 is an
    // unused sentinel; mission records occupy indices 1 through 64.
    xdr_scavenged_fuel_slots = newarray(66);
    xdr_scavenged_fuel_item_ids = newarray(66);
    for(int fuel_reset_index = 0; fuel_reset_index <= 65; fuel_reset_index = fuel_reset_index + 1)
    {
        xdr_scavenged_fuel_slots[fuel_reset_index] = 0;
        xdr_scavenged_fuel_item_ids[fuel_reset_index] = 0;
    }
}

function XDR_ClearScavengedFuelTracking(int fuel_clear_slot)
{
    if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66)
    {
        XDR_ResetScavengedFuelArrays();
        exit;
    }
    for(int fuel_clear_index = 1; fuel_clear_index <= 64; fuel_clear_index = fuel_clear_index + 1)
    {
        int fuel_clear_marker = xdr_scavenged_fuel_slots[fuel_clear_index];
        if(fuel_clear_marker != fuel_clear_slot && fuel_clear_marker != -fuel_clear_slot && fuel_clear_marker != fuel_clear_slot + 10) continue;
        xdr_scavenged_fuel_slots[fuel_clear_index] = 0;
        xdr_scavenged_fuel_item_ids[fuel_clear_index] = 0;
    }
}

function XDR_SetReturnFuelNeed(int fuel_need_slot, int fuel_need_value)
{
    if(fuel_need_slot < 1 || fuel_need_slot > 3) exit;
    if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66) XDR_ResetScavengedFuelArrays();
    int fuel_need_marker = fuel_need_slot + 10;
    int fuel_need_free_index = 0;
    for(int fuel_need_index = 1; fuel_need_index <= 64; fuel_need_index = fuel_need_index + 1)
    {
        if(xdr_scavenged_fuel_slots[fuel_need_index] == fuel_need_marker)
        {
            if(fuel_need_value < 0)
            {
                xdr_scavenged_fuel_slots[fuel_need_index] = 0;
                xdr_scavenged_fuel_item_ids[fuel_need_index] = 0;
            }
            else xdr_scavenged_fuel_item_ids[fuel_need_index] = fuel_need_value + 1;
            exit;
        }
        if(!fuel_need_free_index && xdr_scavenged_fuel_slots[fuel_need_index] == 0) fuel_need_free_index = fuel_need_index;
    }
    if(fuel_need_value < 0 || fuel_need_free_index <= 0) exit;
    xdr_scavenged_fuel_slots[fuel_need_free_index] = fuel_need_marker;
    xdr_scavenged_fuel_item_ids[fuel_need_free_index] = fuel_need_value + 1;
}

function XDR_GetReturnFuelNeed(int fuel_get_slot)
{
    result = -1;
    if(fuel_get_slot < 1 || fuel_get_slot > 3) exit;
    if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66) exit;
    int fuel_get_marker = fuel_get_slot + 10;
    for(int fuel_get_index = 1; fuel_get_index <= 64; fuel_get_index = fuel_get_index + 1)
    {
        if(xdr_scavenged_fuel_slots[fuel_get_index] != fuel_get_marker) continue;
        result = xdr_scavenged_fuel_item_ids[fuel_get_index] - 1;
        exit;
    }
}

function XDR_RecordDisplacedLoot(int displaced_slot, dword displaced_item)
{
    result = 0;
    if(displaced_slot < 1 || displaced_slot > 3 || !displaced_item) exit;
    if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66) XDR_ResetScavengedFuelArrays();
    int displaced_item_id = Id(displaced_item);
    if(displaced_item_id <= 1) exit;
    int displaced_free_index = 0;
    for(int displaced_index = 1; displaced_index <= 64; displaced_index = displaced_index + 1)
    {
        if(xdr_scavenged_fuel_slots[displaced_index] == -displaced_slot && xdr_scavenged_fuel_item_ids[displaced_index] == displaced_item_id)
        {
            result = 1;
            exit;
        }
        if(!displaced_free_index && xdr_scavenged_fuel_slots[displaced_index] == 0) displaced_free_index = displaced_index;
    }
    if(displaced_free_index <= 0) exit;
    xdr_scavenged_fuel_slots[displaced_free_index] = -displaced_slot;
    xdr_scavenged_fuel_item_ids[displaced_free_index] = displaced_item_id;
    result = 1;
}

        xdr_scavenged_fuel_item_ids[displaced_clear_index] = 0;

}

function XDR_HasDisplacedLoot(int displaced_has_slot, dword displaced_has_ship, dword displaced_has_star)
{

    if(displaced_has_slot < 1 || displaced_has_slot > 3 || !displaced_has_ship || !displaced_has_star) exit;
    if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66) exit;
    for(int displaced_has_index = 1; displaced_has_index <= 64; displaced_has_index = displaced_has_index + 1)
    {
        if(xdr_scavenged_fuel_slots[displaced_has_index] != -displaced_has_slot) continue;

            xdr_scavenged_fuel_slots[displaced_has_index] = 0;
            xdr_scavenged_fuel_item_ids[displaced_has_index] = 0;

    {

function XDR_SelectFuelCistern(int cistern_slot, dword cistern_ship, dword cistern_star)
{
    result = 0;
    if(cistern_slot < 1 || cistern_slot > 3 || !cistern_ship || !cistern_star) exit;
    dword cistern_tank = ShipEqInSlot(cistern_ship, t_FuelTanks);
    dword cistern_hook = ShipEqInSlot(cistern_ship, t_CargoHook);
    if(!cistern_tank || !cistern_hook) exit;
    int cistern_room = max(0, GetEquipmentStats(cistern_tank, 0) - ShipFuel(cistern_ship));
    if(cistern_room <= 0) exit;
    int cistern_hook_size = GetEquipmentStats(cistern_hook, 0);
    int cistern_best_fuel = -1;
    for(int cistern_index = 0; cistern_index < StarItems(cistern_star); cistern_index = cistern_index + 1)
    {
        dword cistern_item = StarItems(cistern_star, cistern_index);
        if(!cistern_item || !ItemIn(cistern_item, cistern_star)) continue;
        if(XDR_IsEjectedFuelItem(cistern_slot, cistern_item)) continue;
        if(ItemInScript(cistern_item) == 1 || ItemType(cistern_item) != t_Cistern) continue;
        if(!XDR_IsSafeSalvageItem(cistern_item)) continue;
        if(ItemSize(cistern_item) > cistern_hook_size) continue;
        int cistern_fuel = ContainerFuel(cistern_item);
        if(cistern_fuel <= 0 || cistern_fuel <= cistern_best_fuel) continue;
        cistern_best_fuel = cistern_fuel;
        result = cistern_item;
    }
}

function XDR_EnsureFuelPickupSpace(int space_slot, dword space_ship, dword space_star, dword space_cistern)
{
    result = 0;
    if(space_slot < 1 || space_slot > 3 || !space_ship || !space_star || !space_cistern) exit;
    if(!ItemIn(space_cistern, space_star)) exit;
    int space_required = ItemSize(space_cistern) - ShipFreeSpace(space_ship);
    if(space_required <= 0)
    {
        result = 1;
        exit;
    }
    dword space_best_item = 0;
    int space_best_finishes = -1;
    int space_best_score = 2000000000;
    int space_best_size = 0;
    int space_item_count = ShipItems(space_ship);
    for(int space_item_index = 1; space_item_index < space_item_count; space_item_index = space_item_index + 1)
    {
        dword space_item = ShipItems(space_ship, space_item_index);
        if(!space_item || !ItemIn(space_item, space_ship)) continue;
        if(ItemIsInUse(space_item) || ItemInScript(space_item) == 1) continue;
        int space_item_size = ItemSize(space_item);
        if(space_item_size <= 0) continue;
        int space_item_finishes = space_item_size >= space_required;
        int space_item_score = ItemCost(space_item);
        if(!space_item_finishes) space_item_score = space_item_score / max(1, space_item_size);
        if(space_item_finishes < space_best_finishes) continue;
        if(space_item_finishes == space_best_finishes && space_item_score > space_best_score) continue;
        if(space_item_finishes == space_best_finishes && space_item_score == space_best_score && space_item_size <= space_best_size) continue;
        space_best_item = space_item;
        space_best_finishes = space_item_finishes;
        space_best_score = space_item_score;
        space_best_size = space_item_size;
    }
    if(!space_best_item) exit;
    if(!XDR_RecordDisplacedLoot(space_slot, space_best_item)) exit;
    int space_best_item_id = Id(space_best_item);
    dword space_removed_item = GetItemFromShip(space_ship, space_best_item);
    if(!space_removed_item)
    {
        XDR_ClearDisplacedLootId(space_slot, space_best_item_id);
        exit;
    }
    ShipCalcParam(space_ship);
    int space_drop_x = CoordX(space_ship) - 70;
    int space_drop_y = CoordY(space_ship) + 40;
    DropItemInSystem(space_star, space_removed_item, space_drop_x, space_drop_y, space_drop_x, space_drop_y);
    StopMovingItem(space_star, space_removed_item);
    result = -1;
}

function XDR_RecordScavengedFuel(int fuel_record_slot, dword fuel_record_item)
{
    if(!fuel_record_item) exit;
    int fuel_record_type = ItemType(fuel_record_item);
    if(fuel_record_type != t_Cistern && fuel_record_type != t_FuelTanks) exit;
    if(fuel_record_slot < 1 || fuel_record_slot > 3) exit;
    dword fuel_record_ship = IdToShip(xdr_ids[fuel_record_slot]);
    if(!fuel_record_ship || !ItemIn(fuel_record_item, fuel_record_ship)) exit;
    if(ShipInHyperSpace(fuel_record_ship, 1)) exit;
    if(ShipIsTakeoff(fuel_record_ship)) exit;
    if(!ShipInNormalSpace(fuel_record_ship)) exit;
    dword fuel_record_star = ShipStar(fuel_record_ship);
    if(!fuel_record_star) exit;
    int fuel_record_transferred = 0;
    dword fuel_record_tank = ShipEqInSlot(fuel_record_ship, t_FuelTanks);
    if(fuel_record_tank)
    {
        int fuel_record_available = ContainerFuel(fuel_record_item);
        int fuel_record_room = max(0, GetEquipmentStats(fuel_record_tank, 0) - ShipFuel(fuel_record_ship));
        int fuel_record_amount = min(fuel_record_available, fuel_record_room);
        if(fuel_record_amount > 0)
        {
            ShipFuel(fuel_record_ship, ShipFuel(fuel_record_ship) + fuel_record_amount);
            ContainerFuel(fuel_record_item, fuel_record_available - fuel_record_amount);
            fuel_record_transferred = fuel_record_amount;
        }
    }
    // One positive transfer marks the fuel-service phase as complete for the
    // ordinary salvage pass. The stranded recovery state is still allowed to
    // request additional fuel explicitly.
    if(fuel_record_transferred > 0)
    {
        if(ArrayDim(xdr_scavenged_fuel_slots) != 66 || ArrayDim(xdr_scavenged_fuel_item_ids) != 66) XDR_ResetScavengedFuelArrays();
        int fuel_record_free_index = 0;
        for(int fuel_record_index = 1; fuel_record_index <= 64; fuel_record_index = fuel_record_index + 1)
        {
            if(xdr_scavenged_fuel_slots[fuel_record_index] != 0) continue;
            fuel_record_free_index = fuel_record_index;
            break;
        }
        if(fuel_record_free_index > 0)
        {
            xdr_scavenged_fuel_slots[fuel_record_free_index] = fuel_record_slot;
            xdr_scavenged_fuel_item_ids[fuel_record_free_index] = Id(fuel_record_item);
        }
    }
    // A disposable cistern is ejected even when fuel remains. Equipment fuel
    // tanks stay in the hold as ordinary valuable loot for the vanilla AI.
    dword fuel_record_discard = 0;
    if(fuel_record_type == t_Cistern && ItemIn(fuel_record_item, fuel_record_ship)) fuel_record_discard = GetItemFromShip(fuel_record_ship, fuel_record_item);
    if(fuel_record_discard)
    {
        int fuel_record_drop_x = CoordX(fuel_record_ship) - 40;
        int fuel_record_drop_y = CoordY(fuel_record_ship) + 20;
        DropItemInSystem(fuel_record_star, fuel_record_discard, fuel_record_drop_x, fuel_record_drop_y, fuel_record_drop_x, fuel_record_drop_y);

    }

function XDR_ReturnTransitAllowed(int transit_slot, dword transit_star)
{
    result = 0;
    if(!transit_star) exit;
    if(!XDR_IsStrategicStar(transit_star)) exit;
    int transit_owner = StarOwner(transit_star);
    if(transit_owner == 1)
    {
        if(XDR_IsCoalitionAssault(transit_star)) exit;
        if(StarSeries(transit_star) == xdr_series[transit_slot]) result = 1;
        exit;
    }
    if(StarBattle(transit_star)) exit;
    if(transit_owner == 0 || transit_owner == 2)
    {
        result = 1;
        exit;
    }
}

function XDR_SelectReturnPlace(int return_slot, dword return_ship, dword return_origin_star)
{
    result = 0;
    XDR_ClearRoute(return_slot);
    XDR_SetReturnFuelNeed(return_slot, -1);
    xdr_return_place_ids[return_slot] = 0;
    xdr_return_kinds[return_slot] = 0;
    if(!return_ship || !return_origin_star) exit;

    // Build a fresh shortest-path tree on every return decision. Transit is
    // allowed through safe Coalition/pirate systems and through Dominator
    // systems of the purchased camouflage series. Thus a newly captured
    // same-series system can become a valid escape corridor immediately.
    unknown return_galaxy_indices = newarray(1);
    unknown return_parent_indices = newarray(1);
    unknown return_seen_by_galaxy = newarray(1);
    ArrayClear(return_galaxy_indices);
    ArrayClear(return_parent_indices);
    ArrayClear(return_seen_by_galaxy);
    int return_galaxy_count = GalaxyStars();
    int return_source_index = -1;
    for(int return_source_probe = 0; return_source_probe < return_galaxy_count; return_source_probe = return_source_probe + 1)
    {
        ArrayAdd(return_seen_by_galaxy, 0);
        if(GalaxyStar(return_source_probe) == return_origin_star) return_source_index = return_source_probe;
    }
    if(return_source_index < 0) exit;
    ArrayAdd(return_galaxy_indices, return_source_index);
    ArrayAdd(return_parent_indices, 0);
    return_seen_by_galaxy[return_source_index + 1] = 1;

    int return_found_node = -1;
    int return_found_place_id = 0;
    int return_found_kind = 0;
    for(int return_pair_cursor = 0; return_pair_cursor < return_galaxy_count * return_galaxy_count; return_pair_cursor = return_pair_cursor + 1)
    {
        int return_cursor = return_pair_cursor / return_galaxy_count + 1;
        int return_candidate_index = return_pair_cursor - (return_cursor - 1) * return_galaxy_count;
        if(return_cursor >= ArrayDim(return_galaxy_indices)) continue;
        dword return_current_star = GalaxyStar(return_galaxy_indices[return_cursor]);
        if(!return_current_star) continue;
        if(return_candidate_index == 0)
        {
            dword return_current_base = XDR_UsablePirateBaseAtStar(return_current_star, return_ship);
            if(return_current_base)
            {
                return_found_node = return_cursor;
                return_found_place_id = Id(return_current_base);
                return_found_kind = 1;
                break;
            }
            dword return_current_planet = XDR_FindSafePlanet(return_current_star);
            if(return_current_planet)
            {
                return_found_node = return_cursor;
                return_found_place_id = Id(return_current_planet);
                return_found_kind = 2;
                break;
            }
        }
        dword return_candidate_star = GalaxyStar(return_candidate_index);
        if(!return_candidate_star) continue;
        if(return_candidate_star == return_origin_star || return_seen_by_galaxy[return_candidate_index + 1]) continue;
        if(!XDR_ReturnTransitAllowed(return_slot, return_candidate_star)) continue;
        // The fourth argument checks physical reach independently of the
        // currently loaded fuel. Actual fuel is checked immediately before
        // OrderJump in the state machine.
        if(!ShipCanJump(return_ship, return_current_star, return_candidate_star)) continue;
        return_seen_by_galaxy[return_candidate_index + 1] = 1;
        ArrayAdd(return_galaxy_indices, return_candidate_index);
        ArrayAdd(return_parent_indices, return_cursor);
    }
    if(return_found_node < 1 || return_found_place_id <= 1) exit;
    xdr_return_place_ids[return_slot] = return_found_place_id;
    xdr_return_kinds[return_slot] = return_found_kind;
    if(return_found_node == 1)
    {
        XDR_SetReturnFuelNeed(return_slot, 0);
        result = 1;
        exit;
    }

    int return_backtrack_node = return_found_node;
    int return_first_star_id = 0;
    int return_required_fuel = 0;
    for(int return_backtrack_step = 0; return_backtrack_step < return_galaxy_count; return_backtrack_step = return_backtrack_step + 1)
    {
        if(return_backtrack_node <= 1) break;
        dword return_backtrack_star = GalaxyStar(return_galaxy_indices[return_backtrack_node]);
        if(!return_backtrack_star) exit;
        int return_parent_node = return_parent_indices[return_backtrack_node];
        if(return_parent_node < 1) exit;
        dword return_parent_star = GalaxyStar(return_galaxy_indices[return_parent_node]);
        if(!return_parent_star) exit;
        return_required_fuel = return_required_fuel + round(Dist(return_backtrack_star, return_parent_star));
        return_first_star_id = Id(return_backtrack_star);
        return_backtrack_node = return_parent_node;
    }
    if(return_backtrack_node != 1 || return_first_star_id <= 1)
    {
        xdr_return_place_ids[return_slot] = 0;
        xdr_return_kinds[return_slot] = 0;
        exit;
    }
    XDR_SetNextRouteStarId(return_slot, return_first_star_id);
    XDR_SetReturnFuelNeed(return_slot, return_required_fuel);
    result = 1;
}

function XDR_HasFuelForReturn(int return_fuel_slot, dword return_fuel_ship, dword return_fuel_star)
{
    result = 0;
    if(return_fuel_slot < 1 || return_fuel_slot > 3 || !return_fuel_ship || !return_fuel_star) exit;
    int return_fuel_need = XDR_GetReturnFuelNeed(return_fuel_slot);
    if(return_fuel_need < 0)
    {
        if(!XDR_SelectReturnPlace(return_fuel_slot, return_fuel_ship, return_fuel_star)) exit;
        return_fuel_need = XDR_GetReturnFuelNeed(return_fuel_slot);
    }
    if(return_fuel_need < 0) exit;
    if(ShipFuel(return_fuel_ship) >= return_fuel_need) result = 1;
}

function XDR_PrepareNextLoot(int prepare_slot, dword prepare_ship, dword prepare_star)
{
    result = 0;
    if(prepare_slot < 1 || prepare_slot > 3 || !prepare_ship || !prepare_star) exit;
    if(xdr_pending_item_ids[prepare_slot] > 1) exit;
    int prepare_has_return_fuel = XDR_HasFuelForReturn(prepare_slot, prepare_ship, prepare_star);
    dword prepare_item = 0;
    if(!prepare_has_return_fuel)
    {
        dword prepare_cistern = XDR_SelectFuelCistern(prepare_slot, prepare_ship, prepare_star);
        if(prepare_cistern)
        {
            int prepare_space_result = XDR_EnsureFuelPickupSpace(prepare_slot, prepare_ship, prepare_star, prepare_cistern);
            if(prepare_space_result < 0)
            {
                xdr_no_loot_counts[prepare_slot] = 0;
                if(xdr_test_mode) xdr_test_last_event = 'для нужной цистерны временно выгружен дешёвый лут; после заправки он будет подобран обратно';
                result = 2;
                exit;
            }
            if(prepare_space_result <= 0) exit;
            prepare_item = prepare_cistern;
        }
    }
    if(!prepare_item) prepare_item = XDR_SelectDisplacedLoot(prepare_slot, prepare_ship, prepare_star);
    if(!prepare_item && XDR_HasDisplacedLoot(prepare_slot, prepare_ship, prepare_star)) exit;
    if(!prepare_item)
    {
        if(CurTurn() - xdr_phase_turns[prepare_slot] >= XDR_ConfigInt(13, 20) || ShipFreeSpace(prepare_ship) <= 0) exit;
        int prepare_loot_mode = 0;
        if(prepare_has_return_fuel) prepare_loot_mode = 2;
        prepare_item = XDR_SelectLoot(prepare_ship, prepare_star, prepare_loot_mode);
    }
    if(!prepare_item) exit;
    xdr_pending_item_ids[prepare_slot] = Id(prepare_item);
    xdr_pending_since[prepare_slot] = CurTurn();
    XDR_StartMarkedPickup(prepare_slot, prepare_ship, prepare_item, 1);
    result = 1;
}

function XDR_PrepareRecoveryFuel(int recovery_slot, dword recovery_ship, dword recovery_star)
{
    result = 0;
    if(recovery_slot < 1 || recovery_slot > 3 || !recovery_ship || !recovery_star) exit;
    if(xdr_pending_item_ids[recovery_slot] > 1) exit;
    int recovery_has_return_fuel = XDR_HasFuelForReturn(recovery_slot, recovery_ship, recovery_star);
    dword recovery_item = 0;
    if(!recovery_has_return_fuel)
    {
        dword recovery_cistern = XDR_SelectFuelCistern(recovery_slot, recovery_ship, recovery_star);
        if(recovery_cistern)
        {
            int recovery_space_result = XDR_EnsureFuelPickupSpace(recovery_slot, recovery_ship, recovery_star, recovery_cistern);
            if(recovery_space_result < 0)
            {
                if(xdr_test_mode) xdr_test_last_event = 'аварийная заправка освободила место под цистерну; выгруженный лут сохранён по ID';
                result = 2;
                exit;
            }
            if(recovery_space_result <= 0) exit;
            recovery_item = recovery_cistern;
        }
    }
    if(!recovery_item) recovery_item = XDR_SelectDisplacedLoot(recovery_slot, recovery_ship, recovery_star);
    if(!recovery_item) exit;
    xdr_pending_item_ids[recovery_slot] = Id(recovery_item);
    xdr_pending_since[recovery_slot] = CurTurn();
    XDR_StartMarkedPickup(recovery_slot, recovery_ship, recovery_item, 1);
    result = 1;
}

function XDR_DetachSlot(int detach_slot, dword detach_ship, int keep_specialist)
{
    if(detach_ship)
    {
        XDR_CancelPickup(detach_slot, detach_ship);
        XDR_RestoreMask(detach_slot, detach_ship);
    }
    XDR_ClearRoute(detach_slot);
    XDR_ClearBattleWait(detach_slot);
    xdr_target_star_ids[detach_slot] = 0;
    xdr_series[detach_slot] = -1;
    xdr_charge_expire_turns[detach_slot] = 0;
    xdr_return_place_ids[detach_slot] = 0;
    xdr_return_kinds[detach_slot] = 0;
    xdr_release_flags[detach_slot] = 0;
    XDR_ClearScavengedFuelTracking(detach_slot);
    if(keep_specialist)
    {
        xdr_states[detach_slot] = 11;
        // In state 11 this field is reused only as an optional money target.
        // Ordinary detach reasons have no additional earning requirement.
        xdr_phase_turns[detach_slot] = 0;
        // Vanilla AI may earn the missing money; retry eligibility every day.
        xdr_due_turns[detach_slot] = CurTurn() + 1;
    }
    else XDR_ResetSlotData(detach_slot, CurTurn() + XDR_ConfigInt(6, 1000));
    if(detach_ship && XDR_GroupContains(detach_ship)) ShipOut(detach_ship);
    if(detach_ship)
    {
        OrderLock(detach_ship, 0);
        ShipFreeFlight(detach_ship);
    }
}

function XDR_DetachAfterSpentCharge(int spent_slot, dword spent_ship)
{
    if(!spent_ship) { XDR_DetachSlot(spent_slot, spent_ship, 1); exit; }
    int spent_next_cost = XDR_CamouflageCost(spent_ship);
    int spent_retry_money = ShipMoney(spent_ship) + spent_next_cost;
    XDR_DetachSlot(spent_slot, spent_ship, 1);
    // The failed charge has really been consumed. Do not immediately buy
    // another one from a large existing wallet: the ranger must first earn
    // one additional camouflage price through vanilla activity.
    xdr_phase_turns[spent_slot] = spent_retry_money;
}

function XDR_HasGoodCargoHook(dword good_hook_ship)

function XDR_SlotEligible(dword eligible_ship)
{
    result = 0;
    if(!eligible_ship) exit;
    if(eligible_ship == Player() || ShipTypeN(eligible_ship) != t_Ranger) exit;
    if(ShipOnSidePirateClan(eligible_ship) || ShipIsPartner(eligible_ship) || ShipInGameEvent(eligible_ship) || ShipInScript(eligible_ship, 0)) exit;
    if(ShipInHyperSpace(eligible_ship, 1) || ShipIsTakeoff(eligible_ship)) exit;
    dword eligible_place = GetShipPlanet(eligible_ship);
    if(!eligible_place) eligible_place = GetShipRuins(eligible_ship);
    if(!eligible_place && !ShipInNormalSpace(eligible_ship)) exit;
    dword eligible_star = XDR_CurrentStar(eligible_ship);
    if(!eligible_star) exit;
    if(StarBattle(eligible_star)) exit;
    // A non-zero raw target is enough to reject the candidate; it is never
    // dereferenced because ShipGetBad may retain a stale engine handle.
    if(ShipGetBad(eligible_ship)) exit;
    if(!ShipEqInSlot(eligible_ship, t_Engine) || !ShipEqInSlot(eligible_ship, t_FuelTanks)) exit;
    if(!ShipEqInSlot(eligible_ship, t_CargoHook)) exit;
    if(!XDR_HullSupportsHook(eligible_ship)) exit;
    if(HullDamage(eligible_ship) > XDR_ConfigInt(5, 40) || ShipFreeSpace(eligible_ship) < XDR_ConfigInt(4, 20)) exit;
    if(ShipMoney(eligible_ship) < XDR_CamouflageCost(eligible_ship)) exit;
    result = 1;
}

function XDR_TestSpawnStalker()
{
    if(!xdr_test_mode || !xdr_test_spawn_stalker) exit;
    xdr_test_spawn_stalker = 0;
    if(!xdr_initialized || ArrayDim(xdr_ids) != 4 || ArrayDim(xdr_due_turns) != 4 || ArrayDim(xdr_ids) != ArrayDim(xdr_due_turns))
    {
        xdr_test_last_event = 'спавн сталкера не удался: хранилище не готово';
        exit;
    }

    int stalker_slot = 0;
    for(int stalker_slot_index = 1; stalker_slot_index <= 3; stalker_slot_index = stalker_slot_index + 1)
    {
        if(xdr_ids[stalker_slot_index] == 0 && xdr_due_turns[stalker_slot_index] <= CurTurn())
        {
            stalker_slot = stalker_slot_index;
            break;
        }
    }
    if(!stalker_slot)
    {
        xdr_test_last_event = 'спавн сталкера не удался: все три слота заняты';
        exit;
    }

    dword stalker_player = Player();
    if(!stalker_player)
    {
        xdr_test_last_event = 'спавн сталкера не удался: игрок недоступен';
        exit;
    }
    if(ShipInHyperSpace(stalker_player, 1) || ShipIsTakeoff(stalker_player) || !ShipInNormalSpace(stalker_player))
    {
        xdr_test_last_event = 'спавн сталкера не удался: игрок не в обычном космосе';
        exit;
    }
    dword stalker_star = ShipStar(stalker_player);
    if(!stalker_star)
    {
        xdr_test_last_event = 'спавн сталкера не удался: система игрока недоступна';
        exit;
    }

    dword stalker_factory_planet = 0;
    for(int stalker_local_planet_index = 0; stalker_local_planet_index < StarPlanets(stalker_star); stalker_local_planet_index = stalker_local_planet_index + 1)
    {
        dword stalker_local_planet = StarPlanets(stalker_star, stalker_local_planet_index);
        if(!stalker_local_planet) continue;
        int stalker_local_owner = PlanetOwner(stalker_local_planet);
        if(stalker_local_owner >= 0 && stalker_local_owner <= 4)
        {
            stalker_factory_planet = stalker_local_planet;
            break;
        }
    }
    int stalker_star_count = GalaxyStars();
    for(int stalker_factory_cursor = 0; !stalker_factory_planet && stalker_factory_cursor < stalker_star_count * 20; stalker_factory_cursor = stalker_factory_cursor + 1)
    {
        int stalker_factory_star_index = stalker_factory_cursor / 20;
        int stalker_factory_planet_index = stalker_factory_cursor - stalker_factory_star_index * 20;
        dword stalker_factory_star = GalaxyStar(stalker_factory_star_index);
        if(!stalker_factory_star) continue;
        if(stalker_factory_planet_index >= StarPlanets(stalker_factory_star)) continue;
        dword stalker_candidate_planet = StarPlanets(stalker_factory_star, stalker_factory_planet_index);
        if(!stalker_candidate_planet) continue;
        int stalker_candidate_owner = PlanetOwner(stalker_candidate_planet);
        if(stalker_candidate_owner >= 0 && stalker_candidate_owner <= 4) stalker_factory_planet = stalker_candidate_planet;
    }
    if(!stalker_factory_planet)
    {
        xdr_test_last_event = 'спавн сталкера не удался: нет подходящей планеты Коалиции';
        exit;
    }

    dword stalker_ship = BuyRanger(stalker_factory_planet, 100);
    if(!stalker_ship)
    {
        xdr_test_last_event = 'спавн сталкера не удался: BuyRanger не создал корабль';
        exit;
    }
    TransferShip(stalker_ship, stalker_star);
    ShipSetCoords(stalker_ship, CoordX(stalker_player) + 250, CoordY(stalker_player) + 100);
    int stalker_min_money = max(1, XDR_CamouflageCost(stalker_ship)) * 20;
    if(ShipMoney(stalker_ship) < stalker_min_money) ShipMoney(stalker_ship, stalker_min_money);

    XDR_ResetSlotData(stalker_slot, CurTurn());
    xdr_ids[stalker_slot] = Id(stalker_ship);
    xdr_states[stalker_slot] = 13;
    xdr_due_turns[stalker_slot] = CurTurn() + 1;
    xdr_phase_turns[stalker_slot] = CurTurn();
    xdr_last_hull_damage[stalker_slot] = HullDamage(stalker_ship);
    OrderNone(stalker_ship);
    ShipJoin(XenoDomSpecialists, stalker_ship, 'XenoDomState');
    dword stalker_player_for_relation = Player();
    if(stalker_player_for_relation)
    {
        RelationToRanger(stalker_ship, stalker_player_for_relation, 100);
        if(ShipGetBad(stalker_ship)) ShipSetBad(stalker_ship, 0);
    }
    OrderLock(stalker_ship, 1);
    xdr_test_last_event = 'сталкер создан рядом с игроком; маршрут начнётся на следующем ходу';
}

function XDR_TestQuickScenario()
{
    if(!xdr_test_mode || xdr_test_quick_stage == 0) exit;

    if(xdr_test_quick_stage < 0)
    {
        XDR_TestQuickStopPlayerMask();
        xdr_test_quick_stage = 0;
        xdr_test_last_event = 'тестовая маскировка игрока снята';
        exit;
    }

    dword quick_player = Player();
    if(!quick_player)
    {
        XDR_TestQuickFail('быстрый тест отменён: игрок недоступен');
        exit;
    }

    if(xdr_test_quick_stage == 1)
    {
        if(ShipInHyperSpace(quick_player, 1) || ShipIsTakeoff(quick_player) || !ShipInNormalSpace(quick_player))
        {
            xdr_test_last_event = 'быстрый тест ждёт выхода игрока в обычный космос';
            exit;
        }

        int quick_free_slot = 0;
        for(int quick_slot_probe = 1; quick_slot_probe <= 3; quick_slot_probe = quick_slot_probe + 1)
        {
            if(xdr_ids[quick_slot_probe] == 0 && xdr_due_turns[quick_slot_probe] <= CurTurn())
            {
                quick_free_slot = quick_slot_probe;
                break;
            }
        }
        if(!quick_free_slot)
        {
            XDR_TestQuickFail('быстрый тест отменён: все три слота специалистов заняты');
            exit;
        }

        dword quick_source_star = 0;
        dword quick_target_star = 0;
        int quick_star_count = GalaxyStars();
        for(int quick_pair_cursor = 0; quick_pair_cursor < quick_star_count * quick_star_count; quick_pair_cursor = quick_pair_cursor + 1)
        {
            int quick_target_index = quick_pair_cursor / quick_star_count;
            int quick_source_index = quick_pair_cursor - quick_target_index * quick_star_count;
            dword quick_target_candidate = GalaxyStar(quick_target_index);
            if(!quick_target_candidate) continue;
            if(!XDR_IsTargetValid(quick_target_candidate, -1, 0)) continue;
            dword quick_source_candidate = GalaxyStar(quick_source_index);
            if(!quick_source_candidate) continue;
            if(quick_source_candidate == quick_target_candidate) continue;
            if(!XDR_IsStrategicStar(quick_source_candidate)) continue;
            if(StarBattle(quick_source_candidate)) continue;
            if(StarPlanets(quick_source_candidate) <= 0) continue;
            int quick_source_owner = StarOwner(quick_source_candidate);
            if(quick_source_owner != 0) continue;
            if(!XDR_FindSafePlanet(quick_source_candidate)) continue;
            if(!ShipCanJump(quick_player, quick_source_candidate, quick_target_candidate)) continue;
            quick_source_star = quick_source_candidate;
            quick_target_star = quick_target_candidate;
            break;
        }
        if(!quick_source_star || !quick_target_star)
        {
            XDR_TestQuickFail('быстрый тест отменён: не найдена безопасная система в одном прыжке от доминаторов');
            exit;
        }

        xdr_test_quick_slot = quick_free_slot;
        xdr_test_quick_source_star_id = Id(quick_source_star);
        xdr_test_quick_target_star_id = Id(quick_target_star);
        for(int quick_sector_index = 0; quick_sector_index < GalaxySectors(); quick_sector_index = quick_sector_index + 1)
        {
            dword quick_sector = GalaxySectors(quick_sector_index);
            if(quick_sector) SectorVisible(quick_sector, 1);
        }
        dword quick_arrival_anchor = StarPlanets(quick_source_star, 0);
        if(!quick_arrival_anchor)
        {
            XDR_TestQuickFail('быстрый тест отменён: в системе старта исчезла опорная планета');
            exit;
        }
        int quick_arrival_x = CoordX(quick_arrival_anchor) + 1000;
        int quick_arrival_y = CoordY(quick_arrival_anchor) + 500;
        OrderTeleport(quick_player, quick_source_star, quick_arrival_x, quick_arrival_y, 1, 1);
        xdr_test_quick_stage = 2;
        xdr_test_last_event = 'все сектора открыты; игрок отправлен варпом в соседнюю систему';
        exit;
    }

    if(xdr_test_quick_stage == 2)
    {
        if(ShipInHyperSpace(quick_player, 1) || ShipIsTakeoff(quick_player) || !ShipInNormalSpace(quick_player)) exit;
        dword quick_source_check = XDR_FindStarById(xdr_test_quick_source_star_id);
        dword quick_target_check = XDR_FindStarById(xdr_test_quick_target_star_id);
        if(!quick_source_check || !quick_target_check)
        {
            XDR_TestQuickFail('быстрый тест отменён: сохранённые системы больше не существуют');
            exit;
        }
        dword quick_player_star = ShipStar(quick_player);
        if(!quick_player_star || quick_player_star != quick_source_check)
        {
            xdr_test_last_event = 'быстрый тест ждёт завершения переноса игрока';
            exit;
        }
        if(!XDR_IsTargetValid(quick_target_check, -1, 0))
        {
            XDR_TestQuickFail('быстрый тест отменён: система доминаторов изменилась до запуска');
            exit;
        }

        int quick_series = StarSeries(quick_target_check);

        dword quick_factory_planet = 0;
        int quick_factory_star_count = GalaxyStars();
        for(int quick_factory_cursor = 0; !quick_factory_planet && quick_factory_cursor < quick_factory_star_count * 20; quick_factory_cursor = quick_factory_cursor + 1)
        {
            int quick_factory_star_index = quick_factory_cursor / 20;
            int quick_factory_planet_index = quick_factory_cursor - quick_factory_star_index * 20;
            dword quick_factory_star = GalaxyStar(quick_factory_star_index);
            if(!quick_factory_star) continue;
            if(quick_factory_planet_index >= StarPlanets(quick_factory_star)) continue;
            dword quick_factory_candidate = StarPlanets(quick_factory_star, quick_factory_planet_index);
            if(!quick_factory_candidate) continue;
            int quick_factory_owner = PlanetOwner(quick_factory_candidate);
            if(quick_factory_owner >= 0 && quick_factory_owner <= 4) quick_factory_planet = quick_factory_candidate;
        }
        if(!quick_factory_planet)
        {
            XDR_TestQuickFail('быстрый тест отменён: нет планеты Коалиции для создания рейнджера');
            exit;
        }

        int quick_slot = xdr_test_quick_slot;
        if(quick_slot < 1 || quick_slot > 3 || xdr_ids[quick_slot] != 0)
        {
            XDR_TestQuickFail('быстрый тест отменён: подготовленный слот уже занят');
            exit;
        }
        dword quick_ranger = BuyRanger(quick_factory_planet, 100);
        if(!quick_ranger)
        {
            XDR_TestQuickFail('быстрый тест отменён: BuyRanger не создал корабль');
            exit;
        }
        TransferShip(quick_ranger, quick_source_check);
        ShipSetCoords(quick_ranger, CoordX(quick_player) + 250, CoordY(quick_player) + 100);
        int quick_ranger_money = max(1, XDR_CamouflageCost(quick_ranger)) * 20;
        if(ShipMoney(quick_ranger) < quick_ranger_money) ShipMoney(quick_ranger, quick_ranger_money);
        dword quick_ranger_tank = ShipEqInSlot(quick_ranger, t_FuelTanks);
        if(quick_ranger_tank)
            ShipFuel(quick_ranger, GetEquipmentStats(quick_ranger_tank, 0));
        dword quick_ranger_hook = ShipEqInSlot(quick_ranger, t_CargoHook);
        if(!quick_ranger_tank || !quick_ranger_hook || ShipFreeSpace(quick_ranger) < 3)
        {
            ShipFreeFlight(quick_ranger);
            XDR_TestQuickFail('быстрый тест отменён: штатный рейнджер не имеет установленного бака, захвата или трёх единиц свободного трюма');
            exit;
        }

        XDR_ResetSlotData(quick_slot, CurTurn());
        xdr_ids[quick_slot] = Id(quick_ranger);
        xdr_states[quick_slot] = 3;
        xdr_target_star_ids[quick_slot] = Id(quick_target_check);
        xdr_series[quick_slot] = quick_series;
        xdr_charge_expire_turns[quick_slot] = CurTurn() + XDR_ConfigInt(11, 300);
        xdr_due_turns[quick_slot] = CurTurn();
        xdr_phase_turns[quick_slot] = CurTurn();
        xdr_last_hull_damage[quick_slot] = HullDamage(quick_ranger);
        dword quick_return_planet = XDR_FindSafePlanet(quick_source_check);
        if(!quick_return_planet)
        {
            ShipFreeFlight(quick_ranger);
            XDR_TestQuickFail('быстрый тест отменён: исходная система потеряла безопасную планету возврата');
            exit;
        }
        xdr_return_place_ids[quick_slot] = Id(quick_return_planet);
        xdr_return_kinds[quick_slot] = 2;
        xdr_test_quick_ranger_id = Id(quick_ranger);
        OrderNone(quick_ranger);
        ShipJoin(XenoDomSpecialists, quick_ranger, 'XenoDomState');
        RelationToRanger(quick_ranger, quick_player, 100);
        if(ShipGetBad(quick_ranger)) ShipSetBad(quick_ranger, 0);
        int quick_player_charges = PlayerChameleonCharges(quick_series);
        if(quick_player_charges < 10) PlayerChameleonCharges(quick_series, 10);
        if(ShipGetBad(quick_ranger)) ShipSetBad(quick_ranger, 0);
        OrderLock(quick_ranger, 1);
        xdr_test_quick_stage = 3;
        xdr_test_last_event = 'быстрый сталкер создан с зарядом; назначен прямой прыжок к доминаторам';
        exit;
    }

    if(xdr_test_quick_stage == 3)
    {
        int quick_arriving_ranger_id = xdr_test_quick_ranger_id;
        dword quick_arriving_ranger = 0;
        if(quick_arriving_ranger_id > 1) quick_arriving_ranger = IdToShip(quick_arriving_ranger_id);
        if(!quick_arriving_ranger)
        {
            XDR_TestQuickFail('быстрый тест прерван: сталкер не найден по сохранённому ID');
            exit;
        }
        if(ShipInHyperSpace(quick_arriving_ranger, 1) || ShipIsTakeoff(quick_arriving_ranger) || !ShipInNormalSpace(quick_arriving_ranger)) exit;
        dword quick_arrival_star = ShipStar(quick_arriving_ranger);
        if(!quick_arrival_star) exit;
        if(Id(quick_arrival_star) != xdr_test_quick_target_star_id) exit;

        dword quick_arrival_hook = ShipEqInSlot(quick_arriving_ranger, t_CargoHook);
        if(!quick_arrival_hook)
        {
            XDR_TestQuickFail('быстрый тест прерван: сталкер потерял захват до прибытия');
            exit;
        }
        int quick_loot_size = min(1, min(ShipFreeSpace(quick_arriving_ranger), GetEquipmentStats(quick_arrival_hook, 0)));
        if(quick_loot_size < 1)
        {
            XDR_TestQuickFail('быстрый тест прерван: в трюме сталкера нет места для тестовой цистерны');
            exit;
        }
        dword quick_arrival_tank = ShipEqInSlot(quick_arriving_ranger, t_FuelTanks);
        if(!quick_arrival_tank)
        {
            XDR_TestQuickFail('быстрый тест прерван: сталкер потерял штатный топливный бак до прибытия');
            exit;
        }
        // Simulate the fuel state after a costly outward jump. Mode 0 proves
        // transfer from a picked cistern; mode 1 proves vanilla solar charging.
        ShipFuel(quick_arriving_ranger, 0);
        dword quick_primary_loot = 0;
        if(xdr_test_quick_fuel_mode == 0)
        {
            dword quick_fuel_cistern = CreateEquipment(t_Cistern, quick_loot_size, 500, 5);
            if(!quick_fuel_cistern)
            {
                XDR_TestQuickFail('быстрый тест прерван: тестовая цистерна не создана');
                exit;
            }
            ContainerFuel(quick_fuel_cistern, max(1, GetEquipmentStats(quick_arrival_tank, 0)));
            int quick_fuel_x = CoordX(quick_arriving_ranger) + 60;
            int quick_fuel_y = CoordY(quick_arriving_ranger) + 20;
            DropItemInSystem(quick_arrival_star, quick_fuel_cistern, quick_fuel_x, quick_fuel_y, quick_fuel_x, quick_fuel_y);
            StopMovingItem(quick_arrival_star, quick_fuel_cistern);
            // A second full cistern is the regression guard: after servicing
            // the tank the stalker must leave it in space and collect both
            // dominator wrecks instead.
            dword quick_extra_cistern = CreateEquipment(t_Cistern, quick_loot_size, 500, 5);
            if(!quick_extra_cistern)
            {
                XDR_TestQuickFail('быстрый тест прерван: запасная тестовая цистерна не создана');
                exit;
            }
            ContainerFuel(quick_extra_cistern, max(1, GetEquipmentStats(quick_arrival_tank, 0)));
            int quick_extra_fuel_x = CoordX(quick_arriving_ranger) + 140;
            int quick_extra_fuel_y = CoordY(quick_arriving_ranger) + 55;
            DropItemInSystem(quick_arrival_star, quick_extra_cistern, quick_extra_fuel_x, quick_extra_fuel_y, quick_extra_fuel_x, quick_extra_fuel_y);
            StopMovingItem(quick_arrival_star, quick_extra_cistern);
            quick_primary_loot = quick_fuel_cistern;
        }

        dword quick_dom_wreck_one = CreateEquipment(t_Engine, 1, max(1, GalaxyTechLevel()), 5);
        dword quick_dom_wreck_two = CreateEquipment(t_Radar, 1, max(1, GalaxyTechLevel()), 5);
        if(!quick_dom_wreck_one || !quick_dom_wreck_two)
        {
            XDR_TestQuickFail('быстрый тест прерван: доминаторские обломки не созданы');
            exit;
        }
        int quick_wreck_one_preloaded = 0;
        if(xdr_test_quick_fuel_mode == 0)
        {
            int quick_eviction_free_space = ShipFreeSpace(quick_arriving_ranger);
            int quick_eviction_hook_size = GetEquipmentStats(quick_arrival_hook, 0);
            if(quick_eviction_free_space > 0 && quick_eviction_hook_size >= quick_eviction_free_space)
            {
                ItemSize(quick_dom_wreck_one, quick_eviction_free_space);
                AddItemToShip(quick_arriving_ranger, quick_dom_wreck_one);
                ShipCalcParam(quick_arriving_ranger);
                quick_wreck_one_preloaded = 1;
            }
        }
        int quick_wreck_one_x = CoordX(quick_arriving_ranger) + 85;
        int quick_wreck_one_y = CoordY(quick_arriving_ranger) - 25;
        int quick_wreck_two_x = CoordX(quick_arriving_ranger) + 110;
        int quick_wreck_two_y = CoordY(quick_arriving_ranger) + 35;
        if(!quick_wreck_one_preloaded)
        {
            DropItemInSystem(quick_arrival_star, quick_dom_wreck_one, quick_wreck_one_x, quick_wreck_one_y, quick_wreck_one_x, quick_wreck_one_y);
            StopMovingItem(quick_arrival_star, quick_dom_wreck_one);
        }
        DropItemInSystem(quick_arrival_star, quick_dom_wreck_two, quick_wreck_two_x, quick_wreck_two_y, quick_wreck_two_x, quick_wreck_two_y);
        StopMovingItem(quick_arrival_star, quick_dom_wreck_two);
        if(!quick_primary_loot) quick_primary_loot = quick_dom_wreck_one;

        xdr_test_quick_loot_id = Id(quick_primary_loot);
        xdr_test_quick_stage = 4;
        xdr_states[xdr_test_quick_slot] = 5;
        xdr_phase_turns[xdr_test_quick_slot] = CurTurn();
        xdr_no_loot_counts[xdr_test_quick_slot] = 0;
        xdr_pending_item_ids[xdr_test_quick_slot] = 0;
        xdr_pending_since[xdr_test_quick_slot] = 0;
        // Use the production selector immediately. In fuel mode the hold is
        // deliberately full, so assigning the cistern directly would only
        // wait for the five-turn pickup timeout before the eviction branch got
        // a chance to run.
        XDR_PrepareNextLoot(xdr_test_quick_slot, quick_arriving_ranger, quick_arrival_star);
        if(xdr_test_quick_fuel_mode == 0)
            xdr_test_last_event = 'созданы две цистерны и два обломка: при полном трюме лут временно выгружается под топливо, затем подбирается обратно';
        else
            xdr_test_last_event = 'созданы два обломка доминаторов без топлива; после сбора тестируется зарядка у солнца';
        exit;
    }

    if(xdr_test_quick_stage == 4)
    {
        int quick_finish_slot = xdr_test_quick_slot;
        if(quick_finish_slot < 1 || quick_finish_slot > 3)
        {
            XDR_TestQuickFail('быстрый тест завершён без активного слота');
            exit;
        }
        if(xdr_states[quick_finish_slot] == 11 || xdr_ids[quick_finish_slot] <= 1)
        {
            xdr_test_quick_stage = 0;
            xdr_test_last_event = 'быстрый тест завершён: сталкер вернулся и передан ванильному ИИ до следующего цикла';
        }
        exit;
    }
}

function XDR_Recruit()
{
    if(CurTurn() < xdr_next_recruit_turn || xdr_disabled) exit;
    if(!XDR_NativeReady())
    {
        xdr_next_recruit_turn = CurTurn() + XDR_ConfigInt(1, 100);
        if(xdr_test_mode) xdr_test_last_event = 'набор остановлен: Native Loader не инициализировал маскировку';
        exit;
    }
    if(xdr_test_mode && !xdr_test_force_recruit)
    {
        xdr_next_recruit_turn = CurTurn() + 5;
        exit;
    }
    if(ArrayDim(xdr_ids) != 4) exit;
    if(ArrayDim(xdr_due_turns) != 4) exit;
    if(ArrayDim(xdr_ids) != ArrayDim(xdr_due_turns)) exit;
    int recruit_base_count = XDR_CountPirateBases();
    if(recruit_base_count <= 0)
    {
        xdr_next_recruit_turn = CurTurn() + XDR_ConfigInt(1, 100);
        if(xdr_test_mode) xdr_test_last_event = 'набор пропущен: пиратских баз нет';
        exit;
    }
    xdr_next_recruit_turn = CurTurn() + XDR_ConfigInt(1, 100);
    if(xdr_test_mode) xdr_next_recruit_turn = CurTurn() + 5;
    int recruit_slot = 0;
    int recruit_max_slots = XDR_ConfigInt(3, 3);
    if(xdr_test_mode && xdr_test_force_recruit) recruit_max_slots = 3;
    for(int recruit_slot_index = 1; recruit_slot_index <= recruit_max_slots; recruit_slot_index = recruit_slot_index + 1)
    {
        if(xdr_ids[recruit_slot_index] == 0 && xdr_due_turns[recruit_slot_index] <= CurTurn())
        {
            recruit_slot = recruit_slot_index;
            break;
        }
    }
    if(!recruit_slot) exit;
    int recruit_roll = Rnd(1, 100);
    if(xdr_test_mode && xdr_test_force_recruit) recruit_roll = 1;
    if(recruit_roll > XDR_ConfigInt(2, 10) && !xdr_test_mode) exit;

    int recruit_ranger_count = GalaxyRangers();
    if(recruit_ranger_count <= 0) exit;
    int recruit_ranger_index = Rnd(0, recruit_ranger_count - 1);
    if(xdr_test_mode)
    {
        recruit_ranger_index = xdr_test_ranger_cursor;
        if(recruit_ranger_index < 0 || recruit_ranger_index >= recruit_ranger_count) recruit_ranger_index = 0;
        xdr_test_ranger_cursor = recruit_ranger_index + 1;
        if(xdr_test_ranger_cursor >= recruit_ranger_count) xdr_test_ranger_cursor = 0;
    }
    dword recruit_candidate = GalaxyRangers(recruit_ranger_index);
    xdr_test_force_recruit = 0;
    if(!XDR_SlotEligible(recruit_candidate)) exit;
    int recruit_already_specialist = xdr_ids[1] == Id(recruit_candidate) || xdr_ids[2] == Id(recruit_candidate) || xdr_ids[3] == Id(recruit_candidate);
    if(recruit_already_specialist) exit;
    dword recruit_star = XDR_CurrentStar(recruit_candidate);
    if(!XDR_BuildRouteToBase(recruit_slot, recruit_candidate, recruit_star)) exit;

    xdr_ids[recruit_slot] = Id(recruit_candidate);
    xdr_states[recruit_slot] = 1;
    xdr_due_turns[recruit_slot] = CurTurn();
    xdr_phase_turns[recruit_slot] = CurTurn();
    xdr_last_hull_damage[recruit_slot] = HullDamage(recruit_candidate);
    ShipJoin(XenoDomSpecialists, recruit_candidate, 'XenoDomState');
    OrderLock(recruit_candidate, 1);
    if(xdr_test_mode) xdr_test_last_event = 'специалист выбран и направлен на пиратскую базу';
}

function XDR_HandleThreat(int threat_slot, dword threat_ranger, dword threat_star)
{
    result = 0;
    if(!XDR_ApplyMask(threat_slot, threat_ranger, threat_star))
    {
        xdr_disabled = 1;
        result = 1;
        if(xdr_test_mode) xdr_test_last_event = 'нативная маскировка потеряна; миссия немедленно прекращена';
        exit;
    }
    xdr_last_hull_damage[threat_slot] = HullDamage(threat_ranger);
    // Coalition ships and pirates are intentionally unaffected by camouflage.
    // If one of them acquires the ranger, abort this salvage pass without
    // treating that as a failure of the Dominator-specific mask.
    for(int threat_ship_index = 0; threat_ship_index < StarShips(threat_star); threat_ship_index = threat_ship_index + 1)
    {
        dword threat_ship = StarShips(threat_star, threat_ship_index);
        if(!threat_ship) continue;
        if(threat_ship == threat_ranger) continue;
        if(ShipTypeN(threat_ship) >= t_RC) continue;
        if(!ShipInNormalSpace(threat_ship) || ShipIsTakeoff(threat_ship) || ShipInHyperSpace(threat_ship, 1)) continue;
        int threat_is_dominator = ShipOwner(threat_ship) == 5;
        if(ShipGetBad(threat_ship) == threat_ranger)
        {
            if(threat_is_dominator)
            {
                // Do not mutate TKling orders. A hostile pointer here means an
                // unsupported engine/mod interaction; withdraw while the native
                // protection remains active.
                result = 1;
                if(xdr_test_mode) xdr_test_last_event = 'доминатор обошёл нативную проверку; сталкер отходит без вмешательства в его ИИ';
            }
            else result = 2;
            exit;
        }
    }
    XDR_KeepDefensiveTargetOnly(threat_ranger);
    xdr_threat_counts[threat_slot] = 0;
}

function XDR_ProcessSlot(int slot)
{
    int slot_ship_id = xdr_ids[slot];
    if(slot_ship_id <= 1) exit;
    dword slot_ship = IdToShip(slot_ship_id);
    if(!slot_ship)
    {
        XDR_MarkDead(slot);
        exit;
    }
    int slot_owner = ShipOwner(slot_ship);
    if(ShipTypeN(slot_ship) != t_Ranger || slot_owner == 7)
    {
        XDR_DetachSlot(slot, slot_ship, 0);
        exit;
    }

    int slot_state = xdr_states[slot];

    // Hard state-machine invariant: no outbound/salvage/return branch can run
    // unless state 2 recorded a paid camouflage charge, its series and target.
    // This also safely stops malformed fresh-test slots before any Dominator
    // jump can be issued.
    int slot_requires_charge = slot_state == 3 || slot_state == 4 || slot_state == 5 || slot_state == 6 ||
                               slot_state == 7 || slot_state == 8 || slot_state == 12 || slot_state == 14 ||
                               slot_state == 15 || slot_state == 16 || slot_state == 17 || slot_state == 18;
    if(slot_requires_charge && !XDR_HasPurchasedCharge(slot))
    {
        XDR_CancelPickup(slot, slot_ship);
        XDR_RestoreMask(slot, slot_ship);
        XDR_ClearRoute(slot);
        xdr_target_star_ids[slot] = 0;
        xdr_series[slot] = -1;
        xdr_charge_expire_turns[slot] = 0;
        xdr_states[slot] = 13;
        xdr_due_turns[slot] = CurTurn() + 1;
        if(!ShipInHyperSpace(slot_ship, 1) && !ShipIsTakeoff(slot_ship) && ShipInNormalSpace(slot_ship)) OrderNone(slot_ship);
        if(xdr_test_mode) xdr_test_last_event = 'защитная остановка: вылет без купленного заряда запрещён';
        exit;
    }

    if(slot_state != 11 && !XDR_GroupContains(slot_ship))
    {
        if(ShipInScript(slot_ship, 0))
        {
            // Another script has claimed this ranger. Restore only values still
            // owned by us, cancel our pickup marker and release OrderLock, but
            // never call ShipOut on a group owned by the other script.
            XDR_CancelPickup(slot, slot_ship);
            XDR_RestoreMask(slot, slot_ship);
            OrderLock(slot_ship, 0);
            XDR_ResetSlotData(slot, CurTurn() + XDR_ConfigInt(6, 1000));
            exit;
        }
        ShipJoin(XenoDomSpecialists, slot_ship, 'XenoDomState');
        OrderLock(slot_ship, 1);
    }

    if(slot_state != 11)
    {
        if(xdr_ghost_flags[slot])
        {
            XDR_ReassertGhost(slot, slot_ship);
            // NoTargetToShip explicitly excludes unconditionally hostile
            // Dominators. Reapply the per-Dominator relation and target reset before
            // deciding whether this ranger has a legitimate defensive target.
            if(!ShipInHyperSpace(slot_ship, 1) && !ShipIsTakeoff(slot_ship) && ShipInNormalSpace(slot_ship))
            {
                dword slot_mask_star = ShipStar(slot_ship);
                if(slot_mask_star)
                {
                    if(Id(slot_mask_star) == xdr_target_star_ids[slot] && StarOwner(slot_mask_star) == 1)
                        XDR_ApplyMask(slot, slot_ship, slot_mask_star);
                }
            }
        }
        XDR_KeepDefensiveTargetOnly(slot_ship);
        // Actual shots are intercepted by the dedicated player state. Merely
        // selecting the stalker as a target must not open the warning dialog.
    }

    // The hostile dialog answer is a persistent per-ship abort request. It
    // must never reopen the warning and it must override salvage/combat with
    // an immediate withdrawal order on the same day-controller pass.
    if(xdr_dialog_markers[slot] == 1)
    {
        int slot_can_abort_for_player = slot_state == 3 || slot_state == 4 || slot_state == 5 ||
                                        slot_state == 7 || slot_state == 12 || slot_state == 14 ||
                                        slot_state == 15 || slot_state == 16 || slot_state == 17 || slot_state == 18;
        if(slot_can_abort_for_player)
        {
            XDR_CancelPickup(slot, slot_ship);
            XDR_ClearBattleWait(slot);
            XDR_ClearRoute(slot);
            xdr_states[slot] = 6;
            slot_state = 6;
            xdr_phase_turns[slot] = CurTurn();
            if(xdr_test_mode) xdr_test_last_event = 'игрок продолжил атаку; повторный диалог запрещён, сталкер немедленно отходит';
        }
    }

    if(xdr_release_flags[slot] && (slot_state == 1 || slot_state == 2 || slot_state == 10 || slot_state == 11 || slot_state == 13 || slot_state == 14))
    {
        XDR_DetachSlot(slot, slot_ship, 0);
        exit;
    }
    if(xdr_release_flags[slot] && slot_state != 8 && slot_state != 9)
    {
        xdr_states[slot] = 6;
        slot_state = 6;
    }

    if(slot_state == 1)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        dword route_docked_planet = GetShipPlanet(slot_ship);
        dword route_docked_base = GetShipRuins(slot_ship);
        dword route_home_base = XDR_FindPirateBaseById(xdr_home_base_ids[slot]);
        dword route_docked_star = 0;
        dword route_docked_next_star = 0;
        if(route_docked_planet || route_docked_base)
        {
            route_docked_star = XDR_CurrentStar(slot_ship);
            route_docked_next_star = XDR_NextRouteStar(slot);
            // A zero-edge route uses the current star as a marker. Consume it
            // before deciding whether to take off, otherwise the ship bounces
            // endlessly between the local pirate base and normal space.
            if(route_docked_star && route_docked_next_star)
            {
                if(route_docked_star == route_docked_next_star)
                {
                    XDR_PopRouteStar(slot, route_docked_star);
                    route_docked_next_star = XDR_NextRouteStar(slot);
                }
            }
        }
        if(route_docked_base && route_home_base && route_docked_base == route_home_base && !route_docked_next_star)
        {
            xdr_states[slot] = 2;
            xdr_phase_turns[slot] = CurTurn();
            exit;
        }
        if(route_docked_planet || route_docked_base)
        {
            int route_camo_reserve = XDR_CamouflageCost(slot_ship);
            XDR_ServiceRefuel(slot_ship, route_camo_reserve);
            if(route_docked_star && route_docked_next_star)
            {
                // This is the real departure check, so it must include the
                // fuel currently loaded after servicing at the landing place.
                if(!ShipCanJump(slot_ship, route_docked_star, route_docked_next_star, 1))
                {
                    XDR_DetachSlot(slot, slot_ship, 1);
                    if(xdr_test_mode) xdr_test_last_event = 'не хватает денег на топливо по маршруту; рейнджер вернулся к обычным делам';
                    exit;
                }
            }
            OrderTakeOff(slot_ship);
            xdr_phase_turns[slot] = CurTurn() + 1;
            exit;
        }
        if(!ShipInNormalSpace(slot_ship)) exit;
        dword route_current_star = XDR_CurrentStar(slot_ship);
        dword route_next_star = XDR_NextRouteStar(slot);
        if(route_next_star && route_current_star == route_next_star)
        {
            XDR_PopRouteStar(slot, route_current_star);
            route_next_star = XDR_NextRouteStar(slot);
        }
        if(route_next_star)
        {
            if(StarOwner(route_next_star) == 1)
            {
                // A route to a camouflage shop may never enter a Dominator
                // system before state 2 has paid for a charge.
                XDR_ClearRoute(slot);
                if(XDR_BuildRouteToBase(slot, slot_ship, route_current_star)) exit;
                xdr_states[slot] = 13;
                xdr_due_turns[slot] = CurTurn() + 1;
                OrderNone(slot_ship);
                if(xdr_test_mode) xdr_test_last_event = 'опасный путь к базе отброшен; сталкер ждёт безопасную базу';
                exit;
            }
            if(ShipCanJump(slot_ship, route_current_star, route_next_star, 1))
            {
                XDR_OrderLockedJump(slot_ship, route_next_star);
                exit;
            }
            dword route_refuel_planet = XDR_FindSafePlanet(route_current_star);
            if(route_refuel_planet)
            {
                OrderLanding(slot_ship, route_refuel_planet, 1);
                exit;
            }
            XDR_DetachSlot(slot, slot_ship, 1);
            exit;
        }
        if(!route_home_base)
        {
            if(XDR_CountPirateBases() > 0 && XDR_BuildRouteToBase(slot, slot_ship, route_current_star)) exit;
            XDR_DetachSlot(slot, slot_ship, 1);
            exit;
        }
        dword route_home_star = XDR_BaseStar(route_home_base);
        if(route_home_star == route_current_star)
        {
            OrderLanding(slot_ship, route_home_base, 1);
            exit;
        }
        if(XDR_BuildRouteToBase(slot, slot_ship, route_current_star)) exit;
        XDR_DetachSlot(slot, slot_ship, 1);
        exit;
    }

    if(slot_state == 2)
    {
        if(CurTurn() < xdr_due_turns[slot]) exit;
        dword wait_base = XDR_FindPirateBaseById(xdr_home_base_ids[slot]);
        if(!wait_base || GetShipRuins(slot_ship) != wait_base)
        {
            dword wait_current_star = XDR_CurrentStar(slot_ship);
            if(wait_current_star && XDR_CountPirateBases() > 0 && XDR_BuildRouteToBase(slot, slot_ship, wait_current_star))
            {
                xdr_states[slot] = 1;
                exit;
            }
            XDR_DetachSlot(slot, slot_ship, 1);
            exit;
        }
        dword wait_base_star = XDR_BaseStar(wait_base);
        if(!wait_base_star) exit;
        if(xdr_charge_expire_turns[slot] <= 0 && !XDR_HasGoodCargoHook(slot_ship))
        {
            // RScript has no targeted "buy a cargo hook" order. Give the
            // landed ranger one real class-AI shopping pass at the pirate base.
            // Candidates already have a hook; this pass only lets vanilla AI
            // replace a weak one. If it cannot equip a useful hook, release the
            // ranger before any camouflage money is charged.
            OrderLock(slot_ship, 0);
            ShipFreeFlight(slot_ship);
            if(!XDR_HasGoodCargoHook(slot_ship))
            {
                XDR_DetachSlot(slot, slot_ship, 1);
                xdr_next_recruit_turn = max(xdr_next_recruit_turn, CurTurn() + 100);
                if(xdr_test_mode) xdr_test_last_event = 'ванильный ИИ не купил хороший захват; камуфляж не оплачен, рейнджер освобождён';
                exit;
            }
            OrderLock(slot_ship, 1);
            if(ShipIsTakeoff(slot_ship))
            {
                xdr_states[slot] = 1;
                if(xdr_test_mode) xdr_test_last_event = 'ванильный ИИ улучшил захват и успел взлететь; возврат за камуфляжем будет отдельным заходом';
                exit;
            }
            OrderNone(slot_ship);
            if(xdr_test_mode) xdr_test_last_event = 'ванильный ИИ установил хороший захват; теперь можно покупать камуфляж';
        }
        dword wait_target = 0;
        if(xdr_target_star_ids[slot] > 1) wait_target = XDR_FindStarById(xdr_target_star_ids[slot]);
        if(xdr_charge_expire_turns[slot] > 0)
        {
            if(CurTurn() >= xdr_charge_expire_turns[slot])
            {
                XDR_DetachAfterSpentCharge(slot, slot_ship);
                if(xdr_test_mode) xdr_test_last_event = 'заряд камуфляжа просрочен; рейнджер вернулся к обычным делам';
                exit;
            }
            int wait_stored_target_valid = 0;
            if(wait_target)
            {
                if(XDR_IsTargetValid(wait_target, xdr_series[slot], 0))
                {
                    if(XDR_StarHasValuableLoot(slot_ship, wait_target)) wait_stored_target_valid = 1;
                }
            }
            if(!wait_stored_target_valid)
            {
                wait_target = XDR_FindTarget(slot_ship, wait_base_star, xdr_series[slot]);
                if(wait_target)
                {
                    xdr_target_star_ids[slot] = Id(wait_target);
                    if(xdr_test_mode) xdr_test_last_event = 'заряд сохранён; выбрана новая цель той же серии';
                }
                else
                {
                    xdr_target_star_ids[slot] = 0;
                    xdr_due_turns[slot] = CurTurn() + 1;
                    if(xdr_test_mode) xdr_test_last_event = 'заряд сохранён; подходящей цели этой серии пока нет';
                    exit;
                }
            }
        }
        if(xdr_charge_expire_turns[slot] <= 0)
        {
            wait_target = XDR_FindTarget(slot_ship, wait_base_star, -1);
            if(!wait_target)
            {
                XDR_DetachSlot(slot, slot_ship, 1);
                xdr_due_turns[slot] = CurTurn() + 100;
                if(xdr_test_mode) xdr_due_turns[slot] = CurTurn() + 10;
                if(xdr_test_mode) xdr_test_last_event = 'нет цели с ценными обломками; рейнджер вернулся к обычным делам';
                exit;
            }
            int wait_camo_cost = XDR_CamouflageCost(slot_ship);
            if(ShipMoney(slot_ship) < wait_camo_cost)
            {
                XDR_DetachSlot(slot, slot_ship, 1);
                if(xdr_test_mode) xdr_test_last_event = 'не хватает денег на камуфляж; рейнджер копит обычными делами';
                exit;
            }
            ShipMoney(slot_ship, ShipMoney(slot_ship) - wait_camo_cost);
            xdr_target_star_ids[slot] = Id(wait_target);
            xdr_series[slot] = StarSeries(wait_target);
            xdr_charge_expire_turns[slot] = CurTurn() + XDR_ConfigInt(11, 300);
            if(xdr_test_mode) xdr_test_last_event = 'заряд камуфляжа куплен; подготовка к вылету';
        }
        XDR_TransferFuel(slot, slot_ship);
        XDR_ServiceRefuel(slot_ship, 0);
        xdr_states[slot] = 3;
        xdr_phase_turns[slot] = CurTurn();
        exit;
    }

    if(slot_state == 3)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        dword takeoff_current_star = XDR_CurrentStar(slot_ship);
        if(!takeoff_current_star) exit;
        dword takeoff_target = XDR_FindStarById(xdr_target_star_ids[slot]);
        int takeoff_at_valuable_target = 0;
        if(XDR_IsTargetValid(takeoff_current_star, xdr_series[slot], 0))
        {
            if(XDR_StarHasValuableLoot(slot_ship, takeoff_current_star))
            {
                xdr_target_star_ids[slot] = Id(takeoff_current_star);
                takeoff_target = takeoff_current_star;
                takeoff_at_valuable_target = 1;
            }
        }
        if(takeoff_at_valuable_target)
        {
            XDR_RefreshMaskAfterJump(slot, slot_ship, takeoff_current_star);
            xdr_last_hull_damage[slot] = HullDamage(slot_ship);
            xdr_phase_turns[slot] = CurTurn();
            xdr_no_loot_counts[slot] = 0;
            xdr_states[slot] = 5;
            exit;
        }
        dword takeoff_planet = GetShipPlanet(slot_ship);
        dword takeoff_station = GetShipRuins(slot_ship);
        if(takeoff_planet || takeoff_station)
        {
            XDR_TransferFuel(slot, slot_ship);
            XDR_ServiceRefuel(slot_ship, 0);
            if(!XDR_BuildRouteToTarget(slot, slot_ship, takeoff_current_star, xdr_series[slot]))
            {
                // The charge belongs to a series, not to one concrete star.
                // Keep it and wait at the shop for another reachable target.
                xdr_states[slot] = 2;
                xdr_due_turns[slot] = CurTurn() + 1;
                if(xdr_test_mode) xdr_test_last_event = 'маршрут изменился; заряд сохранён до новой цели той же серии';
                exit;
            }
            OrderTakeOff(slot_ship);
            xdr_phase_turns[slot] = CurTurn() + 1;
            if(xdr_test_mode) xdr_test_last_event = 'корабль заправлен; ближайшая цель нужной серии пересчитана';
            exit;
        }
        if(!ShipInNormalSpace(slot_ship)) exit;
        XDR_TransferFuel(slot, slot_ship);
        if(StarOwner(takeoff_current_star) == 1)
        {
            XDR_ReassertGhost(slot, slot_ship);
            XDR_ApplyMask(slot, slot_ship, takeoff_current_star);
            if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, takeoff_current_star);
        }
        else XDR_RestoreMask(slot, slot_ship);
        if(!XDR_BuildRouteToTarget(slot, slot_ship, takeoff_current_star, xdr_series[slot]))
        {
            // Return to a pirate base without discarding the paid charge.
            // State 2 will keep checking the same series until expiry.
            if(XDR_BuildRouteToBase(slot, slot_ship, takeoff_current_star))
            {
                xdr_states[slot] = 1;
                if(xdr_test_mode) xdr_test_last_event = 'цель исчезла; заряд сохранён, начат возврат на базу';
                exit;
            }
            OrderNone(slot_ship);
            if(xdr_test_mode) xdr_test_last_event = 'цель исчезла; заряд сохранён, безопасного возврата пока нет';
            exit;
        }
        takeoff_target = XDR_FindStarById(xdr_target_star_ids[slot]);
        dword takeoff_next_star = XDR_NextRouteStar(slot);
        if(!takeoff_next_star)
        {
            if(XDR_BuildRouteToBase(slot, slot_ship, takeoff_current_star))
            {
                xdr_states[slot] = 1;
                if(xdr_test_mode) xdr_test_last_event = 'маршрут цели пуст; заряд сохранён, корабль возвращается на базу';
                exit;
            }
            OrderNone(slot_ship);
            exit;
        }
        if(xdr_test_mode) xdr_test_last_event = 'маршрут пересчитан к ближайшему ценному луту нужной серии';
        // Route construction may use a physical-reach probe, but a real jump
        // must include current fuel or the engine can keep an impossible order.
        if(!ShipCanJump(slot_ship, takeoff_current_star, takeoff_next_star, 1))
        {
            dword takeoff_refuel_base = StarRuins(takeoff_current_star, 'PB');
            if(takeoff_refuel_base)
            {
                if(RelationToRanger(takeoff_refuel_base, slot_ship) >= 10)
                {
                    OrderLanding(slot_ship, takeoff_refuel_base, 1);
                    exit;
                }
            }
            dword takeoff_refuel_planet = XDR_FindSafePlanet(takeoff_current_star);
            if(takeoff_refuel_planet)
            {
                OrderLanding(slot_ship, takeoff_refuel_planet, 1);
                exit;
            }
            XDR_DetachAfterSpentCharge(slot, slot_ship);
            if(xdr_test_mode) xdr_test_last_event = 'по пути негде заправиться; камуфляж потрачен, рейнджер освобождён';
            exit;
        }
        if(StarOwner(takeoff_next_star) == 1)
        {
            // OrderJump can take several days while the ship approaches the
            // jump point. State 14 waits until the final calculated day before
            // arming the combat identity; the visible hull changes in hyper.
            XDR_RestoreMask(slot, slot_ship);
            XDR_OrderLockedJump(slot_ship, takeoff_next_star);
            xdr_states[slot] = 14;
            xdr_phase_turns[slot] = CurTurn();
            if(xdr_test_mode) xdr_test_last_event = 'прыжок назначен; маскировка ждёт последнего хода';
            exit;
        }
        XDR_RestoreMask(slot, slot_ship);
        XDR_OrderLockedJump(slot_ship, takeoff_next_star);
        xdr_states[slot] = 4;
        xdr_phase_turns[slot] = CurTurn();
        exit;
    }

    if(slot_state == 14)
    {
        if(ShipInHyperSpace(slot_ship, 1))
        {
            // The combat identity was armed on the final approach day, but the
            // visible hull is changed only after the old system renderer is gone.
            // This avoids both turn-animation skips and a stale duplicate model.
            dword hyper_target_star = XDR_FindStarById(xdr_target_star_ids[slot]);
            if(hyper_target_star)
            {
                if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship))
                    XDR_ApplyVisualIdentity(slot, slot_ship, hyper_target_star);
            }
            xdr_states[slot] = 4;
            exit;
        }
        if(ShipIsTakeoff(slot_ship) || !ShipInNormalSpace(slot_ship)) exit;
        dword staged_target_star = XDR_FindStarById(xdr_target_star_ids[slot]);
        dword staged_order_star = ShipOrderObj(slot_ship);
        if(ShipOrder(slot_ship) != 3 || !staged_target_star || staged_order_star != staged_target_star)
        {
            XDR_RestoreMask(slot, slot_ship);
            xdr_states[slot] = 3;
            exit;
        }
        int staged_target_lost = 0;
        if(!XDR_IsTargetValid(staged_target_star, xdr_series[slot], 0)) staged_target_lost = 1;
        int staged_quick_target = 0;
        if(xdr_test_mode && xdr_test_quick_stage == 3 && slot == xdr_test_quick_slot)
        {
            if(xdr_target_star_ids[slot] == xdr_test_quick_target_star_id) staged_quick_target = 1;
        }
        if(!staged_target_lost && !staged_quick_target)
        {
            if(!XDR_StarHasValuableLoot(slot_ship, staged_target_star)) staged_target_lost = 1;
        }
        if(staged_target_lost)
        {
            OrderNone(slot_ship);
            XDR_RestoreMask(slot, slot_ship);
            xdr_states[slot] = 3;
            if(xdr_test_mode) xdr_test_last_event = 'цель пропала до прыжка; маршрут будет пересчитан';
            exit;
        }
        if(!xdr_ghost_flags[slot] && ShipTurnBeforeEndOrder(slot_ship) <= 1)
        {
            if(!XDR_ArmMaskForJump(slot, slot_ship, staged_target_star))
            {
                OrderNone(slot_ship);
                xdr_states[slot] = 3;
                xdr_disabled = 1;
                if(xdr_test_mode) xdr_test_last_event = 'вылет отменён: нативная маскировка не подтвердилась';
                exit;
            }
            if(xdr_test_mode) xdr_test_last_event = 'боевая маскировка включена на последнем ходу; облик будет сменён в гипере';
            exit;
        }
        if(xdr_ghost_flags[slot]) XDR_ReassertGhost(slot, slot_ship);
        exit;
    }

    if(slot_state == 4)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        if(!ShipInNormalSpace(slot_ship)) exit;
        dword outbound_current_star = XDR_CurrentStar(slot_ship);
        dword outbound_target = XDR_FindStarById(xdr_target_star_ids[slot]);
        XDR_RecordSystemEntryPoint(slot, slot_ship);
        if(StarOwner(outbound_current_star) == 1)
        {
            XDR_RefreshMaskAfterJump(slot, slot_ship, outbound_current_star);
            if(xdr_test_mode) xdr_test_last_event = 'маскировка закреплена сразу после прибытия';
        }
        else XDR_RestoreMask(slot, slot_ship);
        int outbound_wait_for_battle = 0;
        if(outbound_current_star && outbound_target)
        {
            if(outbound_current_star == outbound_target)
            {
                if(StarOwner(outbound_current_star) == 1)
                {
                    if(StarSeries(outbound_current_star) == xdr_series[slot])
                    {
                        if(XDR_IsCoalitionAssault(outbound_current_star)) outbound_wait_for_battle = 1;
                    }
                }
            }
        }
        if(outbound_wait_for_battle)
        {
            XDR_CancelPickup(slot, slot_ship);
            XDR_OrderBattleWait(slot, slot_ship);
            xdr_states[slot] = 16;
            xdr_phase_turns[slot] = CurTurn();
            if(xdr_test_mode) xdr_test_last_event = 'в целевой системе идёт бой; сталкер уходит к краю и ждёт';
            exit;
        }
        int outbound_at_valuable_target = 0;
        if(XDR_IsTargetValid(outbound_current_star, xdr_series[slot], 0))
        {
            if(XDR_StarHasValuableLoot(slot_ship, outbound_current_star))
            {
                xdr_target_star_ids[slot] = Id(outbound_current_star);
                outbound_target = outbound_current_star;
                outbound_at_valuable_target = 1;
            }
        }
        if(outbound_at_valuable_target)
        {
            xdr_last_hull_damage[slot] = HullDamage(slot_ship);
            xdr_phase_turns[slot] = CurTurn();
            xdr_no_loot_counts[slot] = 0;
            xdr_states[slot] = 5;
            exit;
        }
        xdr_states[slot] = 3;
        if(xdr_test_mode) xdr_test_last_event = 'достигнута промежуточная система; маршрут будет проверен перед следующим прыжком';
        exit;
    }

    if(slot_state == 5)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship) || !ShipInNormalSpace(slot_ship)) exit;
        dword salvage_star = XDR_CurrentStar(slot_ship);
        // Also heals active missions loaded from a pre-persistent-ghost save.
        // Such saves can already be in salvage state with ghost_flags == 0.
        if(salvage_star)
        {
            if(StarOwner(salvage_star) == 1)
            {
                XDR_ReassertGhost(slot, slot_ship);
                if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, salvage_star);
                if(ShipGetBad(slot_ship)) ShipSetBad(slot_ship, 0);
            }
        }
        dword salvage_target = XDR_FindStarById(xdr_target_star_ids[slot]);
        int salvage_wait_for_battle = 0;
        if(salvage_target && salvage_star)
        {
            if(salvage_star == salvage_target)
            {
                if(StarOwner(salvage_target) == 1)
                {
                    if(StarSeries(salvage_target) == xdr_series[slot])
                    {
                        if(XDR_IsCoalitionAssault(salvage_target)) salvage_wait_for_battle = 1;
                    }
                }
            }
        }
        if(salvage_wait_for_battle)
        {
            XDR_CancelPickup(slot, slot_ship);
            XDR_OrderBattleWait(slot, slot_ship);
            xdr_states[slot] = 16;
            xdr_phase_turns[slot] = CurTurn();
            if(xdr_test_mode) xdr_test_last_event = 'началось освобождение; сбор приостановлен до конца боя';
            exit;
        }
        int salvage_abort = !salvage_target || salvage_star != salvage_target || !XDR_IsTargetValid(salvage_target, xdr_series[slot], 0);
        if(xdr_test_mode && xdr_test_force_abort)
        {
            salvage_abort = 1;
            xdr_test_force_abort = 0;
            xdr_test_last_event = 'принудительная потеря цели перевела миссию в отход';
        }
        if(salvage_abort)
        {
            XDR_CancelPickup(slot, slot_ship);
            xdr_states[slot] = 6;
            exit;
        }
        if(XDR_HandleThreat(slot, slot_ship, salvage_star))
        {
            XDR_CancelPickup(slot, slot_ship);
            xdr_states[slot] = 6;
            exit;
        }
        if(xdr_pending_item_ids[slot] > 1)
        {
            dword salvage_pending = IdToItem(xdr_pending_item_ids[slot]);
            if(salvage_pending && ItemIn(salvage_pending, slot_ship))
            {
                ShipPicksItem(slot_ship, salvage_pending, 0);
                XDR_RecordScavengedFuel(slot, salvage_pending);
                xdr_pending_item_ids[slot] = 0;
                xdr_pending_since[slot] = 0;
                xdr_no_loot_counts[slot] = 0;
                XDR_TransferFuel(slot, slot_ship);
                int salvage_continue_result = XDR_PrepareNextLoot(slot, slot_ship, salvage_star);
                if(!salvage_continue_result) xdr_no_loot_counts[slot] = xdr_no_loot_counts[slot] + 1;
                if(!salvage_continue_result && xdr_no_loot_counts[slot] >= XDR_ConfigInt(14, 3)) xdr_states[slot] = 6;
                exit;
            }
            int salvage_pending_start_turn = xdr_pending_since[slot];
            if(salvage_pending_start_turn < 0) salvage_pending_start_turn = -salvage_pending_start_turn - 1;
            if(!salvage_pending || !ItemIn(salvage_pending, salvage_star) || CurTurn() - salvage_pending_start_turn > 5)
            {
                XDR_CancelPickup(slot, slot_ship);
                int salvage_has_displaced_after_loss = XDR_HasDisplacedLoot(slot, slot_ship, salvage_star);
                if(salvage_has_displaced_after_loss) xdr_no_loot_counts[slot] = 0;
                else xdr_no_loot_counts[slot] = xdr_no_loot_counts[slot] + 1;
                if(!salvage_has_displaced_after_loss && xdr_no_loot_counts[slot] >= XDR_ConfigInt(14, 3)) xdr_states[slot] = 6;
                exit;
            }
            XDR_StartMarkedPickup(slot, slot_ship, salvage_pending, CurTurn() == salvage_pending_start_turn);
            exit;
        }
        int salvage_prepare_result = XDR_PrepareNextLoot(slot, slot_ship, salvage_star);
        if(salvage_prepare_result) exit;
        if(CurTurn() - xdr_phase_turns[slot] >= XDR_ConfigInt(13, 20) || ShipFreeSpace(slot_ship) <= 0)
        {
            xdr_states[slot] = 6;
            exit;
        }
        xdr_no_loot_counts[slot] = xdr_no_loot_counts[slot] + 1;
        if(xdr_no_loot_counts[slot] >= XDR_ConfigInt(14, 3)) xdr_states[slot] = 6;
        exit;
    }

    if(slot_state == 16)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship) || !ShipInNormalSpace(slot_ship)) exit;
        dword battle_wait_star = XDR_CurrentStar(slot_ship);
        dword battle_wait_target = XDR_FindStarById(xdr_target_star_ids[slot]);
        int battle_wait_identity_valid = 0;
        if(battle_wait_star && battle_wait_target)
        {
            if(battle_wait_star == battle_wait_target)
            {
                if(StarOwner(battle_wait_star) == 1)
                {
                    if(StarSeries(battle_wait_star) == xdr_series[slot]) battle_wait_identity_valid = 1;
                }
            }
        }
        if(!battle_wait_identity_valid)
        {
            XDR_CancelPickup(slot, slot_ship);
            XDR_ClearBattleWait(slot);
            XDR_RestoreMask(slot, slot_ship);
            xdr_target_star_ids[slot] = 0;
            xdr_states[slot] = 6;
            slot_state = 6;
        }
        else
        {
            XDR_ReassertGhost(slot, slot_ship);
            XDR_ApplyMask(slot, slot_ship, battle_wait_star);
            if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, battle_wait_star);
            dword battle_military_attacker = XDR_FindMilitaryAttacker(slot_ship, battle_wait_star);
            if(battle_military_attacker)
            {
                XDR_CancelPickup(slot, slot_ship);
                XDR_ClearBattleWait(slot);
                XDR_RestoreMask(slot, slot_ship);
                if(ShipGetBad(battle_military_attacker) == slot_ship)
                {
                    ShipSetBad(battle_military_attacker, 0);
                    OrderNone(battle_military_attacker);
                }
                if(ShipGetBad(slot_ship) != 0) ShipSetBad(slot_ship, 0);
                xdr_target_star_ids[slot] = 0;
                xdr_states[slot] = 6;
                slot_state = 6;
                xdr_phase_turns[slot] = CurTurn();
                if(xdr_test_mode) xdr_test_last_event = 'военные атаковали сталкера; маскировка снята, начат отход';
            }
            else if(XDR_IsCoalitionAssault(battle_wait_star))
            {
                XDR_KeepDefensiveTargetOnly(slot_ship);
                int battle_wait_x = xdr_route_slots[slot];
                int battle_wait_y = xdr_route_orders[slot];
                if(!battle_wait_x && !battle_wait_y)
                {
                    XDR_OrderBattleWait(slot, slot_ship);
                    exit;
                }
                if(XDR_BattleClearance(battle_wait_star, slot_ship, battle_wait_x, battle_wait_y) < 1200)
                {
                    // The front reached the real entry edge. Do not send the
                    // ranger across the battle in search of a different exit:
                    // cancel opportunistic salvage and hold at its known,
                    // engine-provided arrival point until withdrawal is safe.
                    XDR_CancelPickup(slot, slot_ship);
                    XDR_OrderBattleWait(slot, slot_ship);
                    exit;
                }
                if(xdr_pending_item_ids[slot] > 1)
                {
                    dword battle_pending_item = IdToItem(xdr_pending_item_ids[slot]);
                    if(battle_pending_item && ItemIn(battle_pending_item, slot_ship))
                    {
                        ShipPicksItem(slot_ship, battle_pending_item, 0);
                        XDR_RecordScavengedFuel(slot, battle_pending_item);
                        xdr_pending_item_ids[slot] = 0;
                        xdr_pending_since[slot] = 0;
                        XDR_TransferFuel(slot, slot_ship);
                        exit;
                    }
                    int battle_pending_start_turn = xdr_pending_since[slot];
                    if(battle_pending_start_turn < 0) battle_pending_start_turn = -battle_pending_start_turn - 1;
                    if(!battle_pending_item || !ItemIn(battle_pending_item, battle_wait_star) || CurTurn() - battle_pending_start_turn > 5)
                    {
                        XDR_CancelPickup(slot, slot_ship);
                        exit;
                    }
                    if(XDR_BattleClearance(battle_wait_star, slot_ship, CoordX(battle_pending_item), CoordY(battle_pending_item)) < 1400)
                    {
                        XDR_CancelPickup(slot, slot_ship);
                        XDR_OrderBattleWait(slot, slot_ship);
                        exit;
                    }
                    XDR_StartMarkedPickup(slot, slot_ship, battle_pending_item, CurTurn() == battle_pending_start_turn);
                    exit;
                }
                int battle_ignore_cisterns = XDR_HasFuelForReturn(slot, slot_ship, battle_wait_star);
                dword battle_edge_loot = XDR_SelectBattleEdgeLoot(slot, slot_ship, battle_wait_star, battle_ignore_cisterns);
                if(battle_edge_loot)
                {
                    xdr_pending_item_ids[slot] = Id(battle_edge_loot);
                    xdr_pending_since[slot] = CurTurn();
                    XDR_StartMarkedPickup(slot, slot_ship, battle_edge_loot, 1);
                    if(xdr_test_mode) xdr_test_last_event = 'у точки ожидания найден безопасный лут; сталкер подбирает его под шум боя';
                    exit;
                }
                XDR_OrderBattleWait(slot, slot_ship);
                exit;
            }
            else
            {
                XDR_ClearBattleWait(slot);
                if(XDR_StarHasValuableLoot(slot_ship, battle_wait_star))
                {
                    xdr_states[slot] = 5;
                    xdr_phase_turns[slot] = CurTurn();
                    xdr_no_loot_counts[slot] = 0;
                    XDR_KeepDefensiveTargetOnly(slot_ship);
                    if(xdr_test_mode) xdr_test_last_event = 'военные уничтожены; сталкер возобновил сбор добычи';
                    exit;
                }
                xdr_states[slot] = 6;
                slot_state = 6;
            }
        }
    }

    if(slot_state == 6)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        XDR_ClearBattleWait(slot);
        dword fuel_current_star = XDR_CurrentStar(slot_ship);
        if(!fuel_current_star) exit;
        int fuel_inside_same_series = StarOwner(fuel_current_star) == 1 && StarSeries(fuel_current_star) == xdr_series[slot];
        if(fuel_inside_same_series)
        {
            XDR_ReassertGhost(slot, slot_ship);
            if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, fuel_current_star);
            if(ShipGetBad(slot_ship)) ShipSetBad(slot_ship, 0);
            XDR_HandleThreat(slot, slot_ship, fuel_current_star);
        }
        else XDR_RestoreMask(slot, slot_ship);
        XDR_CancelPickup(slot, slot_ship);
        XDR_TransferFuel(slot, slot_ship);
        if(!XDR_SelectReturnPlace(slot, slot_ship, fuel_current_star))
        {
            xdr_states[slot] = 12;
            exit;
        }
        dword fuel_return_place = XDR_GetReturnPlace(slot);
        dword fuel_return_star = XDR_ReturnPlaceStar(slot);
        if(!fuel_return_place || !fuel_return_star)
        {
            xdr_return_place_ids[slot] = 0;
            exit;
        }
        if(fuel_current_star == fuel_return_star)
        {
            xdr_states[slot] = 9;
            exit;
        }
        dword fuel_next_star = XDR_NextRouteStar(slot);
        if(!fuel_next_star)
        {
            xdr_states[slot] = 12;
            exit;
        }
        if(!XDR_ReturnTransitAllowed(slot, fuel_next_star))
        {
            XDR_ClearRoute(slot);
            xdr_return_place_ids[slot] = 0;
            xdr_return_kinds[slot] = 0;
            exit;
        }
        if(ShipCanJump(slot_ship, fuel_current_star, fuel_next_star, 1))
        {
            XDR_OrderLockedJump(slot_ship, fuel_next_star);
            xdr_states[slot] = 15;
            exit;
        }
        xdr_states[slot] = 7;
        xdr_phase_turns[slot] = CurTurn();
        xdr_last_hull_damage[slot] = HullDamage(slot_ship);
        // State 7 reuses the now-free pickup fields to watch real solar-fuel
        // progress without extending the persistent storage schema.
        xdr_pending_since[slot] = ShipFuel(slot_ship);
        xdr_no_loot_counts[slot] = 0;
        exit;
    }

    if(slot_state == 7)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship) || !ShipInNormalSpace(slot_ship)) exit;
        dword solar_current_star = XDR_CurrentStar(slot_ship);
        if(!solar_current_star) exit;
        // Solar charging is never attempted while a real battle is present.
        // State 16 keeps the ship at a bounded engine-derived edge point and
        // returns here only after the military assault has ended.
        if(XDR_IsMissionBlockingBattle(solar_current_star))
        {
            XDR_CancelPickup(slot, slot_ship);
            XDR_OrderBattleWait(slot, slot_ship);
            xdr_states[slot] = 16;
            xdr_phase_turns[slot] = CurTurn();
            if(xdr_test_mode) xdr_test_last_event = 'солнечная зарядка отложена: в системе идёт настоящий бой, сталкер ждёт у безопасного края';
            exit;
        }
        int solar_current_radius = DistCoords(CoordX(slot_ship), CoordY(slot_ship), 0, 0);
        int solar_current_damage = HullDamage(slot_ship);
        int solar_damage_boundary = 0;
        int solar_boundary_just_detected = 0;
        if(xdr_phase_turns[slot] < 0) solar_damage_boundary = -xdr_phase_turns[slot] - 1;
        if(solar_damage_boundary <= 0 && solar_current_radius <= XDR_ConfigInt(21, 600) && solar_current_damage > xdr_last_hull_damage[slot])
        {
            dword solar_nonstar_attacker = XDR_FindActualAttacker(slot_ship, solar_current_star);
            if(!solar_nonstar_attacker)
            {
                solar_damage_boundary = max(1, solar_current_radius);
                xdr_phase_turns[slot] = -solar_damage_boundary - 1;
                solar_boundary_just_detected = 1;
                if(xdr_test_mode) xdr_test_last_event = 'зафиксирован фактический радиус начала урона звезды; он принят как граница солнечной зарядки';
            }
        }
        xdr_last_hull_damage[slot] = solar_current_damage;
        int solar_inside_same_series = StarOwner(solar_current_star) == 1 && StarSeries(solar_current_star) == xdr_series[slot];
        if(solar_inside_same_series)
        {
            XDR_ReassertGhost(slot, slot_ship);
            if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, solar_current_star);
            if(ShipGetBad(slot_ship)) ShipSetBad(slot_ship, 0);
            XDR_HandleThreat(slot, slot_ship, solar_current_star);
        }
        else XDR_RestoreMask(slot, slot_ship);
        XDR_TransferFuel(slot, slot_ship);
        if(!XDR_SelectReturnPlace(slot, slot_ship, solar_current_star))
        {
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = 0;
            if(HullDamage(slot_ship) >= XDR_ConfigInt(24, 60)) XDR_EnterCriticalShelter(slot, slot_ship, solar_current_star);
            else
            {
                XDR_OrderSolarRetreat(slot_ship);
                xdr_states[slot] = 12;
            }
            exit;
        }
        dword solar_return_star = XDR_ReturnPlaceStar(slot);
        if(!solar_return_star)
        {
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = 0;
            if(HullDamage(slot_ship) >= XDR_ConfigInt(24, 60)) XDR_EnterCriticalShelter(slot, slot_ship, solar_current_star);
            else
            {
                XDR_OrderSolarRetreat(slot_ship);
                xdr_states[slot] = 12;
            }
            exit;
        }
        if(solar_current_star == solar_return_star)
        {
            xdr_states[slot] = 9;
            exit;
        }
        dword solar_next_star = XDR_NextRouteStar(slot);
        if(!solar_next_star)
        {
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = 0;
            if(HullDamage(slot_ship) >= XDR_ConfigInt(24, 60)) XDR_EnterCriticalShelter(slot, slot_ship, solar_current_star);
            else
            {
                XDR_OrderSolarRetreat(slot_ship);
                xdr_states[slot] = 12;
            }
            exit;
        }
        if(!XDR_ReturnTransitAllowed(slot, solar_next_star))
        {
            XDR_ClearRoute(slot);
            xdr_return_place_ids[slot] = 0;
            xdr_return_kinds[slot] = 0;
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = 0;
            if(HullDamage(slot_ship) >= XDR_ConfigInt(24, 60)) XDR_EnterCriticalShelter(slot, slot_ship, solar_current_star);
            else
            {
                XDR_OrderSolarRetreat(slot_ship);
                xdr_states[slot] = 12;
            }
            exit;
        }
        // Solar charging has a deliberately stricter completion rule than a
        // cistern: remain at the real charging boundary until the installed
        // tank is full, then leave on this same controller pass. Cistern logic
        // elsewhere stops as soon as fuel covers the complete return route.
        int solar_tank_full = XDR_FuelTankIsFull(slot_ship);
        if(solar_tank_full && ShipCanJump(slot_ship, solar_current_star, solar_next_star, 1))
        {
            XDR_OrderLockedJump(slot_ship, solar_next_star);
            xdr_states[slot] = 15;
            xdr_pending_since[slot] = 0;
            xdr_no_loot_counts[slot] = 0;
            if(xdr_test_mode) xdr_test_last_event = 'солнечная зарядка заполнила бак; прыжок на выход заказан немедленно';
            exit;
        }
        if(solar_tank_full)
        {
            // A full tank that still cannot make the first safe transition
            // cannot benefit from more solar exposure. Retreat and wait for a
            // route change or real fuel service instead of taking lethal heat.
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = 0;
            if(HullDamage(slot_ship) >= XDR_ConfigInt(24, 60)) XDR_EnterCriticalShelter(slot, slot_ship, solar_current_star);
            else
            {
                XDR_OrderSolarRetreat(slot_ship);
                xdr_states[slot] = 12;
                if(xdr_test_mode) xdr_test_last_event = 'бак полон, но первый безопасный переход недоступен; сталкер отошёл от звезды';
            }
            exit;
        }
        // Only a real increase of ShipFuel counts as successful charging. The
        // first damaged turn establishes the engine-specific boundary and is
        // not counted as a stall. Three later boundary turns without growth
        // terminate the attempt, so a full/unsuitable tank cannot loop here.
        int solar_current_fuel = ShipFuel(slot_ship);
        if(xdr_pending_since[slot] < 0) xdr_pending_since[slot] = solar_current_fuel;
        if(solar_current_fuel > xdr_pending_since[slot])
        {
            xdr_pending_since[slot] = solar_current_fuel;
            xdr_no_loot_counts[slot] = 0;
            if(xdr_test_mode) xdr_test_last_event = 'солнечная зарядка увеличила запас топлива; сталкер ждёт полного бака';
        }
        else
        {
            if(solar_damage_boundary > 0 && !solar_boundary_just_detected && solar_current_radius <= solar_damage_boundary + 20)
            {
                xdr_no_loot_counts[slot] = xdr_no_loot_counts[slot] + 1;
            }
        }
        if(HullDamage(slot_ship) >= XDR_ConfigInt(24, 60))
        {
            XDR_EnterCriticalShelter(slot, slot_ship, solar_current_star);
            exit;
        }
        if(xdr_no_loot_counts[slot] >= XDR_ConfigInt(25, 3))
        {
            // Negative value is a local state-12 sentinel: do not immediately
            // send a repaired ship back into a solar attempt that cannot fuel it.
            xdr_pending_since[slot] = -solar_current_fuel - 1;
            xdr_no_loot_counts[slot] = 0;
            XDR_OrderSolarRetreat(slot_ship);
            xdr_states[slot] = 12;
            if(xdr_test_mode) xdr_test_last_event = 'солнечная зарядка не даёт топлива; сталкер отошёл от звезды и перешёл к поиску реального топлива';
            exit;
        }
        XDR_OrderSolarApproach(slot_ship, solar_damage_boundary);
        exit;
    }

    if(slot_state == 17)
    {
        // Critical-hull shelter. The script API has no CanLand call, so an
        // uninhabited planet is tried once and the actual docked planet is
        // observed for five days. Failure falls back to a bounded point outside
        // the damaging star radius; the ranger never approaches the sun again.
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        dword shelter_current_star = XDR_CurrentStar(slot_ship);
        if(!shelter_current_star) exit;
        int shelter_inside_same_series = StarOwner(shelter_current_star) == 1 && StarSeries(shelter_current_star) == xdr_series[slot];
        if(shelter_inside_same_series)
        {
            XDR_ReassertGhost(slot, slot_ship);
            if(ShipInNormalSpace(slot_ship))
            {
                if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, shelter_current_star);
                if(ShipGetBad(slot_ship)) ShipSetBad(slot_ship, 0);
                XDR_HandleThreat(slot, slot_ship, shelter_current_star);
            }
        }
        else XDR_RestoreMask(slot, slot_ship);

        dword shelter_planet = 0;
        if(xdr_route_slots[slot] > 1) shelter_planet = IdToPlanet(xdr_route_slots[slot]);
        dword shelter_docked_planet = GetShipPlanet(slot_ship);
        if(shelter_docked_planet)
        {
            // Landing on any object other than the attempted shelter is not a
            // valid wait state; take off and continue with the space fallback.
            if(!shelter_planet || shelter_docked_planet != shelter_planet)
            {
                XDR_ClearBattleWait(slot);
                xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
                xdr_no_loot_counts[slot] = -1;
                xdr_states[slot] = 18;
                OrderTakeOff(slot_ship);
                exit;
            }
            XDR_SelectReturnPlace(slot, slot_ship, shelter_current_star);
            int shelter_has_return_fuel = XDR_HasFuelForReturn(slot, slot_ship, shelter_current_star);
            dword shelter_new_cistern = 0;
            if(!shelter_has_return_fuel && !XDR_IsMissionBlockingBattle(shelter_current_star))
                shelter_new_cistern = XDR_SelectFuelCistern(slot, slot_ship, shelter_current_star);
            if(shelter_has_return_fuel || shelter_new_cistern)
            {
                XDR_ClearBattleWait(slot);
                xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
                xdr_no_loot_counts[slot] = -1;
                xdr_states[slot] = 18;
                OrderTakeOff(slot_ship);
                if(xdr_test_mode)
                {
                    if(shelter_has_return_fuel) xdr_test_last_event = 'на аварийной стоянке уже хватает топлива на весь путь; сталкер взлетает';
                    else xdr_test_last_event = 'в системе появилась цистерна; сталкер взлетает для аварийной заправки';
                }
            }
            exit;
        }
        if(!ShipInNormalSpace(slot_ship)) exit;

        XDR_SelectReturnPlace(slot, slot_ship, shelter_current_star);
        if(XDR_HasFuelForReturn(slot, slot_ship, shelter_current_star))
        {
            XDR_ClearBattleWait(slot);
            xdr_pending_since[slot] = 0;
            xdr_no_loot_counts[slot] = 0;
            xdr_states[slot] = 6;
            if(xdr_test_mode) xdr_test_last_event = 'аварийное ожидание завершено: топлива хватает на весь обратный маршрут';
            exit;
        }
        dword shelter_space_cistern = 0;
        if(!XDR_IsMissionBlockingBattle(shelter_current_star))
            shelter_space_cistern = XDR_SelectFuelCistern(slot, slot_ship, shelter_current_star);
        if(shelter_space_cistern)
        {
            XDR_ClearBattleWait(slot);
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = -1;
            xdr_states[slot] = 12;
            if(xdr_test_mode) xdr_test_last_event = 'в системе появилась цистерна; аварийный сталкер перешёл к её подбору';
            exit;
        }
        int shelter_landing_pending = 0;
        if(shelter_planet)
        {
            dword shelter_planet_star = PlanetToStar(shelter_planet);
            if(shelter_planet_star == shelter_current_star && CurTurn() - xdr_route_orders[slot] <= 5) shelter_landing_pending = 1;
        }
        if(shelter_landing_pending)
        {
            // Preserve the original OrderLanding. Reissuing it every day can
            // restart the engine landing sequence and create a takeoff loop.
            exit;
        }
        XDR_ClearBattleWait(slot);
        XDR_OrderSolarRetreat(slot_ship);
        exit;
    }

    if(slot_state == 18)
    {
        // One-turn boundary after an emergency takeoff. Do not run ShipCanJump
        // or item pickup against a ship that is still attached to a planet.
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        dword shelter_takeoff_planet = GetShipPlanet(slot_ship);
        dword shelter_takeoff_base = GetShipRuins(slot_ship);
        if(shelter_takeoff_planet || shelter_takeoff_base)
        {
            OrderTakeOff(slot_ship);
            exit;
        }
        if(!ShipInNormalSpace(slot_ship)) exit;
        dword shelter_takeoff_star = XDR_CurrentStar(slot_ship);
        if(!shelter_takeoff_star) exit;
        XDR_SelectReturnPlace(slot, slot_ship, shelter_takeoff_star);
        if(XDR_HasFuelForReturn(slot, slot_ship, shelter_takeoff_star))
        {
            xdr_pending_since[slot] = 0;
            xdr_no_loot_counts[slot] = 0;
            xdr_states[slot] = 6;
        }
        else
        {
            xdr_pending_since[slot] = -ShipFuel(slot_ship) - 1;
            xdr_no_loot_counts[slot] = -1;
            xdr_states[slot] = 12;
        }
        exit;
    }

    if(slot_state == 8)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        if(!ShipInNormalSpace(slot_ship)) exit;
        dword returning_current_star = XDR_CurrentStar(slot_ship);
        if(!returning_current_star) exit;
        if(StarOwner(returning_current_star) == 0 && !StarBattle(returning_current_star))
        {
            // The mission is over at the first safe Coalition system. Do not
            // keep the ranger locked merely to force a cosmetic landing: restore
            // the real identity and hand it straight back to vanilla behavior.
            XDR_CancelPickup(slot, slot_ship);
            XDR_RestoreMask(slot, slot_ship);
            XDR_DetachSlot(slot, slot_ship, 1);
            xdr_due_turns[slot] = CurTurn() + XDR_ConfigInt(12, 300);
            if(xdr_test_mode) xdr_due_turns[slot] = CurTurn() + 5;
            if(xdr_test_mode) xdr_test_last_event = 'сталкер достиг системы Коалиции и немедленно возвращён ванильному ИИ';
            exit;
        }
        int returning_inside_same_series = StarOwner(returning_current_star) == 1 && StarSeries(returning_current_star) == xdr_series[slot];
        if(returning_inside_same_series)
        {
            XDR_ReassertGhost(slot, slot_ship);
            if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, returning_current_star);
        }
        else XDR_RestoreMask(slot, slot_ship);
        dword returning_return_star = XDR_ReturnPlaceStar(slot);
        if(returning_return_star && returning_current_star == returning_return_star)
        {
            xdr_states[slot] = 9;
            exit;
        }
        XDR_ClearRoute(slot);
        xdr_states[slot] = 6;
        exit;
    }

    if(slot_state == 15)
    {
        if(ShipInHyperSpace(slot_ship, 1))
        {
            // Keep the full combat identity until the old system renderer and
            // every local AI have released the ship. Clearing it on the last
            // normal-space day exposed the ranger to Dominators and briefly
            // let its class AI acquire the player or another ship. Only remove
            // it here, inside hyper, unless the next system belongs to the same
            // Dominator series and therefore needs uninterrupted camouflage.
            dword hyper_return_destination = XDR_NextRouteStar(slot);
            int hyper_keep_same_series_mask = 0;
            if(hyper_return_destination)
            {
                if(StarOwner(hyper_return_destination) == 1)
                {
                    if(!XDR_IsCoalitionAssault(hyper_return_destination))
                    {
                        if(StarSeries(hyper_return_destination) == xdr_series[slot]) hyper_keep_same_series_mask = 1;
                    }
                }
            }
            if(!hyper_keep_same_series_mask)
            {
                XDR_RestoreMask(slot, slot_ship);
                if(xdr_test_mode) xdr_test_last_event = 'маскировка, имя и облик сброшены уже в гиперпространстве';
            }
            xdr_states[slot] = 8;
            exit;
        }
        if(ShipIsTakeoff(slot_ship) || !ShipInNormalSpace(slot_ship)) exit;
        dword unmask_current_star = XDR_CurrentStar(slot_ship);
        if(ShipOrder(slot_ship) != 3 || !ShipOrderObj(slot_ship))
        {
            xdr_states[slot] = 6;
            exit;
        }
        dword unmask_destination_star = ShipOrderObj(slot_ship);
        // The route may have been valid when OrderJump was issued and become
        // unsafe while the ranger was flying to the jump point. Cancel a
        // queued transition into StarBattle and rebuild the BFS immediately.
        if(!XDR_ReturnTransitAllowed(slot, unmask_destination_star))
        {
            OrderNone(slot_ship);
            XDR_ClearRoute(slot);
            xdr_return_place_ids[slot] = 0;
            xdr_return_kinds[slot] = 0;
            xdr_states[slot] = 6;
            if(xdr_ghost_flags[slot] && unmask_current_star)
            {
                if(StarOwner(unmask_current_star) == 1 && StarSeries(unmask_current_star) == xdr_series[slot])
                    XDR_ReassertGhost(slot, slot_ship);
            }
            if(xdr_test_mode) xdr_test_last_event = 'прыжок в систему с боем отменён; обратный маршрут перестроен';
            exit;
        }
        if(!xdr_ghost_flags[slot]) exit;
        if(!unmask_current_star)
        {
            XDR_RestoreMask(slot, slot_ship);
            exit;
        }
        // Never unmask in normal space. The hyper branch above performs the
        // reset after the ship has actually left this system.
        XDR_ReassertGhost(slot, slot_ship);
        exit;
    }

    if(slot_state == 9)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        dword landing_place = XDR_GetReturnPlace(slot);
        if(!landing_place)
        {
            xdr_states[slot] = 6;
            exit;
        }
        dword landing_planet = GetShipPlanet(slot_ship);
        dword landing_base = GetShipRuins(slot_ship);
        if(landing_planet == landing_place || landing_base == landing_place)
        {
            XDR_RestoreMask(slot, slot_ship);
            xdr_target_star_ids[slot] = 0;
            xdr_series[slot] = -1;
            xdr_charge_expire_turns[slot] = 0;
            xdr_return_place_ids[slot] = 0;
            xdr_return_kinds[slot] = 0;
            if(xdr_release_flags[slot])
            {
                XDR_DetachSlot(slot, slot_ship, 0);
                exit;
            }
            if(landing_base) xdr_home_base_ids[slot] = Id(landing_base);
            XDR_DetachSlot(slot, slot_ship, 1);
            xdr_due_turns[slot] = CurTurn() + XDR_ConfigInt(12, 300);
            if(xdr_test_mode) xdr_due_turns[slot] = CurTurn() + 5;
            if(xdr_test_mode) xdr_test_last_event = 'специалист вернулся, снял управление мода и продолжил обычные дела';
            exit;
        }
        if(landing_planet || landing_base)
        {
            XDR_TransferFuel(slot, slot_ship);
            XDR_ServiceRefuel(slot_ship, 0);
            OrderTakeOff(slot_ship);
            exit;
        }
        if(!ShipInNormalSpace(slot_ship)) exit;
        XDR_TransferFuel(slot, slot_ship);
        dword landing_current_star = XDR_CurrentStar(slot_ship);
        dword landing_return_star = XDR_ReturnPlaceStar(slot);
        if(landing_current_star == landing_return_star)
        {
            OrderLanding(slot_ship, landing_place, 1);
            exit;
        }
        xdr_states[slot] = 6;
        exit;
    }

    if(slot_state == 10)
    {
        dword rest_base = XDR_FindPirateBaseById(xdr_home_base_ids[slot]);
        if(!rest_base || XDR_CountPirateBases() <= 0)
        {
            XDR_DetachSlot(slot, slot_ship, 1);
            exit;
        }
        if(CurTurn() < xdr_due_turns[slot]) exit;
        if(GetShipRuins(slot_ship) == rest_base)
        {
            xdr_states[slot] = 2;
            exit;
        }
        dword rest_current_star = XDR_CurrentStar(slot_ship);
        if(rest_current_star && XDR_BuildRouteToBase(slot, slot_ship, rest_current_star))
        {
            xdr_states[slot] = 1;
            exit;
        }
        XDR_DetachSlot(slot, slot_ship, 1);
        exit;
    }

    if(slot_state == 11)
    {
        if(CurTurn() < xdr_due_turns[slot] || XDR_CountPirateBases() <= 0) exit;
        int dormant_camo_cost = XDR_CamouflageCost(slot_ship);
        if(xdr_phase_turns[slot] > 0)
        {
            if(ShipMoney(slot_ship) < xdr_phase_turns[slot])
            {
                xdr_due_turns[slot] = CurTurn() + 1;
                if(xdr_test_mode) xdr_test_last_event = 'потраченный заряд: рейнджер зарабатывает цену следующего камуфляжа';
                exit;
            }
            xdr_phase_turns[slot] = 0;
        }
        if(ShipMoney(slot_ship) < dormant_camo_cost)
        {
            xdr_due_turns[slot] = CurTurn() + 1;
            if(xdr_test_mode) xdr_test_last_event = 'неактивный специалист занимается обычными делами и копит на камуфляж';
            exit;
        }
        if(ShipInScript(slot_ship, 0) || ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        dword dormant_star = XDR_CurrentStar(slot_ship);
        if(!dormant_star) exit;
        if(StarBattle(dormant_star)) exit;
        if(!XDR_BuildRouteToBase(slot, slot_ship, dormant_star))
        {
            xdr_due_turns[slot] = CurTurn() + 1;
            exit;
        }
        ShipJoin(XenoDomSpecialists, slot_ship, 'XenoDomState');
        OrderLock(slot_ship, 1);
        xdr_states[slot] = 1;
        exit;
    }

    if(slot_state == 13)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship)) exit;
        if(!ShipInNormalSpace(slot_ship))
        {
            if(xdr_test_mode) xdr_test_last_event = 'сталкер ждёт появления в обычном космосе перед поиском базы';
            exit;
        }
        if(CurTurn() < xdr_due_turns[slot])
        {
            OrderNone(slot_ship);
            exit;
        }
        dword preview_star = XDR_CurrentStar(slot_ship);
        if(!preview_star)
        {
            XDR_DetachSlot(slot, slot_ship, 1);
            exit;
        }
        if(XDR_IsMissionBlockingBattle(preview_star) || XDR_CountPirateBases() <= 0)
        {
            OrderNone(slot_ship);
            xdr_due_turns[slot] = CurTurn() + 1;
            if(xdr_test_mode) xdr_test_last_event = 'сталкер удерживается рядом: безопасной пиратской базы пока нет';
            exit;
        }
        if(!XDR_BuildRouteToBase(slot, slot_ship, preview_star))
        {
            OrderNone(slot_ship);
            xdr_due_turns[slot] = CurTurn() + 1;
            if(xdr_test_mode) xdr_test_last_event = 'сталкер удерживается рядом: достижимой безопасной базы нет';
            exit;
        }
        dword preview_route_star = XDR_NextRouteStar(slot);
        if(!preview_route_star)
        {
            XDR_ClearRoute(slot);
            OrderNone(slot_ship);
            xdr_due_turns[slot] = CurTurn() + 1;
            if(xdr_test_mode) xdr_test_last_event = 'защита отклонила пустой первый переход к базе';
            exit;
        }
        if(StarOwner(preview_route_star) == 1)
        {
            XDR_ClearRoute(slot);
            OrderNone(slot_ship);
            xdr_due_turns[slot] = CurTurn() + 1;
            if(xdr_test_mode) xdr_test_last_event = 'защита отклонила первый переход к доминаторам без камуфляжа';
            exit;
        }
        xdr_states[slot] = 1;
        if(xdr_test_mode) xdr_test_last_event = 'показ сталкера завершён; проложен маршрут к пиратской базе';
        exit;
    }

    if(slot_state == 12)
    {
        if(ShipInHyperSpace(slot_ship, 1) || ShipIsTakeoff(slot_ship) || !ShipInNormalSpace(slot_ship)) exit;
        dword stranded_star = XDR_CurrentStar(slot_ship);
        if(!stranded_star) exit;
        int stranded_inside_same_series = StarOwner(stranded_star) == 1 && StarSeries(stranded_star) == xdr_series[slot];
        if(stranded_inside_same_series)
        {
            XDR_ReassertGhost(slot, slot_ship);
            if(!xdr_visual_flags[slot] || !IsChameleon(slot_ship)) XDR_ApplyVisualIdentity(slot, slot_ship, stranded_star);
            if(ShipGetBad(slot_ship)) ShipSetBad(slot_ship, 0);
            XDR_HandleThreat(slot, slot_ship, stranded_star);
        }
        else XDR_RestoreMask(slot, slot_ship);
        XDR_TransferFuel(slot, slot_ship);
        XDR_SelectReturnPlace(slot, slot_ship, stranded_star);
        dword stranded_return_star = XDR_ReturnPlaceStar(slot);
        if(stranded_return_star && stranded_star == stranded_return_star)
        {
            xdr_states[slot] = 9;
            exit;
        }
        int stranded_has_return_fuel = XDR_HasFuelForReturn(slot, slot_ship, stranded_star);
        if(stranded_has_return_fuel)
        {
            XDR_CancelPickup(slot, slot_ship);
            xdr_pending_since[slot] = 0;
            xdr_no_loot_counts[slot] = 0;
            xdr_states[slot] = 6;
            exit;
        }
        if(xdr_no_loot_counts[slot] >= 0 && HullDamage(slot_ship) >= XDR_ConfigInt(24, 60))
        {
            XDR_EnterCriticalShelter(slot, slot_ship, stranded_star);
            exit;
        }
        if(xdr_pending_item_ids[slot] > 1)
        {
            dword stranded_pending_item = IdToItem(xdr_pending_item_ids[slot]);
            if(stranded_pending_item && ItemIn(stranded_pending_item, slot_ship))
            {
                ShipPicksItem(slot_ship, stranded_pending_item, 0);
                XDR_RecordScavengedFuel(slot, stranded_pending_item);
                xdr_pending_item_ids[slot] = 0;
                xdr_pending_since[slot] = 0;
                XDR_TransferFuel(slot, slot_ship);
                XDR_SelectReturnPlace(slot, slot_ship, stranded_star);
                if(XDR_HasFuelForReturn(slot, slot_ship, stranded_star))
                {
                    xdr_no_loot_counts[slot] = 0;
                    xdr_states[slot] = 6;
                    exit;
                }
                if(!XDR_IsMissionBlockingBattle(stranded_star)) XDR_PrepareRecoveryFuel(slot, slot_ship, stranded_star);
                exit;
            }
            int stranded_pending_start_turn = xdr_pending_since[slot];
            if(stranded_pending_start_turn < 0) stranded_pending_start_turn = -stranded_pending_start_turn - 1;
            if(stranded_pending_item && ItemIn(stranded_pending_item, stranded_star) && CurTurn() - stranded_pending_start_turn <= 5)
            {
                XDR_StartMarkedPickup(slot, slot_ship, stranded_pending_item, CurTurn() == stranded_pending_start_turn);
                exit;
            }
            XDR_CancelPickup(slot, slot_ship);
        }
        int stranded_has_displaced = XDR_HasDisplacedLoot(slot, slot_ship, stranded_star);
        if(stranded_has_displaced && !XDR_IsMissionBlockingBattle(stranded_star))
        {
            int stranded_displaced_prepare = XDR_PrepareRecoveryFuel(slot, slot_ship, stranded_star);
            if(stranded_displaced_prepare) exit;
        }
        dword stranded_next_star = XDR_NextRouteStar(slot);
        if(stranded_next_star)
        {
            if(!XDR_ReturnTransitAllowed(slot, stranded_next_star))
            {
                XDR_ClearRoute(slot);
                xdr_return_place_ids[slot] = 0;
                xdr_return_kinds[slot] = 0;
            }
        }
        if(!XDR_IsMissionBlockingBattle(stranded_star))
        {
            int stranded_recovery_prepare = XDR_PrepareRecoveryFuel(slot, slot_ship, stranded_star);
            if(stranded_recovery_prepare) exit;
        }
        if(xdr_no_loot_counts[slot] >= 0 && xdr_pending_since[slot] >= 0 && ShipEqInSlot(slot_ship, t_RepairRobot) && HullDamage(slot_ship) <= 60)
        {
            xdr_states[slot] = 7;
            xdr_phase_turns[slot] = CurTurn();
            xdr_last_hull_damage[slot] = HullDamage(slot_ship);
            xdr_pending_since[slot] = ShipFuel(slot_ship);
            xdr_no_loot_counts[slot] = 0;
        }
        exit;
    }
}

function XDR_MayContinueImmediate(int immediate_slot, int immediate_before_id, int immediate_before_state)
{
    result = 0;
    if(xdr_ids[immediate_slot] <= 1 || xdr_ids[immediate_slot] != immediate_before_id) exit;
    if(xdr_states[immediate_slot] == immediate_before_state)
    {
        // A direct hook confirmation deliberately keeps the salvage state.
        // Continue the bounded chain when its marked item is already aboard so
        // confirmation, fuel transfer and selection of the next object happen
        // without consuming additional calendar days.
        int immediate_same_state = xdr_states[immediate_slot];
        if(immediate_same_state != 5 && immediate_same_state != 12 && immediate_same_state != 16) exit;
        if(xdr_pending_item_ids[immediate_slot] <= 1) exit;
        dword immediate_same_ship = IdToShip(xdr_ids[immediate_slot]);
        dword immediate_same_item = IdToItem(xdr_pending_item_ids[immediate_slot]);
        if(!immediate_same_ship || !immediate_same_item) exit;
        if(!ItemIn(immediate_same_item, immediate_same_ship)) exit;
        result = 1;
        exit;
    }
    // States 4 and 8 mean that OrderJump has just been issued. Some engine
    // builds do not expose ShipInHyperSpace until later in the tick, so never
    // run arrival logic immediately and accidentally undo the pre-jump mask.
    if(xdr_states[immediate_slot] == 4 || xdr_states[immediate_slot] == 8 || xdr_states[immediate_slot] == 14) exit;
    dword immediate_ship = IdToShip(xdr_ids[immediate_slot]);
    if(!immediate_ship) exit;
    if(ShipInHyperSpace(immediate_ship, 1) || ShipIsTakeoff(immediate_ship)) exit;
    result = 1;
}

function XDR_ProcessSlotImmediate(int immediate_slot)
{
    // Bookkeeping-only state changes should not consume another calendar day.
    // The chain is deliberately unrolled: an outer runtime loop would multiply
    // every inner galaxy scan and could turn a fast decision into a heavy tick.
    int immediate_id_1 = xdr_ids[immediate_slot];
    int immediate_state_1 = xdr_states[immediate_slot];
    XDR_ProcessSlot(immediate_slot);
    if(!XDR_MayContinueImmediate(immediate_slot, immediate_id_1, immediate_state_1)) exit;

    int immediate_id_2 = xdr_ids[immediate_slot];
    int immediate_state_2 = xdr_states[immediate_slot];
    XDR_ProcessSlot(immediate_slot);
    if(!XDR_MayContinueImmediate(immediate_slot, immediate_id_2, immediate_state_2)) exit;

    int immediate_id_3 = xdr_ids[immediate_slot];
    int immediate_state_3 = xdr_states[immediate_slot];
    XDR_ProcessSlot(immediate_slot);
    if(!XDR_MayContinueImmediate(immediate_slot, immediate_id_3, immediate_state_3)) exit;

    int immediate_id_4 = xdr_ids[immediate_slot];
    int immediate_state_4 = xdr_states[immediate_slot];
    XDR_ProcessSlot(immediate_slot);
    if(!XDR_MayContinueImmediate(immediate_slot, immediate_id_4, immediate_state_4)) exit;

    int immediate_id_5 = xdr_ids[immediate_slot];
    int immediate_state_5 = xdr_states[immediate_slot];
    XDR_ProcessSlot(immediate_slot);
    if(!XDR_MayContinueImmediate(immediate_slot, immediate_id_5, immediate_state_5)) exit;

    int immediate_id_6 = xdr_ids[immediate_slot];
    int immediate_state_6 = xdr_states[immediate_slot];
    XDR_ProcessSlot(immediate_slot);
}

function XDR_ResetIdentityArrays()
{
    xdr_old_factions = newarray(1);
    xdr_old_names = newarray(1);
    ArrayClear(xdr_old_factions);
    ArrayClear(xdr_old_names);
    for(int identity_slot = 1; identity_slot <= 3; identity_slot = identity_slot + 1)
    {
        ArrayAdd(xdr_old_factions, '');
        ArrayAdd(xdr_old_names, '');
    }
}

function XDR_DisableIncompatibleStorage()
{
    // XenoDomRangers has no save migrations before its first stable release.
    // If an experimental save contains another schema, release only resolvable
    // ships and disable this save instead of interpreting old arrays as current
    // state or silently reinitializing them under still-masked NPCs.
    if(ArrayDim(xdr_ids) == 4)
    {
        for(int incompatible_slot = 1; incompatible_slot <= 3; incompatible_slot = incompatible_slot + 1)
        {
            if(xdr_ids[incompatible_slot] <= 1) continue;
            dword incompatible_ship = IdToShip(xdr_ids[incompatible_slot]);
            if(!incompatible_ship) continue;
            XDR_NativeClearMask(incompatible_ship);
            OrderLock(incompatible_ship, 0);
        }
    }
    XDR_NativeReset();
    xdr_disabled = 1;
}

function XDR_InitStorage()
{
    XDR_NativeReset();
    xdr_ids = newarray(1);
    xdr_states = newarray(1);
    xdr_home_base_ids = newarray(1);
    xdr_target_star_ids = newarray(1);
    xdr_series = newarray(1);
    xdr_due_turns = newarray(1);
    xdr_phase_turns = newarray(1);
    xdr_pending_item_ids = newarray(1);
    xdr_pending_since = newarray(1);
    xdr_no_loot_counts = newarray(1);
    xdr_threat_counts = newarray(1);
    xdr_old_owners = newarray(1);
    xdr_old_standings = newarray(1);
    xdr_old_no_targets = newarray(1);
    XDR_ResetIdentityArrays();
    xdr_ghost_flags = newarray(1);
    xdr_visual_flags = newarray(1);
    xdr_charge_expire_turns = newarray(1);
    xdr_last_hull_damage = newarray(1);
    xdr_return_place_ids = newarray(1);
    xdr_return_kinds = newarray(1);
    xdr_release_flags = newarray(1);
    xdr_dialog_markers = newarray(1);
    XDR_ResetRouteArrays();
    XDR_ResetScavengedFuelArrays();
    ArrayClear(xdr_ids); ArrayClear(xdr_states); ArrayClear(xdr_home_base_ids); ArrayClear(xdr_target_star_ids);
    ArrayClear(xdr_series); ArrayClear(xdr_due_turns); ArrayClear(xdr_phase_turns); ArrayClear(xdr_pending_item_ids);
    ArrayClear(xdr_pending_since); ArrayClear(xdr_no_loot_counts); ArrayClear(xdr_threat_counts); ArrayClear(xdr_old_owners);
    ArrayClear(xdr_old_standings); ArrayClear(xdr_old_no_targets); ArrayClear(xdr_ghost_flags); ArrayClear(xdr_visual_flags);
    ArrayClear(xdr_charge_expire_turns); ArrayClear(xdr_last_hull_damage); ArrayClear(xdr_return_place_ids); ArrayClear(xdr_return_kinds);
    ArrayClear(xdr_release_flags); ArrayClear(xdr_dialog_markers);
    for(int init_slot = 1; init_slot <= 3; init_slot = init_slot + 1)
    {
        ArrayAdd(xdr_ids, 0); ArrayAdd(xdr_states, 0); ArrayAdd(xdr_home_base_ids, 0); ArrayAdd(xdr_target_star_ids, 0);
        ArrayAdd(xdr_series, -1); ArrayAdd(xdr_due_turns, CurTurn()); ArrayAdd(xdr_phase_turns, 0); ArrayAdd(xdr_pending_item_ids, 0);
        ArrayAdd(xdr_pending_since, 0); ArrayAdd(xdr_no_loot_counts, 0); ArrayAdd(xdr_threat_counts, 0); ArrayAdd(xdr_old_owners, 0);
        ArrayAdd(xdr_old_standings, 0); ArrayAdd(xdr_old_no_targets, 0); ArrayAdd(xdr_ghost_flags, 0); ArrayAdd(xdr_visual_flags, 0);
        ArrayAdd(xdr_charge_expire_turns, 0); ArrayAdd(xdr_last_hull_damage, 0); ArrayAdd(xdr_return_place_ids, 0); ArrayAdd(xdr_return_kinds, 0);
        ArrayAdd(xdr_release_flags, 0);
        ArrayAdd(xdr_dialog_markers, 0);
    }
    xdr_schema_version = 6;
    xdr_initialized = 1;
    xdr_next_recruit_turn = CurTurn() + XDR_ConfigInt(1, 100);
    if(xdr_test_mode) xdr_next_recruit_turn = CurTurn() + 5;
}

// The production build still waits for the player UI lifecycle. The dedicated
// test build must not depend on t_OnEnteringForm: ExpPilotBridge injects its
// controls through DialogBegin and can show them without firing that state
// event. Its test-mode GenerationBarrier starts only after turn zero.
if(!xdr_ui_ready && !xdr_test_mode) exit;
if(!xdr_test_mode && CurTurn() <= xdr_ui_ready_turn) exit;

// This call is deliberately before every save-state check. It gives the native
// runtime a stable world marker so stale ship pointers from a previously loaded
// save are invalidated before any mask is reasserted or cleared.
XDR_NativeWorldTick();

if(!xdr_initialized)
{
    XDR_InitStorage();
    if(!xdr_test_mode) exit;
}

if(xdr_schema_version != 6)
{
    XDR_DisableIncompatibleStorage();
    exit;
}

if(ArrayDim(xdr_ids) != 4 || ArrayDim(xdr_states) != 4 || ArrayDim(xdr_home_base_ids) != 4 ||
   ArrayDim(xdr_target_star_ids) != 4 || ArrayDim(xdr_series) != 4 || ArrayDim(xdr_due_turns) != 4 ||
   ArrayDim(xdr_old_factions) != 4 || ArrayDim(xdr_old_names) != 4 ||
   ArrayDim(xdr_dialog_markers) != 4 ||
   ArrayDim(xdr_route_slots) != 4 || ArrayDim(xdr_route_orders) != 4 || ArrayDim(xdr_route_star_ids) != 4)
{
    XDR_DisableIncompatibleStorage();
    exit;
}

if(xdr_test_mode && xdr_test_spawn_stalker) XDR_TestSpawnStalker();
if(xdr_test_mode && xdr_test_quick_stage != 0) XDR_TestQuickScenario();

if(xdr_test_mode && xdr_test_release_all && ArrayDim(xdr_release_flags) == 4)
{
    int release_live_count = (xdr_ids[1] > 1) + (xdr_ids[2] > 1) + (xdr_ids[3] > 1);
    if(xdr_ids[1] > 1) xdr_release_flags[1] = 1;
    if(xdr_ids[2] > 1) xdr_release_flags[2] = 1;
    if(xdr_ids[3] > 1) xdr_release_flags[3] = 1;
    if(!release_live_count)
    {
        xdr_test_release_all = 0;
        xdr_test_last_event = 'все специалисты освобождены';
    }
}
